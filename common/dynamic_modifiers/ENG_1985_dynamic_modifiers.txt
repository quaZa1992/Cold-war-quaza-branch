# =============================================================================
# ENG 1985 - The Troubles state modifiers (Imperia RP Cold War start)
# =============================================================================
# Pattern lifted 1:1 from kurdish_resistance_dynamic_modifier + pkk_static_modifier
# in 02_turkish_dynamic_modifiers.txt:
#
# ENG_ira_troubles_state_modifier -> dynamic, scales with the country-scope
# IRA sliders. Drives resistance target, garrison requirement, garrison damage,
# garrison penetration chance, and economy debuffs that climb with militancy.
# ENG_ira_static_modifier -> always-on "state spirit" with fixed severe debuffs
# that simulate a permanent low-intensity insurgency regardless of slider values.
#
# Both are added to state 17 (Northern Ireland) by
# ENG_1985_ira_apply_state_modifier in scripted_effects, only in the 1985 start
# (has_start_date < 1990.1.1). Together they reproduce the PKK feel: Northern
# Ireland is permanently degraded as a province, AND that degradation gets
# worse the higher the IRA militancy rises.
# =============================================================================

# -----------------------------------------------------------------------------
# DYNAMIC scaling modifier - reads from ENG country variables
# -----------------------------------------------------------------------------
ENG_ira_troubles_state_modifier = {
	enable = { always = yes }
	icon = GFX_idea_sectarian_troubles_kurdistan
	custom_modifier_tooltip = ENG_ira_troubles_state_modifier_tt

	# Resistance pressure - mirrors kurdish_resistance_dynamic_modifier.
	resistance_target = ENG.ENG_ira_resistance_target
	required_garrison_factor = ENG.ENG_ira_garrison_requirement
	resistance_damage_to_garrison = ENG.ENG_ira_garrison_damage
	resistance_garrison_penetration_chance = ENG.ENG_ira_garrison_penetration

	# Economic / military drag that climbs with militancy.
	local_org_regain = ENG.ENG_ira_local_org_penalty
	state_resources_factor = ENG.ENG_ira_resources_penalty
	local_building_slots_factor = ENG.ENG_ira_building_slots_penalty
	army_speed_factor_for_controller = ENG.ENG_ira_army_speed_penalty
}

# -----------------------------------------------------------------------------
# STATIC "state spirit" - permanent severe Troubles debuff
# -----------------------------------------------------------------------------
# Even with IRA militancy at zero this fires - it's the baseline cost of
# garrisoning Northern Ireland in 1985. Removed only when state 17 leaves ENG
# control (e.g. via the radical "Troops Out" focus path).

}
