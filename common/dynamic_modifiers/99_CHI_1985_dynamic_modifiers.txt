# =============================================================================
# CHINA 1985 - Dynamic modifier (Imperia RP Cold War)
# =============================================================================
# Single dynamic modifier consuming the multipliers computed by
# CHI_1985_monthly_state_update. Wrapper-only; the math lives in scripted
# effects. Disables itself outside the 1985 cold-war start so the MD-2000
# China game is untouched.
# =============================================================================

CHI_1985_state_dynamic_modifier = {
	icon = "GFX_idea_emblem_of_china"
	enable = {
		original_tag = CHI
		has_country_flag = CHI_1985_state_system_enabled
		has_start_date < 1990.1.1
	}
	remove_trigger = {
		OR = {
			NOT = { original_tag = CHI }
			NOT = { has_country_flag = CHI_1985_state_system_enabled }
			NOT = { has_start_date < 1990.1.1 }
		}
	}

	# --- Reform pressure block ----------------------------------------------
	stability_factor = CHI_reform_pressure_stability_factor
	political_power_factor = CHI_reform_pressure_political_power_factor
	productivity_growth_modifier = CHI_reform_pressure_productivity_growth_modifier
	consumer_goods_factor = CHI_reform_pressure_consumer_goods_factor

	# --- Party control block ------------------------------------------------
	drift_defence_factor = CHI_party_control_drift_defence_factor
	research_speed_factor = CHI_party_control_research_speed_factor

	# --- PLA modernisation block --------------------------------------------
	army_org_factor = CHI_pla_modernization_army_org_factor
	planning_speed = CHI_pla_modernization_planning_speed
	training_time_army_factor = CHI_pla_modernization_training_time_factor
}
