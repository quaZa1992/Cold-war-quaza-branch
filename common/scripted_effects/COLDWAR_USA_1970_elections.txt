COLDWAR_USA_election_begin = {
	if = {
		limit = {
			tag = USA
			original_tag = USA
			is_dynamic_country = no
			IMP_is_1970_scenario = yes
			has_elections = yes
		}
		clear_variable = COLDWAR_USA_pending_election_year
		if = {
			limit = {
				date > 1971.12.31
				date < 1973.1.1
			}
			set_variable = {
				COLDWAR_USA_pending_election_year = 1972
			}
		}
		if = {
			limit = {
				date > 1975.12.31
				date < 1977.1.1
			}
			set_variable = {
				COLDWAR_USA_pending_election_year = 1976
			}
		}
		if = {
			limit = {
				date > 1979.12.31
				date < 1981.1.1
			}
			set_variable = {
				COLDWAR_USA_pending_election_year = 1980
			}
		}
		if = {
			limit = {
				date > 1983.12.31
				date < 1985.1.1
			}
			set_variable = {
				COLDWAR_USA_pending_election_year = 1984
			}
		}
		if = {
			limit = {
				date > 1987.12.31
				date < 1989.1.1
			}
			set_variable = {
				COLDWAR_USA_pending_election_year = 1988
			}
		}
		if = {
			limit = {
				date > 1991.12.31
				date < 1993.1.1
			}
			set_variable = {
				COLDWAR_USA_pending_election_year = 1992
			}
		}
		if = {

				date < 1997.1.1
			}
			set_variable = {
				COLDWAR_USA_pending_election_year = 1996
			}
		}
	}
}
