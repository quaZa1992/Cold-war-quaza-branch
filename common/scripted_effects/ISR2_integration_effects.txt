ISR2_initialize = {
	if = {
		limit = { ISR2_is_scenario = yes }
		set_global_flag = ISR2_legacy_replaced
		if = {
			limit = {
				NOT = { has_country_flag = ISR2_initialized }
			}
			if = {
				limit = { has_country_flag = ISR70_initialized }
				set_country_flag = ISR2_legacy_save
			}
			if = {
				limit = {
					NOT = { has_country_flag = ISR2_legacy_save }
					date < 1970.1.2
				}
				set_variable = { election_threshold = 0.01 }
			}
			set_country_flag = ISR2_enabled
			ISR2_retire_legacy_modifiers = yes
			if = {
				limit = {
					NOT = { has_focus_tree = ISR70_focus }
				}
				load_focus_tree = ISR70_focus
			}
			ISR2_bilateral_init = yes
			ISR2_domestic_init = yes
			ISR2_territories_init = yes
			ISR2_core_init = yes
			ISR70_restrict_party_slots = yes
			set_variable = { ISR70_leader_slot = ruling_party }
			set_country_flag = ISR2_initialized
			clr_country_flag = ISR2_bootstrap_pending
		}
		ISR2_refresh = yes
	}
}
ISR2_refresh = {
	if = {
		limit = { ISR2_is_scenario = yes has_country_flag = ISR2_initialized }
		ISR2_legacy_refund_pending = yes
		ISR2_territories_refresh = yes
		ISR2_bilateral_reconcile = yes
		ISR2_territories_pressure_reconcile = yes
		ISR2_domestic_refresh = yes
		ISR2_core_refresh = yes
	}
}
ISR2_daily = {
	if = {
		limit = { ISR2_is_scenario = yes }

			limit = {

		ISR2_territories_pressure_reconcile = yes

		ISR2_core_daily = yes

		if = {
			limit = {
				NOT = {
					check_variable = { ISR70_leader_slot = ruling_party }
				}
			}
			ISR70_set_party_leader = yes
