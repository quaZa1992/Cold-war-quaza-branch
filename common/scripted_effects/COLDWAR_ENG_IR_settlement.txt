	if = {
		limit = { has_idea = IMP_ENG_1970_industrial_relations_strain }
		swap_ideas = { remove_idea = IMP_ENG_1970_industrial_relations_strain add_idea = COLDWAR_ENG_IR_agreement }
	}
	else_if = {
		limit = { has_idea = COLDWAR_ENG_BALANCE_relations_partial }
		swap_ideas = { remove_idea = COLDWAR_ENG_BALANCE_relations_partial add_idea = COLDWAR_ENG_IR_agreement }
	}
	else = { add_ideas = COLDWAR_ENG_IR_agreement }
	hidden_effect = {
		set_country_flag = COLDWAR_ENG_BALANCE_relations_resolved
		
	}
}

COLDWAR_ENG_IR_weekly_expenses = {
	set_variable = { COLDWAR_ENG_IR_weekly_bill = 0 }
	if = {
		limit = { has_idea = COLDWAR_ENG_IR_floor COLDWAR_ENG_IR_parliament = yes }
		set_variable = { COLDWAR_ENG_IR_weekly_bill = { value = gdp_total multiply = 0.00104 divide = 52 } }
	}
	else_if = {
		limit = { has_idea = COLDWAR_ENG_IR_councils COLDWAR_ENG_IR_parliament = yes }
		set_variable = { COLDWAR_ENG_IR_weekly_bill = { value = gdp_total multiply = 0.00052 divide = 52 } }
	}
	else_if = {
		limit = { has_idea = COLDWAR_ENG_IR_inspectorate COLDWAR_ENG_IR_parliament = yes }
		set_variable = { COLDWAR_ENG_IR_weekly_bill = { value = gdp_total multiply = 0.00156 divide = 52 } }
	}
	else_if = {
		limit = { has_idea = COLDWAR_ENG_IR_boards COLDWAR_ENG_IR_parliament = yes }
		set_variable = { COLDWAR_ENG_IR_weekly_bill = { value = gdp_total multiply = 0.00104 divide = 52 } }
	}
	add_to_variable = { additional_expenses_rate = COLDWAR_ENG_IR_weekly_bill }
}
