# IMPERIA COLD WAR — Recognition catalogue independent of the native unrecognised-state registry.
ICW_register_recognition_candidate = {
	if = {
		limit = {
			exists = yes
			COLDWAR_recognition_subject = yes
			NOT = { is_in_array = { global.ICW_recognition_candidates = THIS } }
		}
		add_to_array = { global.ICW_recognition_candidates = THIS }
	}
}

# IMPERIA COLD WAR — Rebuild only the small recognition catalogue when its tab opens.
ICW_refresh_recognition_candidates = {
	clear_array = global.ICW_recognition_candidates
	for_each_scope_loop = {
		array = global.Unrecognised_States
		ICW_register_recognition_candidate = yes
	}
	if = {
		limit = { IMP_is_1970_scenario = yes }
		if = { limit = { country_exists = GER } GER = { ICW_register_recognition_candidate = yes } }
		if = { limit = { country_exists = DDR } DDR = { ICW_register_recognition_candidate = yes } }
		if = { limit = { country_exists = CHI } CHI = { ICW_register_recognition_candidate = yes } }
		if = { limit = { country_exists = TAI } TAI = { ICW_register_recognition_candidate = yes } }
		if = { limit = { country_exists = NKO } NKO = { ICW_register_recognition_candidate = yes } }
		if = { limit = { country_exists = KOR } KOR = { ICW_register_recognition_candidate = yes } }
		if = { limit = { country_exists = VIE } VIE = { ICW_register_recognition_candidate = yes } }
		if = { limit = { country_exists = SVN } SVN = { ICW_register_recognition_candidate = yes } }
		if = { limit = { country_exists = ISR } ISR = { ICW_register_recognition_candidate = yes } }
		if = { limit = { country_exists = MON } MON = { ICW_register_recognition_candidate = yes } }
		if = { limit = { country_exists = BIA } BIA = { ICW_register_recognition_candidate = yes } }
		if = { limit = { country_exists = ZIM } ZIM = { ICW_register_recognition_candidate = yes } }
	}
}
