# -----------------------------------------------------------------------------
# Compute the multipliers consumed by the dynamic modifier from the sliders.
# Pattern lifted from SOV_1985_monthly_outlook_update — keep numbers small,
# this fires monthly.
# -----------------------------------------------------------------------------

CHI_1985_monthly_state_update = {
	if = {
		limit = { has_country_flag = CHI_1985_state_system_enabled }

		CHI_1985_clamp_state_system = yes

		# --- Reform pressure -> stability swing + productivity boost -----
		# stability_factor = (CHI_reform_momentum - 50) * 0.0015
		set_variable = { CHI_reform_pressure_stability_factor = CHI_reform_momentum }
		subtract_from_variable = { CHI_reform_pressure_stability_factor = 50 }
		multiply_variable = { CHI_reform_pressure_stability_factor = 0.0015 }

		# productivity_growth_modifier = CHI_reform_momentum * 0.0008
		set_variable = { CHI_reform_pressure_productivity_growth_modifier = CHI_reform_momentum }
		multiply_variable = { CHI_reform_pressure_productivity_growth_modifier = 0.0008 }

		# political_power_factor = -(CHI_inflation_pressure + CHI_corruption_pressure - 50) * 0.0015
		set_variable = { CHI_reform_pressure_political_power_factor = CHI_inflation_pressure }
		add_to_variable = { CHI_reform_pressure_political_power_factor = CHI_corruption_pressure }
		subtract_from_variable = { CHI_reform_pressure_political_power_factor = 50 }
		multiply_variable = { CHI_reform_pressure_political_power_factor = -0.0015 }

		# consumer_goods_factor = (CHI_inflation_pressure - 30) * 0.001 (positive = bad)
		set_variable = { CHI_reform_pressure_consumer_goods_factor = CHI_inflation_pressure }
		subtract_from_variable = { CHI_reform_pressure_consumer_goods_factor = 30 }
		multiply_variable = { CHI_reform_pressure_consumer_goods_factor = 0.001 }

		# --- Party control -> stability + drift defence + research drag --
		set_variable = { CHI_party_control_stability_factor = CHI_party_control }
		subtract_from_variable = { CHI_party_control_stability_factor = 50 }
		multiply_variable = { CHI_party_control_stability_factor = 0.001 }

		set_variable = { CHI_party_control_drift_defence_factor = CHI_party_control }
		multiply_variable = { CHI_party_control_drift_defence_factor = 0.0025 }

		set_variable = { CHI_party_control_political_power_factor = CHI_party_control }
		subtract_from_variable = { CHI_party_control_political_power_factor = 60 }
		multiply_variable = { CHI_party_control_political_power_factor = 0.001 }

		# Higher party_control = less academic freedom = less research speed
		set_variable = { CHI_party_control_research_speed_factor = CHI_party_control }
		subtract_from_variable = { CHI_party_control_research_speed_factor = 60 }
		multiply_variable = { CHI_party_control_research_speed_factor = -0.0008 }

		# --- PLA modernization -> army org + planning + training time ----
		set_variable = { CHI_pla_modernization_army_org_factor = CHI_pla_modernization }
		subtract_from_variable = { CHI_pla_modernization_army_org_factor = 30 }
		multiply_variable = { CHI_pla_modernization_army_org_factor = 0.0015 }

		set_variable = { CHI_pla_modernization_planning_speed = CHI_pla_modernization }
		multiply_variable = { CHI_pla_modernization_planning_speed = 0.002 }

		# More modernization = less training time penalty
		set_variable = { CHI_pla_modernization_training_time_factor = CHI_pla_modernization }
		multiply_variable = { CHI_pla_modernization_training_time_factor = -0.0015 }

		# --- Coastal/Inland gap -> stability + productivity + consumer goods
		set_temp_variable = { CHI_temp_gap = CHI_coastal_growth }
		subtract_from_temp_variable = { CHI_temp_gap = CHI_inland_neglect }

		set_variable = { CHI_coastal_gap_stability_factor = CHI_temp_gap }
		multiply_variable = { CHI_coastal_gap_stability_factor = -0.001 }

		set_variable = { CHI_coastal_gap_productivity_growth_modifier = CHI_temp_gap }
		multiply_variable = { CHI_coastal_gap_productivity_growth_modifier = 0.001 }

		set_variable = { CHI_coastal_gap_consumer_goods_factor = CHI_temp_gap }
		multiply_variable = { CHI_coastal_gap_consumer_goods_factor = -0.0008 }
	}
}
