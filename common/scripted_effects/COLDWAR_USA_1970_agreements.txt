	add_doctrine_cost_reduction = { name = COLDWAR_USA_agreements_germany_forward_GER_doctrine cost_reduction = 0.5 uses = 1 category = land_doctrine }

	set_country_flag = COLDWAR_USA_agreements_germany_forward_signed

	GER = {
	set_country_flag = COLDWAR_USA_agreements_germany_forward_signed

	}
