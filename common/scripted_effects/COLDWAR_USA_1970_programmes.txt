COLDWAR_USA_dom_preview_E1 = {
	set_temp_variable = { treasury_change = -30 }
	modify_treasury_effect = yes
	add_ideas = COLDWAR_USA_dom_E1_s0
	random_owned_state = {
		limit = {
			is_core_of = USA
			is_owned_and_controlled_by = ROOT
			OR = { state = 771 state = 775 state = 777 state = 778 }
			check_variable = { var = building_level@industrial_complex value = 48 compare = less_than_or_equals }
			free_building_slots = { building = industrial_complex size > 1 include_locked = no }
		}
		add_extra_state_shared_building_slots = 2
		add_building_construction = { type = industrial_complex level = 2 instant_build = yes }
	}
}
