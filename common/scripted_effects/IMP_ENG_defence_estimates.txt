IMP_ENG_estimates_recalculate = {
	set_variable = { IMP_ENG_estimates_army_org_factor = COLDWAR_ENG_POLISH_defence_army_org_factor }
	set_variable = { IMP_ENG_estimates_navy_max_range_factor = COLDWAR_ENG_POLISH_defence_navy_max_range_factor }
	set_variable = { IMP_ENG_estimates_navy_org_factor = COLDWAR_ENG_POLISH_defence_navy_org_factor }
	set_variable = { IMP_ENG_estimates_navy_screen_defence_factor = COLDWAR_ENG_POLISH_defence_navy_screen_defence_factor }
	set_variable = { IMP_ENG_estimates_navy_submarine_detection_factor = COLDWAR_ENG_POLISH_defence_navy_submarine_detection_factor }
	set_variable = { IMP_ENG_estimates_war_support_factor = COLDWAR_ENG_POLISH_defence_war_support_factor }
	add_to_variable = { IMP_ENG_estimates_navy_org_factor = COLDWAR_ENG_POLISH_economy_navy_org_factor }
	if = {
		limit = { IMP_ENG_estimates_rhine_active = yes }
		add_to_variable = { IMP_ENG_estimates_army_org_factor = 0.02 }
		add_to_variable = { IMP_ENG_estimates_navy_max_range_factor = -0.04 }
		if = {
			limit = { IMP_ENG_estimates_rhine_trained = yes }
			add_to_variable = { IMP_ENG_estimates_army_org_factor = 0.01 }
		}
	}
	if = {
		limit = { IMP_ENG_estimates_atlantic_active = yes }
		add_to_variable = { IMP_ENG_estimates_navy_max_range_factor = 0.04 }
		add_to_variable = { IMP_ENG_estimates_navy_submarine_detection_factor = 0.04 }
		add_to_variable = { IMP_ENG_estimates_army_org_factor = -0.02 }
		if = {
			limit = { IMP_ENG_estimates_atlantic_trained = yes }
			add_to_variable = { IMP_ENG_estimates_navy_submarine_detection_factor = 0.02 }
		}
	}
	if = {
		limit = { IMP_ENG_estimates_patrol_active = yes }
		add_to_variable = { IMP_ENG_estimates_navy_org_factor = 0.03 }
		add_to_variable = { IMP_ENG_estimates_war_support_factor = 0.02 }
		add_to_variable = { IMP_ENG_estimates_navy_screen_defence_factor = -0.04 }
		if = {
			limit = { IMP_ENG_estimates_patrol_trained = yes }
			add_to_variable = { IMP_ENG_estimates_navy_org_factor = 0.01 }
		}
	}
}
