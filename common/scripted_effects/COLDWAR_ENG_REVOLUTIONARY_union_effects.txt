# Repair a false acceptance made by a third-party observer of the old invitation.

 if = {
  limit = {
   original_tag = ENG
   COLDWAR_ENG_REVOLUTIONARY_government = yes
   has_country_flag = COLDWAR_ENG_REVOLUTIONARY_ireland_member_accepted
   NOT = { has_country_flag = COLDWAR_ENG_REVOLUTIONARY_irish_war_observed }
   IRE = {
    exists = yes
    is_subject_of = ENG
    has_country_flag = COLDWAR_ENG_REVOLUTIONARY_ireland_refused
    NOT = { has_country_flag = COLDWAR_ENG_REVOLUTIONARY_ireland_membership_accepted }
   }
  }
  set_autonomy = { target = IRE autonomy_state = autonomy_free end_wars = no end_civil_wars = no }
  clr_country_flag = COLDWAR_ENG_REVOLUTIONARY_ireland_member_accepted
  set_country_flag = COLDWAR_ENG_REVOLUTIONARY_ireland_refused
  IRE = { if = { limit = { has_cosmetic_tag = COLDWAR_ENG_REVOLUTIONARY_IRELAND } drop_cosmetic_tag = yes } }
 }
}

    }
