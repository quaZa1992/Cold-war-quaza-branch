# IMPERIA COLD WAR — Sheep's South African focus programmes and National Spirits.
# COLDWAR | The national constitutional monarchy is a state institution.
COLDWAR_SAF_MONARCHY_sync = {
 # Migrate a previously completed focus from an older save once.
 if = { limit = { has_completed_focus = COLDWAR_SAF_PARTY_ifp_royal_national NOT = { has_country_flag = COLDWAR_SAF_MONARCHY_established } NOT = { has_country_flag = COLDWAR_SAF_MONARCHY_abolished } } set_country_flag = COLDWAR_SAF_MONARCHY_established }
 # SACP and the National Party / apartheid or radical Boer parties abolish the Crown.
 if = {
  limit = { has_country_flag = COLDWAR_SAF_MONARCHY_established OR = { check_variable = { ruling_party = 0 } check_variable = { ruling_party = 4 } check_variable = { ruling_party = 6 } check_variable = { ruling_party = 13 } check_variable = { ruling_party = 20 } check_variable = { ruling_party = 21 } } }
  clr_country_flag = COLDWAR_SAF_MONARCHY_established
  set_country_flag = COLDWAR_SAF_MONARCHY_abolished
  remove_ideas = COLDWAR_SAF_IFP_national_crown
  if = { limit = { check_variable = { ruling_party = 4 } } set_cosmetic_tag = COLDWAR_SAF_COMMUNIST }
  else_if = { limit = { check_variable = { ruling_party = 0 } } set_cosmetic_tag = SAF_1970 }
  else_if = { limit = { check_variable = { ruling_party = 21 } } set_cosmetic_tag = COLDWAR_SAF_HIST_RADICAL }
  else = { set_cosmetic_tag = COLDWAR_SAF_HIST_ORANGE }
 }
 if = { limit = { has_country_flag = COLDWAR_SAF_MONARCHY_established NOT = { has_idea = COLDWAR_SAF_IFP_national_crown } } add_ideas = COLDWAR_SAF_IFP_national_crown }
}
