# The parties historically admitted to the Front keep their native MD coalition slots.
COLDWAR_SYR_RP_admit_front_communists = {
	if = {
		limit = { check_variable = { party_pop_array^4 > 0.04999 } }
		set_variable = { party_banned_array^4 = 0 }
		set_temp_variable = { add_col_one = 4 }
		add_coalition_members_effect = yes
	}
}

		add_coalition_members_effect = yes
	}
