USA_1985_raise_influence = {
	set_temp_variable = { tag_index = USA }
	change_influence_percentage = yes
}

USA_1985_reduce_soviet_influence = {
	set_temp_variable = { tag_index = SOV }
	change_influence_percentage = yes
}

USA_1985_pay_operation_cyclone_equipment = {
	if = {
		limit = { has_equipment = { infantry_weapons_1 > 3999 } }
		add_equipment_to_stockpile = { type = infantry_weapons_1 amount = -4000 producer = USA }
	}
	if = {
		limit = { has_equipment = { AA_Equipment > 249 } }
		add_equipment_to_stockpile = { type = AA_Equipment amount = -250 producer = USA }
	}
	if = {
		limit = { has_equipment = { cnc_equipment_type > 499 } }
		add_equipment_to_stockpile = { type = cnc_equipment_type amount = -500 producer = USA }
	}
}

USA_1985_apply_operation_cyclone = {
	set_country_flag = USA_1985_operation_cyclone_active
	USA_1985_pay_operation_cyclone_equipment = yes
	add_political_power = -35
	if = {
		limit = { country_exists = SOV }
		SOV = {
			add_timed_idea = { idea = USA_1985_operation_cyclone_bleeding days = 365 }
			add_stability = -0.01
		}
	}
	if = {
		limit = { country_exists = AFG }
		set_temp_variable = { influence_target = AFG }
		set_temp_variable = { percent_change = 5 }
		USA_1985_raise_influence = yes
		if = {
			limit = { country_exists = SOV }
			set_temp_variable = { influence_target = AFG }
			set_temp_variable = { percent_change = -6 }
			USA_1985_reduce_soviet_influence = yes
		}
	}
}

USA_1985_apply_contra_pressure = {

			add_timed_idea = { idea = USA_1985_contra_insurgency_pressure days = 365 }

		}
	}
}

				add_offsite_building = { type = industrial_complex level = 2 }

		USA_1985_raise_influence = yes
	}
	if = {
		limit = { country_exists = CUB }

			add_timed_idea = { idea = USA_1985_cuban_overseas_war_costs days = 365 }
			add_stability = -0.01
		}
	}
}
