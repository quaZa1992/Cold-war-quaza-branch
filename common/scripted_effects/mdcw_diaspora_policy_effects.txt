   set_temp_variable = { gw_lookup_origin = ROOT.id } mdcw_gw_find_origin = yes

    ROOT = { set_variable = { gw_dp_stock = PREV.gw_stock^PREV.gw_index } }

 clamp_variable = { var = gw_dp_trade_k min = 0 max = gw_dp_space }

# One bilateral offer per participant. All callbacks revalidate the captured pair.
mdcw_dp_request_trade = {
 mdcw_dp_prepare_selected = yes
 if = { limit = { mdcw_dp_can_trade = yes NOT = { has_country_flag = gw_dialog_window } }
  add_political_power = -50
  set_variable = { gw_dp_request_kind = 2 } set_variable = { gw_dp_request_pp = 50 }
  set_variable = { gw_dp_request_k = gw_dp_trade_k } set_variable = { gw_dp_request_cash = gw_dp_trade_cash }
  mdcw_dp_send_offer = yes
 }
}
mdcw_dp_request_skill = {
 mdcw_dp_prepare_selected = yes
 if = { limit = { mdcw_dp_can_skill = yes NOT = { has_country_flag = gw_dialog_window } }
  add_political_power = -100
  set_variable = { gw_dp_request_kind = 3 } set_variable = { gw_dp_request_pp = 100 }
  set_variable = { gw_dp_request_k = gw_dp_skill_k } set_variable = { gw_dp_request_cash = gw_dp_skill_cash }

mdcw_dp_send_offer = {
 set_country_flag = gw_dp_request
 set_variable = { gw_dp_partner = gw_dp_target }
 add_to_variable = { gw_ui_dirty = 1 }
 country_event = { id = mdcw_dp.90 days = 17 }
 var:gw_dp_partner = {
  set_country_flag = gw_dp_offer
  set_country_flag = { flag = gw_dp_offer_window days = 19 }
  set_variable = { gw_dp_partner = ROOT.id }
  set_variable = { gw_dp_offer_kind = ROOT.gw_dp_request_kind }
  set_variable = { gw_dp_offer_cash = ROOT.gw_dp_request_cash }
  set_variable = { gw_dp_offer_k = ROOT.gw_dp_request_k }
  set_variable = { gw_dp_offer_compensation = gw_dp_offer_cash } divide_variable = { gw_dp_offer_compensation = 2 }
  add_to_variable = { gw_ui_dirty = 1 }
  if = { limit = { check_variable = { gw_dp_offer_kind = 2 } } country_event = { id = mdcw_dp.1 days = 1 } }
  else = { country_event = { id = mdcw_dp.2 days = 1 } }
  country_event = { id = mdcw_dp.91 days = 17 }
 }
}
mdcw_dp_clear_pending = {
 clr_country_flag = gw_dp_request clr_country_flag = gw_dp_offer
 clear_variable = gw_dp_partner
 add_to_variable = { gw_ui_dirty = 1 }
}
mdcw_dp_close_offer = {
 if = { limit = { has_country_flag = gw_dp_offer_window }
  clr_country_flag = gw_dp_offer_window add_to_variable = { gw_ui_dirty = 1 }
 }
}
mdcw_dp_refund_offer = {
 if = { limit = { has_country_flag = gw_dp_request }
  add_political_power = gw_dp_request_pp
  var:gw_dp_partner = {
   if = { limit = { has_country_flag = gw_dp_offer var:gw_dp_partner = { tag = ROOT } } mdcw_dp_clear_pending = yes }
  }
  mdcw_dp_clear_pending = yes
 }
}
mdcw_dp_accept_offer = {
 if = { limit = { mdcw_dp_pair_pending = yes }
  set_variable = { gw_dp_target = gw_dp_partner }
  # Offer's own locks are permitted; no new request can overwrite either party.
  mdcw_dp_prepare_target = yes
  if = { limit = {
   is_ai = no
   check_variable = { gw_dp_active_count < 2 }
   NOT = { check_variable = { treasury < gw_dp_request_cash } }
   OR = {
    AND = { check_variable = { gw_dp_request_kind = 2 } NOT = { check_variable = { gw_dp_trade_k < gw_dp_request_k } } }
    AND = { check_variable = { gw_dp_request_kind = 3 } NOT = { has_country_flag = gw_dp_skill_cooldown } NOT = { check_variable = { gw_dp_skill_k < gw_dp_request_k } } }
   }
   var:gw_dp_target = { NOT = { has_country_flag = gw_dp_cooldown@ROOT } }
  }
   set_variable = { gw_dp_new_type = gw_dp_request_kind }
   set_variable = { gw_dp_new_k = gw_dp_request_k }
   if = { limit = { check_variable = { gw_dp_new_type = 2 } }
    subtract_from_variable = { treasury = gw_dp_request_cash }
    var:gw_dp_target = { add_to_variable = { treasury = ROOT.gw_dp_request_cash } ingame_update_setup = yes }
   }
   else = { set_country_flag = { flag = gw_dp_skill_cooldown days = 730 } }
   mdcw_dp_start_slot = yes
   var:gw_dp_target = { set_country_flag = { flag = gw_dp_cooldown@ROOT days = 365 } }
   var:gw_dp_partner = { mdcw_dp_clear_pending = yes }
   mdcw_dp_clear_pending = yes
   ingame_update_setup = yes
   if = { limit = { check_variable = { gw_dp_new_slot = 1 } } country_event = { id = mdcw_dp.51 } }
   else = { country_event = { id = mdcw_dp.52 } }
  }
  else = { mdcw_dp_refund_offer = yes country_event = { id = mdcw_dp.11 } }
 }
}
