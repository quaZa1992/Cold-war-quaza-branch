# The focus tree deliberately reuses the USA 2000 dynamic
# modifiers instead of creating a second parallel economy.

USA_1970_initialize_systems = {
	if = {
		limit = { NOT = { has_country_flag = USA_1970_systems_initialized } }
		set_country_flag = USA_1970_systems_initialized

		# Replace the broad generic-democracy profile with a US two-party
		# electorate.  Minor parties remain present for roleplay, but do not
		# begin large enough to derail the presidential term modules.
		set_popularities = {
			democratic = 96.0
			communism = 0.5
			fascism = 0.5
			neutrality = 1.0
			nationalist = 2.0
		}
		start_politics_input = yes
		set_variable = { party_pop_array^1 = 0.530 }
		set_variable = { party_pop_array^2 = 0.420 }
		set_variable = { party_pop_array^3 = 0.010 }
		set_variable = { party_pop_array^4 = 0.005 }
		set_variable = { party_pop_array^10 = 0.005 }
		set_variable = { party_pop_array^16 = 0.006 }
		set_variable = { party_pop_array^17 = 0.002 }
		set_variable = { party_pop_array^18 = 0.002 }
		set_variable = { party_pop_array^20 = 0.020 }
		set_variable = { ruling_party = 1 }
		startup_politics = yes
		set_variable = { term_limit = 2 }
		set_variable = { current_term = 1 }
		set_country_flag = USA_1970_nixon_presidency
		country_event = { id = USA_1970_NIX.1 days = 528 }
		country_event = { id = USA_1970.10 days = 1681 }

		# MD law and national-spirit baseline for the 1970 start.  The United
		# States still uses Selective Service here; the military branch can
		# either retain it or enact the historical all-volunteer transition.
		# These ideas must exist before the global startup calls set_law_vars.
		add_ideas = {
			recession
			corruption_level_03
			bureau_03
			defence_05
			police_02
			edu_03
			health_03
			social_03
			semi_consumption_economy
			draft_army
			volunteer_women
			intervention_global_interventionism
			officer_international_education
			NATO_member
			western_country
			p5_member
			christian
			land_of_the_free
			political_establishment
			american_militarism
			regulated_immigration
			nuclear_power_off
			full_first_use
			landmark_ICBMsilominot_hidden
		}

		# Congress: a Republican White House facing Democratic chambers.
		add_ideas = congress_authority
		set_variable = { USA_state_number = 50 }
		set_variable = { senators_total = 100 }
		set_variable = { senators_support = 43 }
		set_variable = { representatives_total = 435 }
		set_variable = { representatives_support = 192 }
		USA_congress_get_quorum = yes

		# Internal factions used by the modern USA tree.
		add_ideas = {
			wall_street
			defense_industry
			intelligence_community
		}
		set_variable = { wall_street_opinion = 58 }
		set_variable = { defense_industry_opinion = 50 }
		set_variable = { intelligence_community_opinion = 45 }
		set_variable = { USA_strength_of_wall_street_var = 24 }
		initialize_internal_faction_dynamic_modifiers = yes

		# Fiscal baseline (abstract MD monetary units).
		set_variable = { debt = 371.0 }
		set_variable = { treasury = 120.0 }
		set_variable = { int_investments = 200.0 }
		set_variable = { corporate_tax_rate = 48 }
		set_variable = { population_tax_rate = 24 }
		set_variable = { USA_economic_regulation_var = 46 }

		# American economic policy.
		add_dynamic_modifier = { modifier = us_economic_policies }
		set_variable = { econ_consumer_goods = 0.04 }
		set_variable = { econ_factory_output = -0.16 }
		set_variable = { econ_dockyard_output = 0.03 }
		set_variable = { econ_production_efficiency_cap = 0.02 }
		set_variable = { econ_starting_factory_efficiency = 0.02 }
		set_variable = { econ_civ_factory_workforce = 0.08 }
		set_variable = { econ_mil_factory_workforce = 0.18 }
		set_variable = { econ_office_workforce = 0.40 }
		set_variable = { econ_productivity = 0.04 }

		# Department of the Interior.
		add_dynamic_modifier = { modifier = USA_dynamic_modifier_department_of_the_interior }
		set_variable = { doti_construction_speed = -0.15 }
		set_variable = { doti_civ_construction_speed = -0.35 }
		set_variable = { doti_office_construction_speed = 0.05 }
		set_variable = { doti_mil_construction_speed = 0.10 }
		set_variable = { doti_dockyard_construction_speed = 0.15 }
		set_variable = { doti_nuclear_reactor_construction_speed = -0.15 }
		set_variable = { doti_resource_extraction = 0.10 }

		# Department of Energy.  These use the exact variables read by
		# 02_american_dynamic_modifiers.txt.
		add_dynamic_modifier = { modifier = USA_dynamic_modifier_department_of_energy }
		set_variable = { doe_energy_consumption = 0.12 }
		set_variable = { doe_population_energy_consumption = 0.15 }
		set_variable = { doe_renewable_energy_production = -0.10 }
		set_variable = { doe_energy_production = 0.00 }
		set_variable = { doe_fossil_fuel_energy_production = 0.15 }
		set_variable = { doe_nuclear_energy_production = -0.15 }

		# Domestic policy.
		add_dynamic_modifier = { modifier = USA_domestic_policies }
		set_variable = { domestic_political_power_gain = -0.10 }
		set_variable = { domestic_drift_defense = 0.35 }
		set_variable = { domestic_encryption_factor = 0.10 }
		set_variable = { domestic_decryption_factor = 0.10 }
		set_variable = { domestic_monthly_pop = 0.40 }
		set_variable = { domestic_research_speed = 0.05 }
		set_variable = { domestic_unemployment_tolerance = 0.04 }
		set_variable = { domestic_offensive_foreign_influence = 0.05 }
		set_variable = { domestic_foreign_influence_defense = -0.05 }
		set_variable = { domestic_civ_intel = 0.15 }
		set_variable = { domestic_democratic_drift = 0.15 }
		set_variable = { domestic_communist_drift = 0.01 }
		set_variable = { domestic_neutrality_drift = 0.03 }
		set_variable = { domestic_nationalist_drift = 0.01 }

		# Federal budget and tax code.
		add_dynamic_modifier = { modifier = USA_federal_budgets }
		set_variable = { federal_budget_bureaucracy_cost = 0.45 }
		set_variable = { federal_budget_personnel_cost = 0.10 }
		set_variable = { federal_budget_police_cost = 0.15 }
		set_variable = { federal_budget_education_cost = 0.25 }
		set_variable = { federal_budget_healthcare_cost = 0.10 }
		set_variable = { federal_budget_welfare_cost = 0.20 }
		set_variable = { federal_budget_market_resources = 0.02 }
		set_variable = { federal_budget_investment_return = 0.03 }
		set_variable = { federal_budget_export_money_return = 0.02 }

		add_dynamic_modifier = { modifier = USA_american_tax_code }
		set_variable = { tax_code_interest_rate = -1.20 }
		set_variable = { tax_code_tax_collection_efficiency = 0.06 }
		set_variable = { tax_code_population_tax_collection_efficiency = 0.04 }
		set_variable = { tax_code_corporate_tax_collection_efficiency = 0.02 }
		set_variable = { tax_code_office_park_tax_collection_efficiency = -0.20 }
		set_variable = { tax_code_dockyard_tax_collection_efficiency = -0.08 }
		set_variable = { tax_code_military_tax_collection_efficiency = -0.12 }
		set_variable = { tax_code_civilian_tax_collection_efficiency = -0.05 }

		# Armed forces.  The 2000 history accidentally initializes several
		# military_* aliases; the dynamic modifiers actually read us_*.
		add_dynamic_modifier = { modifier = USA_dynamic_modifier_us_general_staff }
		set_variable = { genstaff_leader_xp_gain = 0.06 }
		set_variable = { genstaff_max_planning_factor = 0.02 }
		set_variable = { genstaff_supply_consumption_factor = 0.45 }
		set_variable = { genstaff_recruitable_population = 0.08 }
		set_variable = { genstaff_conscription = 0.002 }
		set_variable = { genstaff_daily_army_xp = 0.03 }
		set_variable = { genstaff_daily_air_xp = 0.02 }
		set_variable = { genstaff_daily_navy_xp = 0.04 }

		add_dynamic_modifier = { modifier = USA_dynamic_modifier_us_army }
		set_variable = { us_army_recon_factor = 0.03 }
		set_variable = { us_army_breakthrough_factor = 0.00 }
		set_variable = { us_army_army_org_regain = 0.00 }
		set_variable = { us_army_army_org = 0.00 }
		set_variable = { us_army_army_attack = -0.05 }
		set_variable = { us_army_army_defence = 0.00 }
		set_variable = { us_army_army_training_time = 0.15 }
		set_variable = { us_army_army_unit_xp_gain = 0.00 }
		set_variable = { us_army_coordination = 0.03 }

		add_dynamic_modifier = { modifier = USA_dynamic_modifier_us_navy }
		set_variable = { us_navy_repair_speed_factor = 0.08 }
		set_variable = { us_navy_naval_morale_factor = 0.03 }
		set_variable = { us_navy_naval_speed = 0.02 }
		set_variable = { us_navy_naval_damage = 0.02 }
		set_variable = { us_navy_naval_defense = 0.02 }

		add_dynamic_modifier = { modifier = USA_dynamic_modifier_us_airforce }
		set_variable = { us_airforce_air_mission_efficiency = 0.08 }
		set_variable = { us_airforce_air_attack = 0.00 }
		set_variable = { us_airforce_air_defence = 0.03 }
		set_variable = { us_airforce_air_agility = 0.04 }

		# Influence must exist before the AIPAC and diplomacy helpers run.
		init_influence = yes
		set_variable = { domestic_influence_amount = 1000 }
		add_to_array = { influence_array = ISR.id }
		add_to_array = { influence_array_val = 100 }
		add_to_array = { influence_array = SAU.id }
		add_to_array = { influence_array_val = 40 }
		add_to_array = { influence_array = EGY.id }
		add_to_array = { influence_array_val = 25 }
		add_to_array = { influence_array = CAN.id }
		add_to_array = { influence_array_val = 35 }
		add_to_array = { influence_array = MEX.id }
		add_to_array = { influence_array_val = 30 }
		add_to_array = { influence_array = ENG.id }
		add_to_array = { influence_array_val = 20 }
		add_to_array = { influence_array = SOV.id }
		add_to_array = { influence_array_val = 15 }
		add_to_array = { influence_array = CHI.id }
		add_to_array = { influence_array_val = 20 }
		startup_influence = yes

		# Strong, persistent 1970 crisis burdens and their PP-intensive
		# policy portfolios.  Strategies reshape these costs; they do not
		# silently end Vietnam or erase the drug crisis.
		USA_1970_initialize_policy_decision_systems = yes

		force_update_dynamic_modifier = yes
	}
}

USA_1970_set_scenario_leader = {
	clr_country_flag = USA_1970_nixon_presidency
	if = {
		limit = { AND = { has_variable = ruling_party check_variable = { ruling_party = 1 } } }
		if = {
			limit = { date < 1976.11.1 }
			create_country_leader = {
				name = "Richard Nixon"
				picture = "gfx/Leaders/USA/USA_Coldwar_Nixon.dds"
				ideology = conservatism
				traits = { western_conservatism career_politician }
			}
			set_country_flag = USA_1970_nixon_presidency
		}
		else_if = {
			limit = { date < 1980.11.1 }
			create_country_leader = {
				name = "Gerald Ford"
				ideology = conservatism
				traits = { western_conservatism career_politician }
			}
		}
		else_if = {
			limit = { date < 1988.11.1 }
			create_country_leader = {
				name = "Ronald Reagan"
				picture = "gfx/Leaders/USA/ronald.dds"
				ideology = conservatism
				traits = { western_conservatism career_politician }
			}
		}
		else_if = {
			limit = { date < 1996.11.1 }
			create_country_leader = {
				name = "George H. W. Bush"
				ideology = conservatism
				traits = { western_conservatism career_politician }
			}
		}
		else = {
			create_country_leader = {
				name = "Bob Dole"
				ideology = conservatism
				traits = { western_conservatism career_politician }
			}
		}
	}
	else_if = {
		limit = { AND = { has_variable = ruling_party check_variable = { ruling_party = 2 } } }
		if = {
			limit = { date < 1976.11.1 }
			create_country_leader = {
				name = "George McGovern"
				ideology = liberalism
				traits = { western_liberalism career_politician }
			}
		}
		else_if = {
			limit = { date < 1984.11.1 }
			create_country_leader = {
				name = "Jimmy Carter"
				ideology = liberalism
				traits = { western_liberalism career_politician }
			}
		}
		else_if = {
			limit = { date < 1988.11.1 }
			create_country_leader = {
				name = "Walter Mondale"
				ideology = liberalism
				traits = { western_liberalism career_politician }
			}
		}
		else_if = {
			limit = { date < 1992.11.1 }
			create_country_leader = {
				name = "Michael Dukakis"
				ideology = liberalism
				traits = { western_liberalism career_politician }
			}
		}
		else = {
			create_country_leader = {
				name = "Bill Clinton"
				picture = "gfx/Leaders/USA/bill_clinton.dds"
				ideology = liberalism
				traits = { western_liberalism career_politician }
			}
		}
	}
}

USA_1970_register_theater_acceptance = {
	add_to_variable = { USA_1970_theater_acceptances = 1 }
	set_country_flag = USA_1970_any_theater_accept
	if = {
		limit = { check_variable = { USA_1970_theater_acceptances > 1 } }
		set_country_flag = USA_1970_two_theater_accepts
	}
}

USA_1970_register_atlantic_acceptance = {
	add_to_variable = { USA_1970_atlantic_acceptances = 1 }
	USA_1970_register_theater_acceptance = yes
	if = {
		limit = {
			check_variable = { USA_1970_atlantic_acceptances > 1 }
			NOT = { has_country_flag = USA_1970_atlantic_bargain_applied }
		}
		set_country_flag = USA_1970_atlantic_bargain_applied
		add_political_power = 50
		set_temp_variable = { temp_change = 0.02 }
		USA_change_domestic_foreign_influence_defense_change = yes
		set_temp_variable = { temp_change = -0.03 }
		USA_change_federal_budget_personnel_cost_change = yes
	}
}
