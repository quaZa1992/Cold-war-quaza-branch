# The provisional council inherits the old government's subjects and interest groups.
COLDWAR_ENG_REVOLUTIONARY_reorganise_council = {
 if = {
  limit = { original_tag = ENG has_country_flag = COLDWAR_ENG_CRISIS_revolutionary_won check_variable = { ruling_party = 4 } }
  if = {
   limit = { NOT = { has_country_flag = COLDWAR_ENG_REVOLUTIONARY_break_with_protectorates } }
   set_country_flag = COLDWAR_ENG_REVOLUTIONARY_break_with_protectorates
   every_country = {
    limit = { is_subject_of = ENG NOT = { has_autonomy_state = COLDWAR_ENG_REVOLUTIONARY_union_member } }
    ENG = { set_autonomy = { target = PREV autonomy_state = autonomy_free end_wars = no end_civil_wars = no } }
   }
   COLDWAR_ENG_REVOLUTIONARY_close_overseas_accounts = yes
  }
  if = {
   limit = { NOT = { has_country_flag = COLDWAR_ENG_REVOLUTIONARY_cadres_installed } }
   set_country_flag = COLDWAR_ENG_REVOLUTIONARY_cadres_installed
   if = {
    limit = { has_idea = fossil_fuel_industry }
    swap_ideas = { remove_idea = fossil_fuel_industry add_idea = communist_cadres }
    if_no_longer_has_faction_clear_previous = yes
    setup_init_factions = yes
   }
   else_if = { limit = { NOT = { has_idea = communist_cadres } } add_ideas = communist_cadres setup_init_factions = yes }
  }
  if = {
   limit = { NOT = { has_dynamic_modifier = { modifier = COLDWAR_ENG_REVOLUTIONARY_public_support } } }
   add_dynamic_modifier = { modifier = COLDWAR_ENG_REVOLUTIONARY_public_support }
  }
 }
}

COLDWAR_ENG_REVOLUTIONARY_establish_council = {
 if = { limit = { original_tag = ENG has_country_flag = COLDWAR_ENG_CRISIS_revolutionary_won }
  set_country_flag = COLDWAR_ENG_CRISIS_royal_executions
  set_country_flag = ENG_elizabeth_dead_dead
  set_country_flag = ENG_royal_elizabeth_dead
  set_country_flag = ENG_royal_philip_dead
  COLDWAR_ENG_REVOLUTIONARY_abolish_royal_system = yes
  COLDWAR_ENG_REVOLUTIONARY_apply_council_identity = yes
  COLDWAR_ENG_REVOLUTIONARY_install_provisional_council = yes
  COLDWAR_ENG_REVOLUTIONARY_reorganise_council = yes
  if = { limit = { NOT = { has_country_flag = COLDWAR_ENG_REVOLUTIONARY_coup_news_announced } }
   set_country_flag = COLDWAR_ENG_REVOLUTIONARY_coup_news_announced
   news_event = { id = COLDWAR_ENG_REVOLUTIONARY_NEWS.1 hours = 6 }
  }
 }
}
COLDWAR_ENG_REVOLUTIONARY_recover_council = {

 }
}
