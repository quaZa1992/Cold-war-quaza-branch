LBA_1970_refresh_oil_control = {
	if = {
		limit = { NOT = { has_variable = lba_oil_nationalisation } }
		set_variable = { lba_oil_nationalisation = 0.10 }
	}
	clamp_variable = {
		var = lba_oil_nationalisation
		min = 0.10
		max = 1
	}

	set_variable = { LBA_1970_oil_tax_effect = lba_oil_nationalisation }
	multiply_variable = { LBA_1970_oil_tax_effect = 0.20 }
	set_variable = { LBA_1970_oil_resource_export_effect = lba_oil_nationalisation }
	multiply_variable = { LBA_1970_oil_resource_export_effect = 0.25 }
	set_variable = { LBA_1970_oil_export_effect = lba_oil_nationalisation }
	multiply_variable = { LBA_1970_oil_export_effect = 0.35 }
	set_variable = { LBA_1970_oil_resource_effect = lba_oil_nationalisation }
	multiply_variable = { LBA_1970_oil_resource_effect = 0.15 }
	set_variable = { LBA_1970_oil_workforce_effect = lba_oil_nationalisation }
	multiply_variable = { LBA_1970_oil_workforce_effect = -0.10 }
	set_variable = { LBA_1970_oil_influence_defense_effect = lba_oil_nationalisation }
	multiply_variable = { LBA_1970_oil_influence_defense_effect = 0.20 }

	# The global money pulse consumes this coefficient directly. The final oil
	# revenue state receives a higher coefficient than the standard Libyan NOC.
	if = {
		limit = { has_idea = LBA_1970_oil_revenue_state }
		set_variable = { oil_nationalisation_income = 0.020 }
	}
	else = {
		set_variable = { oil_nationalisation_income = 0.014 }
	}
	multiply_variable = { oil_nationalisation_income = lba_oil_nationalisation }
	set_variable = { libya_oil_nationalisation_additional_income = resource_produced@oil }
	multiply_variable = { libya_oil_nationalisation_additional_income = oil_nationalisation_income }

	if = {
		limit = {
			NOT = { has_dynamic_modifier = { modifier = LBA_1970_oil_sovereignty_dynamic_modifier } }
		}
		add_dynamic_modifier = { modifier = LBA_1970_oil_sovereignty_dynamic_modifier }
	}
	force_update_dynamic_modifier = yes
}

LBA_1970_close_foreign_bases = {
	if = {
		limit = {
			country_exists = ENG
			gives_military_access_to = ENG
		}
		ENG = {
			diplomatic_relation = {
				country = ROOT
				relation = military_access
				active = no
			}
		}
	}
	if = {
		limit = {
			country_exists = USA
			gives_military_access_to = USA
		}
		USA = {
			diplomatic_relation = {
				country = ROOT
				relation = military_access
				active = no
			}
		}
	}

	if = {
		limit = { has_idea = LBA_1970_foreign_bases }
		swap_ideas = {
			remove_idea = LBA_1970_foreign_bases
			add_idea = LBA_1970_full_sovereignty
		}
	}
	else_if = {
		limit = { NOT = { has_idea = LBA_1970_full_sovereignty } }
		add_ideas = LBA_1970_full_sovereignty
	}
	set_country_flag = LBA_1970_foreign_bases_closed
}

			add_idea = LBA_1970_unprepared_army_3
		}
	}
	else_if = {
		limit = { has_idea = LBA_1970_unprepared_army_3 }
		remove_ideas = LBA_1970_unprepared_army_3
	}
}

LBA_1970_proclaim_jamahiriya = {
	if = {
		limit = { NOT = { has_idea = LBA_1970_jamahiriya_declared } }
		add_ideas = LBA_1970_jamahiriya_declared
	}
	set_country_flag = LBA_1970_jamahiriya_proclaimed
	set_country_flag = LBA_1970_party_names_active
	set_cosmetic_tag = LBA_1985_JAMAHIRIYA

	# Keep the Jamahiriya on the mod-native Autocracy party index, without
	# resetting the complete popularity array a second time.
	set_politics = {
		ruling_party = communism
		elections_allowed = no
	}
	if = {
		limit = { NOT = { AND = { has_variable = ruling_party check_variable = { ruling_party = 7 } } } }
		clear_variable = ruling_party
		clear_array = gov_coalition_array
		set_variable = { ruling_party = 7 }
		
		startup_politics = yes
	}
	clr_country_flag = party8_banned
	set_country_flag = free_ban_parties
	set_partyall_banned = yes
}
