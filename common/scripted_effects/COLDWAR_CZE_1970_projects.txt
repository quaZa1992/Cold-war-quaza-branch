COLDWAR_CZE_1970_projects_init = {
	if = { limit = { NOT = { has_variable = COLDWAR_CZE_1970_projects_weekly_expense } } set_variable = { COLDWAR_CZE_1970_projects_weekly_expense = 0 } }
	if = { limit = { NOT = { has_variable = COLDWAR_CZE_1970_nuclear_progress_days } } set_variable = { COLDWAR_CZE_1970_nuclear_progress_days = 0 } }
	if = { limit = { NOT = { has_variable = COLDWAR_CZE_1970_ledger_version } } set_variable = { COLDWAR_CZE_1970_ledger_version = 0 } }
	if = { limit = { NOT = { has_variable = COLDWAR_CZE_1970_ledger_transfer_119 } } set_variable = { COLDWAR_CZE_1970_ledger_transfer_119 = 0 } }
	if = { limit = { NOT = { has_variable = COLDWAR_CZE_1970_ledger_transfer_120 } } set_variable = { COLDWAR_CZE_1970_ledger_transfer_120 = 0 } }
	if = { limit = { tag = CZE }
		set_variable = { COLDWAR_CZE_1970_defence_site_count = 0 }
		if = { limit = { has_country_flag = COLDWAR_CZE_1970_defence_site_prague } add_to_variable = { COLDWAR_CZE_1970_defence_site_count = 1 } }
		if = { limit = { has_country_flag = COLDWAR_CZE_1970_defence_site_brno } add_to_variable = { COLDWAR_CZE_1970_defence_site_count = 1 } }
		if = { limit = { has_country_flag = COLDWAR_CZE_1970_defence_site_ostrava } add_to_variable = { COLDWAR_CZE_1970_defence_site_count = 1 } }
		if = { limit = { has_country_flag = COLDWAR_CZE_1970_defence_site_plzen } add_to_variable = { COLDWAR_CZE_1970_defence_site_count = 1 } }
		if = { limit = { has_country_flag = COLDWAR_CZE_1970_defence_site_bratislava } add_to_variable = { COLDWAR_CZE_1970_defence_site_count = 1 } }
		if = { limit = { has_country_flag = COLDWAR_CZE_1970_defence_site_kosice } add_to_variable = { COLDWAR_CZE_1970_defence_site_count = 1 } }
	}
	119 = {
		if = { limit = { NOT = { has_variable = COLDWAR_CZE_1970_nuclear_building_baseline } }
			set_variable = { COLDWAR_CZE_1970_nuclear_building_baseline = building_level@nuclear_reactor }
		}
	}
	120 = {
		if = { limit = { NOT = { has_variable = COLDWAR_CZE_1970_uranium_building_baseline } }
			set_variable = { COLDWAR_CZE_1970_uranium_building_baseline = building_level@uranium_mine }
		}
	}
	COLDWAR_CZE_1970_projects_calculate_streams = yes
}

COLDWAR_CZE_1970_projects_refresh = {
	COLDWAR_CZE_1970_projects_init = yes
	if = {
		limit = { tag = CZE has_country_flag = COLDWAR_CZE_1970_ledger_proposed NOT = { COLDWAR_CZE_1970_ledger_schedule_matches = yes } }
		clr_country_flag = COLDWAR_CZE_1970_ledger_locked
		clr_country_flag = COLDWAR_CZE_1970_ledger_cze_approved
		clr_country_flag = COLDWAR_CZE_1970_ledger_slo_approved
		clr_country_flag = COLDWAR_CZE_1970_dual_cze_approved
		clr_country_flag = COLDWAR_CZE_1970_dual_slo_approved
		SLO = {
			clr_country_flag = COLDWAR_CZE_1970_ledger_locked
			clr_country_flag = COLDWAR_CZE_1970_ledger_cze_approved
			clr_country_flag = COLDWAR_CZE_1970_ledger_slo_approved
			clr_country_flag = COLDWAR_CZE_1970_dual_cze_approved
			clr_country_flag = COLDWAR_CZE_1970_dual_slo_approved
		}
	}
	119 = {
		if = {
			limit = {
				has_variable = COLDWAR_CZE_1970_nuclear_building_baseline
				is_owned_by = CZE
				is_controlled_by = CZE
				check_variable = { var = building_level@nuclear_reactor value = COLDWAR_CZE_1970_nuclear_building_baseline compare = greater_than }
				OWNER = { has_country_flag = COLDWAR_CZE_1970_nuclear_reactor_ordered }
			}
			OWNER = { set_country_flag = COLDWAR_CZE_1970_civil_reactor_commissioned }
		}
	}
	120 = {
		if = {
			limit = {
				has_variable = COLDWAR_CZE_1970_uranium_building_baseline
				is_owned_by = CZE
				is_controlled_by = CZE
				check_variable = { var = building_level@uranium_mine value = COLDWAR_CZE_1970_uranium_building_baseline compare = greater_than }
				OWNER = { has_country_flag = COLDWAR_CZE_1970_uranium_mine_ordered }
			}
			OWNER = { set_country_flag = COLDWAR_CZE_1970_uranium_mine_built }
		}
	}
}

COLDWAR_CZE_1970_projects_calculate_streams = {
	set_variable = { COLDWAR_CZE_1970_projects_weekly_expense = 0 }
	if = {
		limit = { tag = CZE country_exists = SLO check_variable = { gdp_total > 0 } SLO = { check_variable = { gdp_total > 0 } } OR = { NOT = { has_country_flag = COLDWAR_CZE_1970_ledger_proposed } NOT = { COLDWAR_CZE_1970_ledger_schedule_matches = yes } } }
		set_temp_variable = { COLDWAR_CZE_1970_ledger_preview_gdp_total = gdp_total }
		add_to_temp_variable = { COLDWAR_CZE_1970_ledger_preview_gdp_total = SLO.gdp_total }
		set_variable = { COLDWAR_CZE_1970_ledger_preview_cze_share = gdp_total }
		divide_variable = { COLDWAR_CZE_1970_ledger_preview_cze_share = COLDWAR_CZE_1970_ledger_preview_gdp_total }
		set_variable = { COLDWAR_CZE_1970_ledger_preview_slo_share = 1 }
		subtract_from_variable = { COLDWAR_CZE_1970_ledger_preview_slo_share = COLDWAR_CZE_1970_ledger_preview_cze_share }
	}
}

COLDWAR_CZE_1970_projects_weekly = {
	COLDWAR_CZE_1970_projects_calculate_streams = yes
	if = {
		limit = { has_country_flag = COLDWAR_CZE_1970_uranium_development_in_progress }
		if = {
			limit = { 120 = { COLDWAR_CZE_1970_can_finish_uranium_mine = yes } }
			add_to_variable = { COLDWAR_CZE_1970_uranium_development_days = 7 }
			if = {
				limit = { check_variable = { var = COLDWAR_CZE_1970_uranium_development_days value = 365 compare = greater_than_or_equals } }
				120 = {
					if = { limit = { COLDWAR_CZE_1970_can_finish_uranium_mine = yes }
						set_variable = { COLDWAR_CZE_1970_uranium_building_baseline = building_level@uranium_mine }
						add_extra_state_shared_building_slots = 1
						add_building_construction = { type = uranium_mine level = 1 instant_build = no }
						ROOT = { set_country_flag = COLDWAR_CZE_1970_uranium_mine_ordered }
					}
				}
				clr_country_flag = COLDWAR_CZE_1970_uranium_development_in_progress
			}
		}
	}
	if = {
		limit = {
			tag = CZE
			has_country_flag = COLDWAR_CZE_1970_nuclear_project_active
			COLDWAR_CZE_1970_nuclear_design_active = yes
			COLDWAR_CZE_1970_nuclear_fuel_active = yes
			has_war = no
			119 = {
				is_owned_by = CZE
				is_controlled_by = CZE
				is_fully_controlled_by = CZE
				can_construct_building = nuclear_reactor
				free_building_slots = { building = nuclear_reactor size > 0 include_locked = no }
			}
			country_can_build_nuclear_reactors = yes
		}
		add_to_variable = { COLDWAR_CZE_1970_nuclear_progress_days = 7 }
		if = {
			limit = { check_variable = { var = COLDWAR_CZE_1970_nuclear_progress_days value = 1095 compare = greater_than_or_equals } }
			if = {
				limit = {
					COLDWAR_CZE_1970_nuclear_design_active = yes
					COLDWAR_CZE_1970_nuclear_fuel_active = yes
					has_war = no
					119 = { is_owned_by = CZE is_controlled_by = CZE is_fully_controlled_by = CZE can_construct_building = nuclear_reactor free_building_slots = { building = nuclear_reactor size > 0 include_locked = no } }
					country_can_build_nuclear_reactors = yes
				}
				119 = {
					set_variable = { COLDWAR_CZE_1970_nuclear_building_baseline = building_level@nuclear_reactor }
					add_extra_state_shared_building_slots = 1
					add_building_construction = { type = nuclear_reactor level = 1 instant_build = no }
				}
				set_country_flag = COLDWAR_CZE_1970_nuclear_reactor_ordered
				clr_country_flag = COLDWAR_CZE_1970_nuclear_project_active
			}
		}
	}
	COLDWAR_CZE_1970_projects_refresh = yes
}
