# On 1 January 1970: about800 merchant ships above100gross tons, about4.8mGRT.
# Provisional gameplay conversion: one convoy per qualifying merchant ship.
# MD train_equipment_1 is its ordinary civilian supply-train variant. No new
# technology or military OOB is granted. The stock receipt survives save/load.
COLDWAR_SWE_starting_logistics_effect = {
    if = {
        limit = {
            tag = SWE
            original_tag = SWE
            exists = yes
            IMP_is_1970_scenario = yes
            NOT = { has_country_flag = COLDWAR_SWE_starting_logistics_received }
        }
        set_country_flag = COLDWAR_SWE_starting_logistics_received
        add_equipment_to_stockpile = { type = convoy_1 amount = 800 producer = SWE }
        add_equipment_to_stockpile = { type = train_equipment_1 amount = 100 producer = SWE }
    }
}
