imp_calculate_oil_producers = {
    every_country = {
        set_variable = { imp_oil_export_surplus = resource_exported@oil }
        subtract_from_variable = { imp_oil_export_surplus = resource_imported@oil }

        # Cold War: update existing saves to the stronger premium asking-price tiers.
        imp_sync_premium_oil_asking_price = yes

        remove_ideas = imp_major_oil_exporter
        remove_ideas = imp_minor_oil_exporter
    }

    clear_array = global.major_oil_exporters
    clear_array = global.minor_oil_exporters

    for_loop_effect = {
        end = 15
        value = i

        set_temp_variable = { oil_exp_amount_t = -1000 }

        every_country = {
            if = { 
                limit = { 
                    check_variable = { imp_oil_export_surplus > oil_exp_amount_t }
                    NOT = {
                        OR = {  
                            is_in_array = { array = global.major_oil_exporters value = THIS } 
                            is_in_array = { array = global.minor_oil_exporters value = THIS } 
                        }
                    }   
                }
                
                set_temp_variable = { oil_exp_nation_t = THIS }
                set_temp_variable = { oil_exp_amount_t = imp_oil_export_surplus }
            }
        }

        if = { 
            limit = { 
                check_variable = { oil_exp_amount_t > 0 }
            }

            if = { 
                limit = { 
                    check_variable = { i < 3 } 
                }

                add_to_array = {
                    array = global.major_oil_exporters 
                    value = oil_exp_nation_t
                }
            }
            else = {
                add_to_array = {
                    array = global.minor_oil_exporters 
                    value = oil_exp_nation_t
                }
            }
        }
    }

    for_each_scope_loop = {
        array = global.major_oil_exporters 
        add_ideas = imp_major_oil_exporter
    }

    for_each_scope_loop = {
        array = global.minor_oil_exporters 
        add_ideas = imp_minor_oil_exporter
    }
}

}

imp_calculate_oil_dependency = {
    every_country = {
        # Determine a nation's dependency on oil. The base is 40%. We can never get lower than that unless we have resistance but the absolute cap is 20%.
        set_variable = { imp_oil_dependency = 0.4 }

        # Then the next 0.3 is determined by our share of power plants being fossil fuel
        set_temp_variable = { imp_fossil_p = fossil_pp_energy_generation }
        divide_temp_variable = { imp_fossil_p = energy_sum }
        set_temp_variable = { imp_fossil_mod = 0.3 }
        multiply_temp_variable = { imp_fossil_mod = imp_fossil_p }
        add_to_variable = { imp_oil_dependency = imp_fossil_mod }

        # Then the next 0.3 is determined by pop fuel usage
        set_temp_variable = { imp_pop_use_mod = modifier@non_electric_fuel_consumption_modifier }
        add_to_temp_variable = { imp_pop_use_mod = 1 }
        multiply_temp_variable = { imp_pop_use_mod = 0.3 }
        add_to_variable = { imp_oil_dependency = imp_pop_use_mod }

        # Then we add the modifier.
        subtract_from_variable = { imp_oil_dependency = modifier@imp_oil_price_resistance }

        # clamp it. Add it.
        clamp_variable = { var = imp_oil_dependency min = 0.2 max = 1 }

        # Let's only count nations with more than 100b gdp for the oil dependency.
        if = { 
            limit = { 
                check_variable = { mdcw_era_gdp > 100 } # COLDWAR era reference (2000 $)
            }
            add_to_variable = { global.tot_dep = imp_oil_dependency }
            add_to_variable = { global.numRelCforOilDep = 1 }
        }
    }
}

imp_calculate_oil_equilibrium = {
    ### The Oil Price fluctuates towards the base price.
    ### The base price is decided by the oil policies of the largest oil producers.
    ### The base oil price is $75

    # First get all the oil exports
    for_each_scope_loop = {
        array = global.major_oil_exporters 
        add_to_temp_variable = { total_oil_exports_t = imp_oil_export_surplus }
    }
    for_each_scope_loop = {
        array = global.minor_oil_exporters 
        add_to_temp_variable = { total_oil_exports_t = imp_oil_export_surplus }
    }

    # Set the base price
    set_variable = { global.imp_global_oil_price_equilibrium = 75 }

    # Then let's get the % of each major and minor exporter
    for_each_scope_loop = {
        array = global.major_oil_exporters 

        # Set %
        set_variable = { imp_oil_export_p = imp_oil_export_surplus }
        divide_variable = { imp_oil_export_p = total_oil_exports_t }
        clamp_variable = { var = imp_oil_export_p min = 0.01 max = 1 }

        # Base price
        set_temp_variable = { price_adjustment_t = 75 }
        multiply_temp_variable = { price_adjustment_t = imp_oil_export_p }
        
        # Adjustment of equilibrium price
        multiply_temp_variable = { price_adjustment_t = imp_oil_export_adjustment }
        add_to_variable = { global.imp_global_oil_price_equilibrium = price_adjustment_t }
    }

    for_each_scope_loop = {
        array = global.minor_oil_exporters 

        # Set %
        set_variable = { imp_oil_export_p = imp_oil_export_surplus }
        divide_variable = { imp_oil_export_p = total_oil_exports_t }
        clamp_variable = { var = imp_oil_export_p min = 0.001 max = 1 }

        # Base price
        set_temp_variable = { price_adjustment_t = 75 }
        multiply_temp_variable = { price_adjustment_t = imp_oil_export_p }
        
        # Adjustment of equilibrium price
        multiply_temp_variable = { price_adjustment_t = imp_oil_export_adjustment }
        add_to_variable = { global.imp_global_oil_price_equilibrium = price_adjustment_t }
    }

    # Now multiply it by the global dependence.
    multiply_variable = { global.imp_global_oil_price_equilibrium = global.tot_dep }
    
    # MD Cold War: Cold War Equilibrium risk premium. Cold War Tension above 50 makes buyers
    # pay for the risk of supply disruption (0.3 USD per tension point above 50).
    set_variable = { global.mdcw_cw_oil_premium = 0 }
    if = {
        limit = {
            has_global_flag = mdcw_cw_initialized
            check_variable = { global.mdcw_cw_tension > 50 }
        }
        set_variable = { global.mdcw_cw_oil_premium = global.mdcw_cw_tension }
        subtract_from_variable = { global.mdcw_cw_oil_premium = 50 }
        multiply_variable = { global.mdcw_cw_oil_premium = 0.3 }
        add_to_variable = { global.imp_global_oil_price_equilibrium = global.mdcw_cw_oil_premium }
    }
    # Clamp it to a reasonable value
    clamp_variable = { var = global.imp_global_oil_price_equilibrium min = 30 max = 135 }
}

imp_weekly_oil_price_change = {
    # Now that we have the equilibrium, move the actual price towards the equilibrium
    set_temp_variable = { oil_tick_adjustment_t = global.imp_global_oil_price_equilibrium }
    subtract_from_temp_variable = { oil_tick_adjustment_t = global.imp_global_oil_price }
    multiply_temp_variable = { oil_tick_adjustment_t = 0.075 }

    # We additionally adjust the price a little depending on the equilibrium
    if = {
        limit = {
            OR = { 
                AND = { 
                    check_variable = { oil_tick_adjustment_t < 0 }
                    check_variable = { global.imp_global_oil_price_equilibrium < 50 }
                }
                AND = {
                    check_variable = { oil_tick_adjustment_t > 0 }
                    check_variable = { global.imp_global_oil_price_equilibrium > 100 }
                }
            }
        }
        multiply_temp_variable = { oil_tick_adjustment_t = 1.5 }
    }

    add_to_variable = { global.imp_global_oil_price = oil_tick_adjustment_t }
    clamp_variable = { var = global.imp_global_oil_price min = 15 max = 200 }

    # MD Cold War: effective price includes producer level and the cached world embargo premium.
    set_variable = { global.imp_oil_price_effective = global.imp_global_oil_price }
    if = {
        limit = { has_variable = global.mdcw_cw_oil_level }
        multiply_variable = { global.imp_oil_price_effective = global.mdcw_cw_oil_level }
    }
    if = {
        limit = { has_variable = global.mdcw_cw_oil_embargo_world_multiplier }
        multiply_variable = { global.imp_oil_price_effective = global.mdcw_cw_oil_embargo_world_multiplier }
    }

    # Let's assign dynamic modifiers
    if = { 
        limit = { check_variable = { global.imp_oil_price_effective > 100 } }

        every_country = { 
            if = {
                limit = { 
                    has_dynamic_modifier = { modifier = imp_oil_prices }
                }
                remove_dynamic_modifier = { modifier = imp_oil_prices }
                add_dynamic_modifier = { modifier = imp_oil_prices_high }
            }
            else_if = {
                limit = {
                    has_dynamic_modifier = { modifier = imp_oil_prices_low }
                }
                remove_dynamic_modifier = { modifier = imp_oil_prices_low }
                add_dynamic_modifier = { modifier = imp_oil_prices_high }
            }
            else_if = {
                limit = { 
                    NOT = {  
                        has_dynamic_modifier = { modifier = imp_oil_prices_high }
                    }
                }
                add_dynamic_modifier = { modifier = imp_oil_prices_high }
            }
        }
    }
    else_if = {
        limit = { check_variable = { global.imp_oil_price_effective < 50 } }

        every_country = { 
            if = { 
                limit = { 
                    has_dynamic_modifier = { modifier = imp_oil_prices }
                }
                remove_dynamic_modifier = { modifier = imp_oil_prices }
                add_dynamic_modifier = { modifier = imp_oil_prices_low }
            }
            else_if = {
                limit = {
                    has_dynamic_modifier = { modifier = imp_oil_prices_high }
                }
                remove_dynamic_modifier = { modifier = imp_oil_prices_high }
                add_dynamic_modifier = { modifier = imp_oil_prices_low }
            }
            else_if = {
                limit = { 
                    NOT = {  
                        has_dynamic_modifier = { modifier = imp_oil_prices_low }
                    }
                }
                add_dynamic_modifier = { modifier = imp_oil_prices_low }
            }
        }
    }
    else = {
        every_country = { 
            if = { 
                limit = { 
                    has_dynamic_modifier = { modifier = imp_oil_prices_high }
                }
                remove_dynamic_modifier = { modifier = imp_oil_prices_high }
                add_dynamic_modifier = { modifier = imp_oil_prices }
            }
            else_if = {
                limit = {
                    has_dynamic_modifier = { modifier = imp_oil_prices_low }
                }
                remove_dynamic_modifier = { modifier = imp_oil_prices_low }
                add_dynamic_modifier = { modifier = imp_oil_prices }
            }
            else_if = {
                limit = { 
                    NOT = {  
                        has_dynamic_modifier = { modifier = imp_oil_prices }
                    }
                }
                add_dynamic_modifier = { modifier = imp_oil_prices }
            }
        }
    }

    # Let's assign the values for the dynamic modifiers
    # MD Cold War: decided per country by the price it feels (OPEC price level, embargo shortage).
    every_country = {
        set_temp_variable = { imp_oil_factor_t = global.imp_global_oil_price }
        mdcw_cw_oil_effective_price = yes
        set_variable = { mdcw_cw_oil_crisis_ratio = imp_oil_factor_t }
        divide_variable = { mdcw_cw_oil_crisis_ratio = 75 }
        if = {
            limit = {
                check_variable = { imp_oil_factor_t > 75 }
            }
            imp_calculate_oil_dynamic_modifiers_high = yes
        }
        else = {
            imp_calculate_oil_dynamic_modifiers_low = yes
        }
        # MD Cold War: discrete, dependency-scaled energy-crisis stages supplement Imperia's effects.
        mdcw_cw_update_oil_crisis_tier = yes
    }

    # MD Cold War: price shown to the player in the dollars of the scenario year.
    imp_update_oil_price_display = yes
}

# MD Cold War: the oil model runs in model dollars (75 = normal price). Displayed price = model price
# x global.imp_oil_price_unit, which is base unit x OPEC price level x US consumer prices (1970 = 1).
# Base unit: 1970 game 1.80 / 75 = 0.024 (January 1970 posted price of Arabian Light); other
# scenarios 1 (OPEC level and consumer prices only exist in the 1970 game).
imp_update_oil_price_display = {
    if = {
        limit = { NOT = { has_variable = global.imp_oil_price_base_unit } }
        # Savegames from before the display existed
        if = {
            limit = { IMP_is_1970_scenario = yes }
            set_variable = { global.imp_oil_price_base_unit = 0.024 }
        }
        else = {
            set_variable = { global.imp_oil_price_base_unit = 1 }
        }
    }
    set_variable = { global.imp_oil_price_unit = global.imp_oil_price_base_unit }
    if = {
        limit = { has_variable = global.mdcw_cw_oil_level }
        multiply_variable = { global.imp_oil_price_unit = global.mdcw_cw_oil_level }
    }
    if = {
        limit = { has_variable = global.mdcw_cw_oil_embargo_world_multiplier }
        multiply_variable = { global.imp_oil_price_unit = global.mdcw_cw_oil_embargo_world_multiplier }
    }
    # Price in dollars of the base year (1970 dollars in the 1970 game)
    set_variable = { global.imp_oil_price_real_display = global.imp_global_oil_price }
    multiply_variable = { global.imp_oil_price_real_display = global.imp_oil_price_unit }
    if = {
        limit = { has_variable = global.mdcw_cw_cpi }
        multiply_variable = { global.imp_oil_price_unit = global.mdcw_cw_cpi }
    }
    set_variable = { global.imp_oil_price_display = global.imp_global_oil_price }
    multiply_variable = { global.imp_oil_price_display = global.imp_oil_price_unit }
    set_variable = { global.imp_oil_price_equilibrium_display = global.imp_global_oil_price_equilibrium }
    multiply_variable = { global.imp_oil_price_equilibrium_display = global.imp_oil_price_unit }
}

        set_variable = { imp_oil_export_adjustment = -0.8 }
    }
    else_if = { 
        limit = { has_idea = imp_oil_prices_ask_2 }
        swap_ideas = {
            remove_idea = imp_oil_prices_ask_2
            add_idea = imp_oil_prices_ask_1
        }
        set_variable = { imp_oil_export_adjustment = -1 }
    }
    else = {
        add_ideas = imp_oil_prices_ask_5
        set_variable = { imp_oil_export_adjustment = -0.2 }
    }
}

imp_set_trade_policy = {
    set_temp_variable = { imp_p_t = md_oil_export }
    imp_set_trade_policy_base = yes
    subtract_from_temp_variable = { imp_p_t = md_oil_export }

    ## If temp < new -> Increased oil exports
    ## If temp > new -> Decreased oil exports

    if = { 
        limit = { 
            has_idea = imp_major_oil_exporter
            NOT = { 
                has_country_flag = imp_recent_oil_export_change
            }
        }

        if = { 
            limit = { check_variable = { imp_p_t > 2 } }
            set_country_flag = { flag = imp_recent_oil_export_change days = 720 value = 1 }

            set_temp_variable = { oil_cost_multiplier_t = 0.40 }
            imp_spike_oil_price = yes
            country_event = imp_oil.50
        }
        else_if = {
            limit = { check_variable = { imp_p_t > 0 } }
            set_country_flag = { flag = imp_recent_oil_export_change days = 720 value = 1 }
            set_temp_variable = { oil_cost_multiplier_t = 0.20 }
            imp_spike_oil_price = yes

            limit = { check_variable = { imp_p_t < -2 } }
            set_country_flag = { flag = imp_recent_oil_export_change days = 720 value = 1 }
            set_temp_variable = { oil_cost_multiplier_t = -0.40 }
            imp_spike_oil_price = yes
            country_event = imp_oil.52
        }
        else_if = {
            limit = { check_variable = { imp_p_t < 0 } }
            set_country_flag = { flag = imp_recent_oil_export_change days = 720 value = 1 }
            set_temp_variable = { oil_cost_multiplier_t = -0.20 }
            imp_spike_oil_price = yes
            country_event = imp_oil.53
        }
    }
    else_if = { 
        limit = { 
            has_idea = imp_minor_oil_exporter
            NOT = { 
                has_country_flag = imp_recent_oil_export_change
            }
        }

        if = { 
            limit = { check_variable = { imp_p_t > 2 } }
            set_country_flag = { flag = imp_recent_oil_export_change days = 720 value = 1 }
            set_temp_variable = { oil_cost_multiplier_t = 0.20 }
            imp_spike_oil_price = yes
            country_event = imp_oil.50
        }
        else_if = {
            limit = { check_variable = { imp_p_t > 0 } }
            set_country_flag = { flag = imp_recent_oil_export_change days = 720 value = 1 }
            set_temp_variable = { oil_cost_multiplier_t = 0.10 }
            imp_spike_oil_price = yes
            country_event = imp_oil.51

        else_if = {
