COLDWAR_TUR_1970_apply_council_terms = {
	if = {
		limit = { COLDWAR_TUR_1970_council_target_valid = yes }
		if = { limit = { NOT = { has_idea = the_military } } add_ideas = the_military }
		if = { limit = { NOT = { has_idea = TUR_guards_of_kemalism } } add_ideas = TUR_guards_of_kemalism }





		set_temp_variable = { disable_elections = 0 }
		set_temp_variable = { rul_party_temp = COLDWAR_TUR_1970_council_party }
		change_ruling_party_effect = yes
		clear_variable = military_support
		clear_variable = islamist_support
	}
}
