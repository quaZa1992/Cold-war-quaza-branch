KOR_review_civilian_sync_kist = {
	if = {
		limit = {
			has_country_flag = KOR_review_enacted_kist_research_mandate
			KOR_review_civilian_kist_site = yes
		}
		if = {
			limit = { NOT = { has_idea = KOR_review_civilian_kist_applied_research } }
			add_ideas = KOR_review_civilian_kist_applied_research
		}
	}
	else_if = {
		limit = { has_idea = KOR_review_civilian_kist_applied_research }
		remove_ideas = KOR_review_civilian_kist_applied_research
	}
}

KOR_review_civilian_imf_daily_notified = {
	if = {
		limit = { has_country_flag = KOR_review_civilian_imf_application_pending }
		clr_country_flag = KOR_review_civilian_imf_application_pending
		if = {
			limit = {
				KOR_review_civilian_imf_institution_approves = yes
				NOT = { has_country_flag = KOR_review_civilian_imf_open }
			}
			set_country_flag = KOR_review_civilian_imf_open
			set_variable = { KOR_review_civilian_imf_weeks_remaining = 104 }
			set_temp_variable = { KOR_review_civilian_advance = { value = gdp_total multiply = 0.015 } }
			set_temp_variable = { debt_change = KOR_review_civilian_advance }
			modify_debt_effect = yes
			set_temp_variable = { treasury_change = KOR_review_civilian_advance }
			modify_treasury_effect = yes
			add_ideas = KOR_review_civilian_imf_recovery
			country_event = { id = KOR_review_civilian.1 }
		}
		else = { country_event = { id = KOR_review_civilian.2 } }
	}
}
KOR_review_civilian_imf_weekly = {
	if = {
		limit = {
			has_country_flag = KOR_review_civilian_imf_open
			check_variable = { KOR_review_civilian_imf_weeks_remaining > 0 }
		}
		add_to_variable = {
			var = additional_expenses_rate
			value = { value = gdp_total multiply = 0.00003 }
		}
	}
}
KOR_review_civilian_imf_advance_week = {
	if = {
		limit = { has_country_flag = KOR_review_civilian_imf_open check_variable = { KOR_review_civilian_imf_weeks_remaining > 0 } }
		subtract_from_variable = { KOR_review_civilian_imf_weeks_remaining = 1 }
		if = {
			limit = { check_variable = { var = KOR_review_civilian_imf_weeks_remaining value = 0 compare = less_than_or_equals } }
			clr_country_flag = KOR_review_civilian_imf_open
			clear_variable = KOR_review_civilian_imf_weeks_remaining
			remove_ideas = KOR_review_civilian_imf_recovery
		}
	}
}
