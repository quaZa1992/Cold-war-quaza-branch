# IMPERIA CW — Cold War: Fixed resource discoveries; preserve Imperia eligibility and monthly cadence.
# State scope. Freeze the type and amount before publishing the news.
COLDWAR_resource_discovery_roll = {
	random_list = {
		100 = {
			set_variable = { COLDWAR_discovery_type = 1 }
			set_variable = { COLDWAR_discovery_amount = 16 }
		}
		100 = {
			set_variable = { COLDWAR_discovery_type = 2 }
			set_variable = { COLDWAR_discovery_amount = 12 }
		}
		100 = {
			set_variable = { COLDWAR_discovery_type = 3 }
			set_variable = { COLDWAR_discovery_amount = 8 }
		}
		100 = {
			set_variable = { COLDWAR_discovery_type = 4 }
			set_variable = { COLDWAR_discovery_amount = 8 }
		}
		100 = {
			set_variable = { COLDWAR_discovery_type = 5 }
			set_variable = { COLDWAR_discovery_amount = 4 }
		}
		100 = {
			modifier = { factor = 0 NOT = { COLDWAR_resource_rubber_region = yes } }
			set_variable = { COLDWAR_discovery_type = 6 }
			set_variable = { COLDWAR_discovery_amount = 8 }
		}
	}
	random_list = {
		65 = {
			set_variable = { COLDWAR_discovery_size = 1 }
			multiply_variable = { COLDWAR_discovery_amount = 1 }
		}
		30 = {
			set_variable = { COLDWAR_discovery_size = 2 }
			multiply_variable = { COLDWAR_discovery_amount = 2 }
		}
		5 = {
			set_variable = { COLDWAR_discovery_size = 3 }
			multiply_variable = { COLDWAR_discovery_amount = 3 }
		}
	}
	set_state_flag = COLDWAR_resource_discovery_pending
}

# State scope. Retain the selected result after claiming so observers still see the report.
COLDWAR_resource_discovery_claim = {
	# An already-open event from an older save has no prepared result.
	hidden_effect = {
		if = {
			limit = { NOT = { has_variable = COLDWAR_discovery_type } }
			COLDWAR_resource_discovery_roll = yes
		}
	}
	if = {
		limit = { hidden_trigger = { has_state_flag = COLDWAR_resource_discovery_pending } }
		if = {
			limit = { hidden_trigger = { check_variable = { COLDWAR_discovery_type = 1 } } }
			add_resource = { type = oil amount = COLDWAR_discovery_amount }
		}
		else_if = {
			limit = { hidden_trigger = { check_variable = { COLDWAR_discovery_type = 2 } } }
			add_resource = { type = steel amount = COLDWAR_discovery_amount }
		}
		else_if = {
			limit = { hidden_trigger = { check_variable = { COLDWAR_discovery_type = 3 } } }
			add_resource = { type = aluminium amount = COLDWAR_discovery_amount }
		}
		else_if = {
			limit = { hidden_trigger = { check_variable = { COLDWAR_discovery_type = 4 } } }
			add_resource = { type = tungsten amount = COLDWAR_discovery_amount }
		}
		else_if = {
			limit = { hidden_trigger = { check_variable = { COLDWAR_discovery_type = 5 } } }
			add_resource = { type = chromium amount = COLDWAR_discovery_amount }
		}
		else_if = {
			limit = { hidden_trigger = { check_variable = { COLDWAR_discovery_type = 6 } } }
			add_resource = { type = rubber amount = COLDWAR_discovery_amount }
		}
		hidden_effect = {
			clr_state_flag = COLDWAR_resource_discovery_pending
			set_variable = { global.resource_spawner_counter = 0 }
		}
	}
}
