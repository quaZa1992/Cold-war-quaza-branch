# IMPERIA CW — Cold War: initialise existing cartel mechanics before their first monthly check.
IMP_1970_initialize_cartel_variables = {
	if = {
		limit = {
			IMP_is_1970_scenario = yes
			set_cartel_nation = yes
			NOT = { original_tag = HOL }
			NOT = { has_country_flag = CARTEL_defeated_the_cartels }
		}
		# Preserve MD's country-specific baseline; its dated 2000 histories have not run in 1970.
		set_temp_variable = { IMP_1970_cartel_initial_strength = 40 }
		set_temp_variable = { IMP_1970_cartel_initial_influence = 20 }
		if = {
			limit = { OR = { original_tag = ARG original_tag = CHL original_tag = URG } }
			set_temp_variable = { IMP_1970_cartel_initial_strength = 20 }
			set_temp_variable = { IMP_1970_cartel_initial_influence = 10 }
		}
		else_if = {
			limit = { original_tag = BLZ }
			set_temp_variable = { IMP_1970_cartel_initial_strength = 10 }
			set_temp_variable = { IMP_1970_cartel_initial_influence = 5 }
		}
		else_if = {
			limit = { OR = { original_tag = BOL original_tag = BRA original_tag = PAR original_tag = PRU } }
			set_temp_variable = { IMP_1970_cartel_initial_strength = 30 }
			set_temp_variable = { IMP_1970_cartel_initial_influence = 15 }
		}
		else_if = {
			limit = { original_tag = BRM }
			set_temp_variable = { IMP_1970_cartel_initial_strength = 55 }
			set_temp_variable = { IMP_1970_cartel_initial_influence = 45 }
		}
		else_if = {
			limit = { OR = { original_tag = BRU original_tag = LAO } }
			set_temp_variable = { IMP_1970_cartel_initial_strength = 35 }
			set_temp_variable = { IMP_1970_cartel_initial_influence = 25 }
		}
		else_if = {
			limit = { original_tag = GUA }
			set_temp_variable = { IMP_1970_cartel_initial_strength = 50 }
			set_temp_variable = { IMP_1970_cartel_initial_influence = 25 }
		}
		else_if = {
			limit = { original_tag = MAY }
			set_temp_variable = { IMP_1970_cartel_initial_strength = 35 }
			set_temp_variable = { IMP_1970_cartel_initial_influence = 40 }
		}
		else_if = {
			limit = { original_tag = PHI }
			set_temp_variable = { IMP_1970_cartel_initial_strength = 25 }
		}
		else_if = {
			limit = { original_tag = SIA }
			set_temp_variable = { IMP_1970_cartel_initial_strength = 30 }
		}
		else_if = {
			limit = { original_tag = SIN }
			set_temp_variable = { IMP_1970_cartel_initial_strength = 15 }
			set_temp_variable = { IMP_1970_cartel_initial_influence = 10 }
		}
		else_if = {
			limit = { original_tag = VIE }
			set_temp_variable = { IMP_1970_cartel_initial_strength = 20 }
			set_temp_variable = { IMP_1970_cartel_initial_influence = 35 }
		}
		# Missing history is not a defeated cartel. Keep explicit values, including legitimate zeroes.
		if = {
			limit = { NOT = { has_variable = strength_of_cartels } }
			set_variable = { strength_of_cartels = IMP_1970_cartel_initial_strength }
		}
		if = {
			limit = { NOT = { has_variable = cartel_political_influence } }
			set_variable = { cartel_political_influence = IMP_1970_cartel_initial_influence }
		}
	}
}
