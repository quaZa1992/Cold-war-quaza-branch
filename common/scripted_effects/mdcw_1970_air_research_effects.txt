# Later starts (MD history research): a country that already has an MD tech also gets the jet generation techs in front of it,
# so the tree shows no open gaps behind researched techs. Any start date; harmless when nothing applies.
mdcw_inherit_jet_generation_research = {
	if = {
		limit = { has_dlc = "By Blood Alone" }
		if = {
			limit = { has_tech = early_airframe_designs }
			set_technology = {
				popup = no
				mdcw_early_airframes_1945 = 1
			}
		}
		if = {
			limit = { has_tech = gen_3_light }
			set_technology = {
				popup = no
				mdcw_early_airframes_1945 = 1
				mdcw_gen_1_light = 1
				mdcw_gen_2_light = 1
			}
		}
		if = {
			limit = { has_tech = gen_3_medium }
			set_technology = {
				popup = no
				mdcw_early_airframes_1945 = 1
				mdcw_gen_1_medium = 1
				mdcw_gen_2_medium = 1
			}
		}
		if = {
			limit = { has_tech = variable_wing_design }
			set_technology = {
				popup = no
				mdcw_delta_wing_1956 = 1
			}
		}
		if = {
			limit = { has_tech = flying_wing_design }
			set_technology = {
				popup = no
				mdcw_delta_wing_1956 = 1
			}
		}
		if = {
			limit = { has_tech = avionics_1 }
			set_technology = {
				popup = no
				mdcw_fire_control_1963 = 1
				mdcw_pulse_doppler_1972 = 1
			}
		}
		if = {
			limit = { has_tech = engines_2 }
			set_technology = {
				popup = no
				mdcw_engines_1964 = 1
			}
		}
		if = {
			limit = { has_tech = air_weapons_1 }
			set_technology = {
				popup = no
				mdcw_early_aam_1957 = 1
			}
		}
		if = {
			limit = { has_tech = air_weapons_2 }
			set_technology = {
				popup = no
				mdcw_early_aam_1957 = 1
				mdcw_cooled_ir_1965 = 1
				mdcw_aam_1976 = 1
			}
		}
		if = {
			limit = { has_tech = countermeasures_2 }
			set_technology = {
				popup = no
				mdcw_ew_1975 = 1
			}
		}
		if = {
			limit = { has_tech = ag_weapons_2 }
			set_technology = {
				popup = no
				mdcw_pgm_1972 = 1
			}
		}
	}
}
