# IMPERIA COLD WAR — Historical Iraqi national flags.
COLDWAR_IRQ_remember_previous_flag = {
	clear_variable = COLDWAR_IRQ_previous_flag
	if = {
		limit = { has_cosmetic_tag = IRQ_HIST_HASHEMITE }
		set_variable = { COLDWAR_IRQ_previous_flag = 1 }
	}
	else_if = {
		limit = { has_cosmetic_tag = IRQ_HIST_REPUBLIC_1959 }
		set_variable = { COLDWAR_IRQ_previous_flag = 2 }
	}
	else_if = {
		limit = { has_cosmetic_tag = IRQ_BAATH_1970 }
		set_variable = { COLDWAR_IRQ_previous_flag = 3 }
	}
	else_if = {
		limit = { has_cosmetic_tag = IRQ_HIST_SADDAM_1991 }
		set_variable = { COLDWAR_IRQ_previous_flag = 4 }
	}
	else_if = {
		limit = { has_cosmetic_tag = IRQ_HIST_INTERIM_2004 }
		set_variable = { COLDWAR_IRQ_previous_flag = 5 }
	}
	else_if = {
		limit = { has_cosmetic_tag = IRQ_HIST_REPUBLIC_2008 }
		set_variable = { COLDWAR_IRQ_previous_flag = 6 }
	}
	else_if = {
		limit = { has_cosmetic_tag = IRQ_AUTH_nationalist }
		set_variable = { COLDWAR_IRQ_previous_flag = 7 }
	}
	else_if = {
		limit = { has_cosmetic_tag = IRQ_MNF }
		set_variable = { COLDWAR_IRQ_previous_flag = 8 }
	}
	else_if = {
		limit = { has_cosmetic_tag = IRQ }
		set_variable = { COLDWAR_IRQ_previous_flag = 9 }
	}
}

COLDWAR_IRQ_restore_previous_flag = {
	set_temp_variable = { COLDWAR_IRQ_flag_to_restore = COLDWAR_IRQ_previous_flag }
	COLDWAR_IRQ_remember_previous_flag = yes
	if = {
		limit = { check_variable = { COLDWAR_IRQ_flag_to_restore = 1 } }
		set_cosmetic_tag = IRQ_HIST_HASHEMITE
	}
	else_if = {
		limit = { check_variable = { COLDWAR_IRQ_flag_to_restore = 2 } }
		set_cosmetic_tag = IRQ_HIST_REPUBLIC_1959

	}
	else_if = {
		limit = { check_variable = { COLDWAR_IRQ_flag_to_restore = 5 } }
		set_cosmetic_tag = IRQ_HIST_INTERIM_2004
	}
	else_if = {
		limit = { check_variable = { COLDWAR_IRQ_flag_to_restore = 6 } }
		set_cosmetic_tag = IRQ_HIST_REPUBLIC_2008
	}
	else_if = {
		limit = { check_variable = { COLDWAR_IRQ_flag_to_restore = 7 } }
		set_cosmetic_tag = IRQ_AUTH_nationalist
	}
	else_if = {
		limit = { check_variable = { COLDWAR_IRQ_flag_to_restore = 8 } }

	}
	else_if = {
