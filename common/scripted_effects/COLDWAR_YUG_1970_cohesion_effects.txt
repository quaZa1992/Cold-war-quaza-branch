COLDWAR_YUG_1970_cohesion_init = {
	if = {
		limit = { COLDWAR_YUG_1970_active = yes NOT = { has_country_flag = COLDWAR_YUG_1970_cohesion_initialized } }
		set_variable = { COLDWAR_YUG_1970_cohesion = 72 }
		set_variable = { COLDWAR_YUG_1970_case_fiscal_stage = 0 }
		set_variable = { COLDWAR_YUG_1970_auto_claim_cursor = 0 }
		set_variable = { COLDWAR_YUG_1970_fee_0_5_percent = { value = gdp_total multiply = 0.005 } }
		set_variable = { COLDWAR_YUG_1970_fee_0_75_percent = { value = gdp_total multiply = 0.0075 } }
		set_variable = { COLDWAR_YUG_1970_fee_1_percent = { value = gdp_total multiply = 0.01 } }
		set_variable = { COLDWAR_YUG_1970_last_seen_quarter_gdp = last_quarter_gdp_total }
		set_variable = { COLDWAR_YUG_1970_quarter_sample_count = 0 }
		set_country_flag = COLDWAR_YUG_1970_case_fiscal_active
		set_country_flag = COLDWAR_YUG_1970_cohesion_initialized
		set_country_flag = COLDWAR_YUG_1970_system_started
		set_global_flag = COLDWAR_YUG_1970_system_started_global
	}
}

# Called from the native weekly budget pass in SER scope. Rates are current GDP fractions.
COLDWAR_YUG_1970_cohesion_budget_expense = {
	if = {
		limit = { COLDWAR_YUG_1970_active = yes }
		set_variable = { COLDWAR_YUG_1970_fee_0_5_percent = { value = gdp_total multiply = 0.005 } }
		set_variable = { COLDWAR_YUG_1970_fee_0_75_percent = { value = gdp_total multiply = 0.0075 } }
		set_variable = { COLDWAR_YUG_1970_fee_1_percent = { value = gdp_total multiply = 0.01 } }
		if = {
			limit = { has_country_flag = COLDWAR_YUG_1970_fiscal_funding_active }
			if = {
				limit = { has_country_flag = COLDWAR_YUG_1970_contributor_oversight_active }
				set_variable = { COLDWAR_YUG_1970_fiscal_weekly_cost = { value = gdp_total multiply = 0.00015 } }
			}
			else = { set_variable = { COLDWAR_YUG_1970_fiscal_weekly_cost = { value = gdp_total multiply = 0.00025 } } }
			add_to_variable = { additional_expenses_rate = COLDWAR_YUG_1970_fiscal_weekly_cost }
		}
		if = {
			limit = { has_country_flag = COLDWAR_YUG_1970_provincial_funding_active }
			set_variable = { COLDWAR_YUG_1970_provincial_weekly_cost = { value = gdp_total multiply = 0.00006 } }
			add_to_variable = { additional_expenses_rate = COLDWAR_YUG_1970_provincial_weekly_cost }
		}
		if = {
			limit = { has_country_flag = COLDWAR_YUG_1970_command_funding_active }
			set_variable = { COLDWAR_YUG_1970_command_weekly_cost = { value = gdp_total multiply = 0.00008 } }
			add_to_variable = { additional_expenses_rate = COLDWAR_YUG_1970_command_weekly_cost }
		}
	}
}

	}

COLDWAR_YUG_1970_reconcile_cohesion_band = {
	clamp_variable = { var = COLDWAR_YUG_1970_cohesion min = 0 max = 100 }
	if = {
		limit = { check_variable = { COLDWAR_YUG_1970_cohesion < 20 } }
		remove_ideas = { COLDWAR_YUG_1970_federal_deadlock COLDWAR_YUG_1970_constitutional_crisis }
		add_ideas = COLDWAR_YUG_1970_breaking_point
	}
	else_if = {
		limit = { check_variable = { COLDWAR_YUG_1970_cohesion < 40 } }
		remove_ideas = { COLDWAR_YUG_1970_federal_deadlock COLDWAR_YUG_1970_breaking_point }
		add_ideas = COLDWAR_YUG_1970_constitutional_crisis
	}
	else_if = {
		limit = { check_variable = { COLDWAR_YUG_1970_cohesion < 60 } }
		remove_ideas = { COLDWAR_YUG_1970_constitutional_crisis COLDWAR_YUG_1970_breaking_point }
		add_ideas = COLDWAR_YUG_1970_federal_deadlock
	}
	else = {
		remove_ideas = { COLDWAR_YUG_1970_federal_deadlock COLDWAR_YUG_1970_constitutional_crisis COLDWAR_YUG_1970_breaking_point }
	}
}

COLDWAR_YUG_1970_monthly_cohesion = {
	COLDWAR_YUG_1970_cohesion_init = yes
	if = {
		limit = { has_country_flag = COLDWAR_YUG_1970_cohesion_initialized }
		COLDWAR_YUG_1970_capture_quarter_sample = yes
		COLDWAR_YUG_1970_check_fiscal_distress = yes
		COLDWAR_YUG_1970_open_case = yes
		set_variable = { COLDWAR_YUG_1970_monthly_loss = 0 }
		set_variable = { COLDWAR_YUG_1970_monthly_case_loss = 0 }
		if = {
			limit = { has_country_flag = COLDWAR_YUG_1970_tito_dead }
			set_variable = { COLDWAR_YUG_1970_monthly_loss = 1 }
			if = {
				limit = { has_country_flag = COLDWAR_YUG_1970_successor_confirmed }
				if = {
					limit = { has_country_flag = COLDWAR_YUG_1970_constitutional_centre }
					add_to_variable = { COLDWAR_YUG_1970_monthly_loss = -0.1 }
				}
				else_if = {
					limit = { OR = { has_country_flag = COLDWAR_YUG_1970_constitutional_republic_compact has_country_flag = COLDWAR_YUG_1970_constitutional_confederation } }
					add_to_variable = { COLDWAR_YUG_1970_monthly_loss = -0.25 }
				}
			}
			if = {
				limit = { has_country_flag = COLDWAR_YUG_1970_fiscal_funding_active }
				if = { limit = { has_country_flag = COLDWAR_YUG_1970_contributor_oversight_active } add_to_variable = { COLDWAR_YUG_1970_monthly_loss = -0.15 } }
				else = { add_to_variable = { COLDWAR_YUG_1970_monthly_loss = -0.25 } }
			}
			if = { limit = { has_country_flag = COLDWAR_YUG_1970_provincial_funding_active } add_to_variable = { COLDWAR_YUG_1970_monthly_loss = -0.15 } }
			if = { limit = { has_country_flag = COLDWAR_YUG_1970_command_funding_active } add_to_variable = { COLDWAR_YUG_1970_monthly_loss = -0.1 } }
			COLDWAR_YUG_1970_calculate_economic_relief = yes
			set_variable = { COLDWAR_YUG_1970_economic_relief_delta = COLDWAR_YUG_1970_economic_relief }
			multiply_variable = { COLDWAR_YUG_1970_economic_relief_delta = -1 }
			add_to_variable = { COLDWAR_YUG_1970_monthly_loss = COLDWAR_YUG_1970_economic_relief_delta }
			clamp_variable = { var = COLDWAR_YUG_1970_monthly_loss min = 0 max = 1 }
		}
		if = { limit = { has_country_flag = COLDWAR_YUG_1970_case_fiscal_active } COLDWAR_YUG_1970_add_fiscal_pressure = yes }
		if = { limit = { has_country_flag = COLDWAR_YUG_1970_case_constitutional_active } COLDWAR_YUG_1970_add_constitutional_pressure = yes }
		if = { limit = { has_country_flag = COLDWAR_YUG_1970_case_provincial_active } COLDWAR_YUG_1970_add_provincial_pressure = yes }
		if = { limit = { has_country_flag = COLDWAR_YUG_1970_case_command_active } COLDWAR_YUG_1970_add_command_pressure = yes }
		if = {
			limit = { COLDWAR_YUG_1970_tito_in_office = yes }
			multiply_variable = { COLDWAR_YUG_1970_monthly_case_loss = 0.5 }
			set_variable = { COLDWAR_YUG_1970_monthly_loss = COLDWAR_YUG_1970_monthly_case_loss }
		}
		else = { add_to_variable = { COLDWAR_YUG_1970_monthly_loss = COLDWAR_YUG_1970_monthly_case_loss } }
		if = {
			limit = { has_country_flag = COLDWAR_YUG_1970_tito_mediation_window }
			set_variable = { COLDWAR_YUG_1970_mediation_relief = COLDWAR_YUG_1970_monthly_case_loss }
			multiply_variable = { COLDWAR_YUG_1970_mediation_relief = -0.5 }
			add_to_variable = { COLDWAR_YUG_1970_monthly_loss = COLDWAR_YUG_1970_mediation_relief }
		}
		clamp_variable = { var = COLDWAR_YUG_1970_monthly_loss min = 0 max = 3 }
		set_variable = { COLDWAR_YUG_1970_loss_delta = COLDWAR_YUG_1970_monthly_loss }
		multiply_variable = { COLDWAR_YUG_1970_loss_delta = -1 }
		add_to_variable = { COLDWAR_YUG_1970_cohesion = COLDWAR_YUG_1970_loss_delta }
		clamp_variable = { var = COLDWAR_YUG_1970_cohesion min = 0 max = 100 }
		COLDWAR_YUG_1970_start_warning_windows = yes
		COLDWAR_YUG_1970_advance_case_warnings = yes
		add_to_variable = { COLDWAR_YUG_1970_auto_claim_cursor = 1 }
		if = { limit = { check_variable = { COLDWAR_YUG_1970_auto_claim_cursor > 6 } } set_variable = { COLDWAR_YUG_1970_auto_claim_cursor = 0 } }
		COLDWAR_YUG_1970_auto_file_claim = yes
		COLDWAR_YUG_1970_reconcile_cohesion_band = yes
	}
}

# Inactivity can turn an aged federal dispute into one named sovereignty motion.
# A rotating cursor avoids hard-coding one republic as the automatic claimant.
COLDWAR_YUG_1970_auto_file_claim = {
	if = {
		limit = {
			check_variable = { COLDWAR_YUG_1970_cohesion < 40 }
			NOT = { has_country_flag = COLDWAR_YUG_1970_claimant_SLV }
			NOT = { has_country_flag = COLDWAR_YUG_1970_claimant_CRO }
			NOT = { has_country_flag = COLDWAR_YUG_1970_claimant_BOS }
			NOT = { has_country_flag = COLDWAR_YUG_1970_claimant_FYR }
			NOT = { has_country_flag = COLDWAR_YUG_1970_claimant_MNT }
			NOT = { has_country_flag = COLDWAR_YUG_1970_claimant_KOS }
			NOT = { has_country_flag = COLDWAR_YUG_1970_claimant_VOJ }
			OR = {
				AND = { has_country_flag = COLDWAR_YUG_1970_case_constitutional_active check_variable = { COLDWAR_YUG_1970_case_constitutional_stage > 0 } }
				AND = { has_country_flag = COLDWAR_YUG_1970_case_fiscal_active check_variable = { COLDWAR_YUG_1970_case_fiscal_stage > 0 } }
				AND = { has_country_flag = COLDWAR_YUG_1970_case_provincial_active check_variable = { COLDWAR_YUG_1970_case_provincial_stage > 0 } }
			}
		}
