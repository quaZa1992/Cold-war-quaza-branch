ISR2_project_support_quote = {
	set_variable = { ISR2_project_net_cost = ISR2_project_cost }
	clamp_variable = { var = ISR2_project_net_cost min = 0 }
	set_variable = { ISR2_project_absorption_used = 0 }
	set_variable = { ISR2_project_match_used = 0 }
	set_variable = { ISR2_project_procurement_used = 0 }
	set_variable = { ISR2_project_support_cap = ISR2_project_quote }
	multiply_variable = { ISR2_project_support_cap = 0.005 }
	if = {
		limit = {
			check_variable = { ISR2_project_class = 1 }
		}
		set_variable = { ISR2_project_absorption_used = ISR2_project_net_cost }
		if = {
			limit = {
				check_variable = { ISR2_project_absorption_used > ISR2_domestic_absorption_balance }
			}
			set_variable = { ISR2_project_absorption_used = ISR2_domestic_absorption_balance }
		}
		clamp_variable = { var = ISR2_project_absorption_used min = 0 }
		subtract_from_variable = { ISR2_project_net_cost = ISR2_project_absorption_used }
		if = {
			limit = {
				check_variable = { ISR2_domestic_civilian_projects > 0 }
			}
			set_variable = { ISR2_project_match_used = ISR2_project_net_cost }
			if = {
				limit = {
					check_variable = { ISR2_project_match_used > ISR2_domestic_civilian_balance }
				}
				set_variable = { ISR2_project_match_used = ISR2_domestic_civilian_balance }
			}
			clamp_variable = { var = ISR2_project_match_used min = 0 }
			if = {
				limit = {
					check_variable = { ISR2_project_match_used > ISR2_project_support_cap }
				}
				set_variable = { ISR2_project_match_used = ISR2_project_support_cap }
			}
			subtract_from_variable = { ISR2_project_net_cost = ISR2_project_match_used }
		}
	}
	if = {
		limit = {
			check_variable = { ISR2_project_class = 2 }
		}
		if = {
			limit = {
				check_variable = { ISR2_domestic_settlement_projects > 0 }
			}
			set_variable = { ISR2_project_match_used = ISR2_project_net_cost }
			if = {
				limit = {
					check_variable = { ISR2_project_match_used > ISR2_domestic_settlement_balance }
				}
				set_variable = { ISR2_project_match_used = ISR2_domestic_settlement_balance }
			}
			clamp_variable = { var = ISR2_project_match_used min = 0 }
			if = {
				limit = {
					check_variable = { ISR2_project_match_used > ISR2_project_support_cap }
				}
				set_variable = { ISR2_project_match_used = ISR2_project_support_cap }
			}
			subtract_from_variable = { ISR2_project_net_cost = ISR2_project_match_used }
		}
	}
	if = {
		limit = {
			check_variable = { ISR2_project_class = 0 }
		}
		if = {
			limit = {
				check_variable = { ISR2_project_procurement = 1 }
				has_country_flag = ISR2_domestic_tank_coupon
			}
			set_variable = { ISR2_project_procurement_used = ISR2_project_net_cost }
			if = {
				limit = {
					check_variable = { ISR2_project_procurement_used > ISR2_project_support_cap }
				}
				set_variable = { ISR2_project_procurement_used = ISR2_project_support_cap }
			}
			subtract_from_variable = { ISR2_project_net_cost = ISR2_project_procurement_used }
		}
		if = {
			limit = {
				check_variable = { ISR2_project_procurement = 2 }
				has_country_flag = ISR2_domestic_aircraft_coupon
			}
			set_variable = { ISR2_project_procurement_used = ISR2_project_net_cost }
			if = {
				limit = {
					check_variable = { ISR2_project_procurement_used > ISR2_project_support_cap }
				}
				set_variable = { ISR2_project_procurement_used = ISR2_project_support_cap }
			}
			subtract_from_variable = { ISR2_project_net_cost = ISR2_project_procurement_used }
		}
	}
}
ISR2_project_support_commit = {
	ISR2_project_support_quote = yes
	if = {
		limit = {
			check_variable = { ISR2_project_absorption_used > 0 }
		}
		subtract_from_variable = { ISR2_domestic_absorption_balance = ISR2_project_absorption_used }
	}
	if = {
		limit = {
			check_variable = { ISR2_project_class = 1 }
			check_variable = { ISR2_project_match_used > 0 }
		}
		subtract_from_variable = { ISR2_domestic_civilian_balance = ISR2_project_match_used }
		subtract_from_variable = { ISR2_domestic_civilian_projects = 1 }
	}
	if = {
		limit = {
			check_variable = { ISR2_project_class = 2 }
			check_variable = { ISR2_project_match_used > 0 }
		}
		subtract_from_variable = { ISR2_domestic_settlement_balance = ISR2_project_match_used }
		subtract_from_variable = { ISR2_domestic_settlement_projects = 1 }
	}
	if = {
		limit = {
			check_variable = { ISR2_project_procurement = 1 }
			check_variable = { ISR2_project_procurement_used > 0 }
		}
		clr_country_flag = ISR2_domestic_tank_coupon
	}
	if = {
		limit = {
			check_variable = { ISR2_project_procurement = 2 }
			check_variable = { ISR2_project_procurement_used > 0 }
		}
		clr_country_flag = ISR2_domestic_aircraft_coupon
	}
}

				focus = ISR2_home_front
				uncomplete_children = no
				refund_political_power = no
			}
		}
	}
}
ISR2_domestic_refresh = {
	if = {
		limit = { ISR2_is_scenario = yes }
		ISR2_domestic_refresh_prices = yes
		set_variable = { ISR2_domestic_previous_expense = ISR2_domestic_weekly_expense }
		set_variable = { ISR2_domestic_weekly_income = 0 }
		set_variable = { ISR2_domestic_weekly_expense = 0 }
		if = {
			limit = {
				check_variable = { ISR2_domestic_exercise_days > 0 }
				check_variable = { ISR2_domestic_callup_days < 1 }
			}
			if = {
				limit = {
					NOT = {
						has_dynamic_modifier = { modifier = ISR2_domestic_exercise }
					}
				}
				add_dynamic_modifier = { modifier = ISR2_domestic_exercise }
			}
		}
		else = {
			if = {
				limit = {
					has_dynamic_modifier = { modifier = ISR2_domestic_exercise }
				}
				remove_dynamic_modifier = { modifier = ISR2_domestic_exercise }
			}
		}
		if = {
			limit = {
				ISR2_is_scenario = yes
				check_variable = { ISR2_domestic_callup_days > 0 }
				OR = { has_war = yes has_country_flag = ISR2_domestic_security_emergency }
				NOT = { has_country_flag = ISR2_domestic_done_bounded_callup }
			}
			if = {
				limit = {
					NOT = {
						has_dynamic_modifier = { modifier = ISR2_domestic_callup }
					}
				}
				add_dynamic_modifier = { modifier = ISR2_domestic_callup }
			}
		}
		else = {
			if = {
				limit = {
					has_dynamic_modifier = { modifier = ISR2_domestic_callup }
				}
				remove_dynamic_modifier = { modifier = ISR2_domestic_callup }
			}
		}
		if = {
			limit = {
				ISR2_is_scenario = yes
				check_variable = { ISR2_domestic_callup_days > 0 }
				OR = { has_war = yes has_country_flag = ISR2_domestic_security_emergency }
				has_country_flag = ISR2_domestic_done_bounded_callup
			}
			if = {
				limit = {
					NOT = {
						has_dynamic_modifier = { modifier = ISR2_domestic_callup_calendar }
					}
				}
				add_dynamic_modifier = { modifier = ISR2_domestic_callup_calendar }
			}
		}
		else = {
			if = {
				limit = {
					has_dynamic_modifier = { modifier = ISR2_domestic_callup_calendar }
				}
				remove_dynamic_modifier = { modifier = ISR2_domestic_callup_calendar }
			}
		}
		if = {
			limit = {
				ISR2_is_scenario = yes
				has_country_flag = ISR2_domestic_done_home_front
				owns_state = 204
				controls_state = 204
				has_full_control_of_state = 204
			}
			add_to_variable = { ISR2_domestic_weekly_expense = ISR2_domestic_quote_homefront_weekly }
			204 = {
				if = {
					limit = {
						NOT = {
							has_dynamic_modifier = { modifier = ISR2_domestic_civil_defence_204 }
						}
					}
					add_dynamic_modifier = { modifier = ISR2_domestic_civil_defence_204 scope = ISR }
				}
			}
		}
		if = {
			limit = {
				ISR2_is_scenario = yes
				check_variable = { ISR2_domestic_callup_days > 0 }
				OR = { has_war = yes has_country_flag = ISR2_domestic_security_emergency }
			}
			add_to_variable = { ISR2_domestic_weekly_expense = ISR2_domestic_quote_reserve_weekly }
		}
		if = {
			limit = {
				NOT = {
					check_variable = { ISR2_domestic_weekly_expense = ISR2_domestic_previous_expense }
				}
			}
			update_money_dirty_variable = yes
		}
	}
}
ISR2_domestic_daily = {
	if = {
		limit = { ISR2_is_scenario = yes }
		if = {
			limit = {
				check_variable = { ISR2_domestic_exercise_days > 0 }
			}
			subtract_from_variable = { ISR2_domestic_exercise_days = 1 }
		}
		if = {
			limit = {
				check_variable = { ISR2_domestic_callup_days > 0 }
			}
			subtract_from_variable = { ISR2_domestic_callup_days = 1 }
			if = {
				limit = {
					OR = {
						NOT = {
							OR = { has_war = yes has_country_flag = ISR2_domestic_security_emergency }
						}
						check_variable = { ISR2_domestic_callup_days < 1 }
					}
				}
				ISR2_domestic_end_callup = yes
			}
		}
		if = {
			limit = {
				has_completed_focus = ISR2_development_repayment
				NOT = { has_country_flag = ISR2_domestic_cooldown_development_repayment }
			}
			uncomplete_national_focus = {
				focus = ISR2_development_repayment
				uncomplete_children = no
				refund_political_power = no
			}
		}
		if = {
			limit = {
				has_completed_focus = ISR2_reserve_exercise
				NOT = { has_country_flag = ISR2_domestic_cooldown_reserve_exercise }
			}
			uncomplete_national_focus = {
				focus = ISR2_reserve_exercise
				uncomplete_children = no
				refund_political_power = no
			}
		}
		if = {
			limit = {
				has_completed_focus = ISR2_security_emergency
				NOT = { has_country_flag = ISR2_domestic_cooldown_security_emergency }
			}
			uncomplete_national_focus = {
				focus = ISR2_security_emergency
				uncomplete_children = no
				refund_political_power = no
			}
		}
		if = {
			limit = {
				has_completed_focus = ISR2_reserve_callup
				NOT = { has_country_flag = ISR2_domestic_cooldown_reserve_callup }
			}
			uncomplete_national_focus = {
				focus = ISR2_reserve_callup
				uncomplete_children = no
				refund_political_power = no
			}
		}
		if = {
			limit = {
				has_completed_focus = ISR2_reserve_release
				NOT = { has_country_flag = ISR2_domestic_cooldown_reserve_release }
			}
			uncomplete_national_focus = {
				focus = ISR2_reserve_release
				uncomplete_children = no
				refund_political_power = no
			}
		}
		ISR2_domestic_refresh = yes
	}
}
