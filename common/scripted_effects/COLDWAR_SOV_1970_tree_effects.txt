COLDWAR_SOVR_initialize = {
	if = {
		limit = { NOT = { has_country_flag = COLDWAR_SOVR_initialized } }
		set_country_flag = COLDWAR_SOVR_initialized
		if = { limit = { NOT = { has_global_flag = COLDWAR_PB_initialized } } COLDWAR_PB_setup = yes }
		COLDWAR_UNI_refresh_members = yes
		country_event = { id = COLDWAR_SOVR.1 days = 2 }
	}
}

COLDWAR_SOVR_monthly = {
	COLDWAR_PB_monthly_history = yes
	COLDWAR_PB_balance_drift = yes
	COLDWAR_PB_update_offices = yes
	COLDWAR_PB_build_board = yes
	COLDWAR_UNI_refresh_members = yes
	if = { limit = { has_country_flag = COLDWAR_SOVR_afghan_war has_war_support < 0.5 } country_event = { id = COLDWAR_SOVR.2 days = 3 } }
	if = {
		limit = {
			country_exists = AFG
			has_completed_focus = COLDWAR_SOV_for_friendship_with_kabul
			NOT = { has_global_flag = COLDWAR_SOVR_kabul_asked }
			NOT = { has_global_flag = COLDWAR_SOVR_kabul_refused }
			AFG = { has_government = communism OR = { has_civil_war = yes has_stability < 0.3 } }
		}
		AFG = { country_event = { id = COLDWAR_SOVR.3 days = 5 } }
	}
	if = {
		limit = {
			has_completed_focus = COLDWAR_SOV_for_the_kabul_request
			NOT = { has_completed_focus = COLDWAR_SOV_for_the_limited_contingent }
			NOT = { has_country_flag = COLDWAR_SOVR_flank_fell }
			OR = { NOT = { country_exists = AFG } AFG = { NOT = { has_government = communism } } }
		}
		set_country_flag = COLDWAR_SOVR_flank_fell
		country_event = { id = COLDWAR_SOVR.5 days = 2 }
	}
}
