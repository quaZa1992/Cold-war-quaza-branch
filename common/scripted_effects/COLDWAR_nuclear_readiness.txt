COLDWAR_nr_touch = {
	add_to_variable = { COLDWAR_nr_dirty = 1 }
}
COLDWAR_nr_init = {
	if = {
		limit = {
			NOT = { has_country_flag = COLDWAR_nr_initialized }
		}
		set_country_flag = COLDWAR_nr_initialized
		set_variable = { COLDWAR_nr_level = 5 }
		set_variable = { COLDWAR_nr_order = 5 }
		set_variable = { COLDWAR_nr_build_days = 0 }
		set_variable = { COLDWAR_nr_fatigue = 0 }
		set_variable = { COLDWAR_nr_civil_days = 0 }
	}
	if = {
		limit = { is_ai = no }
		COLDWAR_nr_shelter_census = yes
	}
	COLDWAR_nr_refresh = yes
}

	set_variable = { COLDWAR_nr_reserve_share = 0 }

		}
		set_variable = { COLDWAR_nr_ready_share = 0.15 }
		set_variable = { COLDWAR_nr_reserve_share = 0.1 }
		set_variable = { COLDWAR_nr_cost_factor = 1.0 }
	}
	if = {
		limit = {
			check_variable = { COLDWAR_nr_level = 4 }

		set_variable = { COLDWAR_nr_ready_share = 0.3 }
		set_variable = { COLDWAR_nr_reserve_share = 0.15 }
		set_variable = { COLDWAR_nr_cost_factor = 1.15 }
	}
	if = {
		limit = {
			check_variable = { COLDWAR_nr_level = 3 }
		}
		set_variable = { COLDWAR_nr_ready_share = 0.5 }
		set_variable = { COLDWAR_nr_reserve_share = 0.25 }
		set_variable = { COLDWAR_nr_cost_factor = 1.4 }
	}

		limit = {
			check_variable = { COLDWAR_nr_level = 2 }
		}
		set_variable = { COLDWAR_nr_ready_share = 0.75 }
		set_variable = { COLDWAR_nr_reserve_share = 0.4 }

		limit = {
			check_variable = { COLDWAR_nr_level = 1 }
		}
		set_variable = { COLDWAR_nr_ready_share = 1.0 }
		set_variable = { COLDWAR_nr_reserve_share = 0.5 }

	}

		COLDWAR_nr_ready_count = { value = COLDWAR_nr_arsenal multiply = COLDWAR_nr_ready_share round = yes }
	}
	set_variable = {
		COLDWAR_nr_protected_count = { value = COLDWAR_nr_arsenal multiply = COLDWAR_nr_reserve_share round = yes }
	}
	set_variable = {
		COLDWAR_nr_fatigue_stability = { value = COLDWAR_nr_fatigue multiply = -0.0005 }
	}
	if = {
		limit = { COLDWAR_nr_nuclear_power = yes }
		if = {
			limit = {
				NOT = {
					has_dynamic_modifier = { modifier = COLDWAR_nr_strategic_forces }

			}
			add_dynamic_modifier = { modifier = COLDWAR_nr_strategic_forces }

		set_variable = { COLDWAR_nr_protected_count = 0 }
		set_variable = { COLDWAR_nr_ready_share = 0 }
		set_variable = { COLDWAR_nr_reserve_share = 0 }

				has_dynamic_modifier = { modifier = COLDWAR_nr_strategic_forces }
			}
			remove_dynamic_modifier = { modifier = COLDWAR_nr_strategic_forces }
		}
	}
	COLDWAR_nr_shelter_use = yes
	if = {
		limit = {
			OR = {
				check_variable = { COLDWAR_nr_build_days > 0 }
				check_variable = { COLDWAR_nr_civil_days > 0 }
			}
		}
		COLDWAR_nr_schedule_tick = yes

	COLDWAR_nr_calculate_budget = yes
	COLDWAR_nr_validate_first_use = yes
	COLDWAR_nr_update_display = yes
	force_update_dynamic_modifier = yes
	COLDWAR_nr_touch = yes
}
# IMPERIA COLD WAR update 25.11.2026: GDP-based weekly costs. Annual shares are expressed in units of 0.01% of GDP;
# weekly $M = units x gdp_total / 520 (gdp_total is annual GDP in $B). Charged in $B through defence_gain.
# Strategic forces: 1 unit + 0.005 units per stored missile, at most 25 units (0.25% of GDP), times readiness and fatigue.
# Civil defence mobilisation: 2 units + 40 units x national shelter coverage.
COLDWAR_nr_calculate_budget = {
	set_variable = { COLDWAR_nr_forces_weekly_m = 0 }
	set_variable = { COLDWAR_nr_civil_weekly_m = 0 }
	if = {
		limit = { COLDWAR_nr_nuclear_power = yes has_country_flag = COLDWAR_nr_initialized }
		set_variable = {
			COLDWAR_nr_forces_weekly_m = {
				value = num_equipment@nuclear_missile_equipment
				add = num_equipment@nuclear_ballistic_missile_equipment
				multiply = 0.005
				add = 1
				clamp = { min = 1 max = 25 }
				multiply = COLDWAR_nr_cost_factor
				multiply = {
					value = 1
					add = { value = COLDWAR_nr_fatigue multiply = 0.005 }
				}
				multiply = gdp_total
				divide = 520
				clamp = { min = 0 max = 100000000 }
			}
		}
	}
	if = {
		limit = {
			OR = {
				has_country_flag = COLDWAR_nr_civil_active
				check_variable = { COLDWAR_nr_civil_days > 0 }
			}
		}
		set_variable = {
			COLDWAR_nr_civil_weekly_m = {
				value = COLDWAR_nr_shelter_coverage
				multiply = 40
				add = 2
				multiply = gdp_total
				divide = 520
				clamp = { min = 0 max = 100000000 }
			}
		}
	}
	set_variable = { COLDWAR_nr_forces_weekly = { value = COLDWAR_nr_forces_weekly_m divide = 1000 } }
	set_variable = { COLDWAR_nr_civil_weekly = { value = COLDWAR_nr_civil_weekly_m divide = 1000 } }
	set_variable = {
		COLDWAR_nr_weekly_cost = { value = COLDWAR_nr_forces_weekly add = COLDWAR_nr_civil_weekly }
	}
}
# IMPERIA COLD WAR update 25.11.2026: frame and countdown values read by the Strategic Command v2 window.
COLDWAR_nr_update_display = {
	set_variable = {
		COLDWAR_nr_fatigue_frame = { value = COLDWAR_nr_fatigue divide = 5 round = yes add = 1 clamp = { min = 1 max = 21 } }
	}
	set_variable = { COLDWAR_nr_prep_frame = 1 }
	if = {
		limit = {
			check_variable = { COLDWAR_nr_build_days > 0 }
		}
		if = {
			limit = {
				check_variable = { COLDWAR_nr_build_total < COLDWAR_nr_build_days }
			}
			set_variable = { COLDWAR_nr_build_total = COLDWAR_nr_build_days }
		}
		set_variable = {
			COLDWAR_nr_prep_frame = {
				value = COLDWAR_nr_build_total
				subtract = COLDWAR_nr_build_days
				multiply = 8
				divide = COLDWAR_nr_build_total
				round = yes
				add = 1
				clamp = { min = 1 max = 9 }
			}
		}
	}
	set_variable = { COLDWAR_nr_crisis_days = 0 }
	if = {
		limit = { has_variable = COLDWAR_nr_until }
		set_variable = {
			COLDWAR_nr_crisis_days = { value = COLDWAR_nr_until subtract = global.num_days clamp = { min = 0 max = 1000 } }
		}
	}
	COLDWAR_nr_countdown_frame = yes
	COLDWAR_nr_paranoia_display = yes
}
COLDWAR_nr_countdown_frame = {
	set_variable = { COLDWAR_nr_fu_frame = 1 }
	set_temp_variable = { COLDWAR_nr_fu_show_hours = 0 }
	set_temp_variable = { COLDWAR_nr_fu_show_total = 72 }
	if = {
		limit = {
			OR = { has_country_flag = COLDWAR_nr_fu_waiting has_country_flag = COLDWAR_nr_fu_authorised }
		}
		set_temp_variable = { COLDWAR_nr_fu_show_hours = COLDWAR_nr_fu_hours }
		if = {
			limit = { has_country_flag = COLDWAR_nr_fu_authorised }
			set_temp_variable = { COLDWAR_nr_fu_show_total = 168 }
		}
	}
	else_if = {
		limit = {
			check_variable = { COLDWAR_nr_fu_incoming_hours > 0 }
		}
		set_temp_variable = { COLDWAR_nr_fu_show_hours = COLDWAR_nr_fu_incoming_hours }
		if = {
			limit = { COLDWAR_nr_foreign_release = yes }
			set_temp_variable = { COLDWAR_nr_fu_show_total = 168 }
		}
	}
	if = {
		limit = {
			check_variable = { COLDWAR_nr_fu_show_hours > 0 }
		}
		set_variable = {
			COLDWAR_nr_fu_frame = {
				value = COLDWAR_nr_fu_show_hours
				multiply = 12
				divide = COLDWAR_nr_fu_show_total
				add = 0.49999
				round = yes
				add = 1
				clamp = { min = 1 max = 13 }
			}
		}
	}
}

COLDWAR_nr_start_civil = {
	COLDWAR_nr_shelter_census = yes
	if = {
		limit = { COLDWAR_nr_can_mobilise_civil = yes }
		COLDWAR_nr_init = yes
		add_political_power = -25
		set_variable = { COLDWAR_nr_civil_days = 3 }
		if = {
			limit = {
				check_variable = { COLDWAR_nr_shelter_coverage > 0.59999 }
			}
			set_variable = { COLDWAR_nr_civil_days = 1 }
		}
		else_if = {
			limit = {
				check_variable = { COLDWAR_nr_shelter_coverage > 0.29999 }
			}
			set_variable = { COLDWAR_nr_civil_days = 2 }
		}
		add_dynamic_modifier = { modifier = COLDWAR_nr_civil_defence }
		COLDWAR_nr_refresh = yes
	}
}
COLDWAR_nr_end_civil = {

	set_variable = { COLDWAR_nr_civil_days = 0 }
	if = {
		limit = {
			has_dynamic_modifier = { modifier = COLDWAR_nr_civil_defence }
		}
		remove_dynamic_modifier = { modifier = COLDWAR_nr_civil_defence }
	}
	COLDWAR_nr_refresh = yes
}
COLDWAR_nr_schedule_tick = {
	if = {
		limit = {
			NOT = { has_country_flag = COLDWAR_nr_tick_pending }
		}
		set_country_flag = { flag = COLDWAR_nr_tick_pending days = 2 value = 1 }
		country_event = { id = cw_nuclear.91 days = 1 }
	}
}
COLDWAR_nr_tick = {
	if = {
		limit = {
			check_variable = { COLDWAR_nr_build_days > 0 }
		}
		if = {
			limit = {
				COLDWAR_nr_nuclear_power = yes
				NOT = { has_country_flag = COLDWAR_nr_standdown }
			}
			subtract_from_variable = { COLDWAR_nr_build_days = 1 }
			if = {
				limit = {
					check_variable = { COLDWAR_nr_build_days = 0 }
				}
				set_variable = { COLDWAR_nr_level = COLDWAR_nr_order }
				if = {
					limit = { is_ai = no }
					country_event = { id = cw_nuclear.1 }
				}
			}
		}
		else = {
			set_variable = { COLDWAR_nr_build_days = 0 }
			set_variable = { COLDWAR_nr_order = COLDWAR_nr_level }
		}
	}
	if = {
		limit = {
			check_variable = { COLDWAR_nr_civil_days > 0 }
		}
		subtract_from_variable = { COLDWAR_nr_civil_days = 1 }
		if = {
			limit = {
				check_variable = { COLDWAR_nr_civil_days = 0 }
			}
			set_country_flag = COLDWAR_nr_civil_active
			if = {
				limit = { is_ai = no }
				country_event = { id = cw_nuclear.2 }
			}
		}
	}
	COLDWAR_nr_refresh = yes
}
COLDWAR_nr_check_crisis = {
	if = {
		limit = { has_variable = COLDWAR_nr_opponent }
		if = {
			limit = {
				NOT = { COLDWAR_nr_pair_valid = yes }
			}
			COLDWAR_nr_clear_local_crisis = yes
		}
		else_if = {
			limit = {
				check_variable = { COLDWAR_nr_until < global.num_days }
			}
			save_event_target_as = COLDWAR_nr_sender
			var:COLDWAR_nr_opponent = { save_event_target_as = COLDWAR_nr_recipient }
			COLDWAR_nr_close_pair_minus2 = yes
			if = {
				limit = { is_ai = no }
				country_event = { id = cw_nuclear.9 }
			}
			event_target:COLDWAR_nr_recipient = {
				if = {
					limit = { is_ai = no }
					country_event = { id = cw_nuclear.9 }
				}
			}
		}
	}
}
COLDWAR_nr_weekly = {
	if = {
		limit = {
			NOT = { has_country_flag = COLDWAR_nr_initialized }
		}
		COLDWAR_nr_init = yes
	}
	if = {
		limit = {
			COLDWAR_nr_nuclear_power = yes
			check_variable = { COLDWAR_nr_level = 1 }
		}
		add_to_variable = { COLDWAR_nr_fatigue = 14 }
	}
	else_if = {
		limit = {
			COLDWAR_nr_nuclear_power = yes
			check_variable = { COLDWAR_nr_level = 2 }
		}
		add_to_variable = { COLDWAR_nr_fatigue = 7 }
	}
	else_if = {
		limit = {
			COLDWAR_nr_nuclear_power = yes
			check_variable = { COLDWAR_nr_level = 3 }
		}
		add_to_variable = { COLDWAR_nr_fatigue = 1.75 }
	}
	else_if = {
		limit = {
			check_variable = { COLDWAR_nr_fatigue > 0 }
		}
		subtract_from_variable = { COLDWAR_nr_fatigue = 14 }
	}
	clamp_variable = { var = COLDWAR_nr_fatigue min = 0 max = 100 }
	if = {
		limit = { has_capitulated = yes }
		COLDWAR_nr_stand_down = yes
		COLDWAR_nr_end_civil = yes
	}
	COLDWAR_nr_check_crisis = yes
	COLDWAR_nr_paranoia_weekly = yes
	if = {
		limit = {
			OR = {
				has_country_flag = COLDWAR_nr_civil_active
				check_variable = { COLDWAR_nr_civil_days > 0 }
			}
			NOT = {
				any_owned_state = { civilian_shelter > 0 }
			}
		}
		COLDWAR_nr_end_civil = yes
	}
	if = {
		limit = { is_ai = no }
		COLDWAR_nr_shelter_census = yes
	}
	if = {
		limit = {
			OR = {
				is_ai = no
				COLDWAR_nr_nuclear_power = yes
				has_dynamic_modifier = { modifier = COLDWAR_nr_strategic_forces }
				check_variable = { COLDWAR_nr_fatigue > 0 }
				has_country_flag = COLDWAR_nr_civil_active
				check_variable = { COLDWAR_nr_civil_days > 0 }
				has_variable = COLDWAR_nr_opponent
			}
		}
		COLDWAR_nr_refresh = yes
	}
}
COLDWAR_nr_capitulated = {
	if = {
		limit = { has_variable = COLDWAR_nr_opponent }
		var:COLDWAR_nr_opponent = {
			if = {
				limit = {
					exists = yes
					check_variable = { COLDWAR_nr_opponent = PREV.id }
				}
				COLDWAR_nr_clear_local_crisis = yes
			}
		}
		COLDWAR_nr_clear_local_crisis = yes
	}
	COLDWAR_nr_stand_down = yes
	COLDWAR_nr_end_civil = yes
}

		set_temp_variable = { COLDWAR_nr_state_pop_k = state_population_k }
		add_to_variable = { PREV.COLDWAR_nr_population_k = COLDWAR_nr_state_pop_k }
		if = {
			limit = { civilian_shelter > 0 }
			set_temp_variable = { COLDWAR_nr_state_places_k = building_level@civilian_shelter }
			multiply_temp_variable = { COLDWAR_nr_state_places_k = 250 }
			if = {
				limit = {
					check_variable = { COLDWAR_nr_state_places_k > COLDWAR_nr_state_pop_k }
				}
				set_temp_variable = { COLDWAR_nr_state_places_k = COLDWAR_nr_state_pop_k }
			}
			add_to_variable = { PREV.COLDWAR_nr_shelter_places_k = COLDWAR_nr_state_places_k }
		}
	}
	set_variable = { COLDWAR_nr_shelter_coverage = 0 }
	if = {
		limit = {
			check_variable = { COLDWAR_nr_population_k > 0 }
		}
		set_variable = { COLDWAR_nr_shelter_coverage = COLDWAR_nr_shelter_places_k }
		divide_variable = { COLDWAR_nr_shelter_coverage = COLDWAR_nr_population_k }
		clamp_variable = { var = COLDWAR_nr_shelter_coverage min = 0 max = 1 }
	}
	set_variable = {
		COLDWAR_nr_coverage_frame = { value = COLDWAR_nr_shelter_coverage multiply = 20 round = yes add = 1 clamp = { min = 1 max = 21 } }
	}
	set_variable = { COLDWAR_nr_shelter_cover_pct = COLDWAR_nr_shelter_coverage }
	multiply_variable = { COLDWAR_nr_shelter_cover_pct = 100 }
	set_variable = { COLDWAR_nr_shelter_places_m = COLDWAR_nr_shelter_places_k }
	divide_variable = { COLDWAR_nr_shelter_places_m = 1000 }
	set_variable = { COLDWAR_nr_population_m = COLDWAR_nr_population_k }
	divide_variable = { COLDWAR_nr_population_m = 1000 }
	COLDWAR_nr_touch = yes
}

COLDWAR_nr_apply_shelters = {
	set_temp_variable = { COLDWAR_nr_protect = COLDWAR_nr_shelter_hardness }
	if = {
		limit = {
			check_variable = { COLDWAR_nr_protect = 0 }
		}
		set_temp_variable = { COLDWAR_nr_protect = 0.75 }
	}
	if = {
		limit = { civilian_shelter > 0 }
		set_temp_variable = { COLDWAR_nr_state_pop_k = state_population_k }
		if = {
			limit = {
				check_variable = { COLDWAR_nr_state_pop_k > 0 }
			}
			set_temp_variable = { COLDWAR_nr_cover = building_level@civilian_shelter }
			multiply_temp_variable = { COLDWAR_nr_cover = 250 }
			divide_temp_variable = { COLDWAR_nr_cover = COLDWAR_nr_state_pop_k }
			clamp_temp_variable = { var = COLDWAR_nr_cover min = 0 max = 1 }
			set_temp_variable = { COLDWAR_nr_use = 0.35 }
			CONTROLLER = {
				if = {
					limit = { has_country_flag = COLDWAR_nr_civil_active }
					set_temp_variable = { COLDWAR_nr_use = 1 }
				}
				else_if = {
					limit = { COLDWAR_nr_heightened_warning = yes }
					set_temp_variable = { COLDWAR_nr_use = 0.6 }
				}
			}
			OWNER = {
				if = {
					limit = { COLDWAR_nr_has_hardened_shelters = yes }
					add_to_temp_variable = { COLDWAR_nr_protect = 0.1 }
				}
			}
			clamp_temp_variable = { var = COLDWAR_nr_protect min = 0 max = 0.95 }
			multiply_temp_variable = { COLDWAR_nr_cover = COLDWAR_nr_use }
			multiply_temp_variable = { COLDWAR_nr_cover = COLDWAR_nr_protect }
			set_temp_variable = { COLDWAR_nr_survival = 1 }
			subtract_from_temp_variable = { COLDWAR_nr_survival = COLDWAR_nr_cover }
			multiply_temp_variable = { casualty_rate = COLDWAR_nr_survival }
		}
	}
	CONTROLLER = {
		if = {
			limit = { has_country_flag = COLDWAR_nr_civil_active }
			multiply_temp_variable = { casualty_rate = 0.9 }
		}
	}
}
