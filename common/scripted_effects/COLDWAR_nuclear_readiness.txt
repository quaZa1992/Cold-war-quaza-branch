# IMPERIA CW — Cold War: nuclear readiness and crisis diplomacy.
COLDWAR_nr_touch = {
	add_to_variable = { COLDWAR_nr_dirty = 1 }
}
COLDWAR_nr_init = {
	if = {
		limit = {
			NOT = { has_country_flag = COLDWAR_nr_initialized }
		}
		set_country_flag = COLDWAR_nr_initialized
		set_variable = { COLDWAR_nr_level = 5 }
		set_variable = { COLDWAR_nr_order = 5 }
		set_variable = { COLDWAR_nr_build_days = 0 }
		set_variable = { COLDWAR_nr_fatigue = 0 }
		set_variable = { COLDWAR_nr_civil_days = 0 }
	}
	COLDWAR_nr_refresh = yes
}
COLDWAR_nr_refresh = {
	set_variable = {
		COLDWAR_nr_arsenal = {
			value = num_equipment@nuclear_missile_equipment
			add = num_equipment@nuclear_ballistic_missile_equipment
			clamp = { min = 0 max = 1000000 }
		}
	}
	set_variable = { COLDWAR_nr_ready_share = 0 }
	set_variable = { COLDWAR_nr_reserve_share = 0 }
	set_variable = { COLDWAR_nr_cost_factor = 0 }
	if = {
		limit = {
			check_variable = { COLDWAR_nr_level = 5 }
		}
		set_variable = { COLDWAR_nr_ready_share = 0.15 }
		set_variable = { COLDWAR_nr_reserve_share = 0.1 }
		set_variable = { COLDWAR_nr_cost_factor = 1.0 }
	}
	if = {
		limit = {
			check_variable = { COLDWAR_nr_level = 4 }
		}
		set_variable = { COLDWAR_nr_ready_share = 0.3 }
		set_variable = { COLDWAR_nr_reserve_share = 0.15 }
		set_variable = { COLDWAR_nr_cost_factor = 1.15 }
	}
	if = {
		limit = {
			check_variable = { COLDWAR_nr_level = 3 }
		}
		set_variable = { COLDWAR_nr_ready_share = 0.5 }
		set_variable = { COLDWAR_nr_reserve_share = 0.25 }
		set_variable = { COLDWAR_nr_cost_factor = 1.4 }
	}
	if = {
		limit = {
			check_variable = { COLDWAR_nr_level = 2 }
		}
		set_variable = { COLDWAR_nr_ready_share = 0.75 }
		set_variable = { COLDWAR_nr_reserve_share = 0.4 }
		set_variable = { COLDWAR_nr_cost_factor = 1.8 }
	}
	if = {
		limit = {
			check_variable = { COLDWAR_nr_level = 1 }
		}
		set_variable = { COLDWAR_nr_ready_share = 1.0 }
		set_variable = { COLDWAR_nr_reserve_share = 0.5 }
		set_variable = { COLDWAR_nr_cost_factor = 2.5 }
	}
	set_variable = {
		COLDWAR_nr_ready_count = { value = COLDWAR_nr_arsenal multiply = COLDWAR_nr_ready_share round = yes }
	}
	set_variable = {
		COLDWAR_nr_protected_count = { value = COLDWAR_nr_arsenal multiply = COLDWAR_nr_reserve_share round = yes }
	}
	set_variable = {
		COLDWAR_nr_fatigue_stability = { value = COLDWAR_nr_fatigue multiply = -0.0005 }
	}
	if = {
		limit = { COLDWAR_nr_nuclear_power = yes }
		if = {
			limit = {
				NOT = {
					has_dynamic_modifier = { modifier = COLDWAR_nr_strategic_forces }
				}
			}
			add_dynamic_modifier = { modifier = COLDWAR_nr_strategic_forces }
		}
	}
	else = {
		set_variable = { COLDWAR_nr_level = 5 }
		set_variable = { COLDWAR_nr_order = 5 }
		set_variable = { COLDWAR_nr_build_days = 0 }
		set_variable = { COLDWAR_nr_ready_count = 0 }
		set_variable = { COLDWAR_nr_protected_count = 0 }
		set_variable = { COLDWAR_nr_ready_share = 0 }
		set_variable = { COLDWAR_nr_reserve_share = 0 }
		if = {
			limit = {
				has_dynamic_modifier = { modifier = COLDWAR_nr_strategic_forces }
			}
			remove_dynamic_modifier = { modifier = COLDWAR_nr_strategic_forces }
		}
	}
	COLDWAR_nr_calculate_budget = yes
	COLDWAR_nr_validate_first_use = yes
	force_update_dynamic_modifier = yes
	COLDWAR_nr_touch = yes
}
COLDWAR_nr_calculate_budget = {
	set_variable = { COLDWAR_nr_forces_weekly = 0 }
	set_variable = { COLDWAR_nr_civil_weekly = 0 }
	set_variable = { COLDWAR_nr_weekly_cost = 0 }
	if = {
		limit = { COLDWAR_nr_nuclear_power = yes has_country_flag = COLDWAR_nr_initialized }
		set_variable = {
			COLDWAR_nr_forces_weekly = {
				value = num_equipment@nuclear_missile_equipment
				add = num_equipment@nuclear_ballistic_missile_equipment
				multiply = 0.00001
				add = 0.002
				multiply = COLDWAR_nr_cost_factor
				multiply = {
					value = 1
					add = { value = COLDWAR_nr_fatigue multiply = 0.005 }
				}
				multiply = {
					value = gdp_per_capita
					divide = 36
					clamp = { min = 0.4 max = 1.2 }
				}
				multiply = {
					value = inflation_multiplier
					clamp = { min = 0.01 max = 100 }
				}
			}
		}
	}
	if = {
		limit = {
			OR = {
				has_country_flag = COLDWAR_nr_civil_active
				check_variable = { COLDWAR_nr_civil_days > 0 }
			}
		}
		set_variable = {
			COLDWAR_nr_civil_weekly = {
				value = gdp_total
				multiply = 0.00002
				add = 0.002
				multiply = {
					value = inflation_multiplier
					clamp = { min = 0.01 max = 100 }
				}
			}
		}
	}
	set_variable = {
		COLDWAR_nr_weekly_cost = { value = COLDWAR_nr_forces_weekly add = COLDWAR_nr_civil_weekly }
	}
}

COLDWAR_nr_start_civil = {
	if = {
		limit = {
			COLDWAR_nr_enabled = yes
			is_ai = no
			has_capitulated = no
			has_political_power > 24
			NOT = { has_country_flag = COLDWAR_nr_civil_active }
			check_variable = { COLDWAR_nr_civil_days = 0 }
		}
		COLDWAR_nr_init = yes
		add_political_power = -25
		set_variable = { COLDWAR_nr_civil_days = 3 }
		add_dynamic_modifier = { modifier = COLDWAR_nr_civil_defence }
		COLDWAR_nr_refresh = yes
	}
}

COLDWAR_nr_daily = {
	if = {
		limit = {
			NOT = { has_country_flag = COLDWAR_nr_initialized }
		}
		COLDWAR_nr_init = yes
	}
	if = {
		limit = {
			check_variable = { COLDWAR_nr_build_days > 0 }
		}
		if = {
			limit = {
				COLDWAR_nr_nuclear_power = yes
				NOT = { has_country_flag = COLDWAR_nr_standdown }
			}
			subtract_from_variable = { COLDWAR_nr_build_days = 1 }
			if = {
				limit = {
					check_variable = { COLDWAR_nr_build_days = 0 }
				}
				set_variable = { COLDWAR_nr_level = COLDWAR_nr_order }
				if = {
					limit = { is_ai = no }
					country_event = { id = cw_nuclear.1 }
				}
			}
		}
		else = {
			set_variable = { COLDWAR_nr_build_days = 0 }
			set_variable = { COLDWAR_nr_order = COLDWAR_nr_level }
		}
	}
	if = {
		limit = {
			check_variable = { COLDWAR_nr_civil_days > 0 }
		}
		subtract_from_variable = { COLDWAR_nr_civil_days = 1 }
		if = {
			limit = {
				check_variable = { COLDWAR_nr_civil_days = 0 }
			}
			set_country_flag = COLDWAR_nr_civil_active
			if = {
				limit = { is_ai = no }
				country_event = { id = cw_nuclear.2 }
			}
		}
	}
	if = {
		limit = {
			COLDWAR_nr_nuclear_power = yes
			check_variable = { COLDWAR_nr_level = 1 }
		}
		add_to_variable = { COLDWAR_nr_fatigue = 2 }
	}
	else_if = {
		limit = {
			COLDWAR_nr_nuclear_power = yes
			check_variable = { COLDWAR_nr_level = 2 }
		}
		add_to_variable = { COLDWAR_nr_fatigue = 1 }
	}
	else_if = {
		limit = {
			COLDWAR_nr_nuclear_power = yes
			check_variable = { COLDWAR_nr_level = 3 }
		}
		add_to_variable = { COLDWAR_nr_fatigue = 0.25 }
	}
	else = {
		subtract_from_variable = { COLDWAR_nr_fatigue = 2 }
	}
	clamp_variable = { var = COLDWAR_nr_fatigue min = 0 max = 100 }
	if = {
		limit = { has_capitulated = yes }
		COLDWAR_nr_stand_down = yes
		COLDWAR_nr_end_civil = yes
	}
	if = {
		limit = { has_variable = COLDWAR_nr_opponent }
		if = {
			limit = {
				NOT = { COLDWAR_nr_pair_valid = yes }
			}
			COLDWAR_nr_clear_local_crisis = yes
		}
		else_if = {
			limit = {
				check_variable = { COLDWAR_nr_until < global.num_days }
			}
			save_event_target_as = COLDWAR_nr_sender
			var:COLDWAR_nr_opponent = { save_event_target_as = COLDWAR_nr_recipient }
			COLDWAR_nr_close_pair_minus2 = yes
			if = {
				limit = { is_ai = no }
				country_event = { id = cw_nuclear.9 }
			}
			event_target:COLDWAR_nr_recipient = {
				if = {
					limit = { is_ai = no }
					country_event = { id = cw_nuclear.9 }
				}
			}
		}
	}
	COLDWAR_nr_refresh = yes
}

COLDWAR_nr_begin_crisis = {
	if = {
		limit = { COLDWAR_nr_can_warn_prev = yes }
		save_event_target_as = COLDWAR_nr_sender
		PREV = { save_event_target_as = COLDWAR_nr_recipient }
		COLDWAR_nr_tension_5 = yes
		add_to_variable = { global.COLDWAR_nr_serial = 1 }
		set_variable = { COLDWAR_nr_serial = global.COLDWAR_nr_serial }
		set_variable = { COLDWAR_nr_opponent = event_target:COLDWAR_nr_recipient.id }
		set_variable = { COLDWAR_nr_initiator = THIS.id }
		set_variable = {
			COLDWAR_nr_until = { value = global.num_days add = 90 }
		}
		set_variable = { COLDWAR_nr_pressure = COLDWAR_nr_tension_applied }
		set_country_flag = { flag = COLDWAR_nr_public days = 90 value = 1 }
		clr_country_flag = COLDWAR_nr_countered
		clr_country_flag = COLDWAR_nr_hotline
		event_target:COLDWAR_nr_recipient = {
			COLDWAR_nr_init = yes
			set_variable = { COLDWAR_nr_opponent = event_target:COLDWAR_nr_sender.id }
			set_variable = { COLDWAR_nr_initiator = event_target:COLDWAR_nr_sender.id }
			set_variable = { COLDWAR_nr_serial = global.COLDWAR_nr_serial }
			set_variable = {
				COLDWAR_nr_until = { value = global.num_days add = 90 }
			}
			set_variable = { COLDWAR_nr_pressure = event_target:COLDWAR_nr_sender.COLDWAR_nr_pressure }
			clr_country_flag = COLDWAR_nr_countered
			clr_country_flag = COLDWAR_nr_hotline
			COLDWAR_nr_refresh = yes
			country_event = { id = cw_nuclear.3 }
		}
		add_named_threat = { threat = 2 name = COLDWAR_nr_warning_threat }
		every_country = {
			limit = {
				is_ai = no
				NOT = { tag = event_target:COLDWAR_nr_sender }
				NOT = { tag = event_target:COLDWAR_nr_recipient }
			}
			country_event = { id = cw_nuclear.10 }
		}
		COLDWAR_nr_refresh = yes
	}
}
