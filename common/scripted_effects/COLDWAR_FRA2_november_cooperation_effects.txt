	if = {
		limit = {
			COLDWAR_FRA2_NOV_COOP_active_advisory = yes
		}
		if = { limit = { NOT = { has_idea = COLDWAR_FRA2_NOV_COOP_advisory_host } } add_ideas = COLDWAR_FRA2_NOV_COOP_advisory_host }
		FRA = { set_temp_variable = { COLDWAR_FRA2_NOV_COOP_active_scan_advisory = 1 } }
		if = {
			limit = {
				FRA = { NOT = { has_military_access_to = PREV } }
			}
			give_military_access = FRA
			set_country_flag = COLDWAR_FRA2_NOV_COOP_access_owned_advisory
		}
	}
	else = {
		remove_ideas = COLDWAR_FRA2_NOV_COOP_advisory_host
		if = {
			limit = {
				has_country_flag = COLDWAR_FRA2_NOV_COOP_access_owned_advisory
			}
			if = {
				limit = {
					country_exists = FRA
				}
				FRA = { if = { limit = { has_military_access_to = PREV } diplomatic_relation = { country = PREV relation = military_access active = no } } }
			}
			clr_country_flag = COLDWAR_FRA2_NOV_COOP_access_owned_advisory
		}
	}
	if = {
		limit = {
			COLDWAR_FRA2_NOV_COOP_active_commercial = yes
		}
		if = { limit = { NOT = { has_idea = COLDWAR_FRA2_NOV_COOP_commercial_host } } add_ideas = COLDWAR_FRA2_NOV_COOP_commercial_host }

			COLDWAR_FRA2_NOV_COOP_active_security = yes

	if = {
