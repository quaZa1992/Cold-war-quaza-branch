# IMPERIA COLD WAR — South African project payments
# The same calculation runs in effect_tooltip and again on payment.
COLDWAR_SAF_PROJECT_pay_rail = {
 set_temp_variable = { treasury_change = { value = gdp_total multiply = -0.005 } }
 modify_treasury_effect = yes
}

COLDWAR_SAF_PROJECT_pay_factory = {
 set_temp_variable = { treasury_change = { value = gdp_total multiply = -0.01 } }

}

}

COLDWAR_SAF_PROJECT_complete_fuel_silo = {
 if = {
  limit = { COLDWAR_SAF_PROJECT_can_build_fuel_silo = yes }
  COLDWAR_SAF_PROJECT_pay_fuel_silo = yes
  random_owned_controlled_state = {
   limit = { COLDWAR_SAF_PROJECT_fuel_silo_site = yes }
   add_extra_state_shared_building_slots = 1
   add_building_construction = { type = fuel_silo level = 1 instant_build = yes }
  }
  set_country_flag = COLDWAR_SAF_ECONPLAN_silo_built
 }
}
