# Old events must not replace an authored 1970 tree or erase its progress.
# 1970: countries without an authored tree get the generic 1970 tree (mdcw_1970_generic_focus);
# a country that already runs it keeps its progress. Other start dates keep the empty tree.
mdcw_load_1970_fallback_tree = {
	if = {
		limit = {
			NOT = {
				OR = {
					original_tag = USA
					original_tag = SPR
					original_tag = LBA
					original_tag = EGY # COLDWAR | own 1970 tree
					AND = { original_tag = SOV IMP_is_1970_scenario = yes } # COLDWAR | COLDWAR_SOV_1970_focus
					AND = { original_tag = CYP IMP_is_1970_scenario = yes } # COLDWAR_CYP_1970_focus
					AND = { original_tag = ISR IMP_is_1970_scenario = yes } # ISR70 | own 1970 tree
					AND = { original_tag = GRE IMP_is_1970_scenario = yes } # COLDWAR_GRE_1970_focus
				}
			}
		}
		if = {
			limit = { IMP_is_1970_scenario = yes }
			if = {
				limit = { NOT = { has_focus_tree = mdcw_1970_generic_focus } }
				load_focus_tree = mdcw_1970_generic_focus
			}
		}
		else = {
			load_focus_tree = mdcw_1970_empty_focus
		}
	}
}
