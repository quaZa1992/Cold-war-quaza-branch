					set_variable = { ISR2_private_claim_portion = ISR2_private_claim_remainder }
				}
				else = {
					set_variable = {
						ISR2_private_claim_portion = {
							value = ISR2_private_claim_amount
							multiply = ISR2_private_claim_share_GER
							divide = ISR2_private_claim_total_share
						}
					}
				}
				subtract_from_variable = { ISR2_private_claim_remainder = ISR2_private_claim_portion }
				add_to_variable = { ISR2_private_claim_due_GER = ISR2_private_claim_portion }
				add_to_variable = { ISR2_private_retire_due_GER = ISR2_private_claim_share_GER }
			}
			if = {
				limit = {
					check_variable = { ISR2_private_claim_share_ROM > 0 }
				}
				if = {
					limit = {
						check_variable = { ISR2_private_claim_last_index = 3 }
					}
					set_variable = { ISR2_private_claim_portion = ISR2_private_claim_remainder }
				}
				else = {
					set_variable = {
						ISR2_private_claim_portion = {
							value = ISR2_private_claim_amount
							multiply = ISR2_private_claim_share_ROM
							divide = ISR2_private_claim_total_share
						}
					}
				}
				subtract_from_variable = { ISR2_private_claim_remainder = ISR2_private_claim_portion }
				add_to_variable = { ISR2_private_claim_due_ROM = ISR2_private_claim_portion }
				add_to_variable = { ISR2_private_retire_due_ROM = ISR2_private_claim_share_ROM }
			}
			set_variable = { ISR2_private_claim_registered_amount = ISR2_private_claim_amount }
			ISR2_private_claim_settle = yes
		}
	}
}

ISR2_bilateral_loan_tick = {
	if = {
		limit = { ISR2_b_eastern_credit_pair_clear = yes }
		add_to_variable = { ISR2_b_eastern_credit_payment_days = 1 }
		if = {
			limit = {
				check_variable = {
					var = ISR2_b_eastern_credit_payment_days
					value = 7
					compare = greater_than_or_equals
				}
			}
			set_variable = { ISR2_bilateral_loan_principal_due = ISR2_b_eastern_credit_principal_instalment }
			set_variable = { ISR2_bilateral_loan_interest_due = ISR2_b_eastern_credit_interest_instalment }
			if = {
				limit = {
					check_variable = { ISR2_b_eastern_credit_weeks_remaining < 2 }
				}
				set_variable = { ISR2_bilateral_loan_principal_due = ISR2_bilateral_loan_outstanding }
				set_variable = { ISR2_bilateral_loan_interest_due = ISR2_b_eastern_credit_interest_remaining }
			}
			set_variable = { ISR2_bilateral_loan_bill = ISR2_bilateral_loan_principal_due }
			add_to_variable = { ISR2_bilateral_loan_bill = ISR2_bilateral_loan_interest_due }
			if = {
				limit = {
					check_variable = { var = treasury value = ISR2_bilateral_loan_bill compare = greater_than_or_equals }
				}
				set_temp_variable = {
					treasury_change = { value = ISR2_bilateral_loan_bill multiply = -1 }
				}
				modify_treasury_effect = yes
				SOV = {
					set_temp_variable = { treasury_change = ISR.ISR2_bilateral_loan_bill }
					modify_treasury_effect = yes
				}
				set_variable = { ISR2_bilateral_principal_reduction = ISR2_bilateral_loan_principal_due }
				ISR2_bilateral_apply_principal_reduction = yes
				subtract_from_variable = { ISR2_bilateral_loan_outstanding = ISR2_bilateral_loan_principal_due }
				subtract_from_variable = {
					ISR2_b_eastern_credit_interest_remaining = ISR2_bilateral_loan_interest_due
				}
				subtract_from_variable = { ISR2_b_eastern_credit_weeks_remaining = 1 }
				set_variable = { ISR2_b_eastern_credit_payment_days = 0 }
				if = {
					limit = {
						check_variable = { ISR2_b_eastern_credit_weeks_remaining < 1 }
					}
					set_variable = { ISR2_bilateral_loan_outstanding = 0 }
					set_variable = { ISR2_b_eastern_credit_interest_remaining = 0 }
					ISR2_b_eastern_credit_end = yes
				}
			}
		}
	}
}
ISR2_bilateral_loan_early_repay = {
	set_variable = { ISR2_bilateral_early_paid = 0 }
	set_variable = { ISR2_bilateral_early_due = ISR2_bilateral_early_repayment }
	if = {
		limit = {
			check_variable = { ISR2_bilateral_early_due > ISR2_bilateral_loan_outstanding }
		}
		set_variable = { ISR2_bilateral_early_due = ISR2_bilateral_loan_outstanding }
	}
	if = {
		limit = {
			ISR2_b_eastern_credit_pair_clear = yes
			check_variable = { ISR2_bilateral_early_due > 0 }
			check_variable = { var = treasury value = ISR2_bilateral_early_due compare = greater_than_or_equals }
		}
		set_variable = { ISR2_bilateral_loan_old_principal = ISR2_bilateral_loan_outstanding }
		set_temp_variable = {
			treasury_change = { value = ISR2_bilateral_early_due multiply = -1 }
		}
		modify_treasury_effect = yes
		SOV = {
			set_temp_variable = { treasury_change = ISR.ISR2_bilateral_early_due }
			modify_treasury_effect = yes
		}
		set_variable = { ISR2_bilateral_principal_reduction = ISR2_bilateral_early_due }
		ISR2_bilateral_apply_principal_reduction = yes
		subtract_from_variable = { ISR2_bilateral_loan_outstanding = ISR2_bilateral_early_due }
		set_variable = { ISR2_bilateral_early_paid = ISR2_bilateral_early_due }
		set_variable = {
			ISR2_b_eastern_credit_interest_remaining = {
				value = ISR2_b_eastern_credit_interest_remaining
				multiply = ISR2_bilateral_loan_outstanding
				divide = ISR2_bilateral_loan_old_principal
			}
		}
		if = {
			limit = {
				check_variable = { ISR2_bilateral_loan_outstanding < 0.00001 }
			}
			set_variable = { ISR2_bilateral_loan_outstanding = 0 }
			set_variable = { ISR2_b_eastern_credit_interest_remaining = 0 }
			ISR2_b_eastern_credit_end = yes
		}
		else = {
			set_variable = {
				ISR2_b_eastern_credit_principal_instalment = {
					value = ISR2_bilateral_loan_outstanding
					divide = ISR2_b_eastern_credit_weeks_remaining
				}
			}
			set_variable = {
				ISR2_b_eastern_credit_interest_instalment = {
					value = ISR2_b_eastern_credit_interest_remaining
					divide = ISR2_b_eastern_credit_weeks_remaining
				}
			}
		}
	}
}
