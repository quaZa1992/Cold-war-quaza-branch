# IMPERIA COLD WAR — Sami: Sweden 1970 and Nordic cooperation.
# IMPERIA COLD WAR — Kockums restructuring.
COLDWAR_SWE_kockums_begin_effect = {
	if = {
		limit = {
			tag = SWE
			original_tag = SWE
			exists = yes
			IMP_is_1970_scenario = yes
			has_country_flag = COLDWAR_SWE_systems_initialised
			NOT = {
				has_country_flag = COLDWAR_SWE_kockums_drafted
			}
			NOT = {
				has_country_flag = COLDWAR_SWE_delivered_industrial_lifeline
			}
		}
		COLDWAR_SWE_company_profile_refresh_effect = yes
		if = {
			limit = {
				COLDWAR_SWE_kockums_context_available = yes
			}
			set_country_flag = {
				flag = COLDWAR_SWE_kockums_drafted
				value = 1
			}
			set_country_flag = {
				flag = COLDWAR_SWE_kockums_quoted
				value = 1
			}
			set_country_flag = {
				flag = COLDWAR_SWE_delivered_industrial_lifeline
				value = 1
			}
			set_country_flag = {
				flag = COLDWAR_SWE_current_action_industrial_lifeline
				value = 1
			}
			set_country_flag = {
				flag = COLDWAR_SWE_policy_earned_industrial_lifeline
				value = 1
			}
			clr_country_flag = COLDWAR_SWE_funding_pending_industrial_lifeline
			set_variable = {
				COLDWAR_SWE_kockums_cost = {
					value = gdp_total
					multiply = 0.015
				}
			}
			set_variable = {
				COLDWAR_SWE_kockums_cost_million = {
					value = COLDWAR_SWE_kockums_cost
					multiply = 1000
				}
			}
			COLDWAR_SWE_kockums_review_effect = yes
		}
		else = {
			set_country_flag = {
				flag = COLDWAR_SWE_funding_pending_industrial_lifeline
				value = 1
			}
		}
	}
}
COLDWAR_SWE_kockums_review_effect = {
	if = {
		limit = {
			tag = SWE
			original_tag = SWE
			exists = yes
			IMP_is_1970_scenario = yes
			has_country_flag = COLDWAR_SWE_kockums_drafted
			NOT = {
				has_country_flag = COLDWAR_SWE_kockums_settled
			}
			NOT = {
				has_country_flag = COLDWAR_SWE_kockums_window_open
			}
			has_country_flag = COLDWAR_SWE_kockums_quoted
		}
		set_country_flag = {
			flag = COLDWAR_SWE_kockums_window_open
			value = 1
		}
		if = {
			limit = {
				date > 1974.12.31
			}
			country_event = {
				id = COLDWAR_SWE_KOCKUMS.2
			}
		}
		else = {
			country_event = {
				id = COLDWAR_SWE_KOCKUMS.1
			}
		}
	}
}

				days = 1095
			}
		}
		add_timed_idea = {
			idea = COLDWAR_SWE_kockums_naval_training
			days = 365
		}
	}
	else = {
		hidden_effect = {
			if = {
				limit = {
					tag = SWE

					}
				}

			}
		}
	}
}
