COLDWAR_BRA_offer_preview_us_training = {
	custom_effect_tooltip = COLDWAR_BRA_if_accept_tt
	BRA = {
		set_temp_variable = { treasury_change = gdp_total }
		multiply_temp_variable = { treasury_change = -0.0025 }
		modify_treasury_effect = yes
		set_temp_variable = { COLDWAR_BRA_quote_us_training_brazil_1 = gdp_total }
		multiply_temp_variable = { COLDWAR_BRA_quote_us_training_brazil_1 = 0.0006 }
		divide_temp_variable = { COLDWAR_BRA_quote_us_training_brazil_1 = 52 }
		set_temp_variable = { COLDWAR_BRA_quote_us_training_brazil_1_million = COLDWAR_BRA_quote_us_training_brazil_1 }
		multiply_temp_variable = { COLDWAR_BRA_quote_us_training_brazil_1_million = 1000 }
		custom_effect_tooltip = COLDWAR_BRA_quote_us_training_brazil_1_tt
		army_experience = 30
		add_ideas = COLDWAR_BRA_us_training_BRA
	}
	USA = {
		set_temp_variable = { treasury_change = gdp_total }
		multiply_temp_variable = { treasury_change = -0.0004 }
		modify_treasury_effect = yes
		set_temp_variable = { COLDWAR_BRA_quote_us_training_recipient_1 = gdp_total }
		multiply_temp_variable = { COLDWAR_BRA_quote_us_training_recipient_1 = 0.00015 }
		divide_temp_variable = { COLDWAR_BRA_quote_us_training_recipient_1 = 52 }
		set_temp_variable = { COLDWAR_BRA_quote_us_training_recipient_1_million = COLDWAR_BRA_quote_us_training_recipient_1 }
		multiply_temp_variable = { COLDWAR_BRA_quote_us_training_recipient_1_million = 1000 }
		custom_effect_tooltip = COLDWAR_BRA_quote_us_training_recipient_1_tt
		army_experience = 15
		add_ideas = COLDWAR_BRA_us_training_USA
	}
}

COLDWAR_BRA_propose_us_training = {
	if = {
		limit = {
			COLDWAR_BRA_can_propose_us_training = yes
			NOT = {
				has_country_flag = COLDWAR_BRA_pending_us_training
			}
		}
		set_country_flag = COLDWAR_BRA_pending_us_training
		USA = {
			set_country_flag = COLDWAR_BRA_pending_us_training
			country_event = { id = COLDWAR_BRA_AGREEMENTS.1 days = 1 }
		}
	}
}

COLDWAR_BRA_accept_initiator_us_training = {
	if = {
		limit = {
			tag = BRA
			has_country_flag = COLDWAR_BRA_pending_us_training
			NOT = {
				has_country_flag = COLDWAR_BRA_signed_us_training
			}
			USA = {
				has_country_flag = COLDWAR_BRA_signed_us_training
			}
		}
		if = {
			limit = {
				NOT = {
					has_country_flag = COLDWAR_BRA_paid_us_training
				}
			}
			set_temp_variable = { treasury_change = gdp_total }
			multiply_temp_variable = { treasury_change = -0.0025 }
			modify_treasury_effect = yes
			army_experience = 30
			set_country_flag = COLDWAR_BRA_paid_us_training
		}
		set_country_flag = COLDWAR_BRA_signed_us_training
		clr_country_flag = COLDWAR_BRA_pending_us_training
		add_ideas = COLDWAR_BRA_us_training_BRA
	}
}

COLDWAR_BRA_accept_us_training = {
	if = {
		limit = {
			tag = USA
			has_country_flag = COLDWAR_BRA_pending_us_training
			BRA = { COLDWAR_BRA_can_accept_us_training = yes }
		}
		if = {
			limit = {
				NOT = {
					has_country_flag = COLDWAR_BRA_paid_us_training
				}
			}
			set_temp_variable = { treasury_change = gdp_total }
			multiply_temp_variable = { treasury_change = -0.0004 }
			modify_treasury_effect = yes
			army_experience = 15
			set_country_flag = COLDWAR_BRA_paid_us_training
		}
		set_country_flag = COLDWAR_BRA_signed_us_training
		clr_country_flag = COLDWAR_BRA_pending_us_training
		add_ideas = COLDWAR_BRA_us_training_USA
		BRA = {
			country_event = { id = COLDWAR_BRA_AGREEMENTS.101 }
		}
	}
}

COLDWAR_BRA_decline_us_training = {
	if = {
		limit = {
			tag = USA
			has_country_flag = COLDWAR_BRA_pending_us_training
		}
		BRA = {
			clr_country_flag = COLDWAR_BRA_pending_us_training
			set_country_flag = { flag = COLDWAR_BRA_proposal_cooldown_us_training days = 90 }
		}
		clr_country_flag = COLDWAR_BRA_pending_us_training
	}
}

COLDWAR_BRA_terminate_us_training = {
	if = {
		limit = {
			COLDWAR_BRA_campaign = yes
		}
		set_country_flag = { flag = COLDWAR_BRA_terminated_us_training days = 365 value = 1 }
		clr_country_flag = COLDWAR_BRA_signed_us_training
		clr_country_flag = COLDWAR_BRA_pending_us_training
		if = {
			limit = {
				has_idea = COLDWAR_BRA_us_training_BRA
			}
			remove_ideas = COLDWAR_BRA_us_training_BRA
		}
		USA = {
			country_event = { id = COLDWAR_BRA_AGREEMENTS.201 }
		}
	}
}
