IRQ_1985_check_kuwait_annexation = {
	if = {
		limit = {
			original_tag = IRQ
			NOT = { has_country_flag = IRQ_1985_kuwait_decisions_unlocked }
			OR = {
				AND = {
					country_exists = KUW
					has_war_with = KUW
				}
				has_completed_focus = IRQ_1985_the_nineteenth_province
				has_completed_focus = IRQ_1985_take_kuwait
				172 = { is_controlled_by = IRQ }
			}
		}
		set_country_flag = IRQ_1985_kuwait_decisions_unlocked
	}
	if = {
		limit = {
			original_tag = IRQ
			has_country_flag = IRQ_1985_kuwait_decisions_unlocked
			172 = { is_controlled_by = IRQ }
			NOT = { has_country_flag = IRQ_1985_kuwait_control_confirmed }
		}
		set_country_flag = IRQ_1985_kuwait_control_confirmed
	}
	if = {
		limit = {
			original_tag = IRQ
			has_country_flag = IRQ_1985_kuwait_decisions_unlocked
			NOT = { has_country_flag = IRQ_1985_kuwait_annexed_confirmed }
			172 = { is_owned_and_controlled_by = IRQ }
		}
		set_country_flag = IRQ_1985_kuwait_annexed_confirmed
	}
