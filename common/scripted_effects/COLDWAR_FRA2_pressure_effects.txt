# IMPERIA COLD WAR — France 1970 procurement pressure.
COLDWAR_FRA2_pressure_start_NGR = {
	if = {
		limit = {
			COLDWAR_FRA2_pressure_can_send_NGR = yes
		}
		set_country_flag = COLDWAR_FRA2_pressure_pending_NGR
		NGR = { set_country_flag = COLDWAR_FRA2_pressure_pending_NGR }
		set_country_flag = { flag = COLDWAR_FRA2_pressure_cooldown_NGR value = 1 days = 365 }
		NGR = { country_event = { id = COLDWAR_FRA2_pressure.1 days = 30 } }
	}
}

COLDWAR_FRA2_pressure_accept_NGR = {
	if = {
		limit = {
			COLDWAR_FRA2_pressure_can_accept_NGR = yes
		}
		set_country_flag = COLDWAR_FRA2_pressure_charter_signed_NGR
		FRA = { set_country_flag = COLDWAR_FRA2_pressure_charter_signed_NGR clr_country_flag = COLDWAR_FRA2_pressure_pending_NGR set_temp_variable = { treasury_change = { value = gdp_total multiply = -0.0004 } } modify_treasury_effect = yes update_money_dirty_variable = yes }
		clr_country_flag = COLDWAR_FRA2_pressure_pending_NGR
		set_temp_variable = { treasury_change = { value = gdp_total multiply = -0.001 } }
		modify_treasury_effect = yes
		update_money_dirty_variable = yes
		FRA = { COLDWAR_FRA2_pressure_sync_all = yes }
	}
}

COLDWAR_FRA2_pressure_refuse_NGR = {
	if = {
		limit = {
			COLDWAR_FRA2_pressure_can_resolve_NGR = yes
		}
		COLDWAR_FRA2_pressure_register_refusal_NGR = yes
	}
}

COLDWAR_FRA2_pressure_cancel_NGR = {
	if = {
		limit = {
			has_country_flag = COLDWAR_FRA2_pressure_pending_NGR
		}
		clr_country_flag = COLDWAR_FRA2_pressure_pending_NGR
	}
	FRA = { clr_country_flag = COLDWAR_FRA2_pressure_pending_NGR update_money_dirty_variable = yes }
	update_money_dirty_variable = yes
}

COLDWAR_FRA2_pressure_register_refusal_NGR = {
	if = {
		limit = {
			COLDWAR_FRA2_pressure_can_resolve_NGR = yes
		}
		clr_country_flag = COLDWAR_FRA2_pressure_pending_NGR
		FRA = { clr_country_flag = COLDWAR_FRA2_pressure_pending_NGR }
		if = {
			limit = {
				FRA = { COLDWAR_FRA2_pressure_refusal_qualifies_NGR = yes }
			}
			FRA = { set_country_flag = { flag = COLDWAR_FRA2_pressure_aid_freeze_NGR value = 1 days = 180 } set_country_flag = COLDWAR_FRA2_pressure_aid_freeze_refresh_NGR update_money_dirty_variable = yes }
			add_timed_idea = { idea = COLDWAR_FRA2_pressure_withdrawal_crisis days = 90 }
			update_money_dirty_variable = yes
		}
		FRA = { update_money_dirty_variable = yes }
		update_money_dirty_variable = yes
	}
}
