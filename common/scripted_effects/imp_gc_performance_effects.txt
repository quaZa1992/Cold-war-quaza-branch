# IMPERIA CW — Cold War: Imperia gc performance effects — scripted effects.
# Manual game-controller maintenance. Nothing here runs on a tick or for the AI.

imp_gc_perf_clear_forts = {
	# Province buildings require set_building_level with an explicit province selector.
	every_state = {
		set_building_level = {
			type = bunker
			level = 0
			province = { all_provinces = yes limit_to_border = no level > 0 }
		}
		set_building_level = {
			type = coastal_bunker
			level = 0
			province = { all_provinces = yes limit_to_border = no level > 0 }
		}
	}
}

imp_gc_perf_trim_ai_economy = {
	# A one-off cut to levels, not a persistent modifier or monthly world scan.
	# Only fully AI-controlled owned states are touched; occupied player states are skipped.
	every_country = {
		limit = { exists = yes is_ai = yes }
		every_owned_state = {
			limit = { is_fully_controlled_by = PREV }
			set_temp_variable = { imp_gc_t_trim = building_level@industrial_complex }
			multiply_temp_variable = { imp_gc_t_trim = 0.25 }
			round_temp_variable = imp_gc_t_trim
			if = { limit = { check_variable = { imp_gc_t_trim > 0 } }
				remove_building = { type = industrial_complex level = imp_gc_t_trim }
			}
			set_temp_variable = { imp_gc_t_trim = building_level@arms_factory }
			multiply_temp_variable = { imp_gc_t_trim = 0.25 }
			round_temp_variable = imp_gc_t_trim
			if = { limit = { check_variable = { imp_gc_t_trim > 0 } }
				remove_building = { type = arms_factory level = imp_gc_t_trim }
			}
			set_temp_variable = { imp_gc_t_trim = building_level@offices }
			multiply_temp_variable = { imp_gc_t_trim = 0.25 }
			round_temp_variable = imp_gc_t_trim
			if = { limit = { check_variable = { imp_gc_t_trim > 0 } }
				remove_building = { type = offices level = imp_gc_t_trim }
			}
			set_temp_variable = { imp_gc_t_trim = building_level@agriculture_district }
			multiply_temp_variable = { imp_gc_t_trim = 0.25 }
			round_temp_variable = imp_gc_t_trim
			if = { limit = { check_variable = { imp_gc_t_trim > 0 } }
				remove_building = { type = agriculture_district level = imp_gc_t_trim }
			}
			set_temp_variable = { imp_gc_t_trim = building_level@dockyard }
			multiply_temp_variable = { imp_gc_t_trim = 0.25 }
			round_temp_variable = imp_gc_t_trim
			if = { limit = { check_variable = { imp_gc_t_trim > 0 } }
				remove_building = { type = dockyard level = imp_gc_t_trim }
			}
		}
	}
}

imp_gc_perf_clear_ai_divisions = {
	every_country = {
		limit = { exists = yes is_ai = yes }
		# No template/state filter: all land divisions. No equipment refund.
		delete_unit = { disband = no }
	}
}

imp_gc_perf_halve_ai_divisions = {
	every_country = {
		limit = { exists = yes is_ai = yes }
		# Independent roll per division: approximately half, with no refund.
		every_country_division = {
			random = {
				chance = 50
				destroy_unit = yes
			}
		}
	}
}

imp_gc_perf_clear_ai_equipment = {
	every_country = {
		limit = { exists = yes is_ai = yes }
		set_equipment_fraction = 0
	}
}

imp_gc_perf_halve_ai_equipment = {
	every_country = {
		limit = { exists = yes is_ai = yes }
		set_equipment_fraction = 0.5
	}
}

imp_gc_perf_trim_player_equipment = {
	every_country = {
		limit = { exists = yes is_ai = no }
		set_equipment_fraction = 0.75
	}
}
