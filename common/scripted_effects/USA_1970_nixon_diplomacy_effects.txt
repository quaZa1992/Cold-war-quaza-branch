###############################################################
# Nixon-era bilateral diplomacy, 1970-1972
#
# These dispatchers never grant material rewards.  They only
# validate the partner, mark an offer as pending, and deliver the
# corresponding event.  All benefits and policy costs are applied
# by the recipient's accept option in
# events/USA_1970_nixon_diplomacy_events.txt.
###############################################################

USA_1970_nixon_send_china_backchannel_offer = {
	custom_effect_tooltip = USA_1970_nixon_send_china_backchannel_offer_tt
	hidden_effect = {
		if = {
			limit = {
				original_tag = USA
				IMP_is_1970_scenario = yes
				country_exists = CHI
				CHI = { original_tag = CHI }
				NOT = { has_war_with = CHI }
				NOT = { has_country_flag = USA_1970_nixon_china_backchannel_offered }
				NOT = { has_country_flag = USA_1970_nixon_china_backchannel_accepted }
				NOT = { has_country_flag = USA_1970_nixon_china_backchannel_rejected }
			}
			set_country_flag = USA_1970_nixon_china_backchannel_offered
			CHI = { country_event = { id = USA_1970_nixon_dip.1 days = 1 } }
		}
		else_if = {
			limit = {
				original_tag = USA
				IMP_is_1970_scenario = yes
				NOT = { has_country_flag = USA_1970_nixon_china_backchannel_accepted }
				NOT = { has_country_flag = USA_1970_nixon_china_backchannel_rejected }
			}
			set_country_flag = USA_1970_nixon_china_backchannel_unavailable
		}
	}
}

		if = {
			limit = {
				original_tag = USA
				IMP_is_1970_scenario = yes

			if = {
				limit = {
					country_exists = VIE

						original_tag = VIE

					}
					OR = {
						has_war_with = VIE

					}
