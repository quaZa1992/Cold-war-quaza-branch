ENG_royal_family_monthly_update = {
	if = {
		limit = { has_country_flag = ENG_monarchy_abolished }
		ENG_royal_family_cleanup = yes
	}
	if = {
		limit = {
			NOT = { has_country_flag = ENG_monarchy_abolished }
			has_country_flag = ENG_royal_family_mechanic_active
		}
		ENG_royal_family_register_deaths = yes
		add_to_variable = { ENG_royal_family_press_pressure = 0.2 }
		if = {
			limit = { check_variable = { ENG_royal_family_stability_score > 68 } }
			add_to_variable = { ENG_royal_family_public_appeal = 0.1 }
		}
		if = {
			limit = { check_variable = { ENG_royal_family_press_pressure > 72 } }
			add_to_variable = { ENG_royal_family_stability_score = -0.4 }
		}
		if = {
			limit = {
				check_variable = { ENG_royal_family_press_pressure > 78 }
				NOT = { has_country_flag = ENG_royal_recent_tabloid_storm }
			}
			set_country_flag = { flag = ENG_royal_recent_tabloid_storm days = 180 }
			country_event = { id = eng_royal_family.1 days = 1 random_hours = 48 }
		}
		clamp_variable = { var = ENG_royal_family_stability_score min = 0 max = 100 }
		clamp_variable = { var = ENG_royal_family_public_appeal min = 0 max = 100 }
		clamp_variable = { var = ENG_royal_family_press_pressure min = 0 max = 100 }
		clamp_variable = { var = ENG_royal_family_commonwealth_reach min = 0 max = 100 }
		ENG_update_royal_family_modifier = yes
	}
}
