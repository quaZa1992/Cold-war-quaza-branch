icwp_init_globals = {
	if = {
		limit = { NOT = { has_global_flag = icwp_v2_globals } }
		set_global_flag = icwp_v2_globals
		# Share of GDP per factory and week that sets the domestic cost of one unit (calibration value).
		set_variable = { global.icwp_k_value = 0.0067 }
		set_variable = { global.icwp_world_u = 0 }
		set_variable = { global.icwp_holder = 0 }
		set_variable = { global.icwp_cand = 0 }
		set_variable = { global.icwp_cand_m = 0 }
		set_variable = { global.icwp_next_month = 0 }
		set_variable = { global.icwp_rev = 0 }
	}
}

# Country scope. Idempotent. Called when a human player opens the window, and monthly for human players.
icwp_register = {
	icwp_init_globals = yes
	if = {
		limit = { NOT = { has_country_flag = icwp_participant } }
		set_country_flag = icwp_participant
		add_to_array = { array = global.icwp_participants value = THIS }
		set_variable = { icwp_xp = 0 }
		set_variable = { icwp_tier = 0 }
		set_variable = { icwp_offer_u = 2 }
		set_variable = { icwp_offer_pm = 0 }
		set_variable = { icwp_tab = 1 }
		set_variable = { icwp_sel = 0 }
		set_variable = { icwp_cg_applied = 0 }
		set_variable = { icwp_af_u = 0 }
		set_variable = { icwp_af_s = 0 }
		set_variable = { icwp_af_p12 = 0 }
		set_variable = { icwp_af_p24 = 0 }
		icwp_tab_frames = yes
		icwp_refresh = yes
		icwp_host_tier = yes
		icwp_refresh = yes
	}
}

# Constant time. Called from MD's weekly update_state_variables and after every refresh.
icwp_update_cg = {
	set_variable = { icwp_base = industrial_complex_total }
	add_to_variable = { icwp_base = military_factory_total }
	clamp_variable = { var = icwp_base min = 1 }
	set_variable = { icwp_civs = industrial_complex_total }
	clamp_variable = { var = icwp_civs min = 1 }
	set_variable = { icwp_exp_cg = modifier@consumer_goods_expected_value }
	clamp_variable = { var = icwp_exp_cg min = 0.01 }
	# Own consumer goods factories without this system.
	set_temp_variable = { icwp_t_f = modifier@consumer_goods_factor }
	subtract_from_temp_variable = { icwp_t_f = icwp_cg_applied }
	add_to_temp_variable = { icwp_t_f = 1 }
	clamp_temp_variable = { var = icwp_t_f min = 0 }
	set_variable = { icwp_own_cg_f = icwp_t_f }
	multiply_variable = { icwp_own_cg_f = icwp_exp_cg }
	multiply_variable = { icwp_own_cg_f = icwp_base }
	# Factories per 1.0 of consumer_goods_factor.
	set_temp_variable = { icwp_t_k = icwp_exp_cg }
	multiply_temp_variable = { icwp_t_k = icwp_base }
	clamp_temp_variable = { var = icwp_t_k min = 0.01 }
	# Client: deliveries replace at most 40 % of the own consumer goods factories.
	set_variable = { icwp_relief_f = icwp_out_d }
	set_temp_variable = { icwp_t_cap = icwp_own_cg_f }
	multiply_temp_variable = { icwp_t_cap = 0.4 }
	clamp_variable = { var = icwp_relief_f min = 0 max = icwp_t_cap }
	set_temp_variable = { icwp_t_old = icwp_cg_applied }
	set_variable = { icwp_c_cg = icwp_relief_f }
	divide_variable = { icwp_c_cg = icwp_t_k }
	multiply_variable = { icwp_c_cg = -1 }
	# Host: factories bound for foreign clients (only while the tier modifier is active).
	set_variable = { icwp_h_cg = 0 }
	if = {
		limit = { check_variable = { icwp_in_u > 0 } check_variable = { icwp_tier > 0 } }
		set_variable = { icwp_h_cg = icwp_in_d }
		divide_variable = { icwp_h_cg = icwp_t_k }
	}
	# Aftermath: supply shortage after a lost contract.
	set_variable = { icwp_a_cg = icwp_af_u }
	divide_variable = { icwp_a_cg = icwp_t_k }
	set_variable = { icwp_cg_applied = icwp_c_cg }
	add_to_variable = { icwp_cg_applied = icwp_h_cg }
	add_to_variable = { icwp_cg_applied = icwp_a_cg }
	# Dependence of the client on foreign supply.
	set_variable = { icwp_dep = icwp_relief_f }
	set_temp_variable = { icwp_t_own = icwp_own_cg_f }
	clamp_temp_variable = { var = icwp_t_own min = 1 }
	divide_variable = { icwp_dep = icwp_t_own }
	clamp_variable = { var = icwp_dep min = 0 max = 1 }
	if = {
		limit = { NOT = { check_variable = { icwp_cg_applied = icwp_t_old } } }
		force_update_dynamic_modifier = yes
	}
}

# Location profile (host scope)

icwp_host_profile = {
	set_variable = { icwp_access = globalism_value }
	divide_variable = { icwp_access = 100 }
	clamp_variable = { var = icwp_access min = 0 max = 1 }
	if = { limit = { has_country_flag = icwp_export_zone } clamp_variable = { var = icwp_access min = 0.65 max = 1 } }
	# Update 23.11.2026: Imperia policies for producers (selector, economy column).
	# Special Economic Zones: access at least 50 %, up to 15 clients, 50 % capacity.
	# Open Door Manufacturing: access at least 65 %, up to 30 clients, 60 % capacity.
	set_variable = { icwp_max_clients = 8 }
	set_variable = { icwp_cap_share = 0.40 }
	if = {
		limit = { has_idea = COLDWAR_ni_icwp_sez }
		clamp_variable = { var = icwp_access min = 0.50 max = 1 }
		set_variable = { icwp_max_clients = 15 }
		set_variable = { icwp_cap_share = 0.50 }
	}
	else_if = {
		limit = { has_idea = COLDWAR_ni_icwp_open_door }
		clamp_variable = { var = icwp_access min = 0.65 max = 1 }
		set_variable = { icwp_max_clients = 30 }
		set_variable = { icwp_cap_share = 0.60 }
	}
	# Pro-business policies make the location cheaper: x0.92 or x0.85.
	set_variable = { icwp_polm = 1 }
	if = { limit = { has_idea = COLDWAR_ni_icwp_pro_business } set_variable = { icwp_polm = 0.92 } }
	else_if = { limit = { has_idea = COLDWAR_ni_icwp_investor_labour } set_variable = { icwp_polm = 0.85 } }
	# Labour protection (MD law employment_pressure).
	set_variable = { icwp_lawm = 1 }
	if = { limit = { has_idea = minimal_regulations } set_variable = { icwp_lawm = 0.90 } }
	else_if = { limit = { has_idea = light_regulations } set_variable = { icwp_lawm = 0.95 } }
	else_if = { limit = { has_idea = strong_regulations } set_variable = { icwp_lawm = 1.08 } }
	else_if = { limit = { has_idea = comprehensive_regulations } set_variable = { icwp_lawm = 1.15 } }
	# Environmental protection (MD law mining_policies).
	set_variable = { icwp_envm = 1 }
	if = { limit = { has_idea = strip_mining } set_variable = { icwp_envm = 0.92 } }
	else_if = { limit = { has_idea = economic_motivated_mining } set_variable = { icwp_envm = 0.96 } }
	else_if = { limit = { has_idea = partial_environmental_protections } set_variable = { icwp_envm = 1.04 } }
	else_if = { limit = { has_idea = full_environmental_protections } set_variable = { icwp_envm = 1.08 } }
	# Risk premium below 50 % stability, up to +15 %.
	set_variable = { icwp_riskm = 0.5 }
	subtract_from_variable = { icwp_riskm = stability }
	clamp_variable = { var = icwp_riskm min = 0 max = 0.5 }
	multiply_variable = { icwp_riskm = 0.3 }
	add_to_variable = { icwp_riskm = 1 }
	# Clusters: every client makes the location cheaper, up to -12 %.
	set_variable = { icwp_nclients = icwp_clients^num }
	set_variable = { icwp_clusm = icwp_nclients }
	multiply_variable = { icwp_clusm = -0.02 }
	add_to_variable = { icwp_clusm = 1 }
	clamp_variable = { var = icwp_clusm min = 0.88 max = 1 }
	# Tier price.
	set_variable = { icwp_tierm = 1 }
	if = { limit = { check_variable = { icwp_tier = 2 } } set_variable = { icwp_tierm = 1.15 } }
	else_if = { limit = { check_variable = { icwp_tier = 3 } } set_variable = { icwp_tierm = 1.30 } }
	else_if = { limit = { check_variable = { icwp_tier = 4 } } set_variable = { icwp_tierm = 1.50 } }
	set_variable = { icwp_hmult = icwp_lawm }
	multiply_variable = { icwp_hmult = icwp_envm }
	multiply_variable = { icwp_hmult = icwp_riskm }
	multiply_variable = { icwp_hmult = icwp_clusm }
	multiply_variable = { icwp_hmult = icwp_tierm }
	multiply_variable = { icwp_hmult = icwp_polm }
	# Capacity: factories x 0.4 x operation x labour pool x openness.
	set_variable = { icwp_unemp = total_unemployed_percentage_display }
	clamp_variable = { var = icwp_unemp min = 0 max = 1 }
	set_temp_variable = { icwp_t_pool = icwp_unemp }
	multiply_temp_variable = { icwp_t_pool = 3 }
	clamp_temp_variable = { var = icwp_t_pool min = 0 max = 0.5 }
	add_to_temp_variable = { icwp_t_pool = 1 }
	set_temp_variable = { icwp_t_power = 1 }
	if = {
		limit = { check_variable = { energy_consumption > 0.01 } }
		set_temp_variable = { icwp_t_power = energy_consumption }
		add_to_temp_variable = { icwp_t_power = energy_balance }
		divide_temp_variable = { icwp_t_power = energy_consumption }
		clamp_temp_variable = { var = icwp_t_power min = 0 max = 1 }
	}
	set_temp_variable = { icwp_t_work = civilian_factories_manpower_fulfillment }
	clamp_temp_variable = { var = icwp_t_work min = 0 max = 1 }
	set_temp_variable = { icwp_t_open = icwp_access }
	multiply_temp_variable = { icwp_t_open = 0.5 }
	add_to_temp_variable = { icwp_t_open = 0.5 }
	set_variable = { icwp_cap = industrial_complex_total }
	multiply_variable = { icwp_cap = icwp_cap_share }
	multiply_variable = { icwp_cap = icwp_t_work }
	multiply_variable = { icwp_cap = icwp_t_power }
	multiply_variable = { icwp_cap = icwp_t_pool }
	multiply_variable = { icwp_cap = icwp_t_open }
	set_variable = { icwp_free_cap = icwp_cap }
	subtract_from_variable = { icwp_free_cap = icwp_in_u }
	clamp_variable = { var = icwp_free_cap min = 0 }
	# Client side: at most 40 % of the own consumer goods factories can be placed abroad.
	set_variable = { icwp_free_dem = icwp_own_cg_f }
	multiply_variable = { icwp_free_dem = 0.4 }
	subtract_from_variable = { icwp_free_dem = icwp_out_u }
	clamp_variable = { var = icwp_free_dem min = 0 }
}

icwp_quote = {
	# Domestic cost of one unit per week: GDP per factory x calibration value.
	set_temp_variable = { icwp_t_dom = gdp_total }
	set_temp_variable = { icwp_t_b = icwp_base }
	clamp_temp_variable = { var = icwp_t_b min = 1 }
	divide_temp_variable = { icwp_t_dom = icwp_t_b }
	multiply_temp_variable = { icwp_t_dom = global.icwp_k_value }
	# Wage index: host GDP per capita / client GDP per capita.
	set_temp_variable = { icwp_t_g = gdp_per_capita }
	clamp_temp_variable = { var = icwp_t_g min = 0.001 }
	set_temp_variable = { icwp_t_wage = PREV.gdp_per_capita }
	divide_temp_variable = { icwp_t_wage = icwp_t_g }
	clamp_temp_variable = { var = icwp_t_wage min = 0.02 max = 3 }
	set_temp_variable = { icwp_t_mult = icwp_t_wage }
	multiply_temp_variable = { icwp_t_mult = PREV.icwp_hmult }
	set_temp_variable = { icwp_t_nb = 1 }
	if = { limit = { is_neighbor_of = PREV } set_temp_variable = { icwp_t_nb = 0.95 } }
	multiply_temp_variable = { icwp_t_mult = icwp_t_nb }
	set_temp_variable = { icwp_t_list = icwp_t_dom }
	multiply_temp_variable = { icwp_t_list = icwp_t_mult }
	# Agreed price offer on top of the list price (icwp_t_pm: -0.10, 0, 0.10, 0.20; set by the caller).
	set_temp_variable = { icwp_t_pf = icwp_t_pm }
	add_to_temp_variable = { icwp_t_pf = 1 }
	multiply_temp_variable = { icwp_t_mult = icwp_t_pf }
	set_temp_variable = { icwp_t_price = icwp_t_dom }
	multiply_temp_variable = { icwp_t_price = icwp_t_mult }
	set_temp_variable = { icwp_t_save = 1 }
	subtract_from_temp_variable = { icwp_t_save = icwp_t_mult }
	clamp_temp_variable = { var = icwp_t_save min = -1 max = 1 }
	# Update 23.11.2026: the saving decides how many factories we can rent there: saving x 12, at most 11
	# (96 %: 11, 60 %: 7, 30 %: 3). Every rented factory frees exactly one factory at home (icwp_t_fm = 1;
	# the multiplier is kept in the code so a later balance change only needs this line).
	set_temp_variable = { icwp_t_fm = 1 }
	# Concentration: the fewer producers we use, the more each can take. Counted with this producer:
	# 1 producer x2 (up to 22), 2 producers x1.5 (up to 16), 3 or more x1 (up to 11).
	set_temp_variable = { icwp_t_nh = icwp_hosts^num }
	if = { limit = { NOT = { is_in_array = { array = icwp_hosts value = PREV } } } add_to_temp_variable = { icwp_t_nh = 1 } }
	set_temp_variable = { icwp_t_cf = 1 }
	if = { limit = { check_variable = { icwp_t_nh < 1.5 } } set_temp_variable = { icwp_t_cf = 2 } }
	else_if = { limit = { check_variable = { icwp_t_nh < 2.5 } } set_temp_variable = { icwp_t_cf = 1.5 } }
	set_temp_variable = { icwp_t_capmax = icwp_t_cf }
	multiply_temp_variable = { icwp_t_capmax = 11 }
	set_temp_variable = { icwp_t_cap = icwp_t_save }
	multiply_temp_variable = { icwp_t_cap = 12 }
	multiply_temp_variable = { icwp_t_cap = icwp_t_cf }
	subtract_from_temp_variable = { icwp_t_cap = 0.4999 }
	round_temp_variable = icwp_t_cap
	clamp_temp_variable = { var = icwp_t_cap min = 1 max = icwp_t_capmax }
}

icwp_refresh = {
	# Pair values for our own contracts as client.
	set_variable = { icwp_out_u = 0 }
	set_variable = { icwp_out_d = 0 }
	set_variable = { icwp_pay_wk = 0 }
	clear_array = icwp_dh_u
	clear_array = icwp_dh_d
	clear_array = icwp_dh_pay
	clear_array = icwp_dh_s
	clear_array = icwp_dh_m
	clear_array = icwp_dh_ig
	clear_array = icwp_dh_settle
	clear_array = icwp_dh_short
	clear_array = icwp_dh_bv
	clear_array = icwp_dh_bf
	clear_array = icwp_dh_lf
	for_each_scope_loop = {
		array = icwp_hosts
		PREV = { icwp_pair_values = yes }
	}
	# Our role as host: sums over the clients' pair values.
	set_variable = { icwp_in_u = 0 }
	set_variable = { icwp_in_d = 0 }
	set_variable = { icwp_inc_wk = 0 }
	clear_array = icwp_dc_u
	clear_array = icwp_dc_d
	clear_array = icwp_dc_pay
	clear_array = icwp_dc_s
	clear_array = icwp_dc_m
	clear_array = icwp_dc_ig
	clear_array = icwp_dc_settle
	clear_array = icwp_dc_gap
	clear_array = icwp_dc_tt
	clear_array = icwp_dc_bf
	clear_array = icwp_dc_lpg
	clear_array = icwp_dc_lst
	set_variable = { icwp_tt_sum = 0 }
	set_temp_variable = { icwp_t_hc = industrial_complex_total }
	clamp_temp_variable = { var = icwp_t_hc min = 1 }
	set_temp_variable = { icwp_t_hp = overall_productivity }
	clamp_temp_variable = { var = icwp_t_hp min = 1 }
	for_each_scope_loop = {
		array = icwp_clients
		# Technology transfer: delivered factories / our civilian factories x 40 % x productivity gap (client / us, 1 to 3).
		set_temp_variable = { icwp_t_gap = overall_productivity }
		divide_temp_variable = { icwp_t_gap = icwp_t_hp }
		clamp_temp_variable = { var = icwp_t_gap min = 1 max = 3 }
		set_temp_variable = { icwp_t_tt = icwp_d@PREV }
		divide_temp_variable = { icwp_t_tt = icwp_t_hc }
		multiply_temp_variable = { icwp_t_tt = 0.40 }
		multiply_temp_variable = { icwp_t_tt = icwp_t_gap }
		set_variable = { icwp_tt@PREV = icwp_t_tt }
		add_to_variable = { PREV.icwp_tt_sum = icwp_t_tt }
		add_to_array = { array = PREV.icwp_dc_gap value = icwp_t_gap }
		set_temp_variable = { icwp_t_v = icwp_t_tt }
		multiply_temp_variable = { icwp_t_v = 100 }
		add_to_array = { array = PREV.icwp_dc_tt value = icwp_t_v }
		# Green bar (frames 24 to 34): share of the 25 % transfer cap.
		set_temp_variable = { icwp_t_v = icwp_t_tt }
		divide_temp_variable = { icwp_t_v = 0.025 }
		clamp_temp_variable = { var = icwp_t_v min = 0 max = 10 }
		round_temp_variable = icwp_t_v
		add_to_temp_variable = { icwp_t_v = 24 }
		add_to_array = { array = PREV.icwp_dc_bf value = icwp_t_v }
		set_temp_variable = { icwp_t_v = icwp_t_tt }
		multiply_temp_variable = { icwp_t_v = icwp_ig@PREV }
		multiply_temp_variable = { icwp_t_v = 100 }
		add_to_array = { array = PREV.icwp_dc_lpg value = icwp_t_v }
		add_to_variable = { PREV.icwp_in_u = icwp_u@PREV }
		add_to_variable = { PREV.icwp_in_d = icwp_d@PREV }
		add_to_variable = { PREV.icwp_inc_wk = icwp_pay@PREV }
		set_temp_variable = { icwp_t_v = icwp_u@PREV }
		add_to_array = { array = PREV.icwp_dc_u value = icwp_t_v }
		set_temp_variable = { icwp_t_v = icwp_d@PREV }
		add_to_array = { array = PREV.icwp_dc_d value = icwp_t_v }
		set_temp_variable = { icwp_t_v = icwp_pay@PREV }
		add_to_array = { array = PREV.icwp_dc_pay value = icwp_t_v }
		set_temp_variable = { icwp_t_v = icwp_code@PREV }
		add_to_array = { array = PREV.icwp_dc_s value = icwp_t_v }
		set_temp_variable = { icwp_t_v = icwp_mleft@PREV }
		add_to_array = { array = PREV.icwp_dc_m value = icwp_t_v }
		set_temp_variable = { icwp_t_v = icwp_ig@PREV }
		multiply_temp_variable = { icwp_t_v = 100 }
		add_to_array = { array = PREV.icwp_dc_ig value = icwp_t_v }
		set_temp_variable = { icwp_t_v = icwp_settle@PREV }
		add_to_array = { array = PREV.icwp_dc_settle value = icwp_t_v }
	}
	set_temp_variable = { icwp_t_inu = icwp_in_u }
	clamp_temp_variable = { var = icwp_t_inu min = 1 }
	for_each_scope_loop = {
		array = icwp_clients
		set_temp_variable = { icwp_t_v = icwp_u@PREV }
		multiply_temp_variable = { icwp_t_v = icwp_r@PREV }
		divide_temp_variable = { icwp_t_v = icwp_t_inu }
		clamp_temp_variable = { var = icwp_t_v min = 0 max = 1 }
		multiply_temp_variable = { icwp_t_v = 6 }
		multiply_temp_variable = { icwp_t_v = icwp_ig@PREV }
		add_to_array = { array = PREV.icwp_dc_lst value = icwp_t_v }
	}
	icwp_update_cg = yes
	icwp_host_profile = yes
	icwp_apply_modifiers = yes
	set_variable = { icwp_tierf = icwp_tier }
	add_to_variable = { icwp_tierf = 1 }
	set_variable = { icwp_balance = icwp_inc_wk }
	subtract_from_variable = { icwp_balance = icwp_pay_wk }
	set_variable = { icwp_dep_pct = icwp_dep }
	multiply_variable = { icwp_dep_pct = 100 }
	set_variable = { icwp_wshare_pct = icwp_wshare }
	multiply_variable = { icwp_wshare_pct = 100 }
	set_variable = { icwp_relief_r = icwp_relief_f }
	round_variable = icwp_relief_r
	# Concentration factor of our current producers, for the tooltips (1: x2, 2: x1.5, 3 or more: x1).
	set_variable = { icwp_nhosts = icwp_hosts^num }
	set_variable = { icwp_cf_now = 1 }
	if = { limit = { check_variable = { icwp_nhosts < 1.5 } } set_variable = { icwp_cf_now = 2 } }
	else_if = { limit = { check_variable = { icwp_nhosts < 2.5 } } set_variable = { icwp_cf_now = 1.5 } }
	set_variable = { icwp_in_r = icwp_in_d }
	round_variable = icwp_in_r
	icwp_ui_values = yes
	icwp_touch = yes
}

# Client scope, host in PREV: delivered units, weekly payment, settlement, display row.
icwp_pair_values = {

			limit = { check_variable = { icwp_hosts^num > 0 } }
			clear_array = icwp_kill
			for_each_scope_loop = {
				array = icwp_hosts
				PREV = { icwp_pair_month = yes }
			}
			for_each_scope_loop = {
				array = icwp_kill
				PREV = {
					set_temp_variable = { icwp_t_mode = icwp_km@PREV }
					set_temp_variable = { icwp_t_payer = icwp_kp@PREV }
					clear_variable = icwp_km@PREV
					clear_variable = icwp_kp@PREV
					icwp_close_pair = yes
				}
			}
			clear_array = icwp_kill
		}
	}
	# Clients first, so every host reads this month's pair values.
	for_each_scope_loop = {
		array = global.icwp_participants
		if = { limit = { exists = yes } icwp_refresh = yes }
	}
	# Hosts: experience, aftermath decay, refresh.
	set_variable = { global.icwp_world_u = 0 }
	for_each_scope_loop = {
		array = global.icwp_participants
		if = {
			limit = { exists = yes }
			icwp_decay = yes
			icwp_refresh = yes
			if = {
				limit = { check_variable = { icwp_in_d > 0 } }
				set_temp_variable = { icwp_t_x = icwp_share }
				multiply_temp_variable = { icwp_t_x = 25 }
				clamp_temp_variable = { var = icwp_t_x min = 0.5 }
				add_to_variable = { icwp_xp = icwp_t_x }
				clamp_variable = { var = icwp_xp min = 0 max = 100 }
			}
			add_to_variable = { global.icwp_world_u = icwp_in_d }
		}
	}
	icwp_league = yes
	for_each_scope_loop = {
		array = global.icwp_participants
		if = {
			limit = { exists = yes }
			icwp_host_tier = yes
			icwp_refresh = yes
		}
	}
	add_to_variable = { global.icwp_rev = 1 }
}

# Client scope, host in PREV.
icwp_pair_month = {
	if = {
		limit = { OR = { NOT = { exists = yes } NOT = { PREV = { exists = yes } } } }
		set_variable = { icwp_km@PREV = 0 }
		set_variable = { icwp_kp@PREV = 0 }
		add_to_array = { array = icwp_kill value = PREV }
	}
	else_if = {
		limit = { NOT = { AND = { icwp_trade_clear = yes icwp_finance_clear = yes } } }
		# War or embargo: full aftermath. The side that embargoes pays the settlement (sanctions no longer break contracts).
		set_variable = { icwp_km@PREV = 1 }
		set_variable = { icwp_kp@PREV = 0 }
		if = {
			limit = { is_embargoing = PREV }
			set_variable = { icwp_kp@PREV = 1 }
		}
		else_if = {
			limit = { PREV = { is_embargoing = PREV } }
			set_variable = { icwp_kp@PREV = 2 }
		}
		add_to_array = { array = icwp_kill value = PREV }
	}
	else = {
		if = { limit = { check_variable = { icwp_cd@PREV > 0 } } subtract_from_variable = { icwp_cd@PREV = 1 } }
		if = {
			limit = { check_variable = { icwp_s@PREV = 1 } }
			if = {
				limit = { check_variable = { icwp_p@PREV > 0 } }
				subtract_from_variable = { icwp_p@PREV = 1 }
			}
			else_if = {
				limit = { check_variable = { icwp_r@PREV < 1 } }
				add_to_variable = { icwp_r@PREV = 0.1667 }
				clamp_variable = { var = icwp_r@PREV min = 0 max = 1 }
			}
		}
		else_if = {
			limit = { check_variable = { icwp_s@PREV = 2 } }
			subtract_from_variable = { icwp_n@PREV = 1 }
			if = {
				limit = { NOT = { check_variable = { icwp_n@PREV > 0 } } }
				set_variable = { icwp_km@PREV = 0.5 }
				set_variable = { icwp_kp@PREV = 0 }
				add_to_array = { array = icwp_kill value = PREV }
			}
		}
		else_if = {
			limit = { check_variable = { icwp_s@PREV = 3 } }
			subtract_from_variable = { icwp_sp@PREV = 1 }
			if = {
				limit = { NOT = { check_variable = { icwp_sp@PREV > 0 } } }
				set_variable = { icwp_s@PREV = 1 }
			}
		}
		# Integration grows over 36 months of deliveries.
		if = {
			limit = { check_variable = { icwp_d@PREV > 0 } }
			add_to_variable = { icwp_ig@PREV = 0.0278 }
			clamp_variable = { var = icwp_ig@PREV min = 0 max = 1 }
		}
		# Price floats with wages, laws, stability, clients and tier; the agreed price offer stays.
		set_temp_variable = { icwp_t_pm = icwp_pm@PREV }
		icwp_quote = yes
		set_variable = { icwp_pr@PREV = icwp_t_price }
		# The freed factories follow the saving as well.
		set_variable = { icwp_fm@PREV = icwp_t_fm }
		set_variable = { icwp_sv@PREV = icwp_t_save }
	}
}

icwp_decay = {

		if = { limit = { check_variable = { icwp_af_p12 > -0.0005 } } set_variable = { icwp_af_p12 = 0 } }

	if = {

# League table and the Factory of the World. Insertion into a five-place table.
icwp_league = {
	clear_array = global.icwp_lg
	clear_array = global.icwp_lg_s
	for_each_scope_loop = {
		array = global.icwp_participants
		set_variable = { icwp_wshare = 0 }
		if = {
			limit = { exists = yes check_variable = { global.icwp_world_u > 0 } }
			set_variable = { icwp_wshare = icwp_in_d }
			divide_variable = { icwp_wshare = global.icwp_world_u }
		}
		if = {
			limit = { exists = yes check_variable = { icwp_in_d > 0 } }
			set_temp_variable = { icwp_t_pos = 0 }
			for_each_loop = {
				array = global.icwp_lg_s
				value = icwp_t_sv
				index = icwp_t_li
				if = { limit = { NOT = { check_variable = { icwp_t_sv < icwp_wshare } } } add_to_temp_variable = { icwp_t_pos = 1 } }
			}
			if = {
				limit = { check_variable = { icwp_t_pos < 5 } }
				add_to_array = { array = global.icwp_lg value = THIS index = icwp_t_pos }
				set_temp_variable = { icwp_t_sv = icwp_wshare }
				add_to_array = { array = global.icwp_lg_s value = icwp_t_sv index = icwp_t_pos }
				if = {
					limit = { check_variable = { global.icwp_lg^num > 5 } }
					remove_from_array = { array = global.icwp_lg index = 5 }
					remove_from_array = { array = global.icwp_lg_s index = 5 }
				}
			}
		}
	}
	clear_array = global.icwp_lg_bf
	set_temp_variable = { icwp_t_top_s = 0 }
	if = { limit = { check_variable = { global.icwp_lg_s^num > 0 } } set_temp_variable = { icwp_t_top_s = global.icwp_lg_s^0 } }
	clamp_temp_variable = { var = icwp_t_top_s min = 0.001 }
	for_each_loop = {
		array = global.icwp_lg_s
		value = icwp_t_lsv
		index = icwp_t_lsi
		set_temp_variable = { icwp_t_r = icwp_t_lsv }
		divide_temp_variable = { icwp_t_r = icwp_t_top_s }
		multiply_temp_variable = { icwp_t_r = 10 }
		round_temp_variable = icwp_t_r
		add_to_temp_variable = { icwp_t_r = 2 }
		add_to_array = { array = global.icwp_lg_bf value = icwp_t_r }
	}
	clear_array = global.icwp_lg_rk
	for_each_loop = {
		array = global.icwp_lg
		value = icwp_t_lv
		index = icwp_t_li
		set_temp_variable = { icwp_t_r = icwp_t_li }
		add_to_temp_variable = { icwp_t_r = 1 }
		add_to_array = { array = global.icwp_lg_rk value = icwp_t_r }
	}
	# The holder keeps the title while it qualifies.
	if = {
		limit = { check_variable = { global.icwp_holder > 0 } }
		if = {
			limit = { NOT = { var:global.icwp_holder = { exists = yes check_variable = { icwp_wshare > 0.1999 } check_variable = { icwp_clients^num > 2 } } } }
			set_variable = { global.icwp_holder = 0 }
		}
	}
	# A challenger needs 12 months at the top.
	set_temp_variable = { icwp_t_top = 0 }
	if = {
		limit = { check_variable = { global.icwp_lg^num > 0 } }
		set_temp_variable = { icwp_t_top = global.icwp_lg^0 }
	}
	if = {
		limit = {
			check_variable = { icwp_t_top > 0 }
			NOT = { check_variable = { icwp_t_top = global.icwp_holder } }
			var:icwp_t_top = { check_variable = { icwp_wshare > 0.1999 } check_variable = { icwp_clients^num > 2 } check_variable = { icwp_in_d > 9.99 } }
		}
		if = {
			limit = { check_variable = { global.icwp_cand = icwp_t_top } }
			add_to_variable = { global.icwp_cand_m = 1 }
		}
		else = {
			set_variable = { global.icwp_cand = icwp_t_top }
			set_variable = { global.icwp_cand_m = 1 }
		}
		if = {
			limit = { check_variable = { global.icwp_cand_m > 11 } }
			set_variable = { global.icwp_holder = global.icwp_cand }
			set_variable = { global.icwp_cand = 0 }
			set_variable = { global.icwp_cand_m = 0 }
			for_each_scope_loop = {
				array = global.icwp_participants
				if = { limit = { exists = yes is_ai = no } country_event = { id = icwp.10 } }
			}
		}
	}
	else = {
		set_variable = { global.icwp_cand = 0 }
		set_variable = { global.icwp_cand_m = 0 }
	}
}

# Host scope, client in PREV. Adds one producer to the client's market, sorted by saving (available first).
# Concentration lock (client scope, candidate producer in PREV). Adding one more producer lowers the
# limit of every running contract (1 producer x2, 2 producers x1.5, 3 or more x1). If a running contract
# is bigger than its new limit, the new producer is blocked: icwp_t_cblock = 1, and icwp_cb_c, icwp_cb_u and
# icwp_cb_cap name the contract for the tooltip. At most 6 contracts are checked.
icwp_conc_check = {
	set_temp_variable = { icwp_t_cblock = 0 }
	if = {
		limit = {
			check_variable = { icwp_hosts^num > 0 }
			NOT = { is_in_array = { array = icwp_hosts value = PREV } }
		}
		set_temp_variable = { icwp_t_ccf = 1 }
		if = { limit = { check_variable = { icwp_hosts^num < 1.5 } } set_temp_variable = { icwp_t_ccf = 1.5 } }
		set_temp_variable = { icwp_t_ccmax = icwp_t_ccf }
		multiply_temp_variable = { icwp_t_ccmax = 11 }
		for_each_scope_loop = {
			array = icwp_hosts
			PREV = {
				set_temp_variable = { icwp_t_csv = 1 }
				if = { limit = { check_variable = { icwp_sv@PREV > 0 } } set_temp_variable = { icwp_t_csv = icwp_sv@PREV } }
				set_temp_variable = { icwp_t_ccap = icwp_t_csv }
				multiply_temp_variable = { icwp_t_ccap = 12 }
				multiply_temp_variable = { icwp_t_ccap = icwp_t_ccf }
				subtract_from_temp_variable = { icwp_t_ccap = 0.4999 }
				round_temp_variable = icwp_t_ccap
				clamp_temp_variable = { var = icwp_t_ccap min = 1 max = icwp_t_ccmax }
				if = {
					limit = { check_variable = { icwp_t_cblock = 0 } check_variable = { icwp_u@PREV > icwp_t_ccap } }
					set_temp_variable = { icwp_t_cblock = 1 }
					set_variable = { icwp_cb_c = PREV }
					set_variable = { icwp_cb_u = icwp_u@PREV }
					set_variable = { icwp_cb_cap = icwp_t_ccap }
				}
			}
		}
	}
}

icwp_market_entry = {
	icwp_host_profile = yes
	PREV = {
		set_temp_variable = { icwp_t_pm = 0 }
		icwp_quote = yes
		icwp_conc_check = yes
		# Reason codes (13: a running contract would exceed its limit with one more producer): 1 available, 2 pending, 3 not a player, 4 war or embargo, 5 closed, 6 no saving,
		# 7 no capacity, 8 no free slot, 9 contract full.
		set_temp_variable = { icwp_t_ok = 1 }
		if = { limit = { has_country_flag = icwp_offer_out@PREV } set_temp_variable = { icwp_t_ok = 2 } }
		else_if = { limit = { PREV = { has_country_flag = icwp_offer_open } } set_temp_variable = { icwp_t_ok = 2 } }
		else_if = { limit = { NOT = { has_global_flag = icwp_open_all } PREV = { NOT = { icwp_player_ready = yes } } } set_temp_variable = { icwp_t_ok = 3 } }
		else_if = { limit = { NOT = { AND = { icwp_trade_clear = yes icwp_finance_clear = yes } } } set_temp_variable = { icwp_t_ok = 4 } }
		else_if = { limit = { NOT = { is_in_array = { array = icwp_hosts value = PREV } } PREV = { NOT = { icwp_host_open = yes } } } set_temp_variable = { icwp_t_ok = 5 } }
		else_if = { limit = { check_variable = { icwp_t_save < 0.15 } } set_temp_variable = { icwp_t_ok = 6 } }
		else_if = { limit = { PREV = { check_variable = { icwp_free_cap < 1 } } } set_temp_variable = { icwp_t_ok = 7 } }
		else_if = { limit = { NOT = { is_in_array = { array = icwp_hosts value = PREV } } OR = { check_variable = { icwp_hosts^num > 5 } PREV = { NOT = { icwp_host_has_slot = yes } } } } set_temp_variable = { icwp_t_ok = 8 } }
		else_if = { limit = { check_variable = { icwp_t_cblock = 1 } } set_temp_variable = { icwp_t_ok = 13 } }
		else_if = { limit = { is_in_array = { array = icwp_hosts value = PREV } OR = { NOT = { check_variable = { icwp_s@PREV = 1 } } NOT = { check_variable = { icwp_u@PREV < icwp_t_cap } } } } set_temp_variable = { icwp_t_ok = 9 } }
		# Sort key: saving in percent; unavailable producers 1000 lower.
		set_temp_variable = { icwp_t_key = icwp_t_save }
		multiply_temp_variable = { icwp_t_key = 100 }
		if = { limit = { NOT = { check_variable = { icwp_t_ok = 1 } } } subtract_from_temp_variable = { icwp_t_key = 1000 } }
		set_temp_variable = { icwp_t_pos = 0 }
		for_each_loop = {
			array = icwp_mk_key
			value = icwp_t_kv
			index = icwp_t_ki
			if = { limit = { NOT = { check_variable = { icwp_t_kv < icwp_t_key } } } add_to_temp_variable = { icwp_t_pos = 1 } }
		}
		add_to_array = { array = icwp_mk value = PREV index = icwp_t_pos }
		add_to_array = { array = icwp_mk_key value = icwp_t_key index = icwp_t_pos }
		add_to_array = { array = icwp_mk_ok value = icwp_t_ok index = icwp_t_pos }
		add_to_array = { array = icwp_mk_pr value = icwp_t_price index = icwp_t_pos }
		set_temp_variable = { icwp_t_v = icwp_t_save }
		multiply_temp_variable = { icwp_t_v = 100 }
		clamp_temp_variable = { var = icwp_t_v min = 0 }
		add_to_array = { array = icwp_mk_sv value = icwp_t_v index = icwp_t_pos }
		set_temp_variable = { icwp_t_v = PREV.icwp_free_cap }
		round_temp_variable = icwp_t_v
		add_to_array = { array = icwp_mk_free value = icwp_t_v index = icwp_t_pos }
		set_temp_variable = { icwp_t_v = PREV.icwp_tier }
		add_to_temp_variable = { icwp_t_v = 1 }
		add_to_array = { array = icwp_mk_tierf value = icwp_t_v index = icwp_t_pos }
		set_temp_variable = { icwp_t_v = 1 }
		if = { limit = { PREV = { is_ai = yes } } set_temp_variable = { icwp_t_v = 2 } }
		add_to_array = { array = icwp_mk_ai value = icwp_t_v index = icwp_t_pos }
		set_temp_variable = { icwp_t_v = PREV.icwp_clients^num }
		add_to_array = { array = icwp_mk_cl value = icwp_t_v index = icwp_t_pos }
	}
}

		clamp_temp_variable = { var = icwp_t_fmd min = 0.01 }

			limit = {
				icwp_trade_clear = yes

				check_variable = { icwp_t_units > 0 }
				check_variable = { icwp_t_cblock = 0 }
				NOT = { check_variable = { icwp_t_units > icwp_t_room } }
				PREV = { NOT = { check_variable = { icwp_free_cap < icwp_t_units } } }
				NOT = { check_variable = { icwp_t_setup > treasury } }
				OR = {
					is_in_array = { array = icwp_hosts value = PREV }
					AND = {
						check_variable = { icwp_hosts^num < 6 }
						PREV = { icwp_host_has_slot = yes icwp_host_open = yes }

icwp_row_frames = {
	clear_array = icwp_mk_rowf
	for_each_loop = {
		array = icwp_mk
		value = icwp_t_rv
		index = icwp_t_i
		set_temp_variable = { icwp_t_f = 1 }
		set_temp_variable = { icwp_t_j = icwp_t_i }
		add_to_temp_variable = { icwp_t_j = 1 }
		if = { limit = { check_variable = { icwp_t_j = icwp_sel } } set_temp_variable = { icwp_t_f = 2 } }
		add_to_array = { array = icwp_mk_rowf value = icwp_t_f }
	}
}

######################################################################
# Savegame migration from v1 (once, on startup)
######################################################################

# Old contracts become running v2 contracts with the same volume (at most 10 units) and 50 % integration.
icwp_migrate_v1 = {
	if = {
		limit = { has_global_flag = icwp_initialized NOT = { has_global_flag = icwp_v2_migrated } }
		set_global_flag = icwp_v2_migrated
		every_country = {
			limit = { has_country_flag = icwp_initialized }
			clear_array = icwp_buyers
			clear_array = icwp_v1_hosts
			for_each_loop = {
				array = icwp_hosts
				value = icwp_t_h
				add_to_array = { array = icwp_v1_hosts value = icwp_t_h }
			}
			clear_array = icwp_hosts
			for_each_scope_loop = {
				array = icwp_v1_hosts
				if = {
					limit = { exists = yes PREV = { exists = yes } }
					PREV = {
						icwp_register = yes
						PREV = { icwp_register = yes clear_array = icwp_buyers }
						set_temp_variable = { icwp_t_units = icwp_units@PREV }
						clamp_temp_variable = { var = icwp_t_units min = 0 max = 10 }
						if = {
							limit = { check_variable = { icwp_t_units > 0.5 } NOT = { is_in_array = { array = icwp_hosts value = PREV } } }
							add_to_array = { array = icwp_hosts value = PREV }
							PREV = { add_to_array = { array = icwp_clients value = PREV } }
							set_variable = { icwp_u@PREV = icwp_t_units }
							set_variable = { icwp_r@PREV = 1 }
							set_variable = { icwp_p@PREV = 0 }
							set_variable = { icwp_s@PREV = 1 }
							set_variable = { icwp_n@PREV = 0 }
							set_variable = { icwp_sp@PREV = 0 }
							set_variable = { icwp_cd@PREV = 0 }
							set_variable = { icwp_ig@PREV = 0.5 }
							PREV = { icwp_host_profile = yes }
							icwp_quote = yes
							set_variable = { icwp_pr@PREV = icwp_t_price }
						}
						clear_variable = icwp_units@PREV
						clear_variable = icwp_unmatured@PREV
						clear_variable = icwp_delivered@PREV
						clear_variable = icwp_state@PREV
					}
				}
			}
			clear_array = icwp_v1_hosts
			clr_country_flag = icwp_initialized
		}
		clr_global_flag = icwp_initialized
		for_each_scope_loop = {
			array = global.icwp_participants
			icwp_refresh = yes
		}
		for_each_scope_loop = {
			array = global.icwp_participants
			icwp_host_tier = yes
			icwp_refresh = yes
		}
	}
}
