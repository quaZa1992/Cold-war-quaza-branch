COLDWAR_KOR_pol_open_dossier = {
	COLDWAR_KOR_action_open_constitutional_dossier = yes
	if = {
		limit = {
			NOT = { has_country_flag = COLDWAR_KOR_constitutional_order }
			NOT = { has_country_flag = COLDWAR_KOR_constitution_yushin }
			NOT = { has_country_flag = COLDWAR_KOR_constitution_controlled_fifth }
			NOT = { has_country_flag = COLDWAR_KOR_nationwide_martial_law }
			NOT = { nationalist_military_junta_are_in_power = yes }
		}
		set_country_flag = COLDWAR_KOR_talks_authorized
	}
}

			NOT = { has_country_flag = COLDWAR_KOR_review_native_elections_certified }

			NOT = { has_country_flag = elections_in_progress }
