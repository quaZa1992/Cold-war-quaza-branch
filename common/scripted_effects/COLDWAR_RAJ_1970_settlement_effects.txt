# Use the recorded 1971 frontier and the winner's actual cores and claims.
# Terms are settled once, after peace and before the aftermath focuses open.

COLDWAR_RAJ_1970_pak_war_resolve = {
	if = {
		limit = {
			has_country_flag = COLDWAR_RAJ_1970_pak_war_instance_active
			NOT = { has_country_flag = COLDWAR_RAJ_1970_pak_war_resolved }
			NOT = { has_war_with = PAK }
		}
		set_country_flag = COLDWAR_RAJ_1970_pak_war_resolved
		clr_country_flag = COLDWAR_RAJ_1970_pak_war_instance_active
		if = { limit = { has_idea = COLDWAR_RAJ_1970_war_crisis } remove_ideas = COLDWAR_RAJ_1970_war_crisis }
		if = {
			limit = { is_subject_of = PAK }
			set_country_flag = COLDWAR_RAJ_1970_pak_war_defeat_verified
		}
		else_if = {
			limit = { COLDWAR_RAJ_1970_pak_subject_of_raj = yes }
			set_country_flag = COLDWAR_RAJ_1970_pak_war_victory_verified
		}
		else_if = {
			limit = {
				OR = {
					PAK = { exists = yes any_owned_state = { has_state_flag = COLDWAR_RAJ_1970_pak_war_prewar_india } }
					AND = {
						has_country_flag = COLDWAR_RAJ_1970_pak_war_capitulation_defeat_observed
						NOT = { has_country_flag = COLDWAR_RAJ_1970_pak_war_capitulation_victory_observed }
					}
				}
			}
			set_country_flag = COLDWAR_RAJ_1970_pak_war_defeat_verified
		}
		else_if = {
			limit = {
				OR = {
					COLDWAR_RAJ_1970_pak_western_land_gained = yes
					COLDWAR_RAJ_1970_ban_subject_of_raj = yes
					COLDWAR_RAJ_1970_ban_annex_eligible = yes
				}
			}
			set_country_flag = COLDWAR_RAJ_1970_pak_war_victory_verified
		}
		else_if = {
			limit = {
				COLDWAR_RAJ_1970_pak_western_border_intact = yes
				COLDWAR_RAJ_1970_ban_independent_homeland = yes
			}
			set_country_flag = COLDWAR_RAJ_1970_pak_war_stalemate_verified
		}
		else_if = {
			limit = {
				OR = {
					AND = {
						has_country_flag = COLDWAR_RAJ_1970_pak_war_capitulation_victory_observed
						NOT = { has_country_flag = COLDWAR_RAJ_1970_pak_war_capitulation_defeat_observed }
					}
					AND = {
						has_country_flag = COLDWAR_RAJ_1970_pak_war_peace_winner_observed
						NOT = { country_exists = PAK }
					}
				}
			}
			set_country_flag = COLDWAR_RAJ_1970_pak_war_victory_verified
		}
		else = { set_country_flag = COLDWAR_RAJ_1970_pak_war_stalemate_verified }
		force_update_dynamic_modifier = yes
	}
}
