# IMPERIA COLD WAR — Syrian institutional legacies and national programme updates.
# The January 1970 profile retains the original Syrian dynamic spirit identifiers.
# Sources: Library of Congress, Syria: A Country Study (1987), Politics and Government;
# World Bank, Syria: Balikh Irrigation Project (1974); Human Rights Watch, Syria: The Silenced Kurds (1996).
COLDWAR_SYR_RP_refresh_legacy = {
	if = {
		limit = { original_tag = SYR }

		set_variable = { COLDWAR_SYR_divided_command_political_power_factor = -0.1 }
		set_variable = { COLDWAR_SYR_divided_command_army_org_factor = -0.05 }
		if = {
			limit = { has_country_flag = COLDWAR_SYR_RP_atassi_order }
			set_variable = { COLDWAR_SYR_divided_command_political_power_factor = -0.03 }
			set_variable = { COLDWAR_SYR_divided_command_army_org_factor = -0.02 }
		}
		if = {
			limit = { has_country_flag = COLDWAR_SYR_RP_assad_order }
			set_variable = { COLDWAR_SYR_divided_command_political_power_factor = -0.02 }
			set_variable = { COLDWAR_SYR_divided_command_army_org_factor = -0.03 }
		}
		if = {
			limit = { has_country_flag = COLDWAR_SYR_RP_jadid_order }
			set_variable = { COLDWAR_SYR_divided_command_political_power_factor = -0.04 }
			set_variable = { COLDWAR_SYR_divided_command_army_org_factor = -0.02 }
		}
		if = {
			limit = { NOT = { has_dynamic_modifier = COLDWAR_SYR_divided_command } }
			add_dynamic_modifier = { modifier = COLDWAR_SYR_divided_command }
		}

		set_variable = { COLDWAR_SYR_emergency_state_stability_factor = 0.04 }
		set_variable = { COLDWAR_SYR_emergency_state_research_speed_factor = -0.04 }
		set_variable = { COLDWAR_SYR_emergency_state_political_power_factor = -0.05 }
		if = {
			limit = { has_country_flag = COLDWAR_SYR_RP_special_courts }
			set_variable = { COLDWAR_SYR_emergency_state_stability_factor = 0.04 }
			set_variable = { COLDWAR_SYR_emergency_state_research_speed_factor = -0.02 }
			set_variable = { COLDWAR_SYR_emergency_state_political_power_factor = -0.03 }
		}
		if = {
			limit = { has_country_flag = COLDWAR_SYR_RP_ordinary_appeals }
			set_variable = { COLDWAR_SYR_emergency_state_stability_factor = 0 }
			set_variable = { COLDWAR_SYR_emergency_state_research_speed_factor = 0 }
			set_variable = { COLDWAR_SYR_emergency_state_political_power_factor = -0.01 }
		}
		if = {
			limit = { NOT = { has_dynamic_modifier = COLDWAR_SYR_emergency_state } }
			add_dynamic_modifier = { modifier = COLDWAR_SYR_emergency_state }
		}

		set_variable = { COLDWAR_SYR_legacy_1967_army_org_factor = -0.05 }
		set_variable = { COLDWAR_SYR_legacy_1967_war_support_factor = 0 }
		if = {
			limit = { has_country_flag = COLDWAR_SYR_RP_golan_review }
			set_variable = { COLDWAR_SYR_legacy_1967_army_org_factor = -0.02 }
		}
		if = {
			limit = { COLDWAR_SYR_golan_objective_unresolved = yes }
			set_variable = { COLDWAR_SYR_legacy_1967_war_support_factor = 0.05 }
		}
		if = {
			limit = { NOT = { has_dynamic_modifier = COLDWAR_SYR_legacy_1967 } }
			add_dynamic_modifier = { modifier = COLDWAR_SYR_legacy_1967 }
		}

		set_variable = { COLDWAR_SYR_uneven_services_research_speed_factor = -0.05 }
		set_variable = { COLDWAR_SYR_uneven_services_production_speed_buildings_factor = -0.05 }
		if = {
			limit = { has_country_flag = COLDWAR_SYR_RP_teachers }
			set_variable = { COLDWAR_SYR_uneven_services_research_speed_factor = -0.03 }
		}
		if = {
			limit = { has_country_flag = COLDWAR_SYR_RP_colleges }
			set_variable = { COLDWAR_SYR_uneven_services_research_speed_factor = -0.02 }
		}
		if = {
			limit = { has_country_flag = COLDWAR_SYR_RP_budget_reform }
			set_variable = { COLDWAR_SYR_uneven_services_production_speed_buildings_factor = -0.02 }
		}
		if = {
			limit = { NOT = { has_dynamic_modifier = COLDWAR_SYR_uneven_services } }
			add_dynamic_modifier = { modifier = COLDWAR_SYR_uneven_services }
		}

		set_variable = { COLDWAR_SYR_state_sector_strain_bureaucracy_cost_multiplier_modifier = 0.08 }
		set_variable = { COLDWAR_SYR_state_sector_strain_agricolture_productivity_modifier = -0.05 }
		if = {
			limit = { has_country_flag = COLDWAR_SYR_RP_spare_parts }
			set_variable = { COLDWAR_SYR_state_sector_strain_bureaucracy_cost_multiplier_modifier = 0.05 }
		}
		if = {
			limit = { has_country_flag = COLDWAR_SYR_RP_accounts }
			set_variable = { COLDWAR_SYR_state_sector_strain_bureaucracy_cost_multiplier_modifier = 0.03 }
		}
		if = {
			limit = { has_country_flag = COLDWAR_SYR_RP_budget_reform }
			set_variable = { COLDWAR_SYR_state_sector_strain_agricolture_productivity_modifier = -0.02 }
		}
		if = {
			limit = { NOT = { has_dynamic_modifier = COLDWAR_SYR_state_sector_strain } }
			add_dynamic_modifier = { modifier = COLDWAR_SYR_state_sector_strain }
		}

		set_variable = { COLDWAR_SYR_rain_sensitive_farms_agricolture_productivity_modifier = -0.08 }
		if = {
			limit = { has_country_flag = COLDWAR_SYR_RP_drainage }
			set_variable = { COLDWAR_SYR_rain_sensitive_farms_agricolture_productivity_modifier = -0.04 }
		}
		if = {
			limit = { has_country_flag = COLDWAR_SYR_RP_seed_banks }
			set_variable = { COLDWAR_SYR_rain_sensitive_farms_agricolture_productivity_modifier = -0.03 }
		}
		if = {
			limit = { NOT = { has_dynamic_modifier = COLDWAR_SYR_rain_sensitive_farms } }
			add_dynamic_modifier = { modifier = COLDWAR_SYR_rain_sensitive_farms }
		}

		set_variable = { COLDWAR_SYR_citizenship_exclusion_stability_factor = -0.04 }
		set_variable = { COLDWAR_SYR_citizenship_exclusion_political_power_factor = -0.03 }
		if = {
			limit = { has_country_flag = COLDWAR_SYR_RP_registers }
			set_variable = { COLDWAR_SYR_citizenship_exclusion_stability_factor = -0.03 }
		}
		if = {
			limit = { has_country_flag = COLDWAR_SYR_RP_residence }
			set_variable = { COLDWAR_SYR_citizenship_exclusion_stability_factor = 0 }
		}
		if = {
			limit = { has_country_flag = COLDWAR_SYR_RP_appeals }
			set_variable = { COLDWAR_SYR_citizenship_exclusion_political_power_factor = 0 }
		}
		if = {
			limit = { has_country_flag = COLDWAR_SYR_RP_civil_papers }
			set_variable = { COLDWAR_SYR_citizenship_exclusion_stability_factor = 0 }
			set_variable = { COLDWAR_SYR_citizenship_exclusion_political_power_factor = 0 }
		}
		if = {
			limit = { has_country_flag = COLDWAR_SYR_RP_civil_papers }
			if = {
				limit = { has_dynamic_modifier = COLDWAR_SYR_citizenship_exclusion }
				remove_dynamic_modifier = { modifier = COLDWAR_SYR_citizenship_exclusion }
			}
		}
		else_if = {
			limit = { NOT = { has_dynamic_modifier = COLDWAR_SYR_citizenship_exclusion } }
			add_dynamic_modifier = { modifier = COLDWAR_SYR_citizenship_exclusion }
		}
		force_update_dynamic_modifier = yes
	}
}

# Native MD calls this hook when the ruling party changes.
COLDWAR_SYR_refresh_programmes = {
	if = {
		limit = { tag = SYR IMP_is_1970_scenario = yes }
		COLDWAR_SYR_RP_sync_politics = yes
	}
}

COLDWAR_SYR_initialize = {
	if = {
		limit = { tag = SYR IMP_is_1970_scenario = yes }
		if = {
			limit = { NOT = { has_focus_tree = mdcw_1970_generic_focus } }
			load_focus_tree = mdcw_1970_generic_focus
		}
		COLDWAR_SYR_politics_init = yes
		COLDWAR_SYR_RP_refresh_legacy = yes
		COLDWAR_SYR_RP_sync_politics = yes
		COLDWAR_SYR_RP_sync_deals = yes
	}
}

COLDWAR_SYR_weekly = {
	if = {
		limit = { tag = SYR IMP_is_1970_scenario = yes }
		if = {
			limit = { date > 1982.12.31 has_country_flag = COLDWAR_SYR_leader_khatib }
			set_variable = { COLDWAR_SYR_baath_head = 4 }
			COLDWAR_SYR_1970_set_leader = yes
		}
		if = {
			limit = {
				OR = {
					AND = { COLDWAR_SYR_golan_objective_unresolved = yes check_variable = { COLDWAR_SYR_legacy_1967_war_support_factor = 0 } }
					AND = { NOT = { COLDWAR_SYR_golan_objective_unresolved = yes } check_variable = { COLDWAR_SYR_legacy_1967_war_support_factor > 0 } }
				}
			}
			COLDWAR_SYR_RP_refresh_legacy = yes
		}
		COLDWAR_SYR_RP_sync_politics = yes
		COLDWAR_SYR_RP_sync_deals = yes
	}
}
