# COLDWAR | Scope: SOV. Puts Vladimir Putin into the KGB recruitment pool (operatives need La Resistance).
# Nationalities SOV + DDR: native Russian speaker, fluent in German (historically posted to Dresden 1985-1990).
COLDWAR_SOV_create_putin_operative = {
	if = {
		limit = { has_dlc = "La Resistance" }
		custom_effect_tooltip = COLDWAR_SOV_putin_operative_tt
		hidden_effect = {
			create_operative_leader = {
				name = "Vladimir Putin"
				GFX = GFX_portrait_COLDWAR_SOV_vladimir_putin_kgb
				traits = { operative_tough operative_spyhunter operative_linguist }
				bypass_recruitment = no
				available_to_spy_master = yes
				gender = male
				nationalities = { SOV DDR }
			}
		}
		if = {
			limit = { NOT = { has_intelligence_agency = yes } }
			custom_effect_tooltip = COLDWAR_SOV_putin_needs_agency_tt
		}
	}
	else = {
		custom_effect_tooltip = COLDWAR_SOV_putin_needs_lar_tt
	}
	set_global_flag = COLDWAR_SOV_putin_joined_the_kgb
}
