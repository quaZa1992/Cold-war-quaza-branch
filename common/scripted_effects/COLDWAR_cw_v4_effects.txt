# IMPERIA COLD WAR — Cold War blocs v4: generic bloc leaders and bloc pillars.
# IMPERIA COLD WAR — 20.11.2026: bloc founding removed (no founded blocs, no free seats); the four

# Global. Fixed bloc leaders: USA, SOV, CHI (startup and save migration).
COLDWAR_cw_v4_init_leaders = {
	if = {
		limit = {
			NOT = { has_variable = global.mdcw_cw_leader_1 }
			country_exists = USA
		}
		USA = {
			set_variable = { global.mdcw_cw_leader_1 = THIS.id }
			set_country_flag = mdcw_cw_bloc_leader
			set_country_flag = mdcw_cw_leads_1
		}
	}
	if = {
		limit = {
			NOT = { has_variable = global.mdcw_cw_leader_2 }
			country_exists = SOV
		}
		SOV = {
			set_variable = { global.mdcw_cw_leader_2 = THIS.id }
			set_country_flag = mdcw_cw_bloc_leader
			set_country_flag = mdcw_cw_leads_2
		}
	}
	if = {
		limit = {
			NOT = { has_variable = global.mdcw_cw_leader_3 }
			country_exists = CHI
		}
		CHI = {
			set_variable = { global.mdcw_cw_leader_3 = THIS.id }
			set_country_flag = mdcw_cw_bloc_leader
			set_country_flag = mdcw_cw_leads_3
		}
	}
	set_global_flag = COLDWAR_cw_v4_leaders_set
	set_global_flag = COLDWAR_cw_v4_founding_removed
	if = {
		limit = { NOT = { has_global_flag = COLDWAR_cw_v4_pillars_set } }
		COLDWAR_cw_v4_init_pillars = yes
	}
}

# Global. temp COLDWAR_cw_t_slot: the leader of that bloc is gone. Fixed blocs keep
# running without a leader until their historical leader is a member again.
COLDWAR_cw_v4_leader_lost = {
	if = {
		limit = { check_variable = { COLDWAR_cw_t_slot = 1 } }
		var:global.mdcw_cw_leader_1 = {
			clr_country_flag = mdcw_cw_leads_1
			clr_country_flag = mdcw_cw_bloc_leader
		}
		clear_variable = global.mdcw_cw_leader_1
	}
	else_if = {
		limit = { check_variable = { COLDWAR_cw_t_slot = 2 } }
		var:global.mdcw_cw_leader_2 = {
			clr_country_flag = mdcw_cw_leads_2
			clr_country_flag = mdcw_cw_bloc_leader
		}
		clear_variable = global.mdcw_cw_leader_2
	}
	else_if = {
		limit = { check_variable = { COLDWAR_cw_t_slot = 3 } }
		var:global.mdcw_cw_leader_3 = {
			clr_country_flag = mdcw_cw_leads_3
			clr_country_flag = mdcw_cw_bloc_leader
		}
		clear_variable = global.mdcw_cw_leader_3
	}
}

# Country scope, end of mdcw_cw_update_bloc_effects (temp mdcw_cw_t_scale is set there).
# Applies the active pillars of our bloc. Pillars beyond the slots our share allows rest.
COLDWAR_cw_v4_apply_pillars = {
	set_temp_variable = { COLDWAR_cw_t_bloc = mdcw_cw_alignment }
	set_temp_variable = { COLDWAR_cw_t_allowed = 2 }
	if = {
		limit = { check_variable = { mdcw_cw_alignment = 0 } }
		set_temp_variable = { COLDWAR_cw_t_allowed = global.COLDWAR_cw_pillar_slots_0 }
		set_temp_variable = { COLDWAR_cw_t_p1 = global.COLDWAR_cw_pslot_0_1 }
		set_temp_variable = { COLDWAR_cw_t_p2 = global.COLDWAR_cw_pslot_0_2 }
		set_temp_variable = { COLDWAR_cw_t_p3 = global.COLDWAR_cw_pslot_0_3 }
		set_temp_variable = { COLDWAR_cw_t_p4 = global.COLDWAR_cw_pslot_0_4 }
		set_temp_variable = { COLDWAR_cw_t_drift = 0 }
	}
	else_if = {
		limit = { check_variable = { mdcw_cw_alignment = 1 } }
		set_temp_variable = { COLDWAR_cw_t_allowed = global.COLDWAR_cw_pillar_slots_1 }
		set_temp_variable = { COLDWAR_cw_t_p1 = global.COLDWAR_cw_pslot_1_1 }
		set_temp_variable = { COLDWAR_cw_t_p2 = global.COLDWAR_cw_pslot_1_2 }
		set_temp_variable = { COLDWAR_cw_t_p3 = global.COLDWAR_cw_pslot_1_3 }
		set_temp_variable = { COLDWAR_cw_t_p4 = global.COLDWAR_cw_pslot_1_4 }
		set_temp_variable = { COLDWAR_cw_t_drift = 1 }
	}
	else_if = {
		limit = { check_variable = { mdcw_cw_alignment = 2 } }
		set_temp_variable = { COLDWAR_cw_t_allowed = global.COLDWAR_cw_pillar_slots_2 }
		set_temp_variable = { COLDWAR_cw_t_p1 = global.COLDWAR_cw_pslot_2_1 }
		set_temp_variable = { COLDWAR_cw_t_p2 = global.COLDWAR_cw_pslot_2_2 }
		set_temp_variable = { COLDWAR_cw_t_p3 = global.COLDWAR_cw_pslot_2_3 }
		set_temp_variable = { COLDWAR_cw_t_p4 = global.COLDWAR_cw_pslot_2_4 }
		set_temp_variable = { COLDWAR_cw_t_drift = 2 }
	}
	else_if = {
		limit = { check_variable = { mdcw_cw_alignment = 3 } }
		set_temp_variable = { COLDWAR_cw_t_allowed = global.COLDWAR_cw_pillar_slots_3 }
		set_temp_variable = { COLDWAR_cw_t_p1 = global.COLDWAR_cw_pslot_3_1 }
		set_temp_variable = { COLDWAR_cw_t_p2 = global.COLDWAR_cw_pslot_3_2 }
		set_temp_variable = { COLDWAR_cw_t_p3 = global.COLDWAR_cw_pslot_3_3 }
		set_temp_variable = { COLDWAR_cw_t_p4 = global.COLDWAR_cw_pslot_3_4 }
		set_temp_variable = { COLDWAR_cw_t_drift = 2 }
	}
	if = {
		limit = {
			check_variable = { COLDWAR_cw_t_allowed > 0 }
			check_variable = { COLDWAR_cw_t_p1 > 0 }
		}
		set_temp_variable = { COLDWAR_cw_t_pid = COLDWAR_cw_t_p1 }
		COLDWAR_cw_v4_apply_one_pillar = yes
	}
	if = {
		limit = {
			check_variable = { COLDWAR_cw_t_allowed > 1 }
			check_variable = { COLDWAR_cw_t_p2 > 0 }
		}
		set_temp_variable = { COLDWAR_cw_t_pid = COLDWAR_cw_t_p2 }
		COLDWAR_cw_v4_apply_one_pillar = yes
	}
	if = {
		limit = {
			check_variable = { COLDWAR_cw_t_allowed > 2 }
			check_variable = { COLDWAR_cw_t_p3 > 0 }
		}
		set_temp_variable = { COLDWAR_cw_t_pid = COLDWAR_cw_t_p3 }
		COLDWAR_cw_v4_apply_one_pillar = yes
	}
	if = {
		limit = {
			check_variable = { COLDWAR_cw_t_allowed > 3 }
			check_variable = { COLDWAR_cw_t_p4 > 0 }
		}
		set_temp_variable = { COLDWAR_cw_t_pid = COLDWAR_cw_t_p4 }
		COLDWAR_cw_v4_apply_one_pillar = yes
	}

# Receiver scope. Daily outlook drift from every foreign campaign (MD influence style), at most 0.20 a day.
COLDWAR_cw_pps_drift_update = {
	set_variable = { COLDWAR_cw_pps_d0 = 0 }
	set_variable = { COLDWAR_cw_pps_d1 = 0 }
	set_variable = { COLDWAR_cw_pps_d2 = 0 }
	set_variable = { COLDWAR_cw_pps_d3 = 0 }
	set_variable = { COLDWAR_cw_pps_d4 = 0 }
	if = {
		limit = { check_variable = { mdcw_cw_in^num > 0 } }
		for_each_scope_loop = {
			array = mdcw_cw_in
			if = {
				limit = { check_variable = { mdcw_cw_pps_party@PREV > 0 } }
				set_temp_variable = { COLDWAR_cw_t_drift = 0.04 }
				if = { limit = { check_variable = { PREV.mdcw_cw_pps_access@THIS = 1 } } set_temp_variable = { COLDWAR_cw_t_drift = 0.08 } }
				if = { limit = { check_variable = { PREV.mdcw_cw_pps_access@THIS = 2 } } set_temp_variable = { COLDWAR_cw_t_drift = 0.12 } }
				if = { limit = { check_variable = { PREV.mdcw_cw_pps_access@THIS = 3 } } set_temp_variable = { COLDWAR_cw_t_drift = 0.02 } }
				set_temp_variable = { COLDWAR_cw_t_sup = 1 }
				add_to_temp_variable = { COLDWAR_cw_t_sup = modifier@mdcw_cw_support_effectiveness }
				multiply_temp_variable = { COLDWAR_cw_t_drift = COLDWAR_cw_t_sup }
				set_variable = { COLDWAR_cw_pps_rate@PREV = COLDWAR_cw_t_drift }
				set_temp_variable = { outlook_party_index = mdcw_cw_pps_party@PREV }
				subtract_from_temp_variable = { outlook_party_index = 1 }
				set_outlook_from_party_index = yes
				PREV = {
					if = { limit = { check_variable = { outlook_id = 0 } } add_to_variable = { COLDWAR_cw_pps_d0 = COLDWAR_cw_t_drift } }
					else_if = { limit = { check_variable = { outlook_id = 1 } } add_to_variable = { COLDWAR_cw_pps_d1 = COLDWAR_cw_t_drift } }
					else_if = { limit = { check_variable = { outlook_id = 2 } } add_to_variable = { COLDWAR_cw_pps_d2 = COLDWAR_cw_t_drift } }
					else_if = { limit = { check_variable = { outlook_id = 3 } } add_to_variable = { COLDWAR_cw_pps_d3 = COLDWAR_cw_t_drift } }
					else_if = { limit = { check_variable = { outlook_id = 4 } } add_to_variable = { COLDWAR_cw_pps_d4 = COLDWAR_cw_t_drift } }
				}
			}
		}
	}
	set_temp_variable = { COLDWAR_cw_t_dsum = 0 }
	add_to_temp_variable = { COLDWAR_cw_t_dsum = COLDWAR_cw_pps_d0 }
	add_to_temp_variable = { COLDWAR_cw_t_dsum = COLDWAR_cw_pps_d1 }
	add_to_temp_variable = { COLDWAR_cw_t_dsum = COLDWAR_cw_pps_d2 }
	add_to_temp_variable = { COLDWAR_cw_t_dsum = COLDWAR_cw_pps_d3 }
	add_to_temp_variable = { COLDWAR_cw_t_dsum = COLDWAR_cw_pps_d4 }
	if = {
		limit = { check_variable = { COLDWAR_cw_t_dsum > 0.2 } }
		set_temp_variable = { COLDWAR_cw_t_dscale = 0.2 }
		divide_temp_variable = { COLDWAR_cw_t_dscale = COLDWAR_cw_t_dsum }
		multiply_variable = { COLDWAR_cw_pps_d0 = COLDWAR_cw_t_dscale }
		multiply_variable = { COLDWAR_cw_pps_d1 = COLDWAR_cw_t_dscale }
		multiply_variable = { COLDWAR_cw_pps_d2 = COLDWAR_cw_t_dscale }
		multiply_variable = { COLDWAR_cw_pps_d3 = COLDWAR_cw_t_dscale }
		multiply_variable = { COLDWAR_cw_pps_d4 = COLDWAR_cw_t_dscale }
		set_temp_variable = { COLDWAR_cw_t_dsum = 0.2 }
	}
	set_variable = { COLDWAR_cw_pps_dsum = COLDWAR_cw_t_dsum }
	if = {
		limit = { check_variable = { COLDWAR_cw_t_dsum > 0 } }
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = COLDWAR_cw_pps_drift } } }
			add_dynamic_modifier = { modifier = COLDWAR_cw_pps_drift }
		}
	}
	else_if = {
		limit = { has_dynamic_modifier = { modifier = COLDWAR_cw_pps_drift } }
		remove_dynamic_modifier = { modifier = COLDWAR_cw_pps_drift }
	}
}
