		if = {
			limit = {
				is_owned_and_controlled_by = ROOT
				check_variable = { var = building_level@infrastructure value = 5 compare = less_than_or_equals }
			}
			add_extra_state_shared_building_slots = 1
			add_building_construction = { type = infrastructure level = 1 instant_build = yes }
		}
		else = {
			ROOT = {
				every_owned_state = {
					random_select_amount = 1
					add_extra_state_shared_building_slots = 1
					add_building_construction = { type = infrastructure level = 1 instant_build = yes }
				}
			}
		}
		ROOT = {
			if = {
				limit = { NOT = { check_variable = { skip_payment = 1 } } }
				set_temp_variable = { treasury_change = -3.5 }
				modify_treasury_effect = yes
			}
		}
	}
	hidden_effect = {
		if = {
			limit = {
				OR = {
					AND = {
						is_owned_and_controlled_by = ROOT
						check_variable = { var = building_level@infrastructure value = 5 compare = less_than_or_equals }
					}
					ROOT = {
						any_owned_state = {
							is_owned_and_controlled_by = ROOT
							check_variable = { var = building_level@infrastructure value = 5 compare = less_than_or_equals }
						}
					}
				}
			}
			if = {
				limit = {
					is_owned_and_controlled_by = ROOT
					check_variable = { var = building_level@infrastructure value = 5 compare = less_than_or_equals }
				}
				add_extra_state_shared_building_slots = 1
				add_building_construction = { type = infrastructure level = 1 instant_build = yes }
			}
			else = {
				ROOT = {
					set_temp_variable = { COLDWAR_GRANT_remaining_one_state_infrastructure = 1 }
					every_owned_state = {
						limit = {
							is_owned_and_controlled_by = ROOT
							check_variable = { var = building_level@infrastructure value = 5 compare = less_than_or_equals }
							is_in_home_area = yes
						}
						random_select_amount = 1
						add_extra_state_shared_building_slots = 1
						add_building_construction = { type = infrastructure level = 1 instant_build = yes }
						set_state_flag = COLDWAR_GRANT_selected_one_state_infrastructure
						ROOT = {
							subtract_from_temp_variable = { COLDWAR_GRANT_remaining_one_state_infrastructure = 1 }
						}
					}
					if = {
						limit = { check_variable = { COLDWAR_GRANT_remaining_one_state_infrastructure = 1 } }
						every_owned_state = {
							limit = {
								is_owned_and_controlled_by = ROOT
								check_variable = { var = building_level@infrastructure value = 5 compare = less_than_or_equals }
								NOT = { has_state_flag = COLDWAR_GRANT_selected_one_state_infrastructure }
							}
							random_select_amount = 1
							add_extra_state_shared_building_slots = 1
							add_building_construction = { type = infrastructure level = 1 instant_build = yes }
							set_state_flag = COLDWAR_GRANT_selected_one_state_infrastructure
							ROOT = {
								subtract_from_temp_variable = { COLDWAR_GRANT_remaining_one_state_infrastructure = 1 }
							}
						}
					}
					every_owned_state = {
						limit = { has_state_flag = COLDWAR_GRANT_selected_one_state_infrastructure }
						clr_state_flag = COLDWAR_GRANT_selected_one_state_infrastructure
