COLDWAR_ARG_execute_treaty_brazil_industry = {
	if = {
		limit = {
			tag = BRA
			has_country_flag = COLDWAR_ARG_treaty_brazil_industry_pending
			ARG = { has_country_flag = COLDWAR_ARG_treaty_brazil_industry_pending }
			NOT = { has_country_flag = COLDWAR_ARG_treaty_brazil_industry_signed }
			ARG = {
				NOT = { has_country_flag = COLDWAR_ARG_treaty_brazil_industry_signed }
			}
			ARG = {
				country_exists = BRA
				NOT = { has_war_with = BRA }
				NOT = { is_embargoing = BRA }
				NOT = {
					OR = {
						is_in_array = { array = imp_dip_sanc_on_us value = BRA }
						is_in_array = { array = imp_dip_sanc_by_us value = BRA }
						is_in_array = { array = imp_fin_sanc_on_us value = BRA }
						is_in_array = { array = imp_fin_sanc_by_us value = BRA }
						is_in_array = { array = imp_eco_sanc_on_us value = BRA }
						is_in_array = { array = imp_eco_sanc_by_us value = BRA }
						is_in_array = { array = imp_mil_sanc_on_us value = BRA }
						is_in_array = { array = imp_mil_sanc_by_us value = BRA }
					}
				}
				BRA = {
					NOT = { has_war_with = ARG }
					NOT = { is_embargoing = ARG }
					NOT = {
						OR = {
							is_in_array = { array = imp_dip_sanc_on_us value = ARG }
							is_in_array = { array = imp_dip_sanc_by_us value = ARG }
							is_in_array = { array = imp_fin_sanc_on_us value = ARG }
							is_in_array = { array = imp_fin_sanc_by_us value = ARG }
							is_in_array = { array = imp_eco_sanc_on_us value = ARG }
							is_in_array = { array = imp_eco_sanc_by_us value = ARG }
							is_in_array = { array = imp_mil_sanc_on_us value = ARG }
							is_in_array = { array = imp_mil_sanc_by_us value = ARG }
						}
					}
				}
			}
		}
		set_temp_variable = {
			treasury_change = { value = gdp_total multiply = -0.006 }
		}
		modify_treasury_effect = yes
		add_tech_bonus = { name = COLDWAR_ARG_treaty_brazil_industry_bra category = CAT_industry bonus = 0.25 uses = 1 }
		set_country_flag = COLDWAR_ARG_treaty_brazil_industry_signed
		clr_country_flag = COLDWAR_ARG_treaty_brazil_industry_pending
		add_ideas = COLDWAR_ARG_treaty_brazil_industry_partner
		ARG = {
			country_event = { id = COLDWAR_ARG_TREATY.101 }
		}
	}
}
