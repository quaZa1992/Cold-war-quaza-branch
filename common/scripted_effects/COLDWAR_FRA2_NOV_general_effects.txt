COLDWAR_FRA2_NOV_G_clear_conference = {
	clr_country_flag = COLDWAR_FRA2_NOV_G_conference_pending
	every_country = {
		limit = {
			OR = {
				has_country_flag = COLDWAR_FRA2_NOV_G_conference_invited
				has_country_flag = COLDWAR_FRA2_NOV_G_conference_accepted
			}
		}
		clr_country_flag = COLDWAR_FRA2_NOV_G_conference_invited
		clr_country_flag = COLDWAR_FRA2_NOV_G_conference_accepted
	}
}

COLDWAR_FRA2_NOV_G_finish_conference = {
	if = {
		limit = {
			COLDWAR_FRA2_NOV_G_conference_ready = yes
		}
		every_country = {
			limit = {
				COLDWAR_FRA2_NOV_G_conference_participant = yes
				has_country_flag = COLDWAR_FRA2_NOV_G_conference_accepted
			}
			if = {
				limit = {
					has_war_with = VIE
				}
				VIE = {
					white_peace = PREV
				}
			}
			if = {
				limit = {
					has_war_with = SVN
				}
				SVN = {
					white_peace = PREV
				}
			}
		}
		if = {
			limit = {
				country_exists = VIE
			}
			VIE = {
				add_stability = 0.05
				add_war_support = -0.05
			}
		}
		if = {
			limit = {
				country_exists = SVN
			}
			SVN = {
				add_stability = 0.05
				add_war_support = -0.05
			}
		}
		set_country_flag = COLDWAR_FRA2_NOV_paris_settlement
		set_country_flag = COLDWAR_FRA2_NOV_G_conference_settled
		COLDWAR_FRA2_NOV_G_clear_conference = yes
	}
}

COLDWAR_FRA2_NOV_G_start_conference = {
	if = {
		limit = {
			COLDWAR_FRA2_NOV_G_can_start_conference = yes
		}
		add_to_variable = { COLDWAR_FRA2_NOV_G_conference_round = 1 }
		set_country_flag = COLDWAR_FRA2_NOV_G_conference_pending
		set_country_flag = { flag = COLDWAR_FRA2_NOV_G_conference_cooldown value = 1 days = 365 }
		every_country = {
			COLDWAR_FRA2_NOV_G_invite_conference = yes
		}
		COLDWAR_FRA2_NOV_G_finish_conference = yes
	}
}
