imp_set_investment_caps = 
{
	if = {
		limit = { 
			check_variable = { gdp_total > 15000 }
		}
		set_variable = { imp_max_investment_out = 8 }
		set_variable = { imp_max_investment_in = 7 }
	}
	else_if = {
		limit = { 
			check_variable = { gdp_total > 10000 }
		}
		set_variable = { imp_max_investment_out = 7 }
		set_variable = { imp_max_investment_in = 6 }
	}
	else_if = {
		limit = { 
			check_variable = { gdp_total > 4000 }
		}
		set_variable = { imp_max_investment_out = 6 }
		set_variable = { imp_max_investment_in = 5 }
	}
	else_if = {
		limit = { 
			check_variable = { gdp_total > 2000 }
		}
		set_variable = { imp_max_investment_out = 5 }
		set_variable = { imp_max_investment_in = 4 }
	}
	else_if = {
		limit = { 
			check_variable = { gdp_total > 1000 }
		}
		set_variable = { imp_max_investment_out = 4 }
		set_variable = { imp_max_investment_in = 3 }
	}
	else = {
		set_variable = { imp_max_investment_out = 3 }
		set_variable = { imp_max_investment_in = 3 }
	}

	add_to_variable = { imp_max_investment_out = modifier@imp_investment_out_extra }
	add_to_variable = { imp_max_investment_in = modifier@imp_investment_in_extra }
}

imp_calc_casualty_cost = { 
	# set the nasties. Every casualty counts for 60.
	set_variable = { imp_cas_cost = imp_total_casualties_k }
	multiply_variable = { imp_cas_cost = 60 }
	divide_variable = { imp_cas_cost = max_manpower_k }
	clamp_variable = { var = imp_cas_cost min = 0 max = 1 }
	
	set_variable = { imp_cas_cost_tax = imp_cas_cost }
	divide_variable = { imp_cas_cost_tax = 4 }

	## Add extra social costs.
	if = {
		limit = { 
			NOT = { has_dynamic_modifier = { modifier = imp_casualties_dynamic_modifier } } 
		}
		add_dynamic_modifier = { modifier = imp_casualties_dynamic_modifier }
	}
}
