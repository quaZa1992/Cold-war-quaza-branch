IMP_1985_start_iran_iraq_war = {
	if = {
		limit = {
			date < 1990.1.1
			original_tag = IRQ
			PER = { exists = yes }
			NOT = { has_global_flag = IMP_1985_iran_iraq_war_started }
		}
		if = {
			limit = { NOT = { has_war_with = PER } }
			declare_war_on = {
				target = PER
				type = annex_everything
			}
		}
		add_ideas = {
			IRQ_1985_baathist_total_war_state
			IRQ_1985_war_debt_and_militarized_economy
			IRQ_1985_war_of_attrition
		}
		IRQ_1985_setup_tree_variables = yes
		IRQ_1985_setup_kurdish_front = yes
		PER = {
			add_ideas = {
				PER_1985_sacred_defense_mobilization
				PER_1985_revolutionary_war_economy
			}
		}
		if = {
			limit = { ISR = { exists = yes } }
			ISR = { add_ideas = { ISR_1985_security_belt_war } }
		}
		set_global_flag = IMP_1985_iran_iraq_war_started
		set_country_flag = IRQ_1985_iran_iraq_war_context
	}
}

IMP_1985_cleanup_iran_iraq_war_spirits = {
	if = {
		limit = {
			date < 1990.1.1
			original_tag = IRQ

				has_idea = IRQ_1985_baathist_total_war_state
				has_idea = IRQ_1985_war_debt_and_militarized_economy
				has_idea = IRQ_1985_war_of_attrition
			}
			NOT = { has_war_with = PER }
		}
		set_global_flag = IMP_1985_iran_iraq_war_started
		set_country_flag = IRQ_1985_iran_iraq_war_context
		remove_ideas = IRQ_1985_baathist_total_war_state

	}
	if = {
		limit = {
			date < 1990.1.1
			original_tag = PER

				has_idea = PER_1985_sacred_defense_mobilization

		}
