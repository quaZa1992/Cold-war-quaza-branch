# IMPERIA COLD WAR — Selectable Peacekeeping Contingent uniforms.

COLDWAR_PK_prepare_uniform_template = {
	if = {
		limit = { check_variable = { COLDWAR_PK_uniform_choice = 1 } }
		COLDWAR_PK_create_uniform_medium = yes
	}
	else_if = {
		limit = { check_variable = { COLDWAR_PK_uniform_choice = 2 } }
		COLDWAR_PK_create_uniform_deep = yes
	}
	else_if = {
		limit = { check_variable = { COLDWAR_PK_uniform_choice = 3 } }
		COLDWAR_PK_create_uniform_medium = yes
		COLDWAR_PK_create_uniform_deep = yes
	}
	else = {
		COLDWAR_PK_create_uniform_light = yes
	}
}

COLDWAR_PK_create_uniform_light = {
	if = {
		limit = { OR = { NOT = { has_country_flag = COLDWAR_PK_uniform_prepared_light } NOT = { has_template = "Peacekeeping Contingent (Light)" } } }
		division_template = {
			name = "Peacekeeping Contingent (Light)"
			is_locked = yes
			override_model = COLDWAR_PK_blue_helmet_entity
			priority = 2
			regiments = {
				Mot_Inf_Bat = { x = 0 y = 0 }
				Mot_Inf_Bat = { x = 0 y = 1 }
				Mot_Inf_Bat = { x = 0 y = 2 }
				Mot_Inf_Bat = { x = 0 y = 3 }
			}
			support = {
				Mot_Recce_Comp = { x = 0 y = 0 }
				L_Engi_Comp = { x = 0 y = 1 }
			}
		}
		set_country_flag = COLDWAR_PK_uniform_prepared_light
	}
}

COLDWAR_PK_spawn_uniform_light = {
	create_unit = {
		division = "name = \"Peacekeeping Contingent\" division_template = \"Peacekeeping Contingent (Light)\" start_experience_factor = 0.3 start_equipment_factor = 1.0"
		owner = ROOT
	}
}
