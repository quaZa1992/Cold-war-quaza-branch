COLDWAR_EB_contracts_monthly = {
	# sums the running contracts; expired or vanished partners are collected first and removed after the loop
	set_variable = { COLDWAR_EB_contract_total = 0 }
	clear_temp_array = COLDWAR_EB_t_rm
	for_each_scope_loop = {
		array = COLDWAR_EB_ct_partners
		if = {
			limit = {
				OR = {
					NOT = { exists = yes }
					NOT = { check_variable = { PREV.COLDWAR_EB_ct_end@THIS > global.num_days } }
				}
			}
			add_to_temp_array = { COLDWAR_EB_t_rm = THIS }
		}
		else = {
			PREV = { add_to_variable = { COLDWAR_EB_contract_total = COLDWAR_EB_ct_size@PREV } }
		}
	}
	for_each_loop = {
		array = COLDWAR_EB_t_rm
		value = COLDWAR_EB_t_v
		remove_from_array = { COLDWAR_EB_ct_partners = COLDWAR_EB_t_v }
	}
	clamp_variable = { var = COLDWAR_EB_contract_total min = 0 max = 3 }
	COLDWAR_EB_contracts_display = yes # COLDWAR | contract list for the Social Contract tab (hand patch 01.11.2026)
}

COLDWAR_EB_contracts_weekly = {
	# eastern country pays its Western partners every week
	COLDWAR_EB_tension_tier = yes
	set_temp_variable = { COLDWAR_EB_t_price = 1 }
	if = { limit = { check_variable = { COLDWAR_EB_t_tier = 2 } } set_temp_variable = { COLDWAR_EB_t_price = 1.05 } }
	else_if = { limit = { check_variable = { COLDWAR_EB_t_tier = 3 } } set_temp_variable = { COLDWAR_EB_t_price = 1.1 } }
	else_if = { limit = { check_variable = { COLDWAR_EB_t_tier = 4 } } set_temp_variable = { COLDWAR_EB_t_price = 1.2 } }
	else_if = { limit = { check_variable = { COLDWAR_EB_t_tier = 5 } } set_temp_variable = { COLDWAR_EB_t_price = 1.3 } }
	for_each_scope_loop = {
		array = COLDWAR_EB_ct_partners
		if = {
			limit = { exists = yes check_variable = { PREV.COLDWAR_EB_ct_end@THIS > global.num_days } }
			set_temp_variable = { COLDWAR_EB_t_sz = PREV.COLDWAR_EB_ct_size@THIS }
			set_temp_variable = { COLDWAR_EB_t_amt = PREV.gdp_total }
			if = { limit = { check_variable = { COLDWAR_EB_t_sz = 1 } } multiply_temp_variable = { COLDWAR_EB_t_amt = 0.0002 } }
			else_if = { limit = { check_variable = { COLDWAR_EB_t_sz = 2 } } multiply_temp_variable = { COLDWAR_EB_t_amt = 0.0004 } }
			else = { multiply_temp_variable = { COLDWAR_EB_t_amt = 0.0008 } }
			multiply_temp_variable = { COLDWAR_EB_t_amt = COLDWAR_EB_t_price }
			set_temp_variable = { treasury_change = COLDWAR_EB_t_amt }
			modify_treasury_effect = yes
			PREV = {
				set_temp_variable = { treasury_change = COLDWAR_EB_t_amt }
				multiply_temp_variable = { treasury_change = -1 }
				modify_treasury_effect = yes
			}
		}
	}
}
