LBA_update_western_portfolio = {
	set_variable = { LBA_western_portfolio_partners = 0 }

	if = {
		limit = {
			country_exists = ENG
			ENG = { has_idea = LBA_libyan_capital_inflows }
		}
		add_to_variable = { LBA_western_portfolio_partners = 1 }
	}
	if = {
		limit = {
			country_exists = ITA
			ITA = { has_idea = LBA_libyan_capital_inflows }
		}
		add_to_variable = { LBA_western_portfolio_partners = 1 }
	}
	if = {
		limit = {
			country_exists = FRA
			FRA = { has_idea = LBA_libyan_capital_inflows }
		}
		add_to_variable = { LBA_western_portfolio_partners = 1 }
	}
	if = {
		limit = {
			country_exists = GER
			GER = { has_idea = LBA_libyan_capital_inflows }
		}
		add_to_variable = { LBA_western_portfolio_partners = 1 }
	}
	if = {
		limit = {
			country_exists = SWI
			SWI = { has_idea = LBA_libyan_capital_inflows }
		}
		add_to_variable = { LBA_western_portfolio_partners = 1 }
	}
	if = {
		limit = {
			country_exists = MLT
			MLT = { has_idea = LBA_libyan_capital_inflows }
		}
		add_to_variable = { LBA_western_portfolio_partners = 1 }
	}
	if = {
		limit = {
			country_exists = SPR
			SPR = { has_idea = LBA_libyan_capital_inflows }
		}
		add_to_variable = { LBA_western_portfolio_partners = 1 }
	}

	set_variable = { LBA_western_portfolio_return = LBA_western_portfolio_partners }
	multiply_variable = { LBA_western_portfolio_return = 0.01 }
	set_variable = { LBA_western_portfolio_income = LBA_western_portfolio_partners }
	multiply_variable = { LBA_western_portfolio_income = 0.1 }
	set_variable = { LBA_western_portfolio_influence_defense = LBA_western_portfolio_partners }
	multiply_variable = { LBA_western_portfolio_influence_defense = 0.01 }
	set_variable = { LBA_western_portfolio_diplomatic_cover = LBA_western_portfolio_partners }
	multiply_variable = { LBA_western_portfolio_diplomatic_cover = 0.005 }

	if = {
		limit = { has_country_flag = LBA_western_portfolio_balanced }
		add_to_variable = { LBA_western_portfolio_return = 0.005 }
		add_to_variable = { LBA_western_portfolio_influence_defense = 0.005 }
	}

	if = {
		limit = {
			is_ruled_by_a_gaddafi = yes
			check_variable = { LBA_western_portfolio_partners > 0 }
		}
		if = {
			limit = {
				NOT = {
					has_dynamic_modifier = {
						modifier = LBA_gaddafis_western_portfolio_dynamic_modifier
					}
				}
			}
			add_dynamic_modifier = {
				modifier = LBA_gaddafis_western_portfolio_dynamic_modifier
			}
		}
		force_update_dynamic_modifier = yes
	}
	else = {
		if = {
			limit = {
				has_dynamic_modifier = {
					modifier = LBA_gaddafis_western_portfolio_dynamic_modifier
				}
			}
			remove_dynamic_modifier = {
				modifier = LBA_gaddafis_western_portfolio_dynamic_modifier
			}
		}
	}
}
