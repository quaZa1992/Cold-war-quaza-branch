			}
		}
		if = {
			limit = {
				ISR = { has_country_flag = ISR70_policy_kach_special }
			}
			set_variable = { ISR70_policy_access_wanted = 0.10 }
			if = {
				limit = {
					ISR = { has_idea = ISR70_policy_permanent_emergency }
				}
				add_to_variable = { ISR70_policy_access_wanted = 0.05 }
			}
			else_if = {
				limit = {
					ISR = { has_idea = ISR70_policy_electoral_sunset }
				}
				add_to_variable = { ISR70_policy_access_wanted = 0.02 }
			}
		}
		set_variable = { ISR70_policy_access_room = 1 }
		subtract_from_variable = { ISR70_policy_access_room = modifier@local_factories }
		if = {
			limit = {
				has_dynamic_modifier = { modifier = ISR70_policy_factory_access }
			}
			add_to_variable = { ISR70_policy_access_room = ISR70_policy_access_applied }
		}
		clamp_variable = { var = ISR70_policy_access_room min = 0 max = ISR70_policy_access_wanted }
		set_variable = { ISR70_policy_access_applied = ISR70_policy_access_room }
		if = {
			limit = {
				is_controlled_by = ISR
				check_variable = { ISR70_policy_access_wanted > 0 }
			}
			add_dynamic_modifier = { modifier = ISR70_policy_factory_access }
		}
		else = {
			remove_dynamic_modifier = ISR70_policy_factory_access
			clear_variable = ISR70_policy_access_applied
		}
	}
