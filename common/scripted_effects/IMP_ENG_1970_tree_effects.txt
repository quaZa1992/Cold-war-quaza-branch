# IMPERIA COLD WAR — United Kingdom 1970 1970 tree effects.
# COLDWAR | Observe MD government changes without launching or rescheduling elections.
IMP_ENG_1970_ensure_defence_industry = {
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = ENG_defence_industry_modifier } } }
		add_dynamic_modifier = { modifier = ENG_defence_industry_modifier }
	}
}

IMP_ENG_1970_refresh_government_branches = {
	set_variable = { IMP_ENG_1970_layout_party = ruling_party }
	set_country_flag = IMP_ENG_1970_layout_party_cached
	mark_focus_tree_layout_dirty = yes
	force_update_dynamic_modifier = yes
}

IMP_ENG_1970_announce_government = {
	set_country_flag = IMP_ENG_1970_government_announced
	clr_country_flag = IMP_ENG_1970_native_campaign_seen
	if = {
		limit = { check_variable = { ruling_party = 3 } }
		country_event = { id = IMP_ENG_1970.3 days = 1 }
	}
	else_if = {
		limit = { check_variable = { ruling_party = 1 } }
		country_event = { id = IMP_ENG_1970.4 days = 1 }
	}
}
