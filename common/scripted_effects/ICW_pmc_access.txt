# IMPERIA COLD WAR — PMC start-date access and Game Controller override.
ICW_pmc_gc_open = {
	if = {
		limit = { ICW_pmc_gc_controller = yes ICW_pmc_system_active = no }
		clr_global_flag = ICW_pmc_closed
		set_global_flag = ICW_pmc_opened
		international_systems_force_update = yes
	}
}

# IMPERIA COLD WAR — PMC start-date access and Game Controller override.
ICW_pmc_gc_close = {
	if = {
		limit = { ICW_pmc_gc_controller = yes ICW_pmc_system_active = yes }
		set_global_flag = ICW_pmc_closed
		clr_global_flag = ICW_pmc_opened
		every_country = { ICW_pmc_cancel_pending_recruitment = yes }
		international_systems_force_update = yes
	}
}

# IMPERIA COLD WAR — PMC start-date access and Game Controller override.
ICW_pmc_refund_pending_local_0 = {
	if = {
		limit = { NOT = { has_country_flag = ICW_pmc_local_refunded_0 } check_variable = { pmc_local_light_recruited > 0 } OR = { has_country_flag = ICW_pmc_local_pending_0 has_decision = recruit_light_unit } }
		set_country_flag = ICW_pmc_local_refunded_0
		clr_country_flag = ICW_pmc_local_pending_0
		set_temp_variable = { treasury_change = global.mercs_cost^0 }
		multiply_temp_variable = { treasury_change = 0.8 }
		modify_treasury_effect = yes
		set_temp_variable = { temp_expense = global.mercs_weekly_cost^0 }
		multiply_temp_variable = { temp_expense = 0.8 }
		subtract_from_variable = { pmc_expenses = temp_expense }
		subtract_from_variable = { pmc_bought = 1 }
		subtract_from_variable = { merc_deployed = global.merc_deploy_cost^0 }
		subtract_from_variable = { pmc_local_light_recruited = 1 }
		if = { limit = { check_variable = { pmc_local_light_recruited < 1 } } clr_country_flag = pmc_local_light }
	}
}
