# Adapted from quaZa Feature Test, workshop 3818306694.
# IMPERIA CW — Cold War: complete uranium presentation and mine-based production.
# Cold War production starts at zero. Surveyed deposits unlock mine capacity;
# only completed, working mines add uranium to state resources. The non-1970
# deposit table is retained for compatibility; it does not grant production.
# Supply chain: uranium mines -> trade/reserves -> enrichment -> reactor fuel.
# Nuclear research, warheads and naval reactors also require natural uranium.
# Mine output: 3–12 units per working level, at most 1–3 levels by deposit.
# Damage reduces new mine output; reconciliation must never duplicate it.

@imp_uranium_per_enrichment_facility = 25

# Country scope, called from energy_calc_generation after the enrichment output was calculated
imp_uranium_enrichment_supply = {
	set_variable = {
		uranium_enrichment_need = {
			value = total_enrichment_facilities
			multiply = @imp_uranium_per_enrichment_facility
		}
	}
	# resource@uranium is the net balance after the facilities took their share:
	# a negative value is the missing uranium
	set_variable = {
		uranium_supply_ratio = {
			value = uranium_enrichment_need
			add = resource@uranium
			divide = { value = uranium_enrichment_need clamp = { min = 0.01 max = 999999 } }
			clamp = { min = 0 max = 1 }
		}
	}
	set_variable = {
		uranium_enrichment_output_factor = {
			value = uranium_supply_ratio
			multiply = 1
			add = 0
		}
	}
	multiply_variable = { total_nuclear_reactor_fuel_production = uranium_enrichment_output_factor }
	round_variable = total_nuclear_reactor_fuel_production
}

# State scope: mark this state as a uranium deposit
imp_uranium_add_deposit = {
	if = { limit = { NOT = { has_variable = imp_uranium_levels_applied } } set_variable = { imp_uranium_levels_applied = 0 } }
	if = { limit = { NOT = { has_variable = imp_uranium_mine_output } } set_variable = { imp_uranium_mine_output = 0 } }
	if = { limit = { NOT = { has_variable = imp_uranium_mine_levels } } set_variable = { imp_uranium_mine_levels = 0 } }
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = imp_uranium_deposit } } }
		add_dynamic_modifier = { modifier = imp_uranium_deposit }
	}
}

# State scope: use the engine's state building count for output and the map tooltip.
imp_uranium_count_mine_levels = {
	set_variable = { imp_uranium_mine_levels = non_damaged_building_level@uranium_mine }
}

# State scope: reserve unavailable levels in the mine's construction group.
# Both buildings have state_max = 3, so the native menu sees 3 - reserved levels.
# Terrain caps do not restrict state buildings. Only the actual modifier unlocks slots.
# Fixed levels make new-game initialization and weekly reconciliation idempotent.
imp_uranium_sync_mine_slots = {
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = imp_uranium_deposit } } }
		if = { limit = { uranium_mine > 0 } remove_building = { type = uranium_mine level = 3 } }
		if = { limit = { imp_uranium_reserved_slots < 3 } set_building_level = { type = imp_uranium_reserved_slots level = 3 instant_build = yes } }
	}
	else_if = {
		limit = { check_variable = { imp_uranium_mine_slots > 2 } }
		if = { limit = { imp_uranium_reserved_slots > 0 } remove_building = { type = imp_uranium_reserved_slots level = 3 } }
	}
	else_if = {
		limit = { check_variable = { imp_uranium_mine_slots > 1 } }
		if = { limit = { uranium_mine > 2 } set_building_level = { type = uranium_mine level = 2 instant_build = yes } }
		if = { limit = { NOT = { check_variable = { building_level@imp_uranium_reserved_slots = 1 } } } set_building_level = { type = imp_uranium_reserved_slots level = 1 instant_build = yes } }
	}
	else_if = {
		limit = { check_variable = { imp_uranium_mine_slots > 0 } }
		if = { limit = { uranium_mine > 1 } set_building_level = { type = uranium_mine level = 1 instant_build = yes } }
		if = { limit = { NOT = { check_variable = { building_level@imp_uranium_reserved_slots = 2 } } } set_building_level = { type = imp_uranium_reserved_slots level = 2 instant_build = yes } }
	}
	else = {
		if = { limit = { uranium_mine > 0 } remove_building = { type = uranium_mine level = 3 } }
		if = { limit = { imp_uranium_reserved_slots < 3 } set_building_level = { type = imp_uranium_reserved_slots level = 3 instant_build = yes } }
	}
}

# Country scope, weekly: output of the uranium mines.
# The mines add real state resources (add_resource), so Resource Gain Efficiency, excavation techs,
# infrastructure, damage etc. apply to uranium exactly like to every other resource.
# output = min(mine levels, allowed levels of the deposit) x output per level
imp_uranium_update_mines = {
 # IMPERIA CW — Cold War: one owned-state pass replaces daily and duplicate weekly scans.
 clr_country_flag = COLDWAR_uranium_has_mining_district
 every_owned_state = {
  imp_uranium_sync_mine_slots = yes
  if = {
   limit = { has_variable = imp_uranium_per_level }
   PREV = { set_country_flag = COLDWAR_uranium_has_mining_district }
   imp_uranium_count_mine_levels = yes
   imp_uranium_sync_mine_output = yes
  }
 }
 # No separate country output sum: nothing in the game reads that former cache.
 if = {
  limit = { is_ai = yes has_country_flag = COLDWAR_uranium_has_mining_district }
  imp_uranium_ai_build_mines = yes
 }
}

# State scope: bring the uranium state resource in line with the built mine levels
imp_uranium_sync_mine_output = {
	set_variable = { imp_uranium_levels_eff = imp_uranium_mine_levels }
	if = { limit = { NOT = { has_dynamic_modifier = { modifier = imp_uranium_deposit } } } set_variable = { imp_uranium_levels_eff = 0 } }
	if = {
		limit = { check_variable = { imp_uranium_levels_eff > imp_uranium_mine_slots } }
		set_variable = { imp_uranium_levels_eff = imp_uranium_mine_slots }
	}
	# output per level was changed (rebalance): take the old amounts out first
	if = {
		limit = {
			check_variable = { imp_uranium_levels_applied > 0 }
			NOT = { check_variable = { imp_uranium_per_level_applied = imp_uranium_per_level } }
		}
		set_temp_variable = { imp_u_amt = imp_uranium_per_level_applied }
		if = { limit = { check_variable = { imp_uranium_levels_applied > 0 } } imp_uranium_remove_amount = yes }
		if = { limit = { check_variable = { imp_uranium_levels_applied > 1 } } imp_uranium_remove_amount = yes }
		if = { limit = { check_variable = { imp_uranium_levels_applied > 2 } } imp_uranium_remove_amount = yes }
		set_variable = { imp_uranium_levels_applied = 0 }
	}
	set_variable = { imp_uranium_per_level_applied = imp_uranium_per_level }
	set_temp_variable = { imp_u_amt = imp_uranium_per_level }
	imp_uranium_sync_one_level = yes
	imp_uranium_sync_one_level = yes
	imp_uranium_sync_one_level = yes
	set_variable = { imp_uranium_mine_output = imp_uranium_levels_applied }
	multiply_variable = { imp_uranium_mine_output = imp_uranium_per_level }
}

imp_uranium_sync_one_level = {
	if = {
		limit = { check_variable = { imp_uranium_levels_eff > imp_uranium_levels_applied } }
		imp_uranium_add_amount = yes
		add_to_variable = { imp_uranium_levels_applied = 1 }
	}
	else_if = {
		limit = { check_variable = { imp_uranium_levels_eff < imp_uranium_levels_applied } }
		imp_uranium_remove_amount = yes
		subtract_from_variable = { imp_uranium_levels_applied = 1 }
	}
}

# Country scope (AI): the AI plans uranium mines like a real government would - when it needs
# uranium for its own nuclear programme, when the market price is high, or when it owns a rich
# deposit worth exporting. Construction goes into the normal construction queue.
imp_uranium_ai_build_mines = {
	if = {
		limit = {
			NOT = { has_country_flag = imp_ai_uranium_mine_cooldown }
			num_of_available_civilian_factories > 4
			ai_has_major_economic_problems = no
			OR = {
				check_variable = { resource@uranium < 0 }
				check_variable = { global.uranium_price > 0.25 }
				AND = {
					check_variable = { total_enrichment_facilities > 0 }
					check_variable = { uranium_supply_ratio < 1 }
				}
				AND = {
					check_variable = { global.uranium_price > 0.15 }
					num_of_available_civilian_factories > 9
					any_owned_state = {
						has_dynamic_modifier = { modifier = imp_uranium_deposit }
						check_variable = { imp_uranium_per_level > 4 }
						check_variable = { imp_uranium_mine_slots > building_level@uranium_mine }
						can_construct_building = uranium_mine
					}
				}
			}
			NOT = {
				AND = {
					check_variable = { global.uranium_price < 0.08 }
					check_variable = { resource@uranium > 0 }
				}
			}
			any_owned_state = {
				is_controlled_by = ROOT
				has_dynamic_modifier = { modifier = imp_uranium_deposit }
				has_variable = imp_uranium_per_level
				check_variable = { imp_uranium_mine_slots > building_level@uranium_mine }
				can_construct_building = uranium_mine
				NOT = { has_state_flag = imp_uranium_mine_queued }
			}
		}
		set_country_flag = { flag = imp_ai_uranium_mine_cooldown value = 1 days = 90 }
		# the richest free deposit first
		set_temp_variable = { imp_best_per_level = 0 }
		every_owned_state = {
			limit = {
				is_controlled_by = ROOT
				has_dynamic_modifier = { modifier = imp_uranium_deposit }
				has_variable = imp_uranium_per_level
				check_variable = { imp_uranium_mine_slots > building_level@uranium_mine }
				can_construct_building = uranium_mine
				NOT = { has_state_flag = imp_uranium_mine_queued }
				check_variable = { imp_uranium_per_level > imp_best_per_level }
			}
			set_temp_variable = { imp_best_per_level = imp_uranium_per_level }
			set_temp_variable = { imp_best_state = THIS }
		}
		var:imp_best_state = {
			imp_uranium_queue_mine = yes
		}
	}
}

# State scope: queue one state mine level, after checking the deposit and native cap.
imp_uranium_queue_mine = {
	imp_uranium_sync_mine_slots = yes
	if = {
		limit = {
			has_dynamic_modifier = { modifier = imp_uranium_deposit }
			has_variable = imp_uranium_per_level
			check_variable = { imp_uranium_mine_slots > building_level@uranium_mine }
			can_construct_building = uranium_mine
		}
		set_state_flag = { flag = imp_uranium_mine_queued value = 1 days = 365 }
		add_building_construction = { type = uranium_mine level = 1 instant_build = no }
	}
}
