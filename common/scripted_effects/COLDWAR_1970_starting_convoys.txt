# MD's starting utility stockpile stays off in 1970 (no 2000 railway trains); only the native MD convoy formula runs.
COLDWAR_1970_starting_convoys = {
	if = {
		limit = { NOT = { has_country_flag = COLDWAR_1970_starting_convoys_given } }
		set_country_flag = COLDWAR_1970_starting_convoys_given
		set_temp_variable = { base = 100 }
		set_temp_variable = { dockyard_increase = 25 }
		multiply_temp_variable = { dockyard_increase = num_of_naval_factories }
		add_to_temp_variable = { dockyard_increase = base }
		add_equipment_to_stockpile = {
			type = convoy
			amount = dockyard_increase
		}
	}
}
