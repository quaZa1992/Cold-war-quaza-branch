# IMPERIA COLD WAR — Eastern Bloc consumer goods contracts
# COLDWAR | hand patch 01.11.2026 (Claude): one menu action, cap of 3 size steps, small Western drift. Port to the EB generator before the next build.

# Diplomatic action: THIS = Western seller, ROOT = buyer, temp COLDWAR_EB_t_csize 1..3.
COLDWAR_EB_contract_request = {
	ROOT = {
		set_variable = { COLDWAR_EB_ct_pend@PREV = global.num_days }
		add_to_variable = { COLDWAR_EB_ct_pend@PREV = 30 }
		add_to_variable = { COLDWAR_EB_ct_pend_total = COLDWAR_EB_t_csize }
		set_variable = { COLDWAR_EB_ct_pend_until = global.num_days }
		add_to_variable = { COLDWAR_EB_ct_pend_until = 30 }
	}
	add_to_array = { COLDWAR_EB_rq_b = ROOT }
	add_to_array = { COLDWAR_EB_rq_s = COLDWAR_EB_t_csize }
	if = {
		limit = { NOT = { has_country_flag = COLDWAR_EB_rq_busy } }
		set_country_flag = { flag = COLDWAR_EB_rq_busy days = 60 value = 1 }
		country_event = { id = COLDWAR_EBC.1 days = 1 }
	}
}

# Seller scope: the next request in the queue, or done.
COLDWAR_EB_contract_next_request = {
	if = {
		limit = { check_variable = { COLDWAR_EB_rq_b^num > 0 } }
		set_country_flag = { flag = COLDWAR_EB_rq_busy days = 60 value = 1 }
		country_event = { id = COLDWAR_EBC.1 days = 1 }
	}
	else = {
		clr_country_flag = COLDWAR_EB_rq_busy
	}
}

# Seller scope: take the first request out of the queue.
COLDWAR_EB_contract_pop_request = {
	set_variable = { COLDWAR_EB_rq_cur_b = COLDWAR_EB_rq_b^0 }
	set_variable = { COLDWAR_EB_rq_cur_s = COLDWAR_EB_rq_s^0 }
	remove_from_array = { array = COLDWAR_EB_rq_b index = 0 }
	remove_from_array = { array = COLDWAR_EB_rq_s index = 0 }
	set_temp_variable = { COLDWAR_EB_t_g = 0 }
	var:COLDWAR_EB_rq_cur_b = { set_temp_variable = { COLDWAR_EB_t_g = gdp_total } }
	set_variable = { COLDWAR_EB_rq_amt = COLDWAR_EB_t_g }
	if = { limit = { check_variable = { COLDWAR_EB_rq_cur_s = 1 } } multiply_variable = { COLDWAR_EB_rq_amt = 0.0104 } }
	else_if = { limit = { check_variable = { COLDWAR_EB_rq_cur_s = 2 } } multiply_variable = { COLDWAR_EB_rq_amt = 0.0208 } }
	else = { multiply_variable = { COLDWAR_EB_rq_amt = 0.0416 } }
}

# Buyer scope: running contracts for the Social Contract tab (up to four partners).
COLDWAR_EB_contracts_display = {
	COLDWAR_EB_contract_calc_free = yes
	set_variable = { COLDWAR_EB_ctv_n = 0 }
	for_each_scope_loop = {
		array = COLDWAR_EB_ct_partners
		PREV = {
			add_to_variable = { COLDWAR_EB_ctv_n = 1 }
			if = {
				limit = { check_variable = { COLDWAR_EB_ctv_n = 1 } }
				set_variable = { COLDWAR_EB_ctv_p1 = PREV.id }
				set_variable = { COLDWAR_EB_ctv_s1 = COLDWAR_EB_ct_size@PREV }
				set_variable = { COLDWAR_EB_ctv_d1 = COLDWAR_EB_ct_end@PREV }
				subtract_from_variable = { COLDWAR_EB_ctv_d1 = global.num_days }
			}
			if = {
				limit = { check_variable = { COLDWAR_EB_ctv_n = 2 } }
				set_variable = { COLDWAR_EB_ctv_p2 = PREV.id }
				set_variable = { COLDWAR_EB_ctv_s2 = COLDWAR_EB_ct_size@PREV }
				set_variable = { COLDWAR_EB_ctv_d2 = COLDWAR_EB_ct_end@PREV }
				subtract_from_variable = { COLDWAR_EB_ctv_d2 = global.num_days }
			}
			if = {
				limit = { check_variable = { COLDWAR_EB_ctv_n = 3 } }
				set_variable = { COLDWAR_EB_ctv_p3 = PREV.id }
				set_variable = { COLDWAR_EB_ctv_s3 = COLDWAR_EB_ct_size@PREV }
				set_variable = { COLDWAR_EB_ctv_d3 = COLDWAR_EB_ct_end@PREV }
				subtract_from_variable = { COLDWAR_EB_ctv_d3 = global.num_days }
			}
			if = {
				limit = { check_variable = { COLDWAR_EB_ctv_n = 4 } }
				set_variable = { COLDWAR_EB_ctv_p4 = PREV.id }
				set_variable = { COLDWAR_EB_ctv_s4 = COLDWAR_EB_ct_size@PREV }
				set_variable = { COLDWAR_EB_ctv_d4 = COLDWAR_EB_ct_end@PREV }
				subtract_from_variable = { COLDWAR_EB_ctv_d4 = global.num_days }
			}
		}
	}
}

# Buyer scope: free size steps under the cap of 3 (running plus waiting requests).
COLDWAR_EB_contract_calc_free = {
	if = {
		limit = { check_variable = { global.num_days > COLDWAR_EB_ct_pend_until } }
		set_variable = { COLDWAR_EB_ct_pend_total = 0 }
	}
	set_variable = { COLDWAR_EB_ct_free = 3 }
	subtract_from_variable = { COLDWAR_EB_ct_free = COLDWAR_EB_contract_total }
	subtract_from_variable = { COLDWAR_EB_ct_free = COLDWAR_EB_ct_pend_total }
	clamp_variable = { var = COLDWAR_EB_ct_free min = 0 max = 3 }
}

# Buyer scope: send window opens with the largest size that still fits.
COLDWAR_EB_contract_send_reset = {
	COLDWAR_EB_contract_calc_free = yes
	set_variable = { COLDWAR_EB_ct_send_size = COLDWAR_EB_ct_free }
	clamp_variable = { var = COLDWAR_EB_ct_send_size min = 1 max = 3 }
}

# Seller scope (event answer): the buyer's waiting size is released.
COLDWAR_EB_contract_release_pending = {
	var:COLDWAR_EB_rq_cur_b = {
		set_variable = { COLDWAR_EB_ct_pend@PREV = 0 }
		subtract_from_variable = { COLDWAR_EB_ct_pend_total = PREV.COLDWAR_EB_rq_cur_s }
		clamp_variable = { var = COLDWAR_EB_ct_pend_total min = 0 }
	}
}
