# IMPERIA COLD WAR — Dated Lukashenko selection and portrait wardrobe.
COLDWAR_LUKASHENKO_apply_portrait = {
	if = {
		limit = {
			IMP_is_1970_scenario = yes
			exists = yes
			is_dynamic_country = no
			is_ai = no
			OR = {
				original_tag = SOV
				original_tag = BLR
			}
			COLDWAR_LUKASHENKO_ready = yes
			OR = {
				has_country_leader = { name = "Alexander Lukashenko" ruling_only = yes }
				has_country_leader = { name = "Aleksandr Lukashenko" ruling_only = yes }
				has_country_leader = { name = "Alyaksandr Lukashenka" ruling_only = yes }
			}
		}
		if = {
			limit = {
				has_country_flag = COLDWAR_LUKASHENKO_portrait_90s
				COLDWAR_LUKASHENKO_portrait_90s_ready = yes
			}
			set_country_leader_portrait = { portrait = GFX_COLDWAR_LUKASHENKO_90s_portrait }
		}
		else = {
			set_country_leader_portrait = { portrait = GFX_COLDWAR_LUKASHENKO_young_portrait }
		}
	}
}
