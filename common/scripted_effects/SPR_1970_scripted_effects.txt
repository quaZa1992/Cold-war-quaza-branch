# Internal-faction helpers double positive gains for autocratic governments
# and mutate temp_opinion in place. These wrappers preserve the exact value
# advertised by the static focus tooltip and restore it after every call.
SPR_1970_prepare_exact_internal_faction_opinion = {
	set_temp_variable = { SPR_1970_requested_internal_faction_opinion = temp_opinion }
	if = {
		limit = {
			check_variable = { temp_opinion > 0 }
			OR = {
				AND = { has_variable = ruling_party check_variable = { ruling_party = 0 } }
				AND = { has_variable = ruling_party check_variable = { ruling_party = 7 } }
				AND = { has_variable = ruling_party check_variable = { ruling_party = 13 } }
				AND = { has_variable = ruling_party check_variable = { ruling_party = 21 } }
				AND = { has_variable = ruling_party check_variable = { ruling_party = 22 } }
			}
		}
		multiply_temp_variable = { temp_opinion = 0.5 }
	}
}

SPR_1970_restore_exact_internal_faction_opinion = {
	set_temp_variable = { temp_opinion = SPR_1970_requested_internal_faction_opinion }
}

# These construction effects deliberately contain no Treasury payment. The
# standard MD helpers include much larger fixed costs (and the office helper
# also adds a power plant), which would make the focus tooltip misleading.
SPR_1970_build_one_civilian_factory = {
	random_owned_controlled_state = {
		limit = {
			free_building_slots = {
				building = industrial_complex
				size > 0
				include_locked = yes
			}
			OR = {
				is_in_home_area = yes
				NOT = {
					owner = {
						any_owned_state = {
							free_building_slots = {
								building = industrial_complex
								size > 0
								include_locked = yes
							}
							is_in_home_area = yes
						}
					}
				}
			}
		}
		add_extra_state_shared_building_slots = 1
		add_building_construction = {
			type = industrial_complex
			level = 1
			instant_build = yes
		}
	}
}

SPR_1970_update_franco_dynamic_modifier = {
	set_variable = { SPR_1970_franco_health_pp_gain = 0 }
	set_variable = { SPR_1970_franco_health_stability_factor = 0 }
	set_variable = { SPR_1970_franco_health_bureaucracy_factor = 0 }
	set_variable = { SPR_1970_aperturista_drift = 0 }
	set_variable = { SPR_1970_bunker_drift = 0 }

	if = {
		limit = { check_variable = { SPR_1970_franco_health > 70 } }
		set_variable = { SPR_1970_franco_health_pp_gain = -0.10 }
		set_variable = { SPR_1970_franco_health_stability_factor = -0.015 }
		set_variable = { SPR_1970_franco_health_bureaucracy_factor = 0.01 }
	}
	else_if = {
		limit = { check_variable = { SPR_1970_franco_health > 50 } }
		set_variable = { SPR_1970_franco_health_pp_gain = -0.20 }
		set_variable = { SPR_1970_franco_health_stability_factor = -0.03 }
		set_variable = { SPR_1970_franco_health_bureaucracy_factor = 0.025 }
	}
	else_if = {
		limit = { check_variable = { SPR_1970_franco_health > 30 } }
		set_variable = { SPR_1970_franco_health_pp_gain = -0.35 }
		set_variable = { SPR_1970_franco_health_stability_factor = -0.06 }
		set_variable = { SPR_1970_franco_health_bureaucracy_factor = 0.05 }
	}
	else = {
		set_variable = { SPR_1970_franco_health_pp_gain = -0.50 }
		set_variable = { SPR_1970_franco_health_stability_factor = -0.10 }
		set_variable = { SPR_1970_franco_health_bureaucracy_factor = 0.08 }
	}

	# Factional drift follows the size of the lead rather than treating a narrow
	# plurality and overwhelming institutional control as equivalent.
	set_temp_variable = { SPR_1970_influence_lead = SPR_1970_aperturista_influence }
	set_temp_variable = { SPR_1970_opposing_influence = SPR_1970_bunker_influence }
	multiply_temp_variable = { SPR_1970_opposing_influence = -1 }
	add_to_temp_variable = { SPR_1970_influence_lead = SPR_1970_opposing_influence }

	if = {
		limit = { check_variable = { SPR_1970_influence_lead > 0 } }
		if = {
			limit = { check_variable = { SPR_1970_influence_lead > 29 } }
			set_variable = { SPR_1970_aperturista_drift = 0.05 }
		}
		else_if = {
			limit = { check_variable = { SPR_1970_influence_lead > 9 } }
			set_variable = { SPR_1970_aperturista_drift = 0.03 }
		}
		else = {
			set_variable = { SPR_1970_aperturista_drift = 0.01 }
		}
	}
	else_if = {
		limit = { check_variable = { SPR_1970_influence_lead < 0 } }
		multiply_temp_variable = { SPR_1970_influence_lead = -1 }
		if = {
			limit = { check_variable = { SPR_1970_influence_lead > 29 } }
			set_variable = { SPR_1970_bunker_drift = 0.05 }
		}
		else_if = {
			limit = { check_variable = { SPR_1970_influence_lead > 9 } }
			set_variable = { SPR_1970_bunker_drift = 0.03 }
		}
		else = {
			set_variable = { SPR_1970_bunker_drift = 0.01 }
		}
	}

	if = {
		limit = { has_country_flag = SPR_1970_franco_dead }
		if = {
			limit = { has_dynamic_modifier = { modifier = SPR_1970_failing_health_dynamic_modifier } }
			remove_dynamic_modifier = { modifier = SPR_1970_failing_health_dynamic_modifier }
		}
	}
	else = {
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = SPR_1970_failing_health_dynamic_modifier } } }
			add_dynamic_modifier = { modifier = SPR_1970_failing_health_dynamic_modifier }
		}
		force_update_dynamic_modifier = yes
	}
}

	if = {
		limit = { check_variable = { rul_party_temp < 4 } }
		if = {
			limit = { check_variable = { enable_elections = 0 } }
			set_politics = {
				ruling_party = democratic
				elections_allowed = yes
			}
		}
		else = {
			set_politics = {
				ruling_party = democratic
				elections_allowed = no
			}
		}
	}
	else = {
		if = {
			limit = { check_variable = { enable_elections = 0 } }
			set_politics = {
				ruling_party = nationalist
				elections_allowed = yes
			}
		}
		else = {
			set_politics = {
				ruling_party = nationalist
				elections_allowed = no
			}
		}
	}

	update_government_coalition_strength = yes
	update_party_name = yes
	set_coalition_drift = yes
	SPR_1970_refresh_democratic_cabinet_spirit = yes
}

# Cabinet programmes are election-persistent unlocks, but only the highest
# tier belonging to the party currently in government may be active.
SPR_1970_refresh_democratic_cabinet_spirit = {
	if = {
		limit = {
			original_tag = SPR
			IMP_is_1970_scenario = yes
			has_country_flag = SPR_1970_party_system_1982
			has_country_flag = SPR_1970_democracy_consolidated
			has_elections = yes
		}

		if = { limit = { has_idea = SPR_1970_conservative_programme_1_idea } remove_ideas = SPR_1970_conservative_programme_1_idea }
		if = { limit = { has_idea = SPR_1970_conservative_programme_2_idea } remove_ideas = SPR_1970_conservative_programme_2_idea }
		if = { limit = { has_idea = SPR_1970_conservative_programme_3_idea } remove_ideas = SPR_1970_conservative_programme_3_idea }
		if = { limit = { has_idea = SPR_1970_liberal_renewal_1_idea } remove_ideas = SPR_1970_liberal_renewal_1_idea }
		if = { limit = { has_idea = SPR_1970_liberal_renewal_2_idea } remove_ideas = SPR_1970_liberal_renewal_2_idea }
		if = { limit = { has_idea = SPR_1970_liberal_renewal_3_idea } remove_ideas = SPR_1970_liberal_renewal_3_idea }
		if = { limit = { has_idea = SPR_1970_socialist_modernisation_1_idea } remove_ideas = SPR_1970_socialist_modernisation_1_idea }
		if = { limit = { has_idea = SPR_1970_socialist_modernisation_2_idea } remove_ideas = SPR_1970_socialist_modernisation_2_idea }
		if = { limit = { has_idea = SPR_1970_socialist_modernisation_3_idea } remove_ideas = SPR_1970_socialist_modernisation_3_idea }
		if = { limit = { has_idea = SPR_1970_eurocommunist_programme_1_idea } remove_ideas = SPR_1970_eurocommunist_programme_1_idea }
		if = { limit = { has_idea = SPR_1970_eurocommunist_programme_2_idea } remove_ideas = SPR_1970_eurocommunist_programme_2_idea }
		if = { limit = { has_idea = SPR_1970_eurocommunist_programme_3_idea } remove_ideas = SPR_1970_eurocommunist_programme_3_idea }

		if = {
			limit = { AND = { has_variable = ruling_party check_variable = { ruling_party = 1 } } }
			if = { limit = { has_country_flag = SPR_1970_ap_programme_3_unlocked } add_ideas = SPR_1970_conservative_programme_3_idea }
			else_if = { limit = { has_country_flag = SPR_1970_ap_programme_2_unlocked } add_ideas = SPR_1970_conservative_programme_2_idea }
			else_if = { limit = { has_country_flag = SPR_1970_ap_programme_1_unlocked } add_ideas = SPR_1970_conservative_programme_1_idea }
		}
		else_if = {
			limit = { AND = { has_variable = ruling_party check_variable = { ruling_party = 2 } } }
			if = { limit = { has_country_flag = SPR_1970_cds_programme_3_unlocked } add_ideas = SPR_1970_liberal_renewal_3_idea }
			else_if = { limit = { has_country_flag = SPR_1970_cds_programme_2_unlocked } add_ideas = SPR_1970_liberal_renewal_2_idea }
			else_if = { limit = { has_country_flag = SPR_1970_cds_programme_1_unlocked } add_ideas = SPR_1970_liberal_renewal_1_idea }
		}
		else_if = {
			limit = { AND = { has_variable = ruling_party check_variable = { ruling_party = 3 } } }
			if = { limit = { has_country_flag = SPR_1970_psoe_programme_3_unlocked } add_ideas = SPR_1970_socialist_modernisation_3_idea }
			else_if = { limit = { has_country_flag = SPR_1970_psoe_programme_2_unlocked } add_ideas = SPR_1970_socialist_modernisation_2_idea }
			else_if = { limit = { has_country_flag = SPR_1970_psoe_programme_1_unlocked } add_ideas = SPR_1970_socialist_modernisation_1_idea }
		}
		else_if = {
			limit = { AND = { has_variable = ruling_party check_variable = { ruling_party = 4 } } }
			if = { limit = { has_country_flag = SPR_1970_pce_programme_3_unlocked } add_ideas = SPR_1970_eurocommunist_programme_3_idea }
			else_if = { limit = { has_country_flag = SPR_1970_pce_programme_2_unlocked } add_ideas = SPR_1970_eurocommunist_programme_2_idea }
			else_if = { limit = { has_country_flag = SPR_1970_pce_programme_1_unlocked } add_ideas = SPR_1970_eurocommunist_programme_1_idea }
