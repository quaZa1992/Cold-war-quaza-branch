imp_tac_nuke_effects ={
    # Tactical nukes are fired at military targets, which are often not near population centers.
    # Their purpose isn't to maximize military casualities, and as such shouldn't maximize civilian casualties.
    # They have a lower yield. 2% of the population in casualties seems fair.

    add_dynamic_modifier = {
        modifier = nuclear_contamination
        days = 1095
    }

    hidden_effect = {
	
		set_temp_variable = { temp_state_pop = state_population_k }
		if = {
			limit = { civilian_shelter > 0 }
			set_temp_variable = { shelter_levels_effect = building_level@civilian_shelter }
			multiply_temp_variable = { shelter_levels_effect = -0.10 }
			set_temp_variable = { shelter_sufficiency = 1 }
			add_to_temp_variable = { shelter_sufficiency = shelter_levels_effect }
			multiply_temp_variable = { casualty_rate = shelter_sufficiency }
		}
		
		multiply_temp_variable = { temp_state_pop = casualty_rate }
		multiply_temp_variable = { temp_state_pop = 250 } #Optimization, gets rid of 996 loop iterations, maximum casualty = 8 Million People per state; If by one reason you need bigger limit then you did something wrong.
		for_loop_effect = {
			end = 4
			add_manpower = temp_state_pop
		}
		#multiply_temp_variable = { productivity_damage = missile_interception }
		# Approved historical1970 floor: min(previous productivity,100), never raise on a hit.
		set_temp_variable = { mdcw_nuke_productivity_before = productivity_state_var }
		set_temp_variable = { mdcw_nuke_productivity_floor = mdcw_nuke_productivity_before }
		clamp_temp_variable = { var = mdcw_nuke_productivity_floor max = 100 }
		multiply_variable = { productivity_state_var = productivity_damage }
		clamp_variable = {
			var = productivity_state_var
			min = mdcw_nuke_productivity_floor
			max = mdcw_nuke_productivity_before
		}
	
        #set_temp_variable = { state_pop_t = state_population_k }
        #divide_temp_variable = { state_pop_t = 100 } # We want ~2% of the population to die. However, integer overflow will kick in, so we do 2x1 instead.
        #multiply_temp_variable = { state_pop_t = 1000 }

        #set_temp_variable = { casualties_t = 0 }
        #subtract_from_temp_variable = { casualties_t = state_pop_t }

        #for_loop_effect = {
        #    end = 3
        #    add_manpower = casualties_t
        #}

        save_event_target_as = imp_contam_spreader
        CONTROLLER = {
            country_event = { id = imp_nukes.11 days = 5 }
        }

        # Store this for later use?
        add_to_variable = { global.total_nukes_explodes = 1 }
    }

    if = {
        limit = { NOT = { has_global_flag = imp_first_tact_used has_global_flag = imp_first_nuke_used } }
        set_variable = { global.state_tact_first = THIS }
        every_country = {
            news_event = 
            {
                id = imp_nukes.1
                days = 1
            }

            news_event = 
            {
                id = imp_nukes.3
                days = 7
            }
        }
        set_global_flag = imp_first_tact_used
    }
}
