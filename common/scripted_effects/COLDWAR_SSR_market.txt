COLDWAR_SSR_market_complete_light_industry_auction = {
	set_temp_variable = { COLDWAR_SSR_market_transaction_cost = { value = gdp_total multiply = 0.0005 } }
	set_temp_variable = { COLDWAR_SSR_market_transaction_proceeds = { value = gdp_total multiply = 0.0015 } }
	if = {
		limit = { COLDWAR_SSR_is_independent = yes has_country_flag = COLDWAR_SSR_market_pending_light_industry_auction NOT = { has_country_flag = COLDWAR_SSR_market_lot_light_industry_settled } COLDWAR_SSR_market_has_market_authority = yes always = yes check_variable = { treasury > COLDWAR_SSR_market_transaction_cost } }
		set_temp_variable = { treasury_change = COLDWAR_SSR_market_transaction_cost }
		multiply_temp_variable = { treasury_change = -1 }
		modify_treasury_effect = yes
		set_temp_variable = { treasury_change = COLDWAR_SSR_market_transaction_proceeds }
		modify_treasury_effect = yes
		
		set_country_flag = COLDWAR_SSR_market_lot_light_industry_settled
		set_country_flag = COLDWAR_SSR_market_decision_light_industry_auction_settled
		set_country_flag = { flag = COLDWAR_SSR_market_lot_cooldown value = 1 days = 30 }
		set_variable = { COLDWAR_SSR_market_lot_count = { value = COLDWAR_SSR_market_lot_count add = 1 } }
			add_to_variable = { COLDWAR_SSR_market_settlements_civilian_factories_productivity = 0.01 }
			add_to_variable = { COLDWAR_SSR_market_settlements_corporate_tax_income_multiplier_modifier = -0.01 }
		set_temp_variable = { COLDWAR_SSR_market_relief = { value = COLDWAR_SSR_market_original_burden divide = 6 } }
		subtract_from_variable = { COLDWAR_SSR_market_burden_remaining = COLDWAR_SSR_market_relief }
		if = { limit = { check_variable = { COLDWAR_SSR_market_burden_remaining < 0.00001 } } clear_variable = COLDWAR_SSR_market_burden_remaining }
		set_country_flag = COLDWAR_SSR_market_has_settlement
		set_country_flag = COLDWAR_SSR_market_burden_dirty
		set_country_flag = COLDWAR_SSR_market_settlements_dirty
	}
	clr_country_flag = COLDWAR_SSR_market_pending_light_industry_auction
	clr_country_flag = COLDWAR_SSR_market_pending_transaction
	clr_country_flag = COLDWAR_SSR_market_pending_timer

	COLDWAR_SSR_market_reconcile = yes
}

	if = { limit = { COLDWAR_SSR_is_independent = yes has_country_flag = COLDWAR_SSR_market_course_started OR = { has_country_flag = COLDWAR_SSR_market_shock_deadline has_country_flag = COLDWAR_SSR_market_gradual_deadline has_country_flag = COLDWAR_SSR_market_directed_deadline } OR = { NOT = { has_country_flag = COLDWAR_SSR_market_window_applied } NOT = { has_dynamic_modifier = { modifier = COLDWAR_SSR_market_window_dynamic_modifier } } NOT = { has_dynamic_modifier = { modifier = COLDWAR_SSR_market_fiscal_dynamic_modifier } } } }

		if = { limit = { has_country_flag = COLDWAR_SSR_market_directed_deadline } COLDWAR_SSR_market_set_course_values_directed = yes }
		add_dynamic_modifier = { modifier = COLDWAR_SSR_market_window_dynamic_modifier }
		add_dynamic_modifier = { modifier = COLDWAR_SSR_market_fiscal_dynamic_modifier }
		set_country_flag = COLDWAR_SSR_market_window_applied
	}
	if = { limit = { NOT = { COLDWAR_SSR_is_independent = yes has_country_flag = COLDWAR_SSR_market_course_started OR = { has_country_flag = COLDWAR_SSR_market_shock_deadline has_country_flag = COLDWAR_SSR_market_gradual_deadline has_country_flag = COLDWAR_SSR_market_directed_deadline } } OR = { has_country_flag = COLDWAR_SSR_market_window_applied has_dynamic_modifier = { modifier = COLDWAR_SSR_market_window_dynamic_modifier } has_dynamic_modifier = { modifier = COLDWAR_SSR_market_fiscal_dynamic_modifier } } }
		if = { limit = { has_dynamic_modifier = { modifier = COLDWAR_SSR_market_window_dynamic_modifier } } remove_dynamic_modifier = { modifier = COLDWAR_SSR_market_window_dynamic_modifier } }
		if = { limit = { has_dynamic_modifier = { modifier = COLDWAR_SSR_market_fiscal_dynamic_modifier } } remove_dynamic_modifier = { modifier = COLDWAR_SSR_market_fiscal_dynamic_modifier } }
		clr_country_flag = COLDWAR_SSR_market_window_applied
		clear_variable = COLDWAR_SSR_market_window_civilian_factories_productivity
		clear_variable = COLDWAR_SSR_market_window_production_speed_arms_factory_factor
		clear_variable = COLDWAR_SSR_market_window_production_speed_industrial_complex_factor
		clear_variable = COLDWAR_SSR_market_window_productivity_growth_modifier
		clear_variable = COLDWAR_SSR_market_fiscal_production_speed_buildings_factor
		clear_variable = COLDWAR_SSR_market_fiscal_social_cost_multiplier_modifier
		clear_variable = COLDWAR_SSR_market_fiscal_stability_factor
		clear_variable = COLDWAR_SSR_market_fiscal_tax_gain_multiplier_modifier
	}
	if = { limit = { check_variable = { COLDWAR_SSR_market_burden_remaining > 0 } OR = { NOT = { has_country_flag = COLDWAR_SSR_market_burden_applied } has_country_flag = COLDWAR_SSR_market_burden_dirty NOT = { has_dynamic_modifier = { modifier = COLDWAR_SSR_market_burden_dynamic_modifier } } } }
		if = { limit = { has_dynamic_modifier = { modifier = COLDWAR_SSR_market_burden_dynamic_modifier } } remove_dynamic_modifier = { modifier = COLDWAR_SSR_market_burden_dynamic_modifier } }
		add_dynamic_modifier = { modifier = COLDWAR_SSR_market_burden_dynamic_modifier }
		set_country_flag = COLDWAR_SSR_market_burden_applied
		clr_country_flag = COLDWAR_SSR_market_burden_dirty
	}
	if = { limit = { NOT = { check_variable = { COLDWAR_SSR_market_burden_remaining > 0 } } OR = { has_country_flag = COLDWAR_SSR_market_burden_applied has_dynamic_modifier = { modifier = COLDWAR_SSR_market_burden_dynamic_modifier } } }
		if = { limit = { has_dynamic_modifier = { modifier = COLDWAR_SSR_market_burden_dynamic_modifier } } remove_dynamic_modifier = { modifier = COLDWAR_SSR_market_burden_dynamic_modifier } }
		clr_country_flag = COLDWAR_SSR_market_burden_applied
		clr_country_flag = COLDWAR_SSR_market_burden_dirty
	}
	if = { limit = { has_country_flag = COLDWAR_SSR_market_has_settlement OR = { NOT = { has_country_flag = COLDWAR_SSR_market_settlements_applied } has_country_flag = COLDWAR_SSR_market_settlements_dirty NOT = { has_dynamic_modifier = { modifier = COLDWAR_SSR_market_settlements_dynamic_modifier } } } }

		set_country_flag = COLDWAR_SSR_market_settlements_applied
		clr_country_flag = COLDWAR_SSR_market_settlements_dirty
	}
	if = { limit = { NOT = { has_country_flag = COLDWAR_SSR_market_has_settlement } OR = { has_country_flag = COLDWAR_SSR_market_settlements_applied has_country_flag = COLDWAR_SSR_market_settlements_dirty has_dynamic_modifier = { modifier = COLDWAR_SSR_market_settlements_dynamic_modifier } } }
		if = { limit = { has_dynamic_modifier = { modifier = COLDWAR_SSR_market_settlements_dynamic_modifier } } remove_dynamic_modifier = { modifier = COLDWAR_SSR_market_settlements_dynamic_modifier } }
		clr_country_flag = COLDWAR_SSR_market_settlements_applied
		clr_country_flag = COLDWAR_SSR_market_settlements_dirty
	}
