# IMPERIA CW — Cold War: Guest-worker and diaspora mechanics.
# Native guest-worker helpers: ROOT receives, FROM sends during requests and completion.
# Treasury values are MD billions; resident stock/net migration are thousands of people.
# Resident stock is gross retained arrivals, separately from the anti-loop net-emigration ledger.
# Legacy missing stock is seeded once from max(-old net emigration,0) before any transfer.
# Fresh population_total uses units100,000. Healthcare and bureaucracy read
# cached integration coverage; Welfare support enters the native weekly spending array.
# Origin cohorts are country-level aggregates, without a residence-state matrix.
# Recruitment uses physical state population; the net-export ledger prevents remittance loops.
# Explicit return operations reduce cohorts, coverage and origin remittances together.
# Selected states are saved in the request and revalidated on acceptance; completion never rerolls.
# All pending/UI-window locks, refunds, cooldowns and remittance rules remain independent of stock.

# Successful arrival receipts read actual destination state_population_k before and after
# STATE add_manpower; last_arrival_k is the observed delta, never just the requested tier.
# One success log records the receipt; remittances decay by 0.01% per native weekly tick.

# Population mutations use exact-filtered random_owned_state blocks, matching refugee STATE scope.
# 100k uses two distinct saved origin states at -50k each, with all state checks before mutation.
# Native state withdrawals can increase the sender's free military reserve (observed in game).
# Remove only that immediate positive difference; never refund a genuine reserve reduction.

mdcw_gw_initialize_resident_stock = {
	if = {
		limit = {
			NOT = { has_variable = mdcw_gw_guest_workers_k }
		}
		set_variable = { mdcw_gw_guest_workers_k = 0 }
		subtract_from_variable = { mdcw_gw_guest_workers_k = mdcw_gw_net_emigration_k }
		clamp_variable = { var = mdcw_gw_guest_workers_k min = 0 }
	}
}

mdcw_gw_update_resident_costs = {
 clamp_variable = { var = mdcw_gw_guest_workers_k min = 0 }
 mdcw_gw_ledger_initialize = yes
 # Country-level cohorts remain until explicit repatriation.
 clear_variable = mdcw_gw_pending_export_pop
 mdcw_gw_calculate_diaspora_costs = yes
 # IMPERIA CW — Cold War: Upgrade an existing active save once; ordinary budget passes stay cached.
 if = { limit = { check_variable = { gw_known_k > 0 } NOT = { has_country_flag = gw_migration_ready } }
  mdcw_dp_ensure_arrays = yes mdcw_dp_start_reviews = yes
 }
}

mdcw_gw_apply_immigration_25000 = {
	mdcw_gw_initialize_resident_stock = yes
	add_to_variable = { mdcw_gw_guest_workers_k = 25 }
	set_temp_variable = { mdcw_gw_previous_positive_k = mdcw_gw_net_emigration_k }
	clamp_temp_variable = { var = mdcw_gw_previous_positive_k min = 0 }
	subtract_from_variable = { mdcw_gw_net_emigration_k = 25 }
	if = {
		limit = {
			check_variable = { mdcw_gw_previous_positive_k > 0 }
		}
		set_temp_variable = { mdcw_gw_remaining_share = mdcw_gw_net_emigration_k }
		clamp_temp_variable = { var = mdcw_gw_remaining_share min = 0 }
		divide_temp_variable = { mdcw_gw_remaining_share = mdcw_gw_previous_positive_k }
		multiply_variable = { mdcw_gw_remittance_pool_m = mdcw_gw_remaining_share }
	}
	else = {
		set_variable = { mdcw_gw_remittance_pool_m = 0 }
	}
	clamp_variable = { var = mdcw_gw_remittance_pool_m min = 0 max = 3000 }
}

# IMPERIA CW — Cold War: Refresh both parties' Ministry controls when negotiations end.
mdcw_gw_clear_pending = {
	clr_country_flag = mdcw_gw_request_pending
	clr_country_flag = mdcw_gw_offer_pending
	clear_variable = mdcw_gw_partner
	clear_variable = mdcw_gw_pending_tier
	clear_variable = mdcw_gw_pending_pp

}

mdcw_gw_refund_request = {
	if = {
		limit = { has_country_flag = mdcw_gw_request_pending }
		save_event_target_as = mdcw_gw_receiver
		add_political_power = mdcw_gw_pending_pp
		var:mdcw_gw_partner = {
			save_event_target_as = mdcw_gw_sender
			if = {
				limit = {
					has_country_flag = mdcw_gw_offer_pending
					var:mdcw_gw_partner = { tag = ROOT }
				}
				mdcw_gw_clear_pending = yes
			}
		}
		mdcw_gw_clear_pending = yes
	}
}

# ROOT receives, FROM sends. Called once after all validated source-state withdrawals.
# Compare raw people rather than rounded manpower_k; the country effect only removes
# an observed reserve gain and cannot alter the already transferred state population.
mdcw_gw_remove_export_reserve_gain = {
	set_variable = { mdcw_gw_source_reserve_removed = FROM.manpower }
	subtract_from_variable = { mdcw_gw_source_reserve_removed = mdcw_gw_source_reserve_before }
	clamp_variable = { var = mdcw_gw_source_reserve_removed min = 0 }
	if = {
		limit = { check_variable = { mdcw_gw_source_reserve_removed > 0 } }
		FROM = {
			set_temp_variable = { mdcw_gw_source_reserve_delta = ROOT.mdcw_gw_source_reserve_removed }
			multiply_temp_variable = { mdcw_gw_source_reserve_delta = -1 }
			add_manpower = mdcw_gw_source_reserve_delta
		}
	}
}

					ROOT = {

mdcw_gw_weekly_update = {
	if = {
		limit = {
			check_variable = { mdcw_gw_remittance_pool_m > 0 }
		}
		# Compute a 0.01% loss by division: avoid a four-decimal multiplier
		# rounding to 1 in native fixed-point arithmetic.
		set_temp_variable = { mdcw_gw_remittance_decay_m = mdcw_gw_remittance_pool_m }
		divide_temp_variable = { mdcw_gw_remittance_decay_m = 10000 }
		subtract_from_variable = { mdcw_gw_remittance_pool_m = mdcw_gw_remittance_decay_m }
		clamp_variable = { var = mdcw_gw_remittance_pool_m min = 0 max = 3000 }
		if = {
			limit = {
				OR = {
					NOT = {
						check_variable = { mdcw_gw_net_emigration_k > 0 }
					}
					check_variable = { mdcw_gw_remittance_pool_m < 0.001 }
				}
			}
			set_variable = { mdcw_gw_remittance_pool_m = 0 }
		}
	}
}
