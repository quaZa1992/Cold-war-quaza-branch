# IMPERIA COLD WAR — Israel V2 territorial administration and settlements.
ISR2_territories_quote_controlled_sites = {
	set_variable = { ISR2_territories_preview_public_price = gdp_total }
	multiply_variable = { ISR2_territories_preview_public_price = 0.015 }
	set_variable = { ISR2_private_required = gdp_total }
	multiply_variable = { ISR2_private_required = 0.005 }
	if = {
		limit = {
			has_country_flag = ISR2_territories_yanai_term
			western_social_democrats_are_in_power = yes
			check_variable = { ISR2_territories_yanai_used < 2 }
			ISR2_private_ready = yes
		}
		set_variable = { ISR2_territories_preview_public_price = gdp_total }
		multiply_variable = { ISR2_territories_preview_public_price = 0.01 }
	}
	set_variable = { ISR2_project_cost = ISR2_territories_preview_public_price }
	set_variable = { ISR2_project_class = 2 }
	set_variable = { ISR2_project_quote = gdp_total }
	set_variable = { ISR2_project_procurement = 0 }
	ISR2_project_support_quote = yes
	set_variable = { ISR2_territories_preview_cost_controlled_sites = ISR2_project_net_cost }
}
ISR2_territories_complete_controlled_sites = {
	if = {
		limit = {
			ISR2_is_scenario = yes
			has_variable = gdp_total
			check_variable = { gdp_total > 0 }
		}
		ISR2_private_recalculate = yes
		ISR2_territories_refresh_prices = yes
		if = {
			limit = {
				ISR2_is_scenario = yes
				has_variable = gdp_total
				check_variable = { gdp_total > 0 }
				controls_state = 208
				has_full_control_of_state = 208
				208 = {
					owner = { exists = yes }
				}
				208 = {
					check_variable = { var = building_level@agriculture_district value = 9 compare = less_than_or_equals }
				}
				can_staff_an_agriculture_district = yes
				208 = {
					check_variable = { var = building_level@infrastructure value = 5 compare = less_than_or_equals }
				}
				208 = {
					free_building_slots = { building = bunker size > 0 province = 11620 }
				}
				check_variable = { ISR2_territories_tiers_208 < 3 }
				NOT = { ISR2_territories_no_new_starts = yes }
				NOT = { has_country_flag = ISR2_territories_cooldown_controlled_sites }
			}
			ISR2_territories_refresh_prices = yes
			set_variable = { ISR2_territories_project_quote = gdp_total }
			set_variable = { ISR2_territories_project_price = gdp_total }
			multiply_variable = { ISR2_territories_project_price = 0.015 }
			set_variable = { ISR2_private_required = ISR2_territories_project_quote }
			multiply_variable = { ISR2_private_required = 0.005 }
			if = {
				limit = {
					has_country_flag = ISR2_territories_yanai_term
					western_social_democrats_are_in_power = yes
					check_variable = { ISR2_territories_yanai_used < 2 }
					ISR2_private_ready = yes
				}
				ISR2_private_commit = yes
				if = {
					limit = {
						check_variable = {
							var = ISR2_private_paid
							value = ISR2_private_required
							compare = greater_than_or_equals
						}
					}
					set_variable = { ISR2_territories_project_price = ISR2_territories_project_quote }
					multiply_variable = { ISR2_territories_project_price = 0.01 }
					add_to_variable = { ISR2_territories_yanai_used = 1 }
					if = {
						limit = {
							check_variable = { ISR2_territories_yanai_used = 1 }
						}
						set_variable = { ISR2_territories_yanai_state_1 = 208 }
						set_variable = { ISR2_territories_yanai_USA_1 = ISR2_private_paid_USA }
						set_variable = { ISR2_territories_yanai_GER_1 = ISR2_private_paid_GER }
						set_variable = { ISR2_territories_yanai_ROM_1 = ISR2_private_paid_ROM }
					}
					if = {
						limit = {
							check_variable = { ISR2_territories_yanai_used = 2 }
						}
						set_variable = { ISR2_territories_yanai_state_2 = 208 }
						set_variable = { ISR2_territories_yanai_USA_2 = ISR2_private_paid_USA }
						set_variable = { ISR2_territories_yanai_GER_2 = ISR2_private_paid_GER }
						set_variable = { ISR2_territories_yanai_ROM_2 = ISR2_private_paid_ROM }
					}
				}
			}
			set_variable = { ISR2_project_cost = ISR2_territories_project_price }
			set_variable = { ISR2_project_class = 2 }
			set_variable = { ISR2_project_quote = gdp_total }
			set_variable = { ISR2_project_procurement = 0 }
			ISR2_project_support_commit = yes
			set_temp_variable = { treasury_change = ISR2_project_net_cost }
			multiply_temp_variable = { treasury_change = -1 }
			modify_treasury_effect = yes
			208 = {
				add_extra_state_shared_building_slots = 3
				add_building_construction = { type = agriculture_district level = 1 instant_build = yes }
				add_building_construction = { type = infrastructure level = 1 instant_build = yes }
				add_building_construction = {
					type = bunker
					level = 1
					instant_build = yes
					province = { id = 11620 }
				}
			}
			add_to_variable = { ISR2_territories_tiers_208 = 1 }
			add_to_variable = { ISR2_territories_nahal_208 = 1 }
			set_country_flag = ISR2_territories_done_controlled_sites
			set_country_flag = { flag = ISR2_territories_cooldown_controlled_sites days = 182 value = 1 }
			ISR2_territories_refresh = yes
			update_money_dirty_variable = yes
		}
		else = {
			uncomplete_national_focus = {
				focus = ISR2_controlled_sites
				uncomplete_children = no
				refund_political_power = no
			}
		}
	}
	else = {
		uncomplete_national_focus = {
			focus = ISR2_controlled_sites
			uncomplete_children = no
			refund_political_power = no
		}
	}
}

		if = {
			limit = {
				ISR2_is_scenario = yes
				has_variable = gdp_total
				check_variable = { gdp_total > 0 }
				NOT = { controls_state = 208 }
				208 = {
					controller = { exists = yes }
				}
				NOT = { has_country_flag = ISR2_territories_foreign_208 }
				check_variable = { ISR2_territories_foreign_count < 2 }
				NOT = { ISR2_territories_no_new_starts = yes }
				NOT = { has_country_flag = ISR2_territories_cooldown_foreign_footholds }
			}
			ISR2_territories_refresh_prices = yes
			set_temp_variable = { treasury_change = ISR2_territories_cost_0_01 }
			multiply_temp_variable = { treasury_change = -1 }
			modify_treasury_effect = yes
			set_variable = { ISR2_territories_foreign_quote_208 = gdp_total }
			set_country_flag = ISR2_territories_foreign_208
			set_country_flag = ISR2_territories_done_foreign_footholds
