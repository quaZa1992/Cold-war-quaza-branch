COLDWAR_SYR_RP_sync_EGY = {
	# A dissolved negotiating government cannot return its draft.
	if = {
		limit = {
			NOT = { country_exists = EGY }
			OR = { has_country_flag = COLDWAR_SYR_RP_pending_EGY has_country_flag = COLDWAR_SYR_RP_counter_pending_EGY }
		}
		clr_country_flag = COLDWAR_SYR_RP_pending_EGY
		clr_country_flag = COLDWAR_SYR_RP_counter_pending_EGY
		set_country_flag = COLDWAR_SYR_RP_declined_EGY
	}
	if = {
		limit = {
			has_country_flag = COLDWAR_SYR_RP_signed_EGY_main
			COLDWAR_SYR_RP_pair_clear_EGY = yes
		}
		if = { limit = { NOT = { has_idea = COLDWAR_SYR_RP_EGY_main_SYR_spirit } } add_ideas = COLDWAR_SYR_RP_EGY_main_SYR_spirit }
		EGY = { if = { limit = { NOT = { has_idea = COLDWAR_SYR_RP_EGY_main_EGY_spirit } } add_ideas = COLDWAR_SYR_RP_EGY_main_EGY_spirit } }
	}
	else = {
		if = { limit = { has_idea = COLDWAR_SYR_RP_EGY_main_SYR_spirit } remove_ideas = COLDWAR_SYR_RP_EGY_main_SYR_spirit }
		if = {
			limit = { country_exists = EGY }
			EGY = { if = { limit = { has_idea = COLDWAR_SYR_RP_EGY_main_EGY_spirit } remove_ideas = COLDWAR_SYR_RP_EGY_main_EGY_spirit } }
		}
	}
	if = {
		limit = {
			has_country_flag = COLDWAR_SYR_RP_signed_EGY_counter
			COLDWAR_SYR_RP_pair_clear_EGY = yes
		}
		if = { limit = { NOT = { has_idea = COLDWAR_SYR_RP_EGY_counter_SYR_spirit } } add_ideas = COLDWAR_SYR_RP_EGY_counter_SYR_spirit }
		EGY = { if = { limit = { NOT = { has_idea = COLDWAR_SYR_RP_EGY_counter_EGY_spirit } } add_ideas = COLDWAR_SYR_RP_EGY_counter_EGY_spirit } }
	}
	else = {
		if = { limit = { has_idea = COLDWAR_SYR_RP_EGY_counter_SYR_spirit } remove_ideas = COLDWAR_SYR_RP_EGY_counter_SYR_spirit }
		if = {
			limit = { country_exists = EGY }
			EGY = { if = { limit = { has_idea = COLDWAR_SYR_RP_EGY_counter_EGY_spirit } remove_ideas = COLDWAR_SYR_RP_EGY_counter_EGY_spirit } }
		}
	}
}

# Native GDP and the treasury use billions. Accumulate small claims in
# hundred-thousandths of a billion and settle whole 0.001-billion units.
# Full terms fund 0.104% of annual GDP / 52; counterterms fund 0.052% / 52.
COLDWAR_SYR_RP_quote_week_SOV = {
	set_variable = { COLDWAR_SYR_RP_invoice_SOV = 0 }
	if = {
		limit = {
			OR = { has_country_flag = COLDWAR_SYR_RP_signed_SOV_main has_country_flag = COLDWAR_SYR_RP_signed_SOV_counter }
			COLDWAR_SYR_RP_pair_clear_SOV = yes
		}
		add_to_variable = { COLDWAR_SYR_RP_accrued_SOV = gdp_total }
		if = {
			limit = { has_country_flag = COLDWAR_SYR_RP_signed_SOV_main }
			add_to_variable = { COLDWAR_SYR_RP_accrued_SOV = gdp_total }
		}
		set_temp_variable = { COLDWAR_SYR_RP_whole = COLDWAR_SYR_RP_accrued_SOV }
		divide_temp_variable = { COLDWAR_SYR_RP_whole = 100 }
		round_temp_variable = COLDWAR_SYR_RP_whole
		multiply_temp_variable = { COLDWAR_SYR_RP_whole = 100 }
		if = {
			limit = { check_variable = { COLDWAR_SYR_RP_whole > COLDWAR_SYR_RP_accrued_SOV } }
			subtract_from_temp_variable = { COLDWAR_SYR_RP_whole = 100 }
		}
		subtract_from_variable = { COLDWAR_SYR_RP_accrued_SOV = COLDWAR_SYR_RP_whole }
		set_variable = { COLDWAR_SYR_RP_invoice_SOV = COLDWAR_SYR_RP_whole }
		divide_variable = { COLDWAR_SYR_RP_invoice_SOV = 100 }
		divide_variable = { COLDWAR_SYR_RP_invoice_SOV = 1000 }
	}
}

			OR = {
				has_country_flag = COLDWAR_SYR_RP_signed_SOV_main
				has_country_flag = COLDWAR_SYR_RP_signed_SOV_counter

		}

	}
