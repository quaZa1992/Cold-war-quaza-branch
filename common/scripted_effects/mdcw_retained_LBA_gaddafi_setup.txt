LBA_1985_setup_gaddafi_systems = {
	if = {
		limit = {
			date < 1990.1.1
			NOT = { has_country_flag = LBA_1985_gaddafi_systems_setup }
		}
		set_country_flag = LBA_1985_gaddafi_systems_setup
		set_country_flag = LBA_1985_moderated_jamahiriya
		set_country_flag = LBA_show_muammar_gaddafi_decisions
		set_country_flag = khamis_gaddafi_underage
		clr_country_flag = dynamic_flag
		clr_country_flag = dynamic_rebel_flag
		set_cosmetic_tag = LBA_1985_JAMAHIRIYA

		set_variable = { gaddafi_family_instability = 5 }
		set_variable = { gaddafi_family_instability_decay = -0.35 }
		set_variable = { gaddafi_family_stability_effect = gaddafi_family_instability }
		multiply_variable = { gaddafi_family_stability_effect = -0.003 }
		if = {
			limit = {
				NOT = { has_dynamic_modifier = { modifier = libya_gaddafi_instability_dynamic_modifier } }
			}
			add_dynamic_modifier = { modifier = libya_gaddafi_instability_dynamic_modifier }
		}

		set_variable = { tripolitania_loyalty = 0.95 }
		set_variable = { cyrenaica_loyalty = 0.82 }
		set_variable = { fezzan_loyalty = 0.88 }
		libya_update_tribalism_modifiers = yes

		set_variable = { lba_oil_nationalisation = 0.72 }
		set_variable = { lba_oil_nationalisation_temp = lba_oil_nationalisation }
		multiply_variable = { lba_oil_nationalisation_temp = -1 }
		set_variable = { oil_nationalisation_productivity_effect = lba_oil_nationalisation_temp }
		set_variable = { oil_nationalisation_export_effect = lba_oil_nationalisation_temp }
		set_variable = { oil_nationalisation_income = 0.014 }
		multiply_variable = { oil_nationalisation_income = lba_oil_nationalisation }
		if = {
			limit = {
				NOT = { has_dynamic_modifier = { modifier = libya_oil_nationalisation_dynamic_modifier } }
			}
			add_dynamic_modifier = { modifier = libya_oil_nationalisation_dynamic_modifier }
		}
		clear_variable = lba_oil_nationalisation_temp

		update_gaddafi_family_ideas = yes
	}
}
