	imp_bond_purchase = {
		visible = {
			always = yes
		}

		selectable = {
			if = {
				limit = {
					ROOT = { has_country_flag = imp_moneytool_cd }
				}
				custom_trigger_tooltip = {
					tooltip = IMP_MONEYTOOL_COOLDOWN
					always = no

			}
			else = {
				custom_trigger_tooltip = {
					tooltip = IMP_MONEYTOOL_COOLDOWN
					always = yes
				}
			}
		}

		can_be_sent = {
			check_variable = { ROOT.bond_size > 0 }
		}

		on_sent_effect = {
			ROOT = { set_country_flag = { flag = imp_moneytool_cd days = 35 } }
		}

		complete_effect = {
			subtract_from_variable = { ROOT.treasury = ROOT.bond_size }
			add_to_variable = { treasury = ROOT.bond_size }
			
			# Remember the money owed.
			add_to_variable = { ROOT.given_bonds@THIS = ROOT.bond_total }
			
			# Reset values
			set_variable = { ROOT.bond_total = 0 }
			set_variable = { ROOT.bond_size = 0 }
			set_variable = { ROOT.bond_interest = 0 }

			ROOT = { clr_country_flag = imp_moneytool_cd }

			# Calculate the bond sizes
			ROOT = { calculate_new_bonds_held = yes }
			calculate_new_bonds_owed = yes
		}

		reject_effect = {
			# Reset values
			set_variable = { ROOT.bond_total = 0 }
			set_variable = { ROOT.bond_size = 0 }
			set_variable = { ROOT.bond_interest = 0 }

			ROOT = { clr_country_flag = imp_moneytool_cd }
		}

	imp_bond_sell = {
		visible = {
			check_variable = { ROOT.given_bonds@THIS > 0 }
		}
		
		selectable = {
			if = {
				limit = {
					ROOT = { has_country_flag = imp_moneytool_cd }
				}
				custom_trigger_tooltip = {
					tooltip = IMP_MONEYTOOL_COOLDOWN
					always = no
				}
			}
			else = {
				custom_trigger_tooltip = {
					tooltip = IMP_MONEYTOOL_COOLDOWN
					always = yes
				}
			}
		}
		
		can_be_sent = {
			check_variable = { ROOT.bond_sell_size > 0 }
		}

		on_sent_effect = {
			ROOT = { set_country_flag = { flag = imp_moneytool_cd days = 35 } }
		}
		
		complete_effect = {
			# Add the money to the treasury
			set_temp_variable = { bond_sale_t = ROOT.bond_sell_size }
			divide_temp_variable = { bond_sale_t = 2 }
			add_to_variable = { ROOT.treasury = bond_sale_t }

			# Is the sell-off big enough to matter? 25% of total debt?
			set_temp_variable = { selloff_p = ROOT.bond_sell_size }
			divide_temp_variable = { selloff_p = debt }
			if = { 
				limit = { check_variable = { selloff_p > 0.25 } }
				set_country_flag = { flag = imp_recent_major_bond_selloff days = 45 value = 1 }
			}
			else_if = {
				limit = { check_variable = { selloff_p > 0.10 } }
				set_country_flag = { flag = imp_recent_minor_bond_selloff days = 45 value = 1 }
			}

			# Add the money to the debt.
			add_to_variable = { debt = ROOT.bond_sell_size }

			# Remove the money from the bonds owed.
			subtract_from_variable = { ROOT.given_bonds@THIS = ROOT.bond_sell_size }

			# Reset values
			set_variable = { ROOT.bond_sell_size = 0 }
			ROOT = { clr_country_flag = imp_moneytool_cd }

			# Calculate the bond sizes
			ROOT = { calculate_new_bonds_held = yes }
			calculate_new_bonds_owed = yes
		}

		reject_effect = {
			# Reset values
			set_variable = { ROOT.bond_sell_size = 0 }
			ROOT = { clr_country_flag = imp_moneytool_cd }
		}
		
		receive_description = IMP_BOND_SELL_RECEIVE_DESC
		accept_title = IMP_BOND_SELL_ACCEPT_TITLE
		accept_description = IMP_BOND_SELL_ACCEPT_DESC
		reject_title = IMP_BOND_SELL_REJECT_TITLE
		reject_description = IMP_BOND_SELL_REJECT_DESC

		send_scripted_gui = imp_bond_sell_scripted_gui
	
		requires_acceptance = no

		ai_desire = {
			base = -10
		}
	}
