# IMPERIA CW — Cold War: imperia state actions gui — scripted interface.
scripted_gui = {
	imp_state_remove_buildings_scripted_gui = {
		context_type = selected_state_context
		window_name = "imp_state_remove_buildings_gui"
		parent_window_token = selected_state_view

		visible = {
			is_owned_by = ROOT
			NOT = { ROOT = { has_country_flag = imp_decomissioning_buildings } }
		}

		effects = { 
			imp_state_remove_shelters_control_click = {
				remove_building = { type = civilian_shelter level = 1 }
				ROOT = {
					set_country_flag = {
						flag = imp_decomissioning_buildings
						days = 30
						value = 1
					}
				}
			}
			imp_state_remove_reactor_control_click = {
				remove_building = { type = nuclear_reactor level = 1 }
				ROOT = {
					set_country_flag = {
						flag = imp_decomissioning_buildings
						days = 30
						value = 1
					}
				}
			}
		}
	}
}
