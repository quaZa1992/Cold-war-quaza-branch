scripted_gui = {
	icwp_compact = {
		context_type = player_context
		window_name = "icwp_compact_window"
		parent_window_name = icwp_anchor
		# No dirty variable here: the card is small and must resolve its texts as soon as the economy tab opens.

		visible = { icwp_player_ready = yes }

		ai_enabled = { always = no }

		properties = {
			icwp_c_flag = { image = "[Root.GetFlag]" }
			icwp_c_medal = { frame = icwp_tierf }
			icwp_c_pill = { frame = icwp_pill_f }
			icwp_c_bar = { frame = icwp_cardbar_f }
		}

		effects = {
			icwp_c_open_click = {
				icwp_register = yes
				clr_country_flag = icw_econ_dialog_open
				set_country_flag = icwp_dialog_open
				if = {
					limit = { NOT = { check_variable = { icwp_tab > 0 } } }
					set_variable = { icwp_tab = 1 }
				}
				icwp_tab_frames = yes
				icwp_refresh = yes
				if = { limit = { check_variable = { icwp_tab = 3 } } icwp_scan_market = yes }
			}
		}
	}

	icwp_dialog = {
		context_type = player_context
		window_name = "icwp_dialog_window"
		parent_window_name = icwp_dialog_anchor

		visible = {
			icwp_player_ready = yes
			has_country_flag = icwp_dialog_open
			has_country_flag = icwp_participant
			check_variable = { MD_tradeview_tab_economy_state = 1 }
			NOT = { has_country_flag = icw_econ_dialog_open }
		}

		ai_enabled = { always = no }

		properties = {
			icwp_d_flag2 = { image = "[Root.GetFlag]" }
			icwp_d_medal = { frame = icwp_tierf }
			icwp_d_bar = { frame = icwp_headbar_f }
			icwp_tab1 = { frame = icwp_tab1f }
			icwp_tab2 = { frame = icwp_tab2f }
			icwp_tab3 = { frame = icwp_tab3f }
			icwp_tab4 = { frame = icwp_tab4f }
		}

		effects = {
			icwp_d_close_click = {
				clr_country_flag = icwp_dialog_open
				icwp_touch = yes
			}
			icwp_tab1_click = {
				set_variable = { icwp_tab = 1 }

			}
			icwp_tab2_click = {
				set_variable = { icwp_tab = 2 }
				icwp_tab_frames = yes
				icwp_refresh = yes
			}
			icwp_tab3_click = {
				set_variable = { icwp_tab = 3 }

			}
			icwp_tab4_click = {
				set_variable = { icwp_tab = 4 }
				icwp_tab_frames = yes
				icwp_refresh = yes
			}
		}
	}

	icwp_abroad = {
		context_type = player_context
		parent_scripted_gui = icwp_dialog
		window_name = "icwp_abroad_window"
		dirty = icwp_gui_rev

		visible = {
			has_country_flag = icwp_dialog_open
			check_variable = { icwp_tab = 1 }
		}

		ai_enabled = { always = no }

		dynamic_lists = {
			icwp_hosts_grid = {
				array = icwp_hosts
				entry_container = "icwp_host_row"
			}
		}

		triggers = {
			icwp_hr_end_click_enabled = { NOT = { has_country_flag = icwp_exit_pending } }
		}

		properties = {
			icwp_hr_flag = { image = "[?v.GetFlag]" }
			icwp_hr_status = { frame = icwp_dh_s^i }
			icwp_hr_lbl = { frame = icwp_dh_lf^i }
			icwp_hr_bar = { frame = icwp_dh_bf^i }
		}

		effects = {
			# We are the client, v is the producer.
			icwp_hr_wind_click = {
				set_variable = { icwp_act = v }
				var:icwp_act = {
					if = {
						limit = { ROOT = { is_in_array = { array = icwp_hosts value = PREV } check_variable = { icwp_s@PREV = 1 } } }
						ROOT = { icwp_give_notice = yes }
						country_event = { id = icwp.7 }
					}
				}
			}
			icwp_hr_end_click = {
				set_variable = { icwp_ev_p = v }
				set_variable = { icwp_ev_role = 1 }
				set_country_flag = icwp_exit_pending
				icwp_prepare_exit = yes
				country_event = { id = icwp.5 }
			}
		}
	}

	icwp_home = {
		context_type = player_context
		parent_scripted_gui = icwp_dialog
		window_name = "icwp_home_window"

			icwp_cr_end_click_enabled = { NOT = { has_country_flag = icwp_exit_pending } }
		}

		properties = {
			icwp_cr_flag = { image = "[?v.GetFlag]" }
			icwp_cr_status = { frame = icwp_dc_s^i }
			icwp_cr_bar = { frame = icwp_dc_bf^i }
		}

		effects = {
			# We are the producer, v is the client.
			icwp_cr_susp_click = {
				set_variable = { icwp_act = v }
				var:icwp_act = {
					if = {
						limit = { is_in_array = { array = icwp_hosts value = ROOT } check_variable = { icwp_s@ROOT = 1 } NOT = { check_variable = { icwp_cd@ROOT > 0 } } }
						icwp_suspend = yes
						country_event = { id = icwp.6 }
					}
				}
			}
			icwp_cr_wind_click = {
				set_variable = { icwp_act = v }
				var:icwp_act = {
					if = {
						limit = { is_in_array = { array = icwp_hosts value = ROOT } check_variable = { icwp_s@ROOT = 1 } }
						icwp_give_notice = yes
						country_event = { id = icwp.7 }
					}
				}
			}
			icwp_cr_end_click = {
				set_variable = { icwp_ev_p = v }
				set_variable = { icwp_ev_role = 2 }
				set_country_flag = icwp_exit_pending
				icwp_prepare_exit = yes
				country_event = { id = icwp.5 }
			}

	icwp_market = {
		context_type = player_context
		parent_scripted_gui = icwp_dialog
		window_name = "icwp_market_window"
		dirty = icwp_gui_rev

		visible = {
			has_country_flag = icwp_dialog_open
			check_variable = { icwp_tab = 3 }
		}

		ai_enabled = { always = no }

		dynamic_lists = {
			icwp_mk_grid = {
				array = icwp_mk
				entry_container = "icwp_mk_row"
			}
		}

		properties = {
			icwp_mk_rowbtn = { frame = icwp_mk_rowf^i }
			icwp_mk_flag = { image = "[?v.GetFlag]" }
			icwp_mk_medal = { frame = icwp_mk_tierf^i }
			icwp_mk_ai = { frame = icwp_mk_ai^i }
			icwp_mk_reason = { frame = icwp_mk_ok^i }
			icwp_m_flag = { image = "[?icwp_sel_c.GetFlag]" }
			icwp_m_reason = { frame = icwp_pv_f }
			icwp_pm1 = { frame = icwp_pm1f }
			icwp_pm2 = { frame = icwp_pm2f }
			icwp_pm3 = { frame = icwp_pm3f }
			icwp_pm4 = { frame = icwp_pm4f }
		}

		triggers = {
			icwp_send_click_enabled = {
				check_variable = { icwp_sel > 0 }
				check_variable = { icwp_pv_ok = 1 }
			}
			icwp_less_click_enabled = { check_variable = { icwp_offer_u > 1 } }
			# The saving of the selected producer and how many producers we use set the limit (see icwp_quote).
			icwp_more_click_enabled = {
				check_variable = { icwp_offer_u < 22 }
				OR = {
					NOT = { check_variable = { icwp_sel > 0 } }
					check_variable = { icwp_offer_u < icwp_pv_cap }
				}
			}
		}

		effects = {
			icwp_mk_rowbtn_click = {
				set_variable = { icwp_sel = i }
				add_to_variable = { icwp_sel = 1 }
				set_variable = { icwp_sel_c = v }
				icwp_row_frames = yes
				icwp_offer_preview = yes
				if = {
					limit = { check_variable = { icwp_offer_u > icwp_pv_cap } check_variable = { icwp_pv_cap > 0 } }
					set_variable = { icwp_offer_u = icwp_pv_cap }
					icwp_offer_preview = yes
				}
				icwp_touch = yes
			}
			icwp_less_click = {
				subtract_from_variable = { icwp_offer_u = 1 }
				clamp_variable = { var = icwp_offer_u min = 1 max = 22 }
				icwp_offer_preview = yes
				icwp_touch = yes
			}
			icwp_more_click = {
				add_to_variable = { icwp_offer_u = 1 }
				clamp_variable = { var = icwp_offer_u min = 1 max = 22 }
				icwp_offer_preview = yes
				icwp_touch = yes
			}
			icwp_pm1_click = { set_variable = { icwp_offer_pm = -0.10 } icwp_offer_preview = yes icwp_touch = yes }
			icwp_pm2_click = { set_variable = { icwp_offer_pm = 0 } icwp_offer_preview = yes icwp_touch = yes }
			icwp_pm3_click = { set_variable = { icwp_offer_pm = 0.10 } icwp_offer_preview = yes icwp_touch = yes }
			icwp_pm4_click = { set_variable = { icwp_offer_pm = 0.20 } icwp_offer_preview = yes icwp_touch = yes }
			icwp_m_refresh_click = {
				icwp_scan_market = yes
			}
			icwp_send_click = {
				icwp_send_offer = yes
			}
		}
	}
