# IMPERIA COLD WAR — Cold War: nuclear readiness and crisis diplomacy.
# IMPERIA COLD WAR update 20.11.2026: on_daily removed; weekly upkeep and capitulation hook. Countdowns run through scheduled events (cw_nuclear.91 and .93).
on_actions = {
	on_weekly = {
		effect = {
			if = {
				limit = {
					COLDWAR_nr_enabled = yes
					OR = { has_country_flag = COLDWAR_nr_initialized COLDWAR_nr_nuclear_power = yes }
				}
				COLDWAR_nr_weekly = yes
			}
		}
	}
	on_capitulation = {
		effect = {
			if = {
				limit = {
					COLDWAR_nr_enabled = yes
					has_country_flag = COLDWAR_nr_initialized
				}
				COLDWAR_nr_capitulated = yes
			}
		}
	}
}
