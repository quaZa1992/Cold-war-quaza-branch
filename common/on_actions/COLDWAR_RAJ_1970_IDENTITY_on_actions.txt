# IMPERIA COLD WAR — India's 1970 programmes.
on_actions = {
	on_daily_RAJ = {
		effect = {
			if = {
				limit = { original_tag = RAJ IMP_is_1970_scenario = yes has_country_flag = COLDWAR_RAJ_1970_initialised }
				# Fallback for a government change that did not itself call set_leader.
				if = {
					limit = {
						date < 1998.1.1
						check_variable = { ruling_party = 18 }
						OR = { NOT = { has_variable = COLDWAR_RAJ_PEOPLE_dravidian_leader } check_variable = { COLDWAR_RAJ_PEOPLE_dravidian_leader = 0 } }
					}
					COLDWAR_RAJ_1970_initialise_missing_party_identity = yes
					set_leader = yes
				}
				else = { COLDWAR_RAJ_1970_initialise_missing_party_identity = yes }
			}
		}
	}
}
