# IMPERIA COLD WAR — NATO faction rules for existing 1970 saves.
on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { IMP_is_1970_scenario = yes }
				every_country = {
					limit = {
						is_faction_leader = yes
						has_faction_template = faction_template_1985_nato
					}
					set_faction_rule = call_to_war_rule_nato_article_five
					set_faction_rule = war_declaration_rule_independent_countries_only
				}
			}
		}
	}
}
