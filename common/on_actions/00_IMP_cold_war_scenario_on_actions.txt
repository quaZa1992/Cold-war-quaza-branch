# IMPERIA CW — Cold War: 00 IMP scenario on actions — event hooks.
on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { USA = { IMP_is_1970_scenario = yes } }
				set_global_flag = IMP_SCENARIO_1970
			}
			else_if = {
				limit = { USA = { IMP_is_1985_scenario = yes } }
				set_global_flag = IMP_SCENARIO_1985
			}
		}
	}
}
