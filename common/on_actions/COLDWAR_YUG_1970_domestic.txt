# IMPERIA COLD WAR — Yugoslav federal programmes in the January 1970 scenario.
on_actions = {
	on_government_change = {
		effect = {
			if = {
				limit = {
					COLDWAR_YUG_1970_active = yes
				}
				COLDWAR_YUG_1970_refresh_programmes = yes
			}
		}
	}
	on_ruling_party_change = {
		effect = {
			if = {
				limit = {
					COLDWAR_YUG_1970_active = yes
				}
				COLDWAR_YUG_1970_refresh_programmes = yes
			}
		}
	}
	on_startup = {
		effect = {
			if = {
				limit = {
					country_exists = SER
				}
				SER = {
					COLDWAR_YUG_1970_correct_starting_reactor = yes
				}
			}
		}
	}
	on_weekly_SER = {
		effect = {
			if = {
				limit = {
					COLDWAR_YUG_1970_active = yes
				}
				COLDWAR_YUG_1970_refresh_programmes = yes
			}
		}
	}
}
