# IMPERIA COLD WAR — Apply PMC access to new campaigns and existing saves without a recurring callback.
on_actions = {
	on_startup = {
		effect = {
clr_global_flag = ICW_startup_dispatch_pmc_access
every_country = {
 limit = { NOT = { has_global_flag = ICW_startup_dispatch_pmc_access } }
 set_global_flag = ICW_startup_dispatch_pmc_access
if = {
				limit = { ICW_pmc_system_active = no }
				every_country = { ICW_pmc_cancel_pending_recruitment = yes }
				international_systems_force_update = yes
			}
}
clr_global_flag = ICW_startup_dispatch_pmc_access
}
	}
}
