# IMPERIA COLD WAR — Cuba 1970 on-action hooks.
on_actions = {
	on_startup = {
		effect = {
			if = {
			limit = { country_exists = CUB }
			CUB = {
				if = { limit = { COLDWAR_CUB_is_1970 = yes } COLDWAR_CUB_initialize = yes }
			}
			}
		}
	}
	# COLDWAR | hand patch 07.11.2026 (performance): runs only for the tag it is meant for (on_daily_TAG instead of a tag check in every country).
	on_daily_CUB = {
		effect = {
			if = {
				limit = { COLDWAR_CUB_is_1970 = yes }
				COLDWAR_CUB_daily_update = yes
			}
		}
	}
	on_government_change = {
		effect = {
			if = {
				limit = { COLDWAR_CUB_is_1970 = yes }
				COLDWAR_CUB_initialize = yes
				COLDWAR_CUB_sync_programmes = yes
			}
		}
	}
	on_ruling_party_change = {
		effect = {
			if = {
				limit = { COLDWAR_CUB_is_1970 = yes }
				COLDWAR_CUB_initialize = yes
				COLDWAR_CUB_sync_programmes = yes
			}
		}
	}
}
