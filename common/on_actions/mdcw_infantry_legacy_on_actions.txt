# Narrow startup integration for newly inserted infantry predecessor nodes.
on_actions = {
    on_startup = {
        effect = {
            if = {
                limit = { date > 1969.12.31 }
                every_country = {
                    if = {
                        limit = { date < 1970.1.2 }
                        mdcw_apply_1970_infantry_research = yes
                    }
                    else_if = {
                        limit = { date > 1984.12.31 }
                        mdcw_inherit_early_infantry_technologies = yes
                    }
                }
            }
        }
    }
}
