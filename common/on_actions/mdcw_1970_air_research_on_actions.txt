# IMPERIA CW — Cold War: 1970 air research on actions — event hooks.
# COLDWAR jetgen | Applies the 1970 air research snapshot and the jet generation prerequisite inheritance once at game start.
on_actions = {
	on_startup = {
		effect = {
			every_country = {
				mdcw_apply_1970_air_research = yes
				mdcw_inherit_jet_generation_research = yes
			}
		}
	}
}
