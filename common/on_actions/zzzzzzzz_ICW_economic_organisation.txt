# IMPERIA COLD WAR — Economic organisation startup and saved reform clocks
on_actions = {
 on_startup = { effect = { every_country = { limit = { OR = { NOT = { has_country_flag = icw_econ_initialized } NOT = { has_country_flag = icw_econ_sale_initialized } } } icw_econ_initialize = yes } } }
 on_daily = { effect = { icw_econ_daily = yes } }
 on_weekly = { effect = { if = { limit = { OR = { NOT = { has_country_flag = icw_econ_initialized } NOT = { has_country_flag = icw_econ_sale_initialized } } } icw_econ_initialize = yes } } }
}
