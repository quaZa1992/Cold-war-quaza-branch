# IMPERIA COLD WAR — France November voluntary cooperation.
on_actions = {
	on_startup = {
		effect = {
			FRA = { COLDWAR_FRA2_NOV_COOP_sync_all = yes }
		}
	}

	# COLDWAR | hand patch 07.11.2026 (performance): runs only for the tag it is meant for (on_daily_TAG instead of a tag check in every country).
	on_daily_FRA = {
		effect = {
			COLDWAR_FRA2_NOV_COOP_sync_all = yes
		}
	}
	on_daily = {
		effect = {
			if = {
				limit = {
					NOT = { country_exists = FRA }
					OR = {
						has_country_flag = COLDWAR_FRA2_NOV_COOP_signed_advisory
						has_country_flag = COLDWAR_FRA2_NOV_COOP_signed_commercial
						has_country_flag = COLDWAR_FRA2_NOV_COOP_signed_security
						has_country_flag = COLDWAR_FRA2_NOV_COOP_signed_reconstruction
						has_country_flag = COLDWAR_FRA2_NOV_COOP_signed_francophone
						has_country_flag = COLDWAR_FRA2_NOV_COOP_signed_opening
						has_country_flag = COLDWAR_FRA2_NOV_COOP_signed_council
						has_country_flag = COLDWAR_FRA2_NOV_COOP_signed_recognition
					}
				}
				COLDWAR_FRA2_NOV_COOP_sync_host = yes
			}
		}
	}
}
