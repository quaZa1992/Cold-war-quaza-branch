on_actions = {
	on_weekly_KOR = {
		# IMPERIA CW — Cold War: Scenario lifecycle and country updates.
		# COLDWAR: removed MD 2000s random_events pulse (on_weekly_KOR)
	}

	on_weekly_NKO = {
		# IMPERIA CW — Cold War: Scenario lifecycle and country updates.
		# COLDWAR: removed MD 2000s random_events pulse (on_weekly_NKO)
	}
	on_annex = {
		effect = {
			# IMPERIA CW — Cold War: Scenario lifecycle and country updates.
			# COLDWAR: MD 2000s content, only for MD start dates
			if = {
				limit = { has_start_date > 1999.12.31 }
				if = {
					limit = {
						OR = {
							AND = {
								ROOT = { tag = NKO }
								FROM = { tag = KOR }
							}
							AND = {
								ROOT = { tag = KOR }
								FROM = { tag = NKO }
							}
						}
					}
					if = {
						limit = { country_exists = KOR }
						KOR = {
							set_country_flag = korea_peninsula_reunited
							add_ideas = KOR_reunification_shocks
							add_ideas = KOR_divided_military
							add_ideas = KOR_mass_migration
							add_ideas = KOR_fragmented_governance
							add_ideas = KOR_uneven_economy1
							news_event = { id = korea.50 days = 1 }
						}
					}
					else_if = {
						limit = { country_exists = NKO }
						NKO = {
							set_country_flag = korea_peninsula_reunited
							add_ideas = KOR_reunification_shocks
							add_ideas = KOR_divided_military
							add_ideas = KOR_mass_migration
							add_ideas = KOR_fragmented_governance
							add_ideas = KOR_uneven_economy1
							news_event = { id = korea.50 days = 1 }
						}
						JAP = { add_opinion_modifier = { target = NKO modifier = NKO_forced_reunification } }
						TAI = { add_opinion_modifier = { target = NKO modifier = NKO_forced_reunification } }
						USA = { add_opinion_modifier = { target = NKO modifier = NKO_forced_reunification } }
						PHI = { add_opinion_modifier = { target = NKO modifier = NKO_forced_reunification } }
					}
				}
			}
		}
	}
}
