# IMPERIA CW — Cold War: 1970s ground weapon modules — event hooks.
# COLDWAR groundmod | Gives the 1970s step to countries that start with the MD 1985 successor.
# Later starts only: the 1970 scenario gets its research from the history files.
on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { NOT = { any_country = { IMP_is_1970_scenario = yes } } } # COLDWAR | hand patch 07.11.2026: on_startup has no scope, the check needs a country
				every_country = {
					mdcw_inherit_ground_modules_1970s = yes
				}
			}
		}
	}
}
