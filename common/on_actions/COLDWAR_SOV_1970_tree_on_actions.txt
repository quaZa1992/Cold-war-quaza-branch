# IMPERIA COLD WAR — Soviet Union 1970 v4: start and monthly hooks of the rework.
# IMPERIA COLD WAR update 21.11.2026: on_startup has no scope; the scenario trigger now runs inside SOV (it failed with Invalid Scope: None, so the start hook never passed).
on_actions = {
	on_startup = {
		effect = {
			if = { limit = { country_exists = SOV SOV = { IMP_is_1970_scenario = yes } } SOV = { COLDWAR_SOVR_initialize = yes } }
		}
	}
	on_monthly_SOV = {
		effect = {
			if = { limit = { has_country_flag = COLDWAR_SOVR_initialized } COLDWAR_SOVR_monthly = yes }
		}
	}
}
