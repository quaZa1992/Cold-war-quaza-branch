# IMPERIA COLD WAR — one monthly intelligence sweep; no daily or weekly checks.
on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = { COLDWAR_NI_relevant_country = yes }
				COLDWAR_NI_initialize_country = yes
			}
		}
	}
	on_monthly = {
		effect = {
			# on_monthly is country-scoped. The first country builds one shared cache.
			COLDWAR_NI_refresh_observer_cache = yes
			if = {
				limit = { COLDWAR_NI_relevant_country = yes }
				COLDWAR_NI_monthly_country = yes
			}
		}
	}
}
