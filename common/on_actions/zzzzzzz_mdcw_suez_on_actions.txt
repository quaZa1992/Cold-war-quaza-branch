# IMPERIA COLD WAR — Native country scope for global startup initialization.
# IMPERIA CW — Cold War: zzzzzzz suez on actions — event hooks.
on_actions = {
	on_startup = {
		effect = {
clr_global_flag = ICW_startup_dispatch_suez
# IMPERIA COLD WAR — load time (01.11.2026): light start path, see mdcw_suez_update_world.
set_global_flag = mdcw_suez_startup_light
every_country = {
 limit = { NOT = { has_global_flag = ICW_startup_dispatch_suez } }
 set_global_flag = ICW_startup_dispatch_suez
update_canal_blockage_flags = yes
mdcw_suez_update_world = yes
}
clr_global_flag = ICW_startup_dispatch_suez
clr_global_flag = mdcw_suez_startup_light
}
	}
	on_weekly = {
		effect = {
			# MD_on_actions refreshes the physical flag in its global weekly pass.
			# Also bootstraps old saves and updates current Globalism/geography.
			mdcw_suez_update_country = yes
		}
	}
	on_release_as_free = {
		effect = { mdcw_suez_update_country = yes }
	}
	on_release_as_puppet = {
		effect = { mdcw_suez_update_country = yes }
	}
	on_civil_war_started = {
		effect = {
			mdcw_suez_update_country = yes
			FROM = { mdcw_suez_update_country = yes }
		}
	}
}
