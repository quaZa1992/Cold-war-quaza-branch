# IMPERIA COLD WAR — Israel V2 scenario and legacy integration.
on_actions = {
	on_startup = {
		effect = {
			ISR = {
				if = {
					limit = { exists = yes ISR2_is_scenario = yes }
					ISR2_initialize = yes
				}
			}
		}
	}
	# COLDWAR | hand patch 07.11.2026 (performance): runs only for the tag it is meant for (on_daily_TAG instead of a tag check in every country).
	on_daily_ISR = {
		effect = {
			if = {
				limit = { ISR2_is_scenario = yes }
				ISR2_daily = yes
			}
		}
	}
	on_government_change = {
		effect = {
			if = {
				limit = { ISR2_is_scenario = yes }
				ISR2_refresh = yes
			}
		}
	}
}
