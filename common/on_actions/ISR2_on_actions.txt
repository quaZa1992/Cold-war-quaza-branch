# IMPERIA COLD WAR — Israel V2 scenario and legacy integration.
on_actions = {
	on_startup = {
		effect = {
			ISR = {
				if = {
					limit = { exists = yes ISR2_is_scenario = yes }
					ISR2_initialize = yes
				}
			}
		}
	}
	on_daily = {
		effect = {
			if = {
				limit = { ISR2_is_scenario = yes }
				ISR2_daily = yes
			}
		}
	}
	on_government_change = {
		effect = {
			if = {
				limit = { ISR2_is_scenario = yes }
				ISR2_refresh = yes
			}
		}
	}
}
