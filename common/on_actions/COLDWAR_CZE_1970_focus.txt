# IMPERIA COLD WAR — Czechoslovakia 1970 national programmes.

	on_startup = {
		effect = {
			CZE = { COLDWAR_CZE_1970_init = yes }
		}
	}
	on_daily_CZE = {
		effect = {
			if = { limit = { COLDWAR_CZE_1970_is_scenario = yes tag = CZE } COLDWAR_CZE_1970_projects_refresh = yes COLDWAR_CZE_1970_bilateral_refresh = yes COLDWAR_CZE_1970_refresh_politics = yes }
		}
	}
	on_weekly_CZE = {
		effect = {
			if = { limit = { COLDWAR_CZE_1970_is_scenario = yes tag = CZE } COLDWAR_CZE_1970_politics_weekly = yes COLDWAR_CZE_1970_projects_weekly = yes COLDWAR_CZE_1970_bilateral_weekly = yes }
		}
	}
	on_government_change = {
		effect = {
			if = { limit = { COLDWAR_CZE_1970_is_scenario = yes tag = CZE } COLDWAR_CZE_1970_projects_refresh = yes COLDWAR_CZE_1970_bilateral_refresh = yes COLDWAR_CZE_1970_refresh_politics = yes }
		}
	}
	on_ruling_party_change = {
		effect = {
			if = { limit = { COLDWAR_CZE_1970_is_scenario = yes tag = CZE } COLDWAR_CZE_1970_projects_refresh = yes COLDWAR_CZE_1970_bilateral_refresh = yes COLDWAR_CZE_1970_refresh_politics = yes }
		}
	}
}
