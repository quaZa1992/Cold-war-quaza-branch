# IMPERIA CW — Cold War: Integrated zzzzzz imp global covid on actions — event hooks.
on_actions = {
    on_monthly = {
        effect = {
            if = {
                limit = { 
                    has_idea = imp_COVID_vaccine_producer
                    NOT = { has_global_flag = imp_covid_is_done } 
                }

                add_to_variable = { imp_vaccine_storage = imp_vaccine_monthly }
            }
        }
    }

    on_monthly_LIC = {
        # COVID
        effect = { 
            if = { 
                limit = { 
                    date < 2020.1.1
                    date > 2019.11.20
                    has_global_flag = imp_covid_this_game
                }
                news_event = impcovid.1
                set_global_flag = imp_covid_has_begun
                news_event = { id = impcovid.2 days = 12 }
                country_event = { id = impcovid.99 days = 7 }

                CHI = {
                    news_event = { id = impcovid.3 days = 4 }
                }
            }
        }
    }
}
