# IMPERIA CW — Cold War: 99 ddr currency on actions — event hooks.
on_actions = {
	on_weekly = {
		effect = {
			# on_startup does not run again when an existing save is loaded.
			# The helper is guarded by country, historical dates and a one-time flag.
			if = {
				limit = {
					tag = DDR
					NOT = { has_country_flag = mdcw_ddr_mark_initialized }
				}
				mdcw_initialize_ddr_currency = yes
			}
		}
	}
}
