# IMPERIA COLD WAR — USA 1970 national programmes and negotiated partnerships.
on_actions = {
	on_startup = {
		effect = {
			USA = { if = { limit = { tag = USA IMP_is_1970_scenario = yes } COLDWAR_USA_1970_economy_migrate = yes COLDWAR_USA_expansion_reconcile = yes } }
		}
	}
	on_daily_USA = {
		effect = {
			if = { limit = { tag = USA IMP_is_1970_scenario = yes } COLDWAR_USA_1970_economy_migrate = yes COLDWAR_USA_expansion_reconcile = yes }
		}
	}
	on_weekly_USA = {
		effect = {
			if = { limit = { tag = USA IMP_is_1970_scenario = yes } COLDWAR_USA_expansion_reconcile = yes }
		}
	}
	on_war_relation_added = {
		effect = {
			if = { limit = { country_exists = USA USA = { IMP_is_1970_scenario = yes has_country_flag = COLDWAR_USA_expansion_initialized } } USA = { COLDWAR_USA_early_reconcile_all = yes COLDWAR_USA_agreements_reconcile_all = yes } }
		}
	}
}
