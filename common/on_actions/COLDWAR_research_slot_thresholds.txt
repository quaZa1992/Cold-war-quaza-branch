# IMPERIA COLD WAR — Country-scoped research threshold initialization.
# IMPERIA CW — Cold War: existing development thresholds and revised GDP progression.
on_actions = {
	on_startup = {
		effect = {
			set_variable = { GLOBAL.gdp_c_slot^2 = 10.000 }
			set_variable = { GLOBAL.gdp_c_slot^3 = 30.000 }
			set_variable = { GLOBAL.gdp_c_slot^4 = 40.000 }
clr_global_flag = ICW_startup_dispatch_research_thresholds
every_country = {
 limit = { NOT = { has_global_flag = ICW_startup_dispatch_research_thresholds } }
 set_global_flag = ICW_startup_dispatch_research_thresholds
COLDWAR_configure_research_slot_thresholds = yes
}
clr_global_flag = ICW_startup_dispatch_research_thresholds
		}
	}
}
