on_actions = {
	on_daily = {
		effect = {
			if = {
				limit = { IMP_is_1985_scenario = yes }
				SOV = {
					if = {
						limit = {
							has_country_flag = imp_1985_moscow_leverage_controller
							IMP_1985_eb_should_end_minigame = yes
						}
						IMP_1985_eb_end_moscow_leverage_controller = yes
					}
				}

				every_country = {
					limit = {
						has_country_flag = imp_1985_eastern_bloc_member
						IMP_1985_eb_should_end_minigame = yes
					}
					IMP_1985_eb_end_minigame = yes
				}
			}
		}
	}

	
}
