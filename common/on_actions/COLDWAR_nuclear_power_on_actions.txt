# COLDWAR | Kernenergie-Umbau – one monthly tick per country, no world loops
on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = { IMP_is_1970_scenario = yes }
				COLDWAR_nuc_startup_country = yes
			}
		}
	}

	on_monthly = {
		effect = {
			if = {
				limit = { IMP_is_1970_scenario = yes }
				COLDWAR_nuc_monthly = yes
			}
		}
	}
}
