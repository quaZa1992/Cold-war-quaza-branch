# IMPERIA COLD WAR — Brazil 1970 national programmes.
on_actions = {
	on_startup = {
		effect = {
			BRA = {
				COLDWAR_BRA_initialize = yes
			}
		}
	}
	on_daily = {
		effect = {
			if = {
				limit = {
					COLDWAR_BRA_campaign = yes
				}
				COLDWAR_BRA_initialize = yes
				COLDWAR_BRA_open_organisation_stages = yes
				COLDWAR_BRA_reconcile_politics = yes
				COLDWAR_BRA_reconcile_focus_receipts = yes
				COLDWAR_BRA_sync_programmes = yes
				COLDWAR_BRA_sync_agreements = yes
			}
			COLDWAR_BRA_cleanup_orphaned_agreements = yes
		}
	}
	on_monthly = {
		effect = {
			if = {
				limit = {
					COLDWAR_BRA_campaign = yes
				}
				COLDWAR_BRA_political_pressure_pulse = yes
				COLDWAR_BRA_1970_monthly_politics_reconcile = yes
				if = {
					limit = {
						has_country_flag = COLDWAR_BRA_1970_party_reform_open
						NOT = {
							has_country_flag = COLDWAR_BRA_event_sent_party_split
						}
					}
					COLDWAR_BRA_fire_party_split = yes
				}
				COLDWAR_BRA_1970_mark_prn_leader_available = yes
				if = {
					limit = {
						COLDWAR_BRA_1970_party_leader_route_supported = yes
					}
					COLDWAR_BRA_1970_install_party_leader = yes
				}
			}
		}
	}
	on_government_change = {
		effect = {
			if = {
				limit = {
					COLDWAR_BRA_campaign = yes
				}
				COLDWAR_BRA_reconcile_politics = yes
				COLDWAR_BRA_sync_programmes = yes
			}
		}
	}
	on_ruling_party_change = {
		effect = {
			if = {
				limit = {
					COLDWAR_BRA_campaign = yes
				}
				COLDWAR_BRA_reconcile_politics = yes
				COLDWAR_BRA_sync_programmes = yes
			}
		}
	}
}
