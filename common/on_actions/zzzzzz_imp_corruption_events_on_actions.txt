# IMPERIA CW — Cold War: Integrated zzzzzz imp corruption events on actions — event hooks.
on_actions = {
    on_monthly = {
        effect = {
            ### CORRUPTION EVENTS
            if = { 
                limit = { 
                    check_variable = { imp_corruption_event_cd > 8 }
                }

                random_list = { 
                    10 = {
                        set_variable = { imp_corruption_event_cd = 0}

                        if = { 
                            limit = { 
                                has_dynamic_modifier = { modifier = imp_eco_1_dynamic_modifier }
                            }
                            corruption_event_1 = yes
                        }
                        else_if = { 
                            limit = { 
                                has_dynamic_modifier = { modifier = imp_eco_2_dynamic_modifier }
                            }
                            corruption_event_2 = yes
                        }
                        else_if = { 
                            limit = { 
                                has_dynamic_modifier = { modifier = imp_eco_3_dynamic_modifier }
                            }
                            corruption_event_3 = yes
                        }
                    }
                    90 = {
                        add_to_variable = { imp_corruption_event_cd = 1 }
                    }
                }
            }
            else = {
                add_to_variable = { imp_corruption_event_cd = 1 }
            }
        }
    }
}
