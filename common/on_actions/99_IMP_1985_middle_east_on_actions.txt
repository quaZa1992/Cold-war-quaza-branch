# IMPERIA CW — Cold War: Integrated 99 IMP 1985 middle east on actions — event hooks.
on_actions = {
	

	on_daily_IRQ = {
		effect = {
			if = {
				limit = { IMP_is_1985_scenario = yes }
				IMP_1985_cleanup_iran_iraq_war_spirits = yes
				IRQ_1985_check_kurdish_war_outcome = yes
				IRQ_1985_start_postwar_crisis = yes
				IRQ_1985_check_kuwait_annexation = yes
			}
		}
	}

	on_daily_PER = {
		effect = {
			if = {
				limit = { IMP_is_1985_scenario = yes }
				IMP_1985_cleanup_iran_iraq_war_spirits = yes
			}
		}
	}

	on_monthly = {
		effect = {
			if = {
				limit = { IMP_is_1985_scenario = yes }
				every_country = {
					IMP_1985_cleanup_iran_iraq_war_spirits = yes
					IRQ_1985_check_kurdish_war_outcome = yes
					IRQ_1985_start_postwar_crisis = yes
					IRQ_1985_check_kuwait_annexation = yes
				}
			}
		}
	}
}
