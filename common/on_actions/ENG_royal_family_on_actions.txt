on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = {
					original_tag = ENG
					date < 1990.1.1
				}
				ENG_setup_royal_family_mechanic = yes
			}
		}
	}

	on_monthly = {
		effect = {
			every_country = {
				limit = {
					original_tag = ENG
					has_country_flag = ENG_royal_family_mechanic_active
				}
				ENG_royal_family_monthly_update = yes
			}
		}
	}
}
