# IMPERIA COLD WAR — Recurring action performance integration.
# IMPERIA CW — Cold War: Integrated ENG royal family on actions — event hooks.
on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = {
					original_tag = ENG
					date < 1990.1.1
				}
				ENG_setup_royal_family_mechanic = yes
			}
		}
	}

	on_monthly = {
		effect = {
			# COLDWAR | hand patch 07.11.2026 (performance): on_monthly already runs once per country; the every_country inside made it run about 200 times per country.
			if = {
				limit = {
					original_tag = ENG
					has_country_flag = ENG_royal_family_mechanic_active
				}
				ENG_royal_family_monthly_update = yes
			}
		}
	}
}
