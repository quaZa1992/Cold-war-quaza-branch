# IMPERIA CW — Cold War: Integrated ENG 1985 on actions — event hooks.
on_actions = {
	

	on_monthly = {
		effect = {
				every_country = {
				limit = {
					original_tag = ENG
					IMP_is_1985_scenario = yes
					date < 1995.1.1
				}
				ENG_1985_monthly_update = yes
			}
		}
	}

	on_ruling_party_change = {
		effect = {
			if = {
				limit = {
					original_tag = ENG
					IMP_is_1985_scenario = yes
					date < 1995.1.1
				}
				ENG_1985_validate_governing_route = yes
			}
		}
	}
}
