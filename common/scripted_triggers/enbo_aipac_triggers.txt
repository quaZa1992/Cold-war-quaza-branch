# IMPERIA CW — Cold War: Integrated enbo aipac triggers — scripted triggers.
enbo_can_israel_aipac_action = {
	is_ai = no
	OR = {
		tag = ISR
		original_tag = ISR
	}
	NOT = { has_global_flag = enbo_aipac_terminated }
	country_exists = USA
	custom_trigger_tooltip = {
		tooltip = enbo_aipac_signal_ready_tt
		NOT = { has_country_flag = enbo_aipac_action_cooldown }
	}
	USA = {
		NOT = { has_country_flag = enbo_aipac_ended }
	}
}

enbo_can_usa_end_aipac = {
	is_ai = no
	OR = {
		tag = USA
		original_tag = USA
	}
	NOT = { has_global_flag = enbo_aipac_terminated }
	NOT = { has_country_flag = enbo_aipac_ended }
	custom_trigger_tooltip = {
		tooltip = enbo_aipac_requires_supermajority_tt
		check_variable = {
			var = senators_support
			value = 75
			compare = greater_than_or_equals
		}
		check_variable = {
			var = representatives_support
			value = 330
			compare = greater_than_or_equals
		}
	}
}

	}
	NOT = { has_global_flag = enbo_aipac_terminated }
	NOT = { has_country_flag = enbo_aipac_ended }

}
