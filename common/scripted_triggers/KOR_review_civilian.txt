	is_full_state = yes
	check_variable = { gdp_total > 0 }
	OR = { check_variable = { var = debt_ratio value = 0.60 compare = greater_than_or_equals } has_active_mission = debt_default_main_mission }
	NOT = { has_active_mission = cheap_loan_from_the_imf_mission }
	NOT = { has_country_flag = paused_debt_repayment }

		AND = { gdp_per_capita_greater_than_5 = yes interest_rate_lower_than_15 = yes }
		AND = {
			has_active_mission = debt_default_main_mission
			OR = { has_idea = bureau_02 has_idea = bureau_03 has_idea = bureau_04 has_idea = bureau_05 }
		}
	}
	NOT = { OR = { has_idea = corruption_level_08 has_idea = corruption_level_09 has_idea = corruption_level_10 } }
