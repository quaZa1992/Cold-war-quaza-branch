# IMPERIA COLD WAR — South African project affordability
# Project prices use the MD treasury's billions of dollars. HOI4 scripted
# trigger and effect calls use concrete names for these rates.
COLDWAR_SAF_PROJECT_can_fund_rail = {
 custom_trigger_tooltip = {
  tooltip = COLDWAR_SAF_PROJECT_can_fund_tt
  check_variable = { gdp_total > 0 }
  NOT = { has_country_flag = MD_skip_treasury_cost }
  set_temp_variable = { COLDWAR_SAF_PROJECT_price = { value = gdp_total multiply = 0.005 } }
  check_variable = { var = treasury value = COLDWAR_SAF_PROJECT_price compare = greater_than_or_equals }
 }
}
