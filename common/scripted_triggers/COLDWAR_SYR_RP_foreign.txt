# IMPERIA COLD WAR — Syrian treaty eligibility, consent and bilateral standing.

COLDWAR_SYR_RP_unrestricted_trade_SYR = {
	NOT = { is_embargoing = SYR }
	NOT = {
		OR = {
			is_in_array = { array = imp_dip_sanc_by_us value = SYR }
			is_in_array = { array = imp_dip_sanc_on_us value = SYR }
			is_in_array = { array = imp_fin_sanc_by_us value = SYR }
			is_in_array = { array = imp_fin_sanc_on_us value = SYR }
			is_in_array = { array = imp_eco_sanc_by_us value = SYR }
			is_in_array = { array = imp_eco_sanc_on_us value = SYR }
			is_in_array = { array = imp_mil_sanc_by_us value = SYR }
			is_in_array = { array = imp_mil_sanc_on_us value = SYR }
		}
	}
}

COLDWAR_SYR_RP_pair_clear_EGY = {
	country_exists = SYR
	country_exists = EGY
	SYR = {
		NOT = { has_war_with = EGY }
		COLDWAR_SYR_RP_unrestricted_trade_EGY = yes
	}
	EGY = { COLDWAR_SYR_RP_unrestricted_trade_SYR = yes }
}

	}
}

COLDWAR_SYR_RP_can_offer_EGY = {
	hidden_trigger = {
		tag = SYR
		IMP_is_1970_scenario = yes
		NOT = {
			OR = {
				has_country_flag = COLDWAR_SYR_RP_pending_EGY
				has_country_flag = COLDWAR_SYR_RP_counter_pending_EGY
				has_country_flag = COLDWAR_SYR_RP_signed_EGY_main
				has_country_flag = COLDWAR_SYR_RP_signed_EGY_counter
				has_country_flag = COLDWAR_SYR_RP_declined_EGY
			}
		}
	}
	COLDWAR_SYR_RP_may_sign_EGY = yes
}
