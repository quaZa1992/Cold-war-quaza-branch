# These triggers combine the native MD ruling slot with the registered party identity.
# Focus branches remain visible; call the triggers only for real availability and rewards.

COLDWAR_BRA_military_in_power = {
	custom_trigger_tooltip = {
		tooltip = COLDWAR_BRA_1970_military_in_power_tt
		original_tag = BRA
		IMP_is_1970_scenario = yes
		has_variable = ruling_party
		nationalist_military_junta_are_in_power = yes
	}
}

		tooltip = COLDWAR_BRA_1970_arena_in_power_tt
		original_tag = BRA

		has_variable = ruling_party
		western_conservatism_are_in_power = yes
		OR = {
			AND = { check_variable = { COLDWAR_BRA_1970_party_identity_1 = 1 } has_country_flag = COLDWAR_BRA_1970_party_arena_registered }
			AND = { check_variable = { COLDWAR_BRA_1970_party_identity_1 = 2 } has_country_flag = COLDWAR_BRA_1970_party_pds_registered }
		}
	}
}
