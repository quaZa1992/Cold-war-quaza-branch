}
COLDWAR_ORD_ready_rifles = {
	COLDWAR_ORD_republic = yes
	COLDWAR_ORD_human_country = yes
	check_variable = { COLDWAR_ORD_rifles_state = 2 }
	OR = {
		AND = {
			check_variable = { COLDWAR_ORD_rifles_kind = 1 }
			has_country_flag = COLDWAR_ORD_rifles_certified
		}
		AND = {
			check_variable = { COLDWAR_ORD_rifles_kind = 2 }
			check_variable = { var = COLDWAR_ORD_rifles_delivered value = COLDWAR_ORD_rifles_quantity compare = greater_than_or_equals }
