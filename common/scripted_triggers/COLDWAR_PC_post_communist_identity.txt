# IMPERIA COLD WAR — Post-communist party identity and political eligibility.
COLDWAR_PC_western_elected_government = {
	has_elections = yes
	OR = {
		western_conservatism_are_in_power = yes
		western_liberals_are_in_power = yes
		western_social_democrats_are_in_power = yes
	}
}

COLDWAR_PC_legal_party = {
	custom_trigger_tooltip = {
		tooltip = COLDWAR_PC_legal_party_tt
		NOT = { check_variable = { party_banned_array^$SLOT$ = 1 } }
	}
}

COLDWAR_PC_soviet_register_available = {
	OR = {
		COLDWAR_PC_western_elected_government = yes
		AND = {
			has_global_flag = COLDWAR_SU_end_junta
			nationalist_military_junta_are_in_power = yes
		}
	}
}
