# IMPERIA COLD WAR — French diplomatic influence requirements.
COLDWAR_FRA2_influence_5 = {
	custom_trigger_tooltip = {
		tooltip = COLDWAR_FRA2_influence_5_tt
		any_of = {
			array = influence_array
			value = v
			index = i
			var:v = { tag = FRA }
			check_variable = { influence_array_val^i > 4.999 }
		}
	}
}
