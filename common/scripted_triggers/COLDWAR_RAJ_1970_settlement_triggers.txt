COLDWAR_RAJ_1970_pak_western_land_gained = {
	any_owned_state = {
		is_fully_controlled_by = RAJ
		OR = {
			has_state_flag = COLDWAR_RAJ_1970_pak_war_prewar_west
			has_state_flag = COLDWAR_RAJ_1970_pak_war_west_granted
			AND = {
				# Old saves did not snapshot the starting frontier.
				NOT = { RAJ = { has_country_flag = COLDWAR_RAJ_1970_pak_war_scoped } }
				is_core_of = PAK
				NOT = { OR = { is_core_of = RAJ is_core_of = BAN } }
			}
		}
	}
}
