   tooltip = COLDWAR_SAF_DEFENCE_PLAN_programme_funds_tt

 hidden_trigger = { has_country_flag = COLDWAR_SAF_DEFENCE_PLAN_first_issued }
 if = {
  limit = { hidden_trigger = { check_variable = { COLDWAR_SAF_DEFENCE_PLAN_issued < 6 } } }

   tooltip = COLDWAR_SAF_DEFENCE_PLAN_programme_funds_tt
   set_temp_variable = { COLDWAR_SAF_DEFENCE_PLAN_batch = 6 }
   subtract_from_temp_variable = { COLDWAR_SAF_DEFENCE_PLAN_batch = COLDWAR_SAF_DEFENCE_PLAN_issued }
   clamp_temp_variable = { var = COLDWAR_SAF_DEFENCE_PLAN_batch min = 0 max = 5 }
   multiply_temp_variable = { COLDWAR_SAF_DEFENCE_PLAN_batch = 0.1 }
   check_variable = { var = treasury value = COLDWAR_SAF_DEFENCE_PLAN_batch compare = greater_than_or_equals }
  }
 }
