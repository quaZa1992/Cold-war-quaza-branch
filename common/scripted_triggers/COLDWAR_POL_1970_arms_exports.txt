# IMPERIA COLD WAR — Polish arms export acceptance requirements.

COLDWAR_POL_arms_cash_can_accept = {
	exists = yes
	OR = { tag = LBA tag = IRQ }
	country_exists = POL
	NOT = { has_war_with = POL }
	has_country_flag = COLDWAR_POL_arms_cash_pending
	NOT = { has_country_flag = COLDWAR_POL_arms_cash_accepted }
	NOT = { has_country_flag = COLDWAR_POL_arms_cash_refused }
	NOT = { has_country_flag = COLDWAR_POL_arms_cash_expired }
	NOT = { has_country_flag = MD_skip_treasury_cost }
	check_variable = { gdp_total > 0 }
	POL = {
		original_tag = POL
		IMP_is_1970_scenario = yes
		has_country_flag = COLDWAR_POL_arms_cash_offered
		NOT = { has_country_flag = MD_skip_treasury_cost }
		has_equipment = { infantry_weapons_type > 999 }
		has_equipment = { medium_tank_chassis > 49 }
	}
}
