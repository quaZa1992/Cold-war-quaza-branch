# IMPERIA CW — Cold War: quaZa uranium supply chain and raid integration.
# Adapted from quaZa Feature Test, workshop 3818306694.
# Imperia: enrichment facility construction weights, driven by imp_ai_wants_nukes
# (set monthly by imp_ai_nuclear_need_update). The scripted effect places the 3 facilities
# the nuclear weapons projects need; these weights keep the construction AI from building
# them when the country does not want nuclear weapons.

imp_enrichment_not_wanted = {
	allowed = { ai_for_units_is_not_disabled = yes }
	enable = { NOT = { has_country_flag = imp_ai_wants_nukes } has_nuclear_reactors = no }
	abort_when_not_enabled = yes

	ai_strategy = { type = build_building id = enrichment_facility value = -100 }
}

imp_enrichment_wanted = {
	allowed = { ai_for_units_is_not_disabled = yes }
	enable = { OR = { has_country_flag = imp_ai_wants_nukes has_nuclear_reactors = yes } }
	abort_when_not_enabled = yes

	ai_strategy = { type = build_building id = enrichment_facility value = 20 }
}
