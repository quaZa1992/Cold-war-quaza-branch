# IMPERIA CW — Cold War: surveyed mine capacity; no modern reserve-tonnage claims.
scripted_map_modes = {
	imp_uranium_deposit_map_mode = {
		top = { type = none }
		bottom = {
			type = state
			color = {
				if = { limit = { FROM = { check_variable = { imp_uranium_per_level > 8 } } }
					set_temp_variable = { r_clc = 0.30 }
					set_temp_variable = { g_clc = 1.00 }
					set_temp_variable = { b_clc = 0.10 }
				}
				else_if = { limit = { FROM = { check_variable = { imp_uranium_per_level > 5 } } }
					set_temp_variable = { r_clc = 0.35 }
					set_temp_variable = { g_clc = 0.82 }
					set_temp_variable = { b_clc = 0.10 }
				}
				else_if = { limit = { FROM = { check_variable = { imp_uranium_per_level > 3 } } }
					set_temp_variable = { r_clc = 0.40 }
					set_temp_variable = { g_clc = 0.65 }
					set_temp_variable = { b_clc = 0.12 }
				}
				else_if = { limit = { FROM = { check_variable = { imp_uranium_per_level > 2 } } }
					set_temp_variable = { r_clc = 0.42 }
					set_temp_variable = { g_clc = 0.50 }
					set_temp_variable = { b_clc = 0.15 }
				}
				else_if = { limit = { FROM = { has_resources_amount = { resource = uranium amount > 0 } } }
					# Production without further surveyed mine capacity.
					set_temp_variable = { r_clc = 0.85 }
					set_temp_variable = { g_clc = 0.75 }
					set_temp_variable = { b_clc = 0.10 }
				}
				else = {
					set_temp_variable = { r_clc = 0.18 }
					set_temp_variable = { g_clc = 0.18 }
					set_temp_variable = { b_clc = 0.20 }
				}
				set_temp_variable = { red = r_clc }
				set_temp_variable = { green = g_clc }
				set_temp_variable = { blue = b_clc }
				set_temp_variable = { alpha = 1.00 }
			}
		}

		far_text = state
		near_text = state

		update_daily = no
	}
}
