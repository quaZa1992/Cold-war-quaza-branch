imp_oil_price_decisions = 
{
	d_imp_raise_oil_price = 
	{
		icon = GFX_decision_imp_list
		days_remove = 7
		cost = 30

		visible = 
		{
			is_ai = no
			OR = {
				has_idea = fossil_fuel_industry
				mdcw_cw_is_major_oil_exporter = yes
			}
		}

		available = 
		{
			has_idea = fossil_fuel_industry
			NOT = {
				has_idea = imp_oil_prices_ask_1
				has_decision = d_imp_lower_oil_price
			}
		}

		remove_effect = {
			imp_oil_price_demand_lower = yes
		}
	}

	d_imp_lower_oil_price = 
	{
		icon = GFX_decision_imp_list
		days_remove = 7
		cost = 30

		visible = 
		{
			is_ai = no
			OR = {
				has_idea = fossil_fuel_industry
				mdcw_cw_is_major_oil_exporter = yes
			}
		}

		available = 
		{
			has_idea = fossil_fuel_industry
			NOT = {
				has_idea = imp_oil_prices_ask_11
				has_decision = d_imp_raise_oil_price
			}
		}

		remove_effect = {
			imp_oil_price_demand_raise = yes
		}
	}
}
