# IMPERIA CW — Cold War: guest workers decisions — decisions.
# One native country-targeted decision per size category.
# The native PP fee pays for the request; cash and population move only after approval.
# Eligibility and request effects use explicit no-argument helpers for each tier.

mdcw_guest_workers_small = {
	mdcw_gw_recruit_25000 = {
		icon = GFX_decision_generic_civil_support
		cost = 75
		target_root_trigger = { mdcw_gw_is_receiving_country = yes }
		target_trigger = {
			FROM = {
				mdcw_gw_is_sending_country = yes
				NOT = { tag = ROOT }
			}
		}
		visible = {
			mdcw_gw_is_receiving_country = yes
			FROM = {
				mdcw_gw_is_sending_country = yes
				NOT = { tag = ROOT }
			}
		}
		available = { mdcw_gw_request_available_25000 = yes }
		complete_effect = {
			custom_effect_tooltip = mdcw_gw_recruit_25000_effect_tt
			hidden_effect = { mdcw_gw_request_agreement_25000 = yes }
		}
		ai_will_do = { factor = 0 }
	}
}
