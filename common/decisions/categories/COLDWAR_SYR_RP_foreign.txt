			AND = { tag = EGY SYR = { has_country_flag = COLDWAR_SYR_RP_pending_EGY } }

			AND = {
				tag = SYR
				OR = {
					has_country_flag = COLDWAR_SYR_RP_counter_pending_EGY
					has_country_flag = COLDWAR_SYR_RP_counter_pending_ISR
					has_country_flag = COLDWAR_SYR_RP_counter_pending_SOV
