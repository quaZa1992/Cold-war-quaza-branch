add_namespace = LBA1970Oil

# Requests are sent by the 1970 Libya focus tree. No technical reward is
# granted until the invited country explicitly accepts.
country_event = {
	id = LBA1970Oil.1
	title = LBA1970Oil.1.t
	desc = LBA1970Oil.1.d
	picture = GFX_LBA_1970_oil_technical_assistance
	is_triggered_only = yes
	option = {
		name = LBA1970Oil.accept
		trigger = { NOT = { has_war_with = LBA } }
		ai_chance = { base = 85 }
		# IMPERIA COLD WAR — Libya applies the mission in its own result event
		effect_tooltip = { LBA = { LBA_1970_accept_algerian_oil_mission = yes } }
		LBA = { country_event = { id = LBA1970Oil.11 } }
	}
	option = {
		name = LBA1970Oil.decline
		ai_chance = { base = 15 }
		LBA = { country_event = { id = LBA1970Oil.21 } } # IMPERIA COLD WAR — Libya records the refusal itself
	}
}
