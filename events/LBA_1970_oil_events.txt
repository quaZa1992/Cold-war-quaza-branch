add_namespace = LBA1970Oil

# Requests are sent by the 1970 Libya focus tree. No technical reward is
# granted until the invited country explicitly accepts.
country_event = {
	id = LBA1970Oil.1
	title = LBA1970Oil.1.t
	desc = LBA1970Oil.1.d
	picture = GFX_LBA_1970_oil_technical_assistance
	is_triggered_only = yes
	option = {
		name = LBA1970Oil.accept
		trigger = { NOT = { has_war_with = LBA } }
		ai_chance = { base = 85 }
		LBA = {
			if = {
				limit = { has_country_flag = LBA_1970_oil_offer_pending NOT = { has_country_flag = LBA_1970_oil_partner_agreed } }
				LBA_1970_accept_algerian_oil_mission = yes
				country_event = { id = LBA1970Oil.11 days = 1 }
			}
		}
	}
	option = {
		name = LBA1970Oil.decline
		ai_chance = { base = 15 }
		LBA = {
			clr_country_flag = LBA_1970_oil_offer_pending
			set_country_flag = LBA_1970_oil_offer_algeria_refused
			country_event = { id = LBA1970Oil.21 days = 1 }
		}
	}
}
