add_namespace = COLDWAR_nuclear

# Reactor incident (INES 1-3): one reactor out for 120 days, repair 0.1 % of GDP
country_event = {
	id = COLDWAR_nuclear.1
	title = COLDWAR_nuclear.1.t
	desc = COLDWAR_nuclear.1.d
	picture = GFX_SIN_singapore_builds_first_nuclear_reactor
	is_triggered_only = yes

	immediate = {
		log = "[GetDateText]: [Root.GetName]: event COLDWAR_nuclear.1"
		random_controlled_state = {
			limit = { nuclear_reactor > 0 }
			save_event_target_as = COLDWAR_nuc_site
		}
	}

	option = {
		name = COLDWAR_nuclear.1.a
		set_temp_variable = { treasury_change = gdp_total }
		multiply_temp_variable = { treasury_change = -0.001 }
		modify_treasury_effect = yes
		custom_effect_tooltip = COLDWAR_nuclear_outage_tt
		hidden_effect = {
			set_variable = { COLDWAR_nuc_outage = -1 }
			divide_variable = { COLDWAR_nuc_outage = nuclear_reactors }
			clamp_variable = { var = COLDWAR_nuc_outage min = -0.25 max = 0 }
			add_dynamic_modifier = { modifier = COLDWAR_reactor_outage_modifier days = 120 }
			force_update_dynamic_modifier = yes
		}
	}
}

# Serious accident (INES 4-5): one reactor lost, local contamination, clean-up, anti-nuclear movement, world news

	is_triggered_only = yes

	immediate = {

		}
	}

	option = {
		name = COLDWAR_nuclear.2.a
		event_target:COLDWAR_nuc_site = {

		hidden_effect = {
			every_other_country = {
				limit = { exists = yes }
				news_event = { id = COLDWAR_nuclear.4 days = 2 }
			}
		}
	}
}

		custom_effect_tooltip = COLDWAR_nuclear.10.a_tt

		name = COLDWAR_nuclear.10.b

		FROM = { country_event = { id = COLDWAR_nuclear.12 days = 1 } }
	}
}

		name = COLDWAR_nuclear.20.a

				add = 30

			}
		}

			var:COLDWAR_fuel_supplier = { country_event = { id = COLDWAR_nuclear.14 days = 1 } }
		}
		set_variable = { COLDWAR_fuel_supplier = FROM.id }
		custom_effect_tooltip = COLDWAR_nuclear.20.a_tt
		hidden_effect = {
			country_event = { id = COLDWAR_nuclear.23 days = 1460 }
			FROM = { country_event = { id = COLDWAR_nuclear.21 days = 1 } }
		}
