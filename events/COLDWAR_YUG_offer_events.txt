country_event = {
	id = COLDWAR_YUG_OFR.1
	title = COLDWAR_YUG_OFR.1.t
	desc = COLDWAR_YUG_OFR.1.d
	picture = GFX_political_deal

	is_triggered_only = yes

	option = {
		name = COLDWAR_YUG_accept_credits_for_belgrade
		trigger = {
			country_exists = SER
					SER = { has_country_flag = COLDWAR_YUG_offer_credits_for_belgrade }
					NOT = { has_country_flag = COLDWAR_YUG_partner_signed_credits_for_belgrade }
					NOT = { has_country_flag = COLDWAR_YUG_declined_credits_for_belgrade }
					NOT = {
						has_global_flag = COLDWAR_YUG_deal_broken_credits_for_belgrade_USA
					}
			COLDWAR_YUG_deal_clear_SER = yes
		}
		ai_chance = { base = 1 }
		log = "[GetDateText]: [Root.GetName]: event COLDWAR_YUG_OFR.1 option a"
					set_country_flag = COLDWAR_YUG_partner_signed_credits_for_belgrade
					set_country_flag = COLDWAR_YUG_offer_partner
					add_ideas = COLDWAR_YUG_offer_credits_for_belgrade_partner
					custom_effect_tooltip = COLDWAR_YUG_deal_credits_for_belgrade_partner_tt
					SER = { country_event = { id = COLDWAR_YUG.101 days = 1 } }
	}
	option = {
		name = COLDWAR_YUG_decline_credits_for_belgrade
		trigger = {
			country_exists = SER
					SER = { has_country_flag = COLDWAR_YUG_offer_credits_for_belgrade }
					NOT = { has_country_flag = COLDWAR_YUG_partner_signed_credits_for_belgrade }
					NOT = { has_country_flag = COLDWAR_YUG_declined_credits_for_belgrade }
		}
		ai_chance = { base = 0 }
		log = "[GetDateText]: [Root.GetName]: event COLDWAR_YUG_OFR.1 option b"
					set_country_flag = { flag = COLDWAR_YUG_declined_credits_for_belgrade days = 60 value = 1 }
					SER = { country_event = { id = COLDWAR_YUG.102 days = 1 } }
	}
	option = {
		name = COLDWAR_offer_event_lapsed
		trigger = { NOT = { SER = { has_country_flag = COLDWAR_YUG_offer_credits_for_belgrade } } }
		ai_chance = { base = 1 }
	}
}
