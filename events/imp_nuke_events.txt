# Radiation Spreading
country_event = {
	id = imp_nukes.10
	is_triggered_only = yes

	hidden = yes
	
	option = {
		event_target:imp_contam_spreader = {
			every_neighbor_state = {
				
				limit = { 
					OR = {
						NOT = { has_dynamic_modifier = { modifier = nuclear_contamination } }
						NOT = { has_dynamic_modifier = { modifier = nuclear_devastation } }
					}
				}

				random = {
					chance = 60

					save_event_target_as = imp_contam_spreader
					ROOT = { country_event = { id = imp_nukes.11 days = 5 } }
					add_dynamic_modifier = {
						modifier = nuclear_contamination
						days = 365
					}

					CONTROLLER = { 
						if = {
							limit = { NOT = { has_country_flag = contimation_warning } }
							country_event = { id = imp_nukes.13 }
							set_country_flag = { flag = contimation_warning days = 1 }
						}
					}
				}
			}
		}
	}
}
