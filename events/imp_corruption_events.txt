# The Mining Operations
country_event = { 
	id = imp_corruption.11
	title = imp_corruption.11.t
	desc = imp_corruption.11.d
	picture = GFX_imp_cor_11
	is_triggered_only = yes

	trigger = {
		check_variable = { gdp_total < 200 }
		check_variable = { globalism_value > 20 }
		any_owned_state = {
			chromium > 5
		}
	}

	option = {
		name = imp_corruption.11.o1

		random_owned_state = {
			limit = { 
				chromium > 5
			}
			add_resource = {
   				type = chromium
    			amount = -2
			}
		}
	}

	option = {
		name = imp_corruption.11.o2
 
		increase_corruption = yes
	}
}
