# Licence: part of the Imperia Cold War mod, based on Millennium Dawn. Not for reuse outside the mod without permission of the Imperia team.
# Update 2026-11-13: dedicated events for factions that had none of their own (landowners, the military, fossil fuel industry,
#   international bankers, Wall Street, chaebols, donju, clergy, priesthood). Fired from the MD on_monthly random_events pool
#   (common/on_actions/MD_event_on_actions.txt). Favour costs follow the Cold War pass: +5 = 75 PP or 0.3% GDP, +10 = 150 PP or 0.6% GDP.

add_namespace = coldwar_ifac

# IMPERIA COLD WAR — Landowners Resist Land Reform
country_event = {
	id = coldwar_ifac.1
	title = coldwar_ifac.1.t
	desc = coldwar_ifac.1.d
	picture = GFX_internal_faction_political_conflict

	is_triggered_only = yes

	trigger = {
		has_idea = landowners
		NOT = { has_country_flag = COLDWAR_ifac_1_recent }
		check_variable = { agriculture_district_total > 0 }
	}

	immediate = {
		set_country_flag = { flag = COLDWAR_ifac_1_recent value = 1 days = 365 }
	}

	option = {
		name = coldwar_ifac.1.o1
		log = "[GetDateText]: [This.GetName]: coldwar_ifac.1.o1 executed"
		set_temp_variable = { temp_opinion = 10 }
		change_landowners_opinion = yes
		add_political_power = -150
		if = {
			limit = { has_idea = farmers }
			set_temp_variable = { temp_opinion = -10 }
			change_farmers_opinion = yes
		}
		ai_chance = {
			base = 1
			modifier = {
				factor = 0
				has_political_power < 200
			}
		}
	}
	option = {
		name = coldwar_ifac.1.o2
		log = "[GetDateText]: [This.GetName]: coldwar_ifac.1.o2 executed"
		set_temp_variable = { temp_opinion = -15 }
		change_landowners_opinion = yes
		if = {
			limit = { has_idea = farmers }
			set_temp_variable = { temp_opinion = 5 }
			change_farmers_opinion = yes
		}
		add_stability = -0.02
		ai_chance = { base = 1 }
	}
	option = {
		name = coldwar_ifac.1.o3
		log = "[GetDateText]: [This.GetName]: coldwar_ifac.1.o3 executed"
		set_temp_variable = { temp_opinion = 5 }
		change_landowners_opinion = yes
		if = {
			limit = { has_idea = farmers }
			set_temp_variable = { temp_opinion = 5 }
			change_farmers_opinion = yes
		}
		set_temp_variable = { treasury_change = { value = gdp_total multiply = -0.009 } }
		modify_treasury_effect = yes
		ai_chance = {
			base = 1
			modifier = {
				factor = 0
				check_variable = { interest_rate > 9 }
			}
		}
	}
}
