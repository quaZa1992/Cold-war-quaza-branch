# Volunteers returning
country_event = {
	id = imp_minis.3
	title = imp_minis.3.t
	desc = imp_minis.3.d
	
	picture = GFX_imp_mini_vol_back

	major = yes
	is_triggered_only = yes

	immediate = { 
		hidden_effect = {
			## 75% come back unscathed.
			set_temp_variable = { t_volunteers_return = FROM.volunteers_from_k@ROOT }
			multiply_temp_variable = { t_volunteers_return = 750 }
			add_manpower = t_volunteers_return

			## 25% are wounded
			set_temp_variable = { t_volunteers_cas_k = FROM.volunteers_from_k@ROOT }
			multiply_temp_variable = { t_volunteers_cas_k = 0.25 }

			## 25% of those are dead dead
			set_temp_variable = { t_volunteers_dead = t_volunteers_cas_k }
			multiply_temp_variable = { t_volunteers_dead = 0.25 }
			add_to_variable = { imp_total_dead_k = t_volunteers_dead } # Add to death count
			multiply_temp_variable = { t_volunteers_dead = 1000 } 

			set_temp_variable = { cas_t = 0 }
			subtract_from_temp_variable = { cas_t = t_volunteers_dead }
			random_core_state = 
			{
				limit = { state_population_k > t_volunteers_dead_k }
				add_manpower = cas_t
			}

			## The rest are "just" casualties
			multiply_temp_variable = { t_volunteers_cas_k = 0.75 }
			add_to_variable = { imp_total_casualties_k = t_volunteers_cas_k }
	
			imp_calc_casualty_cost = yes

			FROM = { 
				remove_from_array = { 
					array = volunteers_receiving
					value = ROOT
				}
				set_variable = { volunteers_from_k@ROOT = 0 }
			}
		}
	}

	option = {
		name = imp_minis.3.o1
	}
}
