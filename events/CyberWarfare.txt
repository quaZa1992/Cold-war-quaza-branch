country_event = {
	id = MD_cyber.18
	title = MD_cyber.18.t
	desc = MD_cyber.18.d
	picture = GFX_cyber_attack
	is_triggered_only = yes

	option = {
		name = MD_cyber.18.a
		log = "[GetDateText]: [This.GetName]: MD_cyber.18.a"
		if = {
			limit = { has_idea = cyber_election_interference_malus }
			modify_timed_idea = { idea = cyber_election_interference_malus days = 135 }
		}
		else = {
			add_timed_idea = { idea = cyber_election_interference_malus days = 180 }
		}
		# COLDWAR covert v2: the government loses, the party of the backer's own outlook gains (scaled by strength)
		set_party_index_to_ruling_party = yes
		set_temp_variable = { party_popularity_increase = -0.04 }
		multiply_temp_variable = { party_popularity_increase = COLDWAR_ca_strength }
		set_temp_variable = { temp_outlook_increase = party_popularity_increase }
		set_temp_variable = { COLDWAR_ca_ruling = party_index }
		change_relative_party_popularity = yes
		if = {
			limit = {
				has_variable = COLDWAR_ca_backer_party
				NOT = { check_variable = { COLDWAR_ca_backer_party = COLDWAR_ca_ruling } }
			}
			set_temp_variable = { party_index = COLDWAR_ca_backer_party }
			set_temp_variable = { party_popularity_increase = 0.05 }
			multiply_temp_variable = { party_popularity_increase = COLDWAR_ca_strength }
			set_temp_variable = { temp_outlook_increase = party_popularity_increase }
			change_relative_party_popularity = yes
		}
		recalculate_party = yes
	}
}
