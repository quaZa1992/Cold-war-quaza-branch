# Regular Country Events
country_event = {
	id = imp_oil.3
	title = imp_oil.3.t
	desc = imp_oil.3.d
	picture = GFX_news_imp_oil_bad
	is_triggered_only = yes

	option = {
		name = imp_oil.3.o1

		set_party_index_to_ruling_party = yes
		set_temp_variable = { party_popularity_increase = -0.04 }
		set_temp_variable = { temp_outlook_increase = -0.04 }
		change_relative_party_popularity = yes
	}

	option = {
		name = imp_oil.3.o2

		set_temp_variable = { treasury_change = 0 }
		subtract_from_temp_variable = { treasury_change = gdp_total }
		multiply_temp_variable = { treasury_change = 0.01 }
		modify_treasury_effect = yes
	}

	option = {
		name = imp_oil.3.o3

		trigger = {
			OR = {
				has_idea = imp_major_oil_exporter
				has_idea = imp_minor_oil_exporter
			}
			has_idea = fossil_fuel_industry
		}

		add_timed_idea = {
			idea = imp_oil_internal_market_first
			days = 90
		}

		set_temp_variable = { oil_cost_multiplier_t = 0.005 }
        imp_spike_oil_price = yes

		custom_effect_tooltip = {
    		localization_key = oil_price_up_extra_tt
		}
	}
}
