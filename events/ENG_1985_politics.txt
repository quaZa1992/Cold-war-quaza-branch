	id = eng_big_bang_preview.1
	title = eng_big_bang_preview.1.t
	desc = eng_big_bang_preview.1.d
	picture = GFX_city
	is_triggered_only = yes

	option = {
		name = eng_big_bang_preview.1.a
		if = {
			limit = {
				has_country_flag = ENG_big_bang_watchdogs
				check_variable = { ENG_market_confidence > 55 }
			}
			add_political_power = 40
			add_to_variable = { ENG_market_confidence = 12 }
			add_to_variable = { ENG_conservative_office_tax = 0.02 }
			set_temp_variable = { ENG_city_workshop_delta = 0.06 }
			ENG_shift_city_workshop_bop = yes
			country_event = { id = eng_big_bang_preview.2 days = 1 }
		}
		else_if = {
			limit = { has_country_flag = ENG_big_bang_fast_track }
			add_stability = -0.015
			add_to_variable = { ENG_market_confidence = 4 }
			add_to_variable = { ENG_conservative_office_tax = 0.04 }
			add_to_variable = { ENG_conservative_corporate_tax = 0.02 }
			set_temp_variable = { ENG_city_workshop_delta = 0.04 }
			ENG_shift_city_workshop_bop = yes
			set_temp_variable = { ENG_union_strain_delta = -3 }
			ENG_shift_union_strain = yes
			country_event = { id = eng_big_bang_preview.3 days = 1 }
		}
		else = {
			add_stability = -0.035
			add_to_variable = { ENG_market_confidence = -12 }
			add_to_variable = { ENG_party_cohesion = -5 }
			add_to_variable = { ENG_conservative_office_tax = -0.02 }
			set_temp_variable = { ENG_union_strain_delta = -5 }
			ENG_shift_union_strain = yes
			country_event = { id = eng_big_bang_preview.4 days = 1 }
		}
	}
}
