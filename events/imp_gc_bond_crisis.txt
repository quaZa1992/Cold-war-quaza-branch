news_event = { 
	id = imp_bond_crisis.4
	picture = GFX_news_event_bond_crisis_4

	title = imp_bond_crisis.4.t
	desc = imp_bond_crisis.4.d
	is_triggered_only = yes

	immediate = { 
		if = { 
			limit = { 
				tag = LIC
			}

			set_temp_variable = { oil_cost_multiplier_t = -1.25 }
			imp_spike_oil_price = yes
		}
	}

	option = {
		name = imp_bond_crisis.4.o1

		trigger = { 
			NOT = { 
				has_idea = imp_major_oil_exporter
				has_idea = imp_minor_oil_exporter
			}
		}
		add_ideas = imp_reduced_oil_demand
	}
	
	option = {
		name = imp_bond_crisis.4.o2

		trigger = { 
			OR = { 
				has_idea = imp_major_oil_exporter
				has_idea = imp_minor_oil_exporter
			}
		}

		add_ideas = imp_reduced_oil_demand
		decrease_economic_growth = yes
	}
}
