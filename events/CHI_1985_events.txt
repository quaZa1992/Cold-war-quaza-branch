# -----------------------------------------------------------------------------

# -----------------------------------------------------------------------------

	id = china_1985.2
	title = china_1985.2.t
	desc = china_1985.2.d
	picture = GFX_report_event_generic_factory_strike

	trigger = {
		tag = CHI
		has_country_flag = CHI_1985_state_system_enabled
		check_variable = { CHI_inflation_pressure > 60 }
		NOT = { has_country_flag = CHI_1985_inflation_event_fired }
