# Watergate: optional crisis after a high-risk campaign-intelligence course
###############################################################

country_event = {
	id = USA_1970.10
	title = USA_1970.10.t
	desc = USA_1970.10.d
	picture = GFX_political_deal
	is_triggered_only = yes

	trigger = {
		original_tag = USA
		IMP_is_1970_scenario = yes
		has_country_flag = USA_1970_nixon_presidency
		has_country_flag = USA_1970_watergate_exposure
		AND = { has_variable = ruling_party check_variable = { ruling_party = 1 } }
	}

	option = {
		name = USA_1970.10.a
		ai_chance = { base = 85 }
		hidden_effect = { kill_country_leader = yes }
		create_country_leader = {
			name = "Gerald Ford"
			ideology = conservatism
			traits = { western_conservatism career_politician }
		}
		clr_country_flag = USA_1970_nixon_presidency
		set_country_flag = USA_1970_watergate_resignation
		add_stability = 0.03
		add_political_power = -75
		USA_congress_small_support = yes
		set_temp_variable = { temp_change = -0.02 }
		USA_change_domestic_political_power_change = yes
	}
	option = {
		name = USA_1970.10.b
		ai_chance = { base = 15 }
		hidden_effect = { kill_country_leader = yes }
		create_country_leader = {
			name = "Richard Nixon"
			picture = "gfx/Leaders/USA/USA_Coldwar_Nixon.dds"
			ideology = conservatism
			traits = { western_conservatism career_politician }
		}
		clr_country_flag = USA_1970_nixon_presidency
		set_country_flag = USA_1970_watergate_survived
		add_stability = -0.07
		add_political_power = 100
		USA_congress_medium_opposition = yes
		set_temp_variable = { temp_change = -0.05 }
		USA_change_domestic_drift_defense_change = yes
		set_temp_variable = { temp_change = 0.03 }
		USA_change_domestic_civ_intel_change = yes
	}
}
