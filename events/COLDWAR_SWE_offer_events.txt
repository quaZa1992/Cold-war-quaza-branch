country_event = {
	id = COLDWAR_SWE_OFR.1
	title = COLDWAR_SWE_OFR.1.t
	desc = COLDWAR_SWE_OFR.1.d
	picture = GFX_political_deal

	is_triggered_only = yes

	option = {
		name = COLDWAR_SWE_national_accept_procurement
		trigger = {
			has_country_flag = COLDWAR_SWE_requested_procurement
			tag = event_target:COLDWAR_SWE_partner_procurement
			tag = event_target:COLDWAR_SWE_partner_procurement
			SWE = {
				exists = yes
				has_country_flag = COLDWAR_SWE_agreement_pending_procurement
			}
			NOT = {
				has_war_with = SWE
			}
			SWE = {
				AND = {
					set_temp_variable = {
						COLDWAR_SWE_national_required_budget = {
							value = gdp_total
							multiply = 0.01
						}
					}
					check_variable = {
						var = treasury
						value = COLDWAR_SWE_national_required_budget
						compare = greater_than_or_equals
					}
				}
			}
			has_equipment = {
				infantry_weapons_1 > 999
			}
		}
		ai_chance = { base = 1 }
		if = {
				limit = {
					has_country_flag = COLDWAR_SWE_requested_procurement
					tag = event_target:COLDWAR_SWE_partner_procurement
					tag = event_target:COLDWAR_SWE_partner_procurement
					SWE = {
						exists = yes
						has_country_flag = COLDWAR_SWE_agreement_pending_procurement
					}
					NOT = {
						has_war_with = SWE
					}
					SWE = {
						AND = {
							set_temp_variable = {
								COLDWAR_SWE_national_required_budget = {
									value = gdp_total
									multiply = 0.01
								}
							}
							check_variable = {
								var = treasury
								value = COLDWAR_SWE_national_required_budget
								compare = greater_than_or_equals
							}
						}
					}
					has_equipment = {
						infantry_weapons_1 > 999
					}
				}
				COLDWAR_SWE_national_accept_procurement_effect = yes
			}
	}
	option = {
		name = COLDWAR_SWE_national_refuse_procurement
		trigger = {
			has_country_flag = COLDWAR_SWE_requested_procurement
			tag = event_target:COLDWAR_SWE_partner_procurement
		}
		ai_chance = { base = 1 }
		if = {
				limit = {
					has_country_flag = COLDWAR_SWE_requested_procurement
					tag = event_target:COLDWAR_SWE_partner_procurement
				}
				clr_country_flag = COLDWAR_SWE_requested_procurement
				SWE = {
					if = {
						limit = {
							has_country_flag = COLDWAR_SWE_agreement_pending_procurement
						}
						subtract_from_variable = {
							COLDWAR_SWE_pending_count = 1
						}
						clamp_variable = {
							var = COLDWAR_SWE_pending_count
							min = 0
						}
						clr_country_flag = COLDWAR_SWE_political_external_open
						clr_country_flag = COLDWAR_SWE_agreement_pending_procurement
					}
				}
				COLDWAR_SWE_national_reply_status_effect = yes
			}
	}
	option = {
		name = COLDWAR_offer_event_lapsed
		trigger = { NOT = { has_country_flag = COLDWAR_SWE_requested_procurement } }
		ai_chance = { base = 1 }
	}
