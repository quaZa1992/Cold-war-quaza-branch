add_namespace = RW_event

# Request Resources
country_event = {
	id = RW_event.1
	title = RW_event.1.t
	desc = RW_event.1.d
	picture = GFX_imp_resource_mining
	is_triggered_only = yes
	
	option = {
		name = RW_event.1.o1

		trigger = 
		{
			NOT = 
			{
				any_country = {
					NOT = { TAG = root }
					has_resources_rights = 
					{
						state = from.imp_resource_target
					}
				}
			}
		}

		give_resource_rights = 
		{
			receiver = FROM
			state = from.imp_resource_target
		}
		FROM = { country_event = { id = RW_event.2 originator = root } }
	}

	option = {
		name = RW_event.1.o2

		trigger = 
		{
			NOT = 
			{
				any_country = {
					NOT = { TAG = root }
					has_resources_rights = 
					{
						state = from.imp_resource_target
					}
				}
			}
		}

		FROM = { country_event = { id = RW_event.3 originator = root } }
		
		ai_chance = {
			base = 100
		}
	}

	option = {
		name = RW_event.1.o3

		trigger = 
		{
			any_country = {
				NOT = { TAG = root }
				has_resources_rights = 
				{
					state = from.imp_resource_target
				}
			}
		}

		FROM = { country_event = { id = RW_event.4 originator = root } }

		ai_chance = {
			base = 100
		}
	}
}

# Take back Resource Rights. This is done from the country that owns the State.
country_event = {
	id = RW_event.6
	title = RW_event.6.t
	desc = RW_event.6.d
	picture = GFX_imp_resource_mining
	is_triggered_only = yes
	
	option = {
		name = RW_event.6.o1

		trigger = 
		{
			any_country = {
				NOT = { TAG = root }
				has_resources_rights = 
				{
					state = event_target:TB_target_province
				}
			}
		}
		
		every_country = {
			limit = {
				NOT = { TAG = Root }
				OR = {
					has_resources_rights = {
						state = event_target:TB_target_province
						resources = { oil }
					}
					has_resources_rights = {
						state = event_target:TB_target_province
						resources = { rubber }
					}
					has_resources_rights = {
						state = event_target:TB_target_province
						resources = { aluminium }
					}
					has_resources_rights = {
						state = event_target:TB_target_province
						resources = { tungsten }
					}
					has_resources_rights = {
						state = event_target:TB_target_province
						resources = { steel }
					}
					has_resources_rights = {
						state = event_target:TB_target_province
						resources = { chromium }
					}
					has_resources_rights = { state = event_target:TB_target_province resources = { microchips } }
					has_resources_rights = { state = event_target:TB_target_province resources = { composites } }
				}
			}

			remove_resource_rights = event_target:TB_target_province
			country_event = { id = RW_event.7 originator = root } 
		}
	}

	option = {
		name = RW_event.6.o2
	}
}
