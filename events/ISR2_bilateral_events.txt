	title = ISR2_bilateral.1.t
	desc = ISR2_bilateral.1.d
	picture = GFX_COLDWAR_CUB_1970_event_signing
	is_triggered_only = yes
	trigger = {
		tag = ETH
		has_country_flag = ISR2_b_irrigation_exports_pending
		ISR = { ISR2_is_scenario = yes has_country_flag = ISR2_b_irrigation_exports_pending }
	}
	option = {
		name = ISR2_bilateral_accept
		trigger = {
			ISR = { ISR2_b_irrigation_exports_accept_valid = yes }
		}
		ISR = { ISR2_b_irrigation_exports_preview = yes }
		hidden_effect = {
			ISR = { ISR2_b_irrigation_exports_accept = yes }
		}
		ai_chance = { factor = 0 }
	}
	option = {
		name = ISR2_bilateral_refuse
		hidden_effect = {
			ISR = { ISR2_b_irrigation_exports_decline = yes }
		}
		ai_chance = { factor = 100 }
	}
}
