country_event = {
	id = Imp_sanctions.1
	title = Imp_sanctions.1.t
	desc = Imp_sanctions.1.d
	picture = GFX_news_imp_sanctions
	is_triggered_only = yes
	immediate = { IMP_sanctions_by_us_page = yes }
	option = {
		name = IMP_sanctions_previous
		trigger = { check_variable = { imp_sanc_by_us_offset > 0 } }
		hidden_effect = {
			subtract_from_variable = { imp_sanc_by_us_offset = 4 }
			country_event = { id = Imp_sanctions.1 }
		}
	}
	option = {
		name = IMP_sanctions_next
		trigger = { OR = { check_variable = { imp_dip_sanc_by_us^num > imp_sanc_by_us_next_offset } check_variable = { imp_fin_sanc_by_us^num > imp_sanc_by_us_next_offset } check_variable = { imp_eco_sanc_by_us^num > imp_sanc_by_us_next_offset } check_variable = { imp_mil_sanc_by_us^num > imp_sanc_by_us_next_offset } } }
		hidden_effect = {
			add_to_variable = { imp_sanc_by_us_offset = 4 }
			country_event = { id = Imp_sanctions.1 }
		}
	}
	option = { name = Imp_sanctions.1.o1 }
}
