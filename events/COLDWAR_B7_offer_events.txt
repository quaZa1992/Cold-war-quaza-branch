# The GDR resists fraternal assistance: Moscow decides (FROM = DDR).

	id = COLDWAR_B7.1
	title = COLDWAR_B7.1.t
	desc = COLDWAR_B7.1.d
	picture = GFX_Good_bye_lenin
	is_triggered_only = yes
	immediate = { log = "[GetDateText]: [Root.GetName]: event COLDWAR_B7.1" }
	option = {
		name = COLDWAR_B7.1.a
		ai_chance = { base = 30 }
		if = { limit = { FROM = { is_subject_of = ROOT } } end_puppet = FROM }
		FROM = { country_event = { id = COLDWAR_B7.2 days = 1 } }
	}
	option = {
		name = COLDWAR_B7.1.b
		ai_chance = { base = 70 }
		if = { limit = { NOT = { has_war_with = FROM } } declare_war_on = { target = FROM type = topple_government } }
		FROM = { country_event = { id = COLDWAR_B7.3 days = 1 } }
	}
}
