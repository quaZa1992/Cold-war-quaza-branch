## IDEOLOGIES

# I don't know if ideologies works on other type of generic portraits but I know it works for political leaders.

# nationalist = Nationalist
# communism = Emerging
# democratic = Western Outlook
# fascism = Salafist
# neutrality = Non-Aligned


## PORTRAIT TYPES

# political = Country Leader
# operative = Spy
# scientist = Scientist (for research facilities)
# army = Military leaders (Random generated division commanders, they can be promoted to general)
# navy = Naval leaders (Random generated admirals, when you recruit a new admiral it can generate from here)


## GENDERS

# male / female
