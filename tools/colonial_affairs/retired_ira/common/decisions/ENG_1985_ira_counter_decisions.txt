# IMPERIA CW — Cold War: Integrated ENG 1985 ira counter decisions — decisions.
# =============================================================================
# ENG 1985 - British counter-IRA decisions
# =============================================================================
# All decisions live inside the ENG_1985_ira_mechanic category so they appear
# directly under the live IRA tracker GUI - the British player gets the
# 5-slider readout, the threat-phase status, and the click-able counters
# in one place.
#
# Each decision:
#   - has its own ENG_1985_<x>_recent country flag for cooldowns
#   - costs PP (always) + at least one secondary cost (CP / stability / convoys)
#   - mutates one or more IRA sliders by meaningful amounts (>= 5 typically)
#   - many also award something concrete (recovered equipment, opinion, etc.)
#
# Designed to be the LATE half of the British player's tool-kit. The early
# half (security sweep, community outreach, etc.) live in the existing
# ENG_1985_decisions.txt; these are the heavier, more committed responses.
# =============================================================================

ENG_1985_ira_mechanic = {
	# -------------------------------------------------------------------------
	# Convoy Interdiction - the answer to foreign sponsors
	# -------------------------------------------------------------------------
	# Visible only when at least one non-British country is in the cooldown
	# of a recent IRA-supply action AND IRA Foreign Funding is over 30. That
	# way the player can't spam-click it as a free militancy reduction - it's
	# a counter-move tied to actual ongoing foreign support.
	ENG_1985_convoy_interdiction = {
		icon = GFX_decision_generic_naval
		cost = 30

		visible = {
			has_country_flag = ENG_1985_ira_mechanic_active
		}

		available = {
			NOT = { has_country_flag = ENG_1985_convoy_interdiction_recent }
			check_variable = { ENG_ira_foreign_funding > 30 }
			command_power > 14
			any_country = {
				NOT = { original_tag = ENG }
				OR = {
					has_country_flag = ENG_1985_ira_arms_recent
					has_country_flag = ENG_1985_ira_funds_recent
					has_country_flag = ENG_1985_ira_intel_recent
				}
			}
		}

		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision ENG_1985_convoy_interdiction"
			custom_effect_tooltip = ENG_1985_convoy_interdiction_tt
			add_command_power = -15
			set_country_flag = {
				flag = ENG_1985_convoy_interdiction_recent
				days = 90
				value = 1
			}
			add_to_variable = { ENG_ira_foreign_funding = -15 }
			add_to_variable = { ENG_ira_militancy = -3 }
			add_to_variable = { ENG_ira_intelligence_penetration = 8 }
			add_to_variable = { ENG_ira_loyalist_tension = -2 }
			add_equipment_to_stockpile = {
				type = infantry_weapons_type
				amount = 50
				producer = ENG
			}
			add_equipment_to_stockpile = {
				type = L_AT_Equipment
				amount = 15
				producer = ENG
			}
			ENG_1985_ira_clamp_values = yes
		}

		ai_will_do = { factor = 0 }
	}
