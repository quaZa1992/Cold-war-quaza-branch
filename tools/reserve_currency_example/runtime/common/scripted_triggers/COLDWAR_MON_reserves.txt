# IMPERIA COLD WAR — Reserve and debt-denomination prototype requirements.
COLDWAR_MON_seed_valid = {
	NOT = { has_country_flag = COLDWAR_MON_initialized }
	NOT = { check_variable = { debt < 0 } }
	check_variable = { currency_strength > 0 }
	has_variable = COLDWAR_MON_seed_debt_usd_share
	has_variable = COLDWAR_MON_seed_reserve_usd_share
	has_variable = COLDWAR_MON_seed_new_usd_share
	NOT = { check_variable = { COLDWAR_MON_seed_debt_usd_share < 0 } }
	NOT = { check_variable = { COLDWAR_MON_seed_debt_usd_share > 1 } }
	NOT = { check_variable = { COLDWAR_MON_seed_reserve_usd_share < 0 } }
	NOT = { check_variable = { COLDWAR_MON_seed_reserve_usd_share > 1 } }
	NOT = { check_variable = { COLDWAR_MON_seed_new_usd_share < 0 } }
	NOT = { check_variable = { COLDWAR_MON_seed_new_usd_share > 1 } }
	# An explicit receipt prevents absent inputs from becoming a zero-exposure migration.
	has_country_flag = COLDWAR_MON_seed_approved
}

# IMPERIA COLD WAR — Execution guards are shared with the decision's availability.
COLDWAR_MON_can_diversify = {
	has_country_flag = COLDWAR_MON_initialized
	custom_trigger_tooltip = {
		tooltip = COLDWAR_MON_clearing_required
		has_country_flag = COLDWAR_MON_clearing_access
	}
	NOT = { has_country_flag = COLDWAR_MON_diversification_cooldown }
	NOT = { has_country_flag = MD_skip_treasury_cost }
	check_variable = { COLDWAR_MON_months_left < 1 }
	check_variable = { COLDWAR_MON_reserve_usd_share > 0.4 }
	NOT = { political_power < 75 }
	NOT = { check_variable = { treasury < 0.025 } }
}
