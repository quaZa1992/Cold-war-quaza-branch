# IMPERIA COLD WAR — Brazil 1970 party identity runtime effects.
# Native elections continue to choose the MD ruling slot; these helpers register identities.

COLDWAR_BRA_init_politics = {
	if = {
		limit = { original_tag = BRA IMP_is_1970_scenario = yes NOT = { has_country_flag = COLDWAR_BRA_1970_politics_initialized } }
		set_variable = { COLDWAR_BRA_1970_party_identity_1 = 1 }
		set_variable = { COLDWAR_BRA_1970_party_identity_2 = 1 }
		set_variable = { COLDWAR_BRA_1970_party_identity_3 = 0 }
		set_variable = { COLDWAR_BRA_1970_party_identity_20 = 1 }
		set_country_flag = COLDWAR_BRA_1970_party_arena_registered
		set_country_flag = COLDWAR_BRA_1970_party_mdb_registered
		set_country_flag = COLDWAR_BRA_1970_party_pcb_clandestine
		set_country_flag = COLDWAR_BRA_1970_party_pcdob_clandestine
		set_country_flag = COLDWAR_BRA_1970_lacerdist_movement_active
		set_country_flag = COLDWAR_BRA_1970_armed_forces_identity
		if = { limit = { has_variable = ruling_party } set_variable = { COLDWAR_BRA_1970_last_ruling_party = ruling_party } }
		set_country_flag = COLDWAR_BRA_1970_politics_initialized
		update_party_name = yes
	}
}

COLDWAR_BRA_1970_open_party_reform = {
	if = { limit = { original_tag = BRA IMP_is_1970_scenario = yes }
		set_country_flag = COLDWAR_BRA_1970_party_reform_open
	}
}

COLDWAR_BRA_1970_reconcile_politics = {
	if = { limit = { original_tag = BRA IMP_is_1970_scenario = yes }
		COLDWAR_BRA_init_politics = yes
		if = { limit = { has_variable = ruling_party NOT = { has_variable = COLDWAR_BRA_1970_last_ruling_party } }
			set_variable = { COLDWAR_BRA_1970_last_ruling_party = ruling_party }
			update_party_name = yes
		}
		else_if = { limit = { has_variable = ruling_party NOT = { check_variable = { COLDWAR_BRA_1970_last_ruling_party = ruling_party } } }
			set_variable = { COLDWAR_BRA_1970_last_ruling_party = ruling_party }
			update_party_name = yes
		}
	}
}

# Call when a posthumous political opening permits the Lacerdist movement to continue.
COLDWAR_BRA_1970_mark_lacerdist_successor = {
	if = { limit = { original_tag = BRA IMP_is_1970_scenario = yes has_country_flag = COLDWAR_BRA_1970_party_reform_open date > 1977.5.21 NOT = { has_country_flag = COLDWAR_BRA_1970_lacerdist_successor_installed } }
		if = { limit = { NOT = { has_country_leader = { name = "Sandra Cavalcanti" } } }
			create_country_leader = { name = "Sandra Cavalcanti" ideology = Nat_Populism traits = { } }
		}
		if = { limit = { has_country_leader = { name = "Sandra Cavalcanti" } }
			set_country_flag = COLDWAR_BRA_1970_lacerdist_successor_installed
		}
	}
}
